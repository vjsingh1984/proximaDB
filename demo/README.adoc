# High-performance gRPC insertion
vectors = []
for product in REAL_PRODUCTS:
    embedding = self.create_embeddings([f"{product['name']} {product['description']}"])[0]
    
    vector = VectorRecord(
        id=product["id"],
        vector=embedding.tolist(),
        metadata=[
            {"key": "name", "value": product["name"]},
            {"key": "category", "value": product["category"]},
            {"key": "brand", "value": product["brand"]},
            {"key": "price", "value": str(product["price"])},
            {"key": "rating", "value": str(product["rating"])}
        ]
    )
    vectors.append(vector)

result = self.grpc_client.insert_vectors(self.collection_id, vectors)

== ⚡ Storage Engine Demo Deep Dive

=== Overview

In-depth comparison of ProximaDB's storage engines with performance analysis and use case recommendations.

=== VIPER Engine (Columnar)

**Architecture:**
- Column-oriented storage
- Advanced compression (ZSTD)
- Predicate pushdown optimization
- Optimized for analytics workloads

**Strengths:**
- 60-90% I/O reduction for filtered queries
- Superior compression ratios
- Fast aggregate operations
- Excellent for read-heavy workloads

**Code Example:**
[source,python]
----
# Optimized for analytics
viper_collection = client.create_collection(
    "analytics_data",
    CollectionConfig(
        dimension=384,
        distance_metric=DistanceMetric.COSINE,
        engine="viper"
    )
)
----

=== SST Engine (Row-based)

**Architecture:**
- Row-oriented storage with LSM trees
- Block-level compression
- Three-stage filtering pipeline
- Optimized for transactional workloads

**Strengths:**
- Fast row reconstruction
- Excellent update performance
- Lower latency for small queries
- Better for mixed read/write patterns

**Code Example:**
[source,python]
----
# Optimized for OLTP
sst_collection = client.create_collection(
    "transactional_data",
    CollectionConfig(
        dimension=384,
        distance_metric=DistanceMetric.COSINE,
        engine="sst"
    )
)
----

=== Performance Benchmarks

[source,text]
----
Benchmark Results (1000 vectors, 384D):

Engine    Insert Rate    Search Latency    Compression    Memory Usage
VIPER     10.3K/sec      1.9ms            3:1           45MB
SST       10.1K/sec      2.1ms            2:1           67MB

Query Performance (metadata filtering):
VIPER: 60-90% I/O reduction via predicate pushdown
SST: 30% processing savings via bloom filters
----

=== Use Case Guidelines

**Choose VIPER when:**
- Analytics and BI workloads
- Read-heavy applications
- Large datasets requiring compression
- Complex metadata filtering
- Batch processing workflows

**Choose SST when:**
- Real-time applications
- Frequent updates and deletes
- Small, latency-sensitive queries
- Mixed read/write patterns
- OLTP-style workloads

# Solution: Manual download
python -c "from sentence_transformers import SentenceTransformer; SentenceTransformer('all-MiniLM-L6-v2')"

==== 2. Connection Issues

**Error:** `Connection refused (gRPC port 5679)`
[source,bash]
----
# Check if server is running
pgrep proximadb-server

# Start server if needed
./target/release/proximadb-server --config demo/docker-config.toml

# Verify gRPC port is open
netstat -an | grep 5679
----

**Error:** `REST API not responding (port 5678)`
[source,bash]
----
# Test health endpoint
curl http://localhost:5678/health

# Check server logs
tail -f /path/to/proximadb/logs/server.log

# Restart server with debug logging
RUST_LOG=debug ./target/release/proximadb-server --config demo/docker-config.toml
----

==== 3. Collection and Data Issues

**Error:** `Collection already exists`
[source,bash]
----
# Delete existing collection
curl -X DELETE http://localhost:5678/collections/ecommerce_products

# Or use different collection name in demo
----

**Error:** `Failed to store collection metadata`
[source,bash]
----
# Check storage permissions
ls -la /data/proximadb/

# Verify configuration
cat demo/docker-config.toml | grep -A5 "\[storage\]"

# Clear storage and restart
rm -rf /data/proximadb/*
./target/release/proximadb-server --config demo/docker-config.toml
----

**Error:** `No results found` in searches
[source,bash]
----
# Verify collection has data
curl http://localhost:5678/collections/ecommerce_products | jq .

# Check vector count
curl -X POST http://localhost:5678/collections/ecommerce_products/search \
  -d '{"vector": [0.1], "top_k": 1}' | jq '.results | length'

# Re-run demo to reload data
python ecommerce_demo.py
----

[source,bash]

# Kill existing server
pkill -f serve_ui.py

python serve_ui.py --port 8091
----

**Error:** `Failed to connect to ProximaDB` in browser
[source,bash]

# Check API_BASE configuration

==== 5. Performance Issues

**Issue:** Low insertion rates (<1000 vectors/sec)
[source,bash]
----
# Solutions:
1. Use gRPC instead of REST
2. Increase batch sizes (100-500 vectors)
3. Check system resources (CPU, memory)
4. Verify storage is not I/O bound
----

**Issue:** High search latency (>10ms)
[source,bash]
----
# Solutions:
1. Reduce top_k values
2. Use appropriate distance metric
3. Check collection size and indexes
4. Monitor system resources
----

import time
import numpy as np
from proximadb import connect_grpc

def benchmark_inserts(client, collection_id, num_vectors, dimension):
    vectors = []
    for i in range(num_vectors):
        vectors.append({
            'id': f'vec_{i}',
            'vector': np.random.rand(dimension).tolist(),
            'metadata': [{'key': 'index', 'value': str(i)}]
        })
    
    start_time = time.time()
    client.insert_vectors(collection_id, vectors)
    elapsed = time.time() - start_time
    
    rate = num_vectors / elapsed
    print(f"Insert rate: {rate:.0f} vectors/sec")
    return rate

def benchmark_search(client, collection_id, dimension, num_queries):
    latencies = []
    
    for i in range(num_queries):
        query_vector = np.random.rand(dimension).tolist()
        
        start_time = time.time()
        client.search_vectors(collection_id, query_vector, top_k=10)
        elapsed = time.time() - start_time
        
        latencies.append(elapsed * 1000)  # Convert to ms
    
    avg_latency = sum(latencies) / len(latencies)
    p95_latency = sorted(latencies)[int(0.95 * len(latencies))]
    
    print(f"Average latency: {avg_latency:.2f}ms")
    print(f"P95 latency: {p95_latency:.2f}ms")
    
    return avg_latency, p95_latency
----

**Example: Document Search System**
[source,python]

# documents.py - Custom demo
from sentence_transformers import SentenceTransformer
from proximadb import connect_grpc

class DocumentSearchSystem:
    def __init__(self):
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
        self.client = connect_grpc("grpc://localhost:5679")
    
    def index_documents(self, documents):
        vectors = []
        for doc_id, content in documents.items():
            embedding = self.model.encode([content])[0]
            vectors.append({
                'id': doc_id,
                'vector': embedding.tolist(),
                'metadata': [
                    {'key': 'content_preview', 'value': content[:100]},
                    {'key': 'length', 'value': str(len(content))}
                ]
            })
        return self.client.insert_vectors("documents", vectors)
    
    def search(self, query, top_k=5):
        query_embedding = self.model.encode([query])[0]
        return self.client.search_vectors("documents", query_embedding.tolist(), top_k=top_k)
----

# api_server.py
from fastapi import FastAPI
from proximadb import connect_grpc
from sentence_transformers import SentenceTransformer

app = FastAPI()
client = connect_grpc("grpc://localhost:5679")
model = SentenceTransformer('all-MiniLM-L6-v2')

@app.post("/search")
async def search_products(query: str, top_k: int = 5):
    embedding = model.encode([query])[0]
    results = client.search_vectors("ecommerce_products", embedding.tolist(), top_k=top_k)
    return {"query": query, "results": results}

@app.post("/products")
async def add_product(name: str, description: str, metadata: dict):
    text = f"{name} {description}"
    embedding = model.encode([text])[0]
    
    vector = {
        'id': f"product_{metadata.get('id', 'auto')}",
        'vector': embedding.tolist(),
        'metadata': [{'key': k, 'value': str(v)} for k, v in metadata.items()]
    }
    
    result = client.insert_vectors("ecommerce_products", [vector])
    return {"status": "success", "result": result}
----
