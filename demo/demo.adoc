== Overview

ProximaDB provides multiple demonstration scripts showcasing different features:

* **working_demo.py** - Basic ProximaDB functionality and performance
* **full_demo.py** - Advanced features with chunking, quantization, HNSW, and dual protocols
* **chunking_demo.py** - Comprehensive text chunking strategies
* **storage_engines_demo.py** - LSM vs VIPER storage engine comparison with real benchmarks
* **financial_analysis_demo.py** - 13F filings analysis with BERT embeddings and financial intelligence

Each demo can be run either:

1. **With Docker** (recommended for isolated environment)
2. **Direct execution** (requires local Python setup)

== ProximaDB Configuration

=== Server Configuration (config.toml)

ProximaDB uses TOML configuration files for server setup. Here's the complete configuration reference:

[source,toml]
----
# /path/to/proximaDB/config.toml
[server]
# Server binding configuration
host = "0.0.0.0"              # Bind to all interfaces
rest_port = 5678              # REST API port
grpc_port = 5679              # gRPC server port
admin_port = 9090             # Admin/metrics port

# Performance tuning
max_connections = 1000        # Maximum concurrent connections
request_timeout_ms = 30000    # Request timeout in milliseconds
worker_threads = 4            # Number of worker threads

[storage]
# Storage engine configuration
data_directory = "./data"     # Base data directory
temp_directory = "./temp"     # Temporary files directory

# Memtable configuration
memtable_size_mb = 128        # Memtable size in MB
memory_flush_size_bytes = 67108864  # 64MB flush threshold

# WAL (Write-Ahead Log) configuration
[storage.wal_config]
wal_urls = [
    "file:///app/data/wal/disk1",
    "file:///app/data/wal/disk2"
]
sync_mode = "PerBatch"        # Options: Always, PerBatch, Periodic, Never
wal_strategy = "Avro"         # Options: Avro, Bincode

# Storage layout for multi-disk setup
[storage.storage_layout]
wal_disks = [
    "/app/data/wal/disk1",
    "/app/data/wal/disk2"
]
storage_disks = [
    "/app/data/storage/disk1",
    "/app/data/storage/disk2"
]

# Disk pairing strategy
[storage.storage_pairing]
strategy = "RoundRobin"       # Options: RoundRobin, Performance

# VIPER engine (Parquet-based columnar storage)
[storage.viper]
enabled = true
compression = "snappy"        # Options: none, snappy, gzip, lz4, zstd
batch_size = 10000           # Vectors per Parquet file

# LSM (Log-Structured Merge) configuration  
[storage.lsm]
enabled = true
max_level = 7                # Maximum LSM levels
level_size_multiplier = 10   # Size multiplier between levels

[indexing]
# Default indexing configuration
default_index = "HNSW"       # Options: HNSW, IVF, Flat

# HNSW index parameters
[indexing.hnsw]
m = 16                       # Maximum connections per node
ef_construction = 200        # Construction time search width
ef_search = 50              # Search time width
max_m = 16                  # Maximum M value
max_m0 = 32                 # Maximum M0 value

# IVF index parameters  
[indexing.ivf]
nlist = 1024                # Number of clusters
nprobe = 64                 # Number of clusters to search

[api_config]
# REST API configuration
enable_cors = true          # Enable CORS for web clients
cors_origins = ["*"]        # Allowed origins
max_request_size_mb = 100   # Maximum request size

# gRPC configuration
enable_reflection = true    # Enable gRPC reflection
max_message_size_mb = 100   # Maximum gRPC message size

[logging]
# Logging configuration
level = "info"              # Options: error, warn, info, debug, trace
format = "json"             # Options: json, text
file_path = "./logs/proximadb.log"
max_file_size_mb = 100      # Log rotation size
max_files = 10              # Number of log files to keep

[metrics]
# Prometheus metrics
enabled = true
endpoint = "/metrics"
port = 9090

[security]
# Security configuration (for production)
enable_tls = false          # Enable TLS encryption
cert_file = ""              # TLS certificate file
key_file = ""               # TLS private key file
auth_enabled = false        # Enable authentication
----

[WARNING]
====
🔒 *Security Notice*: The default configuration is optimized for development. For production deployments:

* Enable TLS encryption
* Configure authentication
* Restrict CORS origins
* Use secure file permissions
====

==== Platform-Specific Setup Scripts

===== Windows Setup (PowerShell)

Create `setup-windows.ps1`:
[source,powershell]
----
# setup-windows.ps1 - Windows ProximaDB Setup Script
param(
    [string]$InstallPath = "C:\ProximaDB",
    [switch]$DevMode = $false
)

Write-Host "🚀 Setting up ProximaDB on Windows..." -ForegroundColor Green

# Create directories
$DataPath = "$InstallPath\data"
$LogsPath = "$InstallPath\logs"
New-Item -ItemType Directory -Force -Path $DataPath, $LogsPath, "$DataPath\wal\disk1", "$DataPath\wal\disk2", "$DataPath\storage\disk1", "$DataPath\storage\disk2"

# Set up Python virtual environment
Write-Host "📦 Setting up Python environment..." -ForegroundColor Yellow
python -m venv "$InstallPath\venv"
& "$InstallPath\venv\Scripts\Activate.ps1"

# Install dependencies
pip install --upgrade pip
Set-Location demo
pip install -r requirements.txt

# Install ProximaDB SDK
Set-Location ..\clients\python
pip install -e .
Set-Location ..\..\

# Create configuration file
$ConfigContent = @"
[server]
host = "127.0.0.1"
rest_port = 5678
grpc_port = 5679
admin_port = 9090

[storage]
data_directory = "$($DataPath -replace '\\', '/')"
[storage.wal_config]
wal_urls = ["file:///$($DataPath -replace '\\', '/')/wal/disk1", "file:///$($DataPath -replace '\\', '/')/wal/disk2"]
sync_mode = "PerBatch"

[logging]
level = "$(if ($DevMode) { 'debug' } else { 'info' })"
file_path = "$($LogsPath -replace '\\', '/')/proximadb.log"
"@

$ConfigContent | Out-File -FilePath "config-windows.toml" -Encoding UTF8

Write-Host "✅ Windows setup complete!" -ForegroundColor Green
Write-Host "💡 Use 'start-proximadb-windows.ps1' to start the server" -ForegroundColor Cyan
----

Create `start-proximadb-windows.ps1`:
[source,powershell]
----
# start-proximadb-windows.ps1 - Start ProximaDB Server on Windows
param(
    [string]$ConfigFile = "config-windows.toml",
    [switch]$Release = $false
)

Write-Host "🚀 Starting ProximaDB Server on Windows..." -ForegroundColor Green

# Activate virtual environment
& "C:\ProximaDB\venv\Scripts\Activate.ps1"

# Set environment variables
$env:RUST_LOG = "info"
$env:PROXIMADB_CONFIG = $ConfigFile

# Build and start server
if ($Release) {
    Write-Host "🏗️ Building release version..." -ForegroundColor Yellow
    cargo build --release --bin proximadb-server
    Write-Host "🌟 Starting ProximaDB in release mode..." -ForegroundColor Green
    & ".\target\release\proximadb-server.exe" --config $ConfigFile
} else {
    Write-Host "🏗️ Building debug version..." -ForegroundColor Yellow
    cargo build --bin proximadb-server
    Write-Host "🔧 Starting ProximaDB in debug mode..." -ForegroundColor Green
    & ".\target\debug\proximadb-server.exe" --config $ConfigFile
}
----

                "backoff_multiplier": 1.5
            }
        )
    )
    
    # Concurrent operations
    async with client.batch_context() as batch:
        # Parallel vector operations
        insert_task = batch.insert_vectors(collection, vectors, ids)
        search_task = batch.search(collection, query_vector, k=100)
        
        # Wait for completion
        insert_result, search_results = await asyncio.gather(
            insert_task, search_task
        )
    
    # Connection automatically cleaned up
----

- Automatic retry with exponential backoff
- Circuit breaker pattern for failing services
- Graceful degradation under load
- Comprehensive error classification and handling

# Resilient Error Handling Example
from proximadb import ProximaDBError, NetworkError, RateLimitError

try:
    result = await client.insert_vectors(collection, vectors, ids)

    # Automatic backoff and retry
    await asyncio.sleep(e.retry_after)
    result = await client.insert_vectors(collection, vectors, ids)

# Recovery script
#!/bin/bash
# restore-proximadb.sh

BACKUP_FILE="$1"
DATA_DIR="/app/data"

if [ -z "$BACKUP_FILE" ]; then
    echo "Usage: $0 <backup_file.tar.gz>"
    exit 1

# Stop ProximaDB
systemctl stop proximadb

# Backup current data
mv "$DATA_DIR" "$DATA_DIR.backup.$(date +%Y%m%d_%H%M%S)"

mkdir -p "$DATA_DIR"

# Restart ProximaDB

echo "Recovery completed from: $BACKUP_FILE"

#!/bin/bash
# maintenance.sh - ProximaDB Maintenance Script

LOG_DIR="/var/log/proximadb"
BACKUP_DIR="/backup/proximadb"

cleanup_logs() {
    echo "🧹 Cleaning up old logs..."

    echo "✅ Log cleanup completed"

    echo "🧹 Cleaning up old backups..."
    find "$BACKUP_DIR" -name "*.tar.gz" -mtime +$RETENTION_DAYS -delete
    echo "✅ Backup cleanup completed"
}

# Function: Optimize indexes
optimize_indexes() {
    echo "🔧 Optimizing indexes..."
    curl -X POST http://localhost:9090/admin/optimize
    echo "✅ Index optimization triggered"
}

# Function: Health check
health_check() {
    echo "🔍 Performing health check..."
    HEALTH=$(curl -s http://localhost:5678/health | jq -r '.status')
    if [ "$HEALTH" = "healthy" ]; then
        echo "✅ ProximaDB is healthy"
        return 0
    else
        echo "❌ ProximaDB health check failed"
        return 1
    fi
}

# Main execution
echo "🚀 Starting ProximaDB maintenance..."

if health_check; then
    cleanup_logs
    cleanup_backups
    optimize_indexes
    echo "🎉 Maintenance completed successfully"

pinecone.init(api_key="your-key", environment="us-west1-gcp")
index = pinecone.Index("example-index")

# ProximaDB equivalent
from proximadb import connect_rest, CollectionConfig, DistanceMetric
client = connect_rest("http://localhost:5678")
collection = client.create_collection("example-index", CollectionConfig(
    dimension=1536, 
    distance_metric=DistanceMetric.COSINE
))

# Vector operations are similar
# Pinecone: index.upsert(vectors=[(id, vector, metadata)])
# ProximaDB: client.insert_vectors(collection_name, vectors, ids, metadata)
----
