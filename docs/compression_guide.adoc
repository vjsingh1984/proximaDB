function updatePresets() {
  const pattern = document.getElementById('dataPattern').value;
  const data = compressionData[pattern];
  if (data) {
    document.getElementById('dataSize').value = data.presets.size;
    document.getElementById('queryFreq').value = data.presets.qps;
  }
}

function calculateCompression() {
  const pattern = document.getElementById('dataPattern').value;
  const dataSize = parseFloat(document.getElementById('dataSize').value) || 100;
  const priority = document.getElementById('priority').value;
  
  const data = compressionData[pattern];
  if (!data) return;
  
  let bestAlg = null, bestScore = -Infinity;
  
  // Get sparsity level from pattern name for better scoring
  const sparsityLevel = pattern.includes('90') ? 0.9 : 
                       pattern.includes('75') ? 0.75 :
                       pattern === 'user_behavior' ? 0.8 :
                       pattern === 'image_features' ? 0.4 :
                       pattern === 'text_vectors' ? 0.3 : 0.1;
  
  for (const [name, alg] of Object.entries(data.algorithms)) {
    if (name === 'none') continue;
    
    const savings = (1 - alg.r) * 100;
    const storageCost = dataSize * 0.023 * 12 * (1 - alg.r); // Annual S3 savings
    const cpuCost = dataSize * 0.05 * (alg.c / 100) * 24 * 365; // CPU overhead
    const netBenefit = storageCost - cpuCost;
    
    // Sparsity-aware scoring - sparse data benefits more from high compression
    const sparsityBonus = sparsityLevel > 0.5 ? savings * 0.2 : 0;
    
    let score = 0;
    switch (priority) {
      case 'storage': score = savings + sparsityBonus; break;
      case 'speed': score = -alg.l * 1000 + sparsityBonus * 0.1; break;
      case 'bandwidth': score = savings * 2 + sparsityBonus; break;
      case 'balanced': score = savings - alg.l * 100 - alg.c + sparsityBonus * 0.5; break;
    }
    
    if (score > bestScore) {
      bestScore = score;
      bestAlg = {name, ...alg, savings, storageCost, cpuCost, netBenefit};
    }
  }
  
  if (bestAlg) {
    const algName = bestAlg.name.replace('_', ' Level ').toUpperCase();
    document.getElementById('bestAlgorithm').textContent = algName;
    document.getElementById('compressionSavings').textContent = `${Math.round(bestAlg.savings)}%`;
    document.getElementById('annualSavings').textContent = `$${Math.round(bestAlg.netBenefit/1000)}K`;
    document.getElementById('latencyImpact').textContent = `+${bestAlg.l}ms`;
    
    document.getElementById('originalSize').textContent = `${dataSize} GB`;
    document.getElementById('compressedSize').textContent = `${Math.round(dataSize * bestAlg.r)} GB`;
    document.getElementById('cpuOverhead').textContent = `+${bestAlg.c}%`;
    
    document.getElementById('storageCost').textContent = `$${Math.round(bestAlg.storageCost).toLocaleString()}/year saved`;
    document.getElementById('networkCost').textContent = `$${Math.round(bestAlg.storageCost * 0.15).toLocaleString()}/year saved`;
    document.getElementById('cpuCost').textContent = `-$${Math.round(bestAlg.cpuCost).toLocaleString()}/year overhead`;
  }
}

  calculateCompression();
});

== Two-Tier Architecture

=== Protocol Level (Transport)
* **Purpose**: Network bandwidth reduction
* **Default**: Disabled (for debugging)
* **Configuration**: Server-wide

```toml
[api]
rest_compression = false  # Keep disabled for debugging
grpc_compression = true   # Enable for production
```

=== Collection Level (Storage)
* **Purpose**: Storage optimization
* **Default**: Per-collection configuration
* **Configuration**: SDK-driven

```python
compression = CompressionConfig(
    algorithm="zstd",
    level=6,  # 1-9, higher = better compression
    enable_quantization=True
)
```

=== Compression Strategy Guidelines

==== When to Use Compression

✅ **Always compress when**:
* Vectors have >50% sparsity
* Storage costs exceed compute costs
* Batch processing is acceptable
* Working with large dimensions (>1024)

⚠️ **Consider carefully when**:
* Ultra-low latency required (<1ms)
* Vectors are already quantized
* Dimensions are small (<256)
* CPU resources are limited

❌ **Avoid compression when**:
* Debugging or development
* Real-time streaming with <100μs latency
* Vectors are pre-compressed
* Random access patterns dominate

**VIPER Engine**: ✅ Quantization recommended
* Columnar format benefits from uniform types
* Better compression ratios
* Faster SIMD operations

**SST Engine**: ❌ Quantization not recommended
* Row format doesn't benefit
* Added complexity without gains
* Use native FP32

== Context-Aware Compression

The unified compression module automatically optimizes for different contexts:

=== SST Blocks
* Algorithm: LZ4 or ZSTD
* Block-level compression
* Preserves random access

=== Vector Serialization
* Algorithm: Any supported
* Header with checksums
* Streaming support

=== Parquet Columns
* Algorithm: 7 supported (ZSTD, LZ4, Snappy, Gzip, Brotli, LZO, None)
* Column-level compression
* Predicate pushdown preserved

=== Performance Tuning

**Low compression ratio (<30%)**
* Switch to ZSTD from LZ4
* Increase compression level
* Enable quantization

**High latency impact (>20%)**
* Switch to LZ4 from ZSTD
* Reduce compression level
* Disable quantization

**Memory pressure**
* Enable compression for all collections
* Use higher compression levels
* Enable tiering for cold data

== Best Practices

1. **Start with defaults** (ZSTD level 6)
2. **Measure impact** before optimization
3. **Test with your data** - compression varies by content
4. **Consider workload** - read vs write patterns
5. **Monitor metrics** - track ratio and performance
