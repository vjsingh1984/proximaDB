Comprehensive guide to using the ProximaDB Python SDK for vector search, graph operations, document storage, and observability.

== Installation

[source,bash]
----
pip install proximadb-sdk
----

For development with all optional features:

[source,bash]
----
pip install proximadb-sdk[full]
----

== Quick Start

[source,python]
----
from proximadb_sdk import ProximaDBClient, VectorRecord, DistanceMetric

# Connect to ProximaDB
client = ProximaDBClient(url="http://localhost:5678")

# Create a collection
client.create_collection(
    name="my_vectors",
    dimension=768,
    distance_metric=DistanceMetric.COSINE
)

# Insert vectors
client.insert_vectors("my_vectors", [
    VectorRecord(id="doc1", vector=[0.1] * 768, metadata={"title": "Document 1"}),
    VectorRecord(id="doc2", vector=[0.2] * 768, metadata={"title": "Document 2"}),
])

# Search
results = client.search("my_vectors", vector=[0.15] * 768, top_k=10)
for r in results:
    print(f"{r.id}: {r.score:.4f}")
----

=== Embedded Mode

[source,python]
----
from proximadb_sdk import connect_embedded

# In-process database (no network overhead)
db = connect_embedded(data_dir="/tmp/proximadb")

== Storage Engine Selection

ProximaDB offers 6 specialized storage engines. Choose based on your workload:

[cols="1,2,2", options="header"]
|===
| Engine | Best For | Python Config

| SST
| Write-heavy, real-time ingestion
| `engine="sst"`

| HELIX
| Locality-optimized, spatial clustering
| `engine="helix"`

| VIPER
| Analytics, columnar queries
| `engine="viper"`

| SWIFT
| Ultra-low latency (<5K vectors)
| `engine="swift"`

| NOVA
| Mixed workloads
| `engine="nova"`

| RAPTOR
| Adaptive, dynamic workloads
| `engine="raptor"`
|===

[source,python]
----
from proximadb_sdk import StorageEngine

client.create_collection(
    name="analytics_vectors",
    dimension=768,
    engine=StorageEngine.VIPER,  # Best for analytics queries
)
----

=== Search Vectors

[source,python]
----
# Basic search
results = client.search(

    collection="my_collection",
    vector=query_embedding,
    top_k=10,

== Graph Operations

=== Basic Graph Operations

[source,python]
----
# Create a graph
client.create_graph(
    name="knowledge",
    engine="orion",  # In-memory, sub-ms latency
)

# Create nodes
client.create_nodes("knowledge", [
    {"id": "entity_1", "label": "Person", "properties": {"name": "Alice"}},
    {"id": "entity_2", "label": "Company", "properties": {"name": "TechCorp"}},
])

# Create edges
client.create_edges("knowledge", [
    {"source": "entity_1", "target": "entity_2", "type": "WORKS_AT", "weight": 1.0}
])

# Traverse
results = client.traverse(
    "knowledge",
    start_node="entity_1",
    edge_types=["WORKS_AT", "KNOWS"],
    max_depth=2
)
----

=== Graph Analytics

[source,python]

from proximadb_sdk import GraphAnalytics

analytics = GraphAnalytics(client)

# PageRank
pagerank_scores = analytics.pagerank("knowledge", damping_factor=0.85)

    max_depth=3,

=== Pattern Matching

[source,python]
----
from proximadb_sdk import GraphPattern, node, relationship

# Find triangles: A -> B -> C -> A
pattern = (GraphPattern()
    .node("a", label="Person")
    .relationship("r1", type="KNOWS")
    .node("b", label="Person")
    .relationship("r2", type="KNOWS")
    .node("c", label="Person")
    .relationship("r3", type="KNOWS", target="a"))

matches = analytics.pattern_match("knowledge", pattern)

== Multi-Modal Queries

Combine vector, graph, document, and observability queries:

[source,python]
----
from proximadb_sdk import (
    MultiModalQueryBuilder,
    MultiModalQueryExecutor,
    FusionStrategy,
    TimeDecayFunction,
)

# Build a multi-modal query
query = (MultiModalQueryBuilder()
    # Vector similarity search
    .vector("products", query_embedding, top_k=20)
    # Graph traversal from vector results
    .graph_from_vector_results("knowledge", edge_types=["RELATED_TO"])
    # Join by semantic similarity
    .join_semantic("vector.embedding", "graph.node.embedding", threshold=0.8)
    # Fuse with Reciprocal Rank Fusion
    .fuse(FusionStrategy.RRF)
    # Apply time decay
    .with_time_decay(TimeDecayFunction.EXPONENTIAL, halflife_hours=24)
    .limit(10)
    .build())

# Execute
executor = MultiModalQueryExecutor(client)
results = executor.execute(query)

for record in results:
    print(f"{record['id']}: score={record.get('_rrf_score', 0):.4f}")
----

== AutoML Features

Automatic engine selection and hyperparameter optimization:

[source,python]
----
from proximadb_sdk import AutoML, WorkloadType, OptimizationGoal

automl = AutoML(client)

# Automatic engine recommendation
recommendation = automl.recommend_engine(
    dimension=768,
    expected_count=1_000_000,
    workload_type=WorkloadType.READ_HEAVY,
    latency_requirement_ms=10,
    memory_budget_gb=8,
)

print(f"Recommended: {recommendation.engine}")
print(f"Confidence: {recommendation.confidence:.2f}")
print(f"Reasoning: {recommendation.reasoning}")

# Apply recommendation
client.create_collection(
    name="optimized_vectors",
    dimension=768,
    engine=recommendation.engine,
    engine_config=recommendation.config,
)

# Hyperparameter optimization
optimized = automl.optimize_hyperparameters(
    collection="my_collection",
    goal=OptimizationGoal.LATENCY,
    budget_minutes=10,
)
----

== Observability

=== Metrics and Tracing

[source,python]
----
from proximadb_sdk import Observability, traced, metered

# Initialize observability
obs = Observability(client, enable_metrics=True, enable_tracing=True)

# Decorate functions for automatic instrumentation
@traced
@metered
def search_products(query):
    return client.search("products", vector=query, top_k=10)

# Export metrics (Prometheus format)
prometheus_metrics = obs.metrics.export_prometheus()

# Export traces (OpenTelemetry format)
traces = obs.tracer.export_otlp()
----

== Security

=== OAuth2 Authentication

[source,python]
----
from proximadb_sdk import OAuth2Config, OAuth2Provider, OAuth2TokenManager

# Configure OAuth2
oauth_config = OAuth2Config(
    provider=OAuth2Provider.OKTA,
    client_id="my-client-id",
    client_secret="my-client-secret",
    token_url="https://my-org.okta.com/oauth2/default/v1/token",
    scopes=["openid", "profile", "proximadb:read", "proximadb:write"],
    auto_refresh=True,
)

# Get token manager
token_manager = OAuth2TokenManager(oauth_config)

# Client credentials flow (for service accounts)
token = token_manager.client_credentials()

# Use with client
client = ProximaDBClient(
    url="http://localhost:5678",
    api_key=token.access_token,
)
----

=== RBAC Permissions

[source,python]
----
from proximadb_sdk import (
    RBACManager, SecurityManager, SecurityContext,
    security_context, Role
)

# Initialize RBAC
rbac = RBACManager()

# Check permissions
allowed = rbac.check_permission(
    roles=["developer"],
    required_permission="vector:search"
)

# Create security context
security = SecurityManager(rbac_manager=rbac)
ctx = security.create_context(
    user_id="user123",
    tenant_id="tenant_abc",
    roles=["analyst"]
)

# Use context for operations
with security_context(ctx):
    # Operations run with analyst permissions
    results = client.search("vectors", query, top_k=10)
----

== Filtering

=== Filter Builder

[source,python]
----
from proximadb_sdk import FilterBuilder, eq, gt, lt, in_list, and_filters, or_filters

# Simple filters
filter1 = FilterBuilder().add(eq("category", "electronics")).build()
filter2 = FilterBuilder().add(gt("price", 100)).add(lt("price", 500)).build()

# IN operator
filter3 = FilterBuilder().add(in_list("status", ["active", "pending"])).build()

# Logical combinations
complex_filter = or_filters(
    and_filters(eq("category", "electronics"), gt("price", 200)),
    and_filters(eq("category", "books"), lt("price", 50))
)
----

== Batch Operations

[source,python]
----
from proximadb_sdk import RequestBatcher, BatchStrategy

# Create batcher
batcher = RequestBatcher(
    client=client,
    strategy=BatchStrategy.SIZE,
    batch_size=100,
    flush_interval_ms=1000,
)

# Queue operations
for record in records:
    batcher.queue_insert("my_collection", record)

# Flush remaining
batcher.flush()

# Get metrics
metrics = batcher.get_metrics()
print(f"Batches sent: {metrics.batches_sent}")
print(f"Average batch size: {metrics.avg_batch_size}")
----

== Error Handling

[source,python]
----
from proximadb_sdk import (
    ProximaDBError,
    CollectionNotFoundError,
    VectorDimensionError,
    AuthenticationError,
    RateLimitError,
    TimeoutError,
)

try:
    results = client.search("my_collection", query, top_k=10)
except CollectionNotFoundError:
    print("Collection does not exist")
except VectorDimensionError as e:
    print(f"Dimension mismatch: expected {e.expected}, got {e.actual}")
except AuthenticationError:
    print("Authentication failed - check API key")
except RateLimitError as e:
    print(f"Rate limited - retry after {e.retry_after}s")
except TimeoutError:
    print("Request timed out")
except ProximaDBError as e:
    print(f"ProximaDB error: {e}")
----

== Best Practices

=== Connection Management

[source,python]
----
# Use connection pooling for high throughput
client = ProximaDBClient(
    url="http://localhost:5678",
    pool_size=20,
    timeout=30.0,
)

# Reuse client instance across requests
# Don't create new clients for each operation
----

=== Batch Inserts

[source,python]
----
# Good: Batch inserts
records = [VectorRecord(id=f"v{i}", vector=vecs[i]) for i in range(1000)]
client.insert_vectors("collection", records)

# Bad: Individual inserts
for i in range(1000):
    client.insert_vectors("collection", [VectorRecord(id=f"v{i}", vector=vecs[i])])
----

=== Engine Selection

[source,python]
----
# Use AutoML for optimal engine selection
from proximadb_sdk import AutoML

automl = AutoML(client)
rec = automl.recommend_engine(
    dimension=768,
    expected_count=1_000_000,
)
# Apply recommendation
----

=== Resource Cleanup

[source,python]
----
# Embedded mode - always flush before exit
db = connect_embedded("/tmp/data")
try:
    # ... operations ...
finally:
    db.flush()
    db.close()
