function calculateROI() {
    const monthlyCost = parseFloat(document.getElementById('monthlyCost').value) || 10000;
    const vectorCount = parseFloat(document.getElementById('vectorCount').value) || 100;
    
    // Infrastructure: 70% savings
    const infraSavings = Math.round(monthlyCost * 12 * 0.70);
    
    // API calls: Based on vector count and operations
    const apiSavings = Math.round(vectorCount * 180); // $180 per million vectors/year
    
    // Total savings
    const totalSavings = infraSavings + apiSavings;
    
    // Payback period (implementation cost ~$10K)
    const paybackPeriod = (10000 / (totalSavings / 12)).toFixed(1);
    
    // Update display
    document.getElementById('infraSavings').textContent = infraSavings.toLocaleString();
    document.getElementById('apiSavings').textContent = apiSavings.toLocaleString();
    document.getElementById('totalSavings').textContent = totalSavings.toLocaleString();
    document.getElementById('paybackPeriod').textContent = paybackPeriod;
}

document.addEventListener('DOMContentLoaded', calculateROI);
</script>

== Key Value Drivers

=== 1. Infrastructure Consolidation (70% Savings)
* **From**: 3-5 server cluster + load balancers + replication
* **To**: Single container with cloud-native resilience
* **Result**: $210K/year saved on $300K infrastructure

=== 2. Spot Instance Enablement (80% Compute Savings)
* **Stateless architecture**: <1 minute failover
* **70-80% spot usage**: vs 0% for traditional databases
* **Result**: $450/month vs $5,000/month for 10TB workload

=== 3. Intelligent Caching (85% API Reduction)
* **Bloom filters**: 95% metadata scan reduction
* **Tiered storage**: 10% hot, 90% cold optimization
* **Result**: 85M fewer database queries per day

=== 4. Hardware Acceleration (2.9× Performance)

* **Zero-copy operations**: Proto-first architecture
