== Overview

This guide covers all vector operations in ProximaDB including insertion, retrieval, updates, deletion, and search.

== Inserting Vectors

=== Single Vector Insert

[source,python]
----
import numpy as np
from proximadb import Client

client = Client("localhost:5678")
collection = client.get_collection("products")

# Insert a single vector
vector = np.random.rand(512).astype('float32')
collection.insert(
    id="vec_001",
    vector=vector,
    metadata={
        "name": "Product A",
        "price": 29.99,
        "category": "electronics"
    }
)
----

=== Batch Insert

[source,python]
----
# Prepare batch data
num_vectors = 1000
dimension = 512
vectors = np.random.rand(num_vectors, dimension).astype('float32')
ids = [f"vec_{i:04d}" for i in range(num_vectors)]
metadata = [
    {"category": "electronics", "price": np.random.uniform(10, 1000)}
    for _ in range(num_vectors)
]

# Batch insert
results = collection.insert_batch(
    ids=ids,
    vectors=vectors,
    metadata=metadata,
    batch_size=100  # Process in chunks
)

print(f"Inserted: {results.success_count}/{results.total_count}")
----

=== Streaming Insert

[source,python]
----
# For continuous data streams
def vector_generator():
    """Generate vectors from a data source"""
    while True:
        # Simulate data source
        vector = np.random.rand(512).astype('float32')
        metadata = {"timestamp": time.time()}
        yield vector, metadata

# Stream insert with auto-batching
stream = vector_generator()
collection.insert_stream(
    stream=stream,
    batch_size=100,
    flush_interval=1.0  # Flush every second
)
----

== Searching Vectors

=== Basic Search

[source,python]
----
# Search for similar vectors
query_vector = np.random.rand(512).astype('float32')
results = collection.search(
    vector=query_vector,
    k=10  # Return top 10 results
)

for result in results:
    print(f"ID: {result.id}")
    print(f"Score: {result.score:.4f}")
    print(f"Metadata: {result.metadata}")
----

=== Advanced Search Parameters

[source,python]
----
results = collection.search(
    vector=query_vector,
    k=20,
    search_params={
        "ef": 200,           # HNSW search quality
        "nprobe": 10,        # IVF clusters to search
        "radius": 0.8,       # Distance threshold
        "include_vectors": True,  # Return vectors
        "timeout": 5.0       # Timeout in seconds
    }
)
----

=== Filtered Search

[source,python]
----
# Search with metadata filters
results = collection.search(
    vector=query_vector,
    k=10,
    filter={
        "category": "electronics",
        "price": {"$gte": 50, "$lte": 500},
        "tags": {"$in": ["wireless", "bluetooth"]}
    }
)

# Complex filter with AND/OR
results = collection.search(
    vector=query_vector,
    k=10,
    filter={
        "$and": [
            {"category": "electronics"},
            {
                "$or": [
                    {"brand": "Apple"},
                    {"brand": "Samsung"}
                ]
            }
        ]
    }
)
----

=== Multi-Vector Search

[source,python]
----
# Search with multiple query vectors
query_vectors = np.random.rand(5, 512).astype('float32')

batch_results = collection.search_batch(

    k=10

for i, results in enumerate(batch_results):

== Retrieving Vectors

=== Get by ID

[source,python]
----
# Retrieve single vector
vector_record = collection.get("vec_001")

print(f"Metadata: {vector_record.metadata}")

# Retrieve multiple vectors
ids = ["vec_001", "vec_002", "vec_003"]

=== Fetch with Projection

[source,python]
----
# Fetch only specific metadata fields
record = collection.get(
    "vec_001",
    projection=["name", "price"]  # Only return these fields
)

# Exclude vector data
record = collection.get(
    "vec_001",
    include_vector=False  # Metadata only
)
----

== Vector Transformations

=== Normalization

[source,python]

# Normalize vectors for cosine similarity
from sklearn.preprocessing import normalize

vectors = np.random.rand(100, 512)
normalized_vectors = normalize(vectors, norm='l2', axis=1)

collection.insert_batch(
    vectors=normalized_vectors,

=== Dimensionality Reduction

[source,python]

from sklearn.decomposition import PCA

# Reduce dimension before insertion
high_dim_vectors = np.random.rand(100, 1024)
pca = PCA(n_components=512)
reduced_vectors = pca.fit_transform(high_dim_vectors)

collection.insert_batch(
    vectors=reduced_vectors,
    metadata=[{"original_dim": 1024} for _ in range(100)]
)

== Performance Optimization

import time

# Test different batch sizes
batch_sizes = [10, 50, 100, 500, 1000]
vectors = np.random.rand(10000, 512).astype('float32')

for batch_size in batch_sizes:

    collection.insert_batch(

=== Async Operations

[source,python]
----
import asyncio

async def async_insert(collection, vectors):
    """Asynchronous vector insertion"""
    tasks = []
    for i in range(0, len(vectors), 100):
        batch = vectors[i:i+100]
        task = collection.insert_batch_async(vectors=batch)
        tasks.append(task)
    
    results = await asyncio.gather(*tasks)
    return results

# Run async inserts
vectors = np.random.rand(1000, 512).astype('float32')
results = asyncio.run(async_insert(collection, vectors))
----

== Error Handling

=== Handling Insert Errors

[source,python]
----
try:
    collection.insert(
        id="vec_001",
        vector=np.random.rand(512).astype('float32'),
        metadata={"price": "invalid"}  # Type error
    )
except ValidationError as e:
    print(f"Validation error: {e}")
except DuplicateKeyError as e:
    print(f"Duplicate ID: {e}")
except Exception as e:
    print(f"Unexpected error: {e}")
----

=== Retry Logic

[source,python]
----
from tenacity import retry, stop_after_attempt, wait_exponential

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=1, max=10)
)
def insert_with_retry(collection, vector, metadata):
    """Insert with automatic retry"""
    return collection.insert(
        vector=vector,
        metadata=metadata
    )

# Use with retry
vector = np.random.rand(512).astype('float32')
result = insert_with_retry(collection, vector, {"name": "Product"})
----

from tqdm import tqdm

# Track batch insert progress
vectors = np.random.rand(10000, 512).astype('float32')
batch_size = 100

with tqdm(total=len(vectors), desc="Inserting vectors") as pbar:
    for i in range(0, len(vectors), batch_size):

        collection.insert_batch(vectors=batch)
        pbar.update(len(batch))

== Best Practices

=== ID Management

* Use UUIDs for unique identification
* Include timestamp in ID for time-series data
* Use consistent prefixes for categorization

=== Batch Operations

* Optimal batch size: 100-1000 vectors
* Use streaming for continuous data
* Enable compression for large batches

=== Memory Management

* Clear large arrays after insertion
* Use generators for streaming data
* Monitor memory usage during bulk operations
