=== Embedded Mode Tests

[source,rust]
----
// src/embedded/tests.rs
#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::TempDir;

    #[tokio::test]
    async fn test_embedded_create_collection() {
        let tmp = TempDir::new().unwrap();
        let db = EmbeddedProximaDB::new(tmp.path()).await.unwrap();

        db.create_collection("test", 128, "sst").await.unwrap();

        let collections = db.list_collections().await.unwrap();
        assert_eq!(collections.len(), 1);
    }

    #[tokio::test]
    async fn test_embedded_insert_search() {
        let tmp = TempDir::new().unwrap();
        let db = EmbeddedProximaDB::new(tmp.path()).await.unwrap();

        db.create_collection("test", 4, "sst").await.unwrap();
        db.insert("test", vec!["v1"], vec![vec![1.0, 0.0, 0.0, 0.0]]).await.unwrap();

        let results = db.search("test", vec![1.0, 0.0, 0.0, 0.0], 1).await.unwrap();
        assert_eq!(results.len(), 1);
        assert_eq!(results[0].id, "v1");
    }
}

== CI/CD Integration

Add to `.github/workflows/ci.yml`:

[source,yaml]
----
coverage:
  runs-on: ubuntu-latest
  steps:
    - uses: actions/checkout@v4
    - uses: dtolnay/rust-toolchain@stable
      with:
        components: llvm-tools-preview

    - name: Install cargo-llvm-cov
      run: cargo install cargo-llvm-cov

    - name: Generate coverage
      run: cargo llvm-cov --lib --lcov --output-path lcov.info

    - name: Upload to Codecov
      uses: codecov/codecov-action@v4
      with:
        files: lcov.info
        fail_ci_if_error: true

    - name: Check coverage threshold
      run: |
        COVERAGE=$(cargo llvm-cov report 2>&1 | grep "^TOTAL" | awk '{print $NF}' | tr -d '%')
        if (( $(echo "$COVERAGE < 50" | bc -l) )); then
          echo "Coverage $COVERAGE% is below 50% threshold"
          exit 1
        fi
----

== Appendix: Coverage Commands Reference

[source,bash]
----
# Full HTML report
cargo llvm-cov --lib --html --output-dir coverage

# Text summary
cargo llvm-cov report

# Single module coverage
cargo llvm-cov --lib -- --test-threads=1 storage::

# Coverage with specific test
cargo llvm-cov --lib -- test_name

# JSON output for CI
cargo llvm-cov --lib --json --output-path coverage.json

# LCOV format for Codecov/Coveralls
cargo llvm-cov --lib --lcov --output-path lcov.info

# Exclude test files from report
cargo llvm-cov --lib --ignore-filename-regex "tests?"
----
