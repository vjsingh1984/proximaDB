=== Design Philosophy

The refactored design creates specialized cache types for different data patterns while maximizing code reuse through shared infrastructure components:

* **Separation of Concerns**: Each cache type optimized for its specific use case
* **Code Reuse**: Shared base traits and utilities across all cache types
* **Synergies**: Common invalidation, metrics, and storage mechanisms

=== Core Design Patterns

==== 1. Template Method Pattern for Base Cache

[source,rust]
----
// Shared base trait with template methods
pub trait BaseCache: Send + Sync {
    type Key: Hash + Eq + Clone;
    type Value: Clone;
    
    // Template method - defines algorithm structure
    async fn get_with_hooks(&self, key: &Self::Key) -> Option<Self::Value> {
        self.pre_get_hook(key).await;
        
        // Check each tier
        if let Some(value) = self.check_l1(key).await {
            self.record_hit(CacheTier::L1);
            return Some(value);
        }
        
        if let Some(value) = self.check_l2(key).await {
            self.promote_to_l1(key, &value).await;
            self.record_hit(CacheTier::L2);
            return Some(value);
        }
        
        if let Some(value) = self.check_l3(key).await {
            self.promote_to_l2(key, &value).await;
            self.record_hit(CacheTier::L3);
            return Some(value);
        }
        
        self.record_miss();
        None
    }
    
    // Hooks for specialization
    async fn pre_get_hook(&self, key: &Self::Key) {}
    async fn check_l1(&self, key: &Self::Key) -> Option<Self::Value>;
    async fn check_l2(&self, key: &Self::Key) -> Option<Self::Value>;
    async fn check_l3(&self, key: &Self::Key) -> Option<Self::Value>;
    
    fn metrics(&self) -> &CacheMetrics;
}
----

pub struct VectorDataCache {
    // Shared infrastructure
    base: BaseCacheImpl,
    
    // Vector-specific optimizations
    vector_index: Arc<VectorIndex>,
    compression: VectorCompression,
    prefetch_strategy: VectorPrefetchStrategy,
}

impl VectorDataCache {
    pub async fn batch_get(&self, ids: &[String]) -> Vec<Option<VectorRecord>> {
        // Optimize batch retrieval with locality grouping
        let grouped = self.group_by_locality(ids);
        let mut results = Vec::with_capacity(ids.len());
        
        for group in grouped {
            let batch = self.fetch_batch(group).await;
            results.extend(batch);
        }
        
        results
    }
    
    pub async fn similarity_prefetch(&self, query: &[f32], k: usize) {
        // Prefetch likely candidates based on index
        let candidates = self.vector_index.find_candidates(query, k * 2).await;
        
        for candidate_id in candidates {
            if !self.base.contains(&candidate_id).await {
                tokio::spawn(self.clone().prefetch_vector(candidate_id));
            }
        }
    }
}

==== QueryResultCache

[source,rust]
----
pub struct QueryResultCache {
    // Shared infrastructure
    base: BaseCacheImpl,
    
    // Query-specific optimizations
    query_normalizer: QueryNormalizer,
    result_merger: ResultMerger,
    similarity_index: LSHIndex,
}

impl QueryResultCache {
    pub async fn get_similar_results(&self, query: &Query, threshold: f32) -> Option<SearchResults> {
        // Use LSH to find similar cached queries
        let query_hash = self.query_normalizer.normalize_and_hash(query);
        let similar_keys = self.similarity_index.find_similar(query_hash, threshold).await;
        
        for key in similar_keys {
            if let Some(results) = self.base.get(&key).await {
                if self.results_applicable(query, &results) {
                    return Some(self.adapt_results(query, results));
                }
            }
        }
        
        None
    }
    
    pub async fn cache_with_subqueries(&self, query: &Query, results: SearchResults) {
        // Cache main query and useful subqueries
        let key = self.query_normalizer.normalize_and_hash(query);
        self.base.put(key, results.clone()).await;
        
        for subquery in self.extract_subqueries(query) {
            let subkey = self.query_normalizer.normalize_and_hash(&subquery);
            let subresults = self.filter_results_for_subquery(&results, &subquery);
            self.base.put(subkey, subresults).await;
        }
    }
}

==== FilterBitmapCache

[source,rust]
----
pub struct FilterBitmapCache {
    // Shared infrastructure
    base: BaseCacheImpl,
    
    // Filter-specific optimizations
    bitmap_store: RoaringBitmapStore,
    filter_optimizer: FilterOptimizer,
    incremental_updater: IncrementalUpdater,
}

impl FilterBitmapCache {
    pub async fn evaluate_complex_filter(&self, filter: &Filter) -> RoaringBitmap {
        // Decompose filter into cacheable components
        let components = self.filter_optimizer.decompose(filter);
        
        let mut bitmaps = Vec::new();
        for component in components {
            let bitmap = if let Some(cached) = self.base.get(&component.key()).await {
                cached
            } else {
                let computed = self.compute_filter_bitmap(&component).await;
                self.base.put(component.key(), computed.clone()).await;
                computed
            };
            bitmaps.push((component.operator, bitmap));
        }
        
        self.filter_optimizer.combine_bitmaps(bitmaps)
    }
}

==== Unified Memory Management

[source,rust]
----
pub struct UnifiedMemoryManager {
    caches: Vec<Arc<dyn MemoryManaged>>,
    total_limit: usize,
    allocation_strategy: AllocationStrategy,
}

impl UnifiedMemoryManager {
    pub async fn rebalance(&self) {
        let usage_stats = self.collect_usage_stats().await;
        let new_allocations = self.allocation_strategy.compute_allocations(
            &usage_stats,
            self.total_limit
        );
        
        for (cache, new_limit) in self.caches.iter().zip(new_allocations) {
            cache.resize(new_limit).await;
        }
    }
    
    pub async fn handle_memory_pressure(&self) {
        // Coordinate eviction across all caches
        let eviction_candidates = self.collect_eviction_candidates().await;
        let to_evict = self.select_global_victims(eviction_candidates);
        
        for (cache_id, keys) in to_evict {
            self.caches[cache_id].evict_batch(keys).await;
        }
    }
}

==== Cross-Cache Prefetching

[source,rust]
----
pub struct CrossCachePrefetcher {
    cache_coordinator: Arc<CacheCoordinator>,
    pattern_analyzer: PatternAnalyzer,
}

impl CrossCachePrefetcher {
    pub async fn on_vector_access(&self, vector_id: &str) {
        // When a vector is accessed, prefetch related data
        
        // Prefetch metadata
        self.cache_coordinator
            .metadata_cache()
            .prefetch_metadata(vector_id)
            .await;
        
        // Prefetch common filters for this vector
        let likely_filters = self.pattern_analyzer
            .predict_filters(vector_id)
            .await;
        
        for filter in likely_filters {
            self.cache_coordinator
                .filter_cache()
                .prefetch_filter(filter)
                .await;
        }
        
        // Prefetch index nodes if vector is frequently searched
        if self.pattern_analyzer.is_hot_vector(vector_id).await {
            self.cache_coordinator
                .index_cache()
                .prefetch_vector_index_path(vector_id)
                .await;
        }
    }
}

== Conclusion

This unified cache enhancement design provides:

1. **Specialized Performance**: Each cache type optimized for its specific data patterns
2. **Code Reuse**: Shared infrastructure reduces duplication and maintenance burden
3. **Cross-Cache Synergies**: Coordinated prefetching and invalidation improve overall hit rates
4. **Flexibility**: Clear extension points through traits and patterns
5. **Production Ready**: Phased approach ensures smooth transition with minimal risk

The architecture maintains backward compatibility while significantly improving query performance through intelligent caching strategies tailored to vector database operations.
