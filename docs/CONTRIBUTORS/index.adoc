== 🚀 Quick Start for Contributors

=== Development Environment Setup (15 minutes)

==== Prerequisites
[source,bash]
----
# Required tools
rustup install stable
cargo install cargo-watch cargo-tarpaulin
git clone https://github.com/vjsingh1984/proximadb.git
cd proximadb

# Optional but recommended
cargo install cargo-edit         # For managing dependencies
cargo install cargo-audit        # Security audit
cargo install cargo-outdated     # Dependency updates
----

==== Build and Test
[source,bash]
----
# Build all components
cargo build --all --release

# Run the full test suite (takes ~10-15 minutes)
cargo test --all --release

# Run quick smoke tests
cargo test --all --lib --release

# Format and lint
cargo fmt && cargo clippy -- -D warnings

# Run benchmarks
cargo bench optimization_benchmarks
----

==== Development Server
[source,bash]
----
# Start development server with hot reload
cargo watch -x 'run --bin proximadb-server -- --config demo/local-demo-config.toml'

# Verify it's working
curl http://localhost:5678/health
curl http://localhost:5678/metrics
----

==== Required Test Patterns

[source,rust]
----
// REQUIRED: All tests must initialize hardware capabilities
#[test]
fn test_example() {
    let _ = proximadb::core::hardware_capabilities::initialize_hardware_capabilities_default();
    // Test implementation here
}

// REQUIRED: Use real services, no mocking
#[tokio::test]
async fn test_vector_operations() {
    let _ = hardware_capabilities::initialize_hardware_capabilities_default();
    
    // Create real service instance, not mock
    let service = VectorOperationsService::new_test_instance().await;
    
    // Test against real implementation
    let result = service.search(test_query).await;
    assert!(result.is_ok());
}

// PREFERRED: Test against multiple storage engines  
#[test]
fn test_across_engines() {
    let _ = hardware_capabilities::initialize_hardware_capabilities_default();
    
    for engine in ["sst", "viper", "prism"] {
        let service = create_service_with_engine(engine);
        let result = test_operation(&service);
        assert!(result.is_ok(), "Failed with engine: {}", engine);
    }
}
----

=== Code Quality Standards

==== Rust Best Practices
[source,rust]
----
// PREFERRED: Use DashMap over Arc<RwLock>
use dashmap::DashMap;
pub struct ConcurrentService {
    collections: DashMap<String, Collection>, // Not Arc<RwLock<HashMap>>
}

// REQUIRED: Comprehensive error handling
use thiserror::Error;

#[derive(Error, Debug)]
pub enum ServiceError {
    #[error("Storage error: {0}")]
    Storage(#[from] StorageError),
    
    #[error("Invalid dimension: expected {expected}, got {actual}")]  
    InvalidDimension { expected: usize, actual: usize },
}

// PREFERRED: Proto-first data structures
use proximadb_proto::VectorRecord; // Not custom structs

pub struct VectorOperations {
    // Work directly with proto types
    records: Vec<VectorRecord>,
}
----

==== Performance Guidelines
[source,rust]
----
// REQUIRED: Hardware capability awareness
impl DistanceComputation {
    pub fn new(capabilities: &HardwareCapabilities) -> Self {
        let kernel = match capabilities.simd_support {
            SimdCapability::Avx512 => Box::new(Avx512Kernel::new()),
            SimdCapability::Avx2 => Box::new(Avx2Kernel::new()),
            _ => Box::new(ScalarKernel::new()),
        };
        Self { kernel }
    }
}

// PREFERRED: Zero-copy patterns where possible
pub fn process_vectors(vectors: &[VectorRecord]) -> Result<Vec<&VectorRecord>> {
    // Return references, not clones when possible
    Ok(vectors.iter().filter(|v| v.is_valid()).collect())
}

// REQUIRED: Async for I/O, blocking for CPU-intensive
pub struct VectorService {
    thread_pool: ThreadPool, // For CPU-intensive work
}

impl VectorService {
    // I/O operations: async
    pub async fn read_vectors(&self) -> Result<Vec<VectorRecord>> {
        tokio::fs::read(&self.path).await?;
        // ... 
    }
    
    // CPU-intensive: spawn_blocking
    pub async fn compute_distances(&self, vectors: Vec<VectorRecord>) -> Result<DistanceMatrix> {
        let handle = self.thread_pool.spawn_blocking(move || {
            compute_distances_simd(vectors) // CPU-intensive work
        });
        handle.await?
    }
}
