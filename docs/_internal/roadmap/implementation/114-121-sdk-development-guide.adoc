== Overview

This guide provides ready-to-use project templates, CI/CD configurations, and development workflows for implementing the 8 language SDKs (114-121).

== Priority Order

[cols="1,2,1,2,2", options="header"]
|===
| Order | SDK | ID | Rationale | Dependencies
| 1 | Go | 117 | Infra ecosystem, simple concurrency | net/http, grpc-go
| 2 | Java | 114 | Enterprise, JVM ecosystem | OkHttp, gRPC-Java
| 3 | Node.js | 119 | Web developers, async | node-fetch, @grpc/grpc-js
| 4 | Rust | 116 | Performance, native | reqwest, tonic
| 5 | .NET | 118 | Enterprise Windows | HttpClient, Grpc.Net
| 6 | C++ | 115 | Embedded, games | libcurl, grpc++
| 7 | Ruby | 120 | Rails apps | faraday, grpc
| 8 | Swift | 121 | iOS/macOS | URLSession, grpc-swift
|===

	golangci-lint run ./...

	go test -bench=. -benchmem ./...

integration-test:
	PROXIMADB_URL=http://localhost:5678 go test -v -tags=integration ./...

=== CI/CD Pipeline

[source,yaml]
----
# .github/workflows/go.yml
name: Go SDK

on:
  push:
    paths:
      - 'go/**'
      - '.github/workflows/go.yml'
  pull_request:
    paths:
      - 'go/**'

jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        go-version: ['1.21', '1.22']

    steps:
      - uses: actions/checkout@v4

      - name: Set up Go
        uses: actions/setup-go@v5
        with:
          go-version: ${{ matrix.go-version }}

      - name: Install dependencies
        run: |
          cd go
          go mod download

      - name: Lint
        uses: golangci/golangci-lint-action@v3
        with:
          working-directory: go

      - name: Test
        run: |
          cd go
          make test

      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          files: go/coverage.out
          flags: go

  integration:
    runs-on: ubuntu-latest
    needs: test
    services:
      proximadb:
        image: proximadb/proximadb:latest
        ports:
          - 5678:5678
          - 5679:5679

    steps:
      - uses: actions/checkout@v4

      - name: Set up Go
        uses: actions/setup-go@v5
        with:
          go-version: '1.22'

      - name: Wait for ProximaDB
        run: |
          for i in {1..30}; do
            curl -s http://localhost:5678/health && break
            sleep 1
          done

      - name: Integration tests
        run: |
          cd go
          make integration-test

  release:
    runs-on: ubuntu-latest
    needs: [test, integration]
    if: startsWith(github.ref, 'refs/tags/go-v')

    steps:
      - uses: actions/checkout@v4

      - name: Set up Go
        uses: actions/setup-go@v5
        with:
          go-version: '1.22'

      - name: Publish
        run: |
          # Go modules are published by pushing tags
          echo "Go SDK published via git tag"
----

=== CI/CD Pipeline

[source,yaml]
----
# .github/workflows/java.yml
name: Java SDK

on:
  push:
    paths:
      - 'java/**'
      - '.github/workflows/java.yml'
  pull_request:
    paths:
      - 'java/**'

jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        java-version: ['17', '21']

    steps:
      - uses: actions/checkout@v4

      - name: Set up JDK
        uses: actions/setup-java@v4
        with:
          java-version: ${{ matrix.java-version }}
          distribution: 'temurin'
          cache: 'maven'

      - name: Build and test
        run: |
          cd java
          mvn -B verify

      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          files: java/target/site/jacoco/jacoco.xml
          flags: java

  integration:
    runs-on: ubuntu-latest
    needs: test

    steps:
      - uses: actions/checkout@v4

      - name: Set up JDK
        uses: actions/setup-java@v4
        with:
          java-version: '21'
          distribution: 'temurin'
          cache: 'maven'

      - name: Integration tests
        run: |
          cd java
          mvn -B verify -Pintegration

  release:
    runs-on: ubuntu-latest
    needs: [test, integration]
    if: startsWith(github.ref, 'refs/tags/java-v')

    steps:
      - uses: actions/checkout@v4

      - name: Set up JDK
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'
          server-id: ossrh
          server-username: MAVEN_USERNAME
          server-password: MAVEN_PASSWORD
          gpg-private-key: ${{ secrets.GPG_PRIVATE_KEY }}
          gpg-passphrase: MAVEN_GPG_PASSPHRASE

      - name: Publish to Maven Central
        run: |
          cd java
          mvn -B deploy -Prelease
        env:
          MAVEN_USERNAME: ${{ secrets.OSSRH_USERNAME }}
          MAVEN_PASSWORD: ${{ secrets.OSSRH_TOKEN }}
          MAVEN_GPG_PASSPHRASE: ${{ secrets.GPG_PASSPHRASE }}
----

== Cross-SDK Integration Tests

=== Test Runner

[source,python]
----
#!/usr/bin/env python3
# integration-tests/test_all_sdks.py

import subprocess
import sys
import time
import requests

SDKS = {
    'go': 'make -C ../go integration-test',
    'java': 'mvn -f ../java/pom.xml verify -Pintegration',
    'nodejs': 'npm --prefix ../nodejs run test:integration',
    'rust': 'cargo test --manifest-path ../rust/Cargo.toml --features integration',
    'dotnet': 'dotnet test ../dotnet --filter Category=Integration',
    'cpp': 'ctest --test-dir ../cpp/build -R integration',
    'ruby': 'bundle exec rspec --tag integration ../ruby/spec',
    'swift': 'swift test --filter IntegrationTests ../swift',
}

def wait_for_proximadb(url='http://localhost:5678/health', timeout=60):
    """Wait for ProximaDB to be ready"""
    start = time.time()
    while time.time() - start < timeout:
        try:
            r = requests.get(url, timeout=5)
            if r.status_code == 200:
                print("ProximaDB is ready")
                return True
        except requests.RequestException:
            pass
        time.sleep(1)
    return False

def run_sdk_tests(sdk_name):
    """Run integration tests for a specific SDK"""
    if sdk_name not in SDKS:
        print(f"Unknown SDK: {sdk_name}")
        return False

    cmd = SDKS[sdk_name]
    print(f"\n{'='*60}")
    print(f"Running {sdk_name} SDK integration tests")
    print(f"{'='*60}\n")

    result = subprocess.run(cmd, shell=True)
    return result.returncode == 0

def main():
    if not wait_for_proximadb():
        print("ProximaDB not available")
        sys.exit(1)

    sdks_to_test = sys.argv[1:] if len(sys.argv) > 1 else SDKS.keys()

    results = {}
    for sdk in sdks_to_test:
        results[sdk] = run_sdk_tests(sdk)

    print("\n" + "="*60)
    print("RESULTS")
    print("="*60)
    for sdk, passed in results.items():
        status = "PASS" if passed else "FAIL"
        print(f"  {sdk}: {status}")

    if not all(results.values()):
        sys.exit(1)

if __name__ == '__main__':
    main()
----
