// src/cdc/event.rs

use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use uuid::Uuid;

/// Unified change event format for all CDC sources
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ChangeEvent {
    /// Unique event ID
    pub id: Uuid,
    /// Source system information
    pub source: SourceInfo,
    /// Event timestamp
    pub timestamp: DateTime<Utc>,
    /// Logical sequence number
    pub lsn: u64,
    /// Operation type
    pub operation: Operation,
    /// Collection/table name
    pub collection: String,
    /// Primary key/ID
    pub key: String,
    /// Before state (for updates/deletes)
    #[serde(skip_serializing_if = "Option::is_none")]
    pub before: Option<RecordState>,
    /// After state (for inserts/updates)
    #[serde(skip_serializing_if = "Option::is_none")]
    pub after: Option<RecordState>,
    /// Transaction context
    #[serde(skip_serializing_if = "Option::is_none")]
    pub transaction: Option<TransactionInfo>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SourceInfo {
    /// Connector type
    pub connector: ConnectorType,
    /// Source database name
    pub database: String,
    /// Schema/namespace
    #[serde(skip_serializing_if = "Option::is_none")]
    pub schema: Option<String>,
    /// Server identifier
    pub server_id: String,
}

#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq)]
#[serde(rename_all = "lowercase")]
pub enum ConnectorType {
    PostgreSQL,
    MySQL,
    MongoDB,
    ProximaDB,
}

#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq)]
#[serde(rename_all = "lowercase")]
pub enum Operation {
    Insert,
    Update,
    Delete,
    Truncate,
    SchemaChange,
    Snapshot,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RecordState {
    /// Vector data
    #[serde(skip_serializing_if = "Option::is_none")]
    pub vector: Option<Vec<f32>>,
    /// Metadata fields
    pub metadata: HashMap<String, serde_json::Value>,
    /// Raw source data
    #[serde(skip_serializing_if = "Option::is_none")]
    pub raw: Option<serde_json::Value>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TransactionInfo {
    pub id: String,
    pub begin_time: DateTime<Utc>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub commit_time: Option<DateTime<Utc>>,
    pub total_events: u32,
    pub event_position: u32,
}

impl ChangeEvent {
    pub fn new_insert(
        source: SourceInfo,
        collection: String,
        key: String,
        after: RecordState,
    ) -> Self {
        Self {
            id: Uuid::new_v4(),
            source,
            timestamp: Utc::now(),
            lsn: 0,
            operation: Operation::Insert,
            collection,
            key,
            before: None,
            after: Some(after),
            transaction: None,
        }
    }

    pub fn new_update(
        source: SourceInfo,
        collection: String,
        key: String,
        before: RecordState,
        after: RecordState,
    ) -> Self {
        Self {
            id: Uuid::new_v4(),
            source,
            timestamp: Utc::now(),
            lsn: 0,
            operation: Operation::Update,
            collection,
            key,
            before: Some(before),
            after: Some(after),
            transaction: None,
        }
    }

    pub fn new_delete(
        source: SourceInfo,
        collection: String,
        key: String,
        before: RecordState,
    ) -> Self {
        Self {
            id: Uuid::new_v4(),
            source,
            timestamp: Utc::now(),
            lsn: 0,
            operation: Operation::Delete,
            collection,
            key,
            before: Some(before),
            after: None,
            transaction: None,
        }
    }

    pub fn with_lsn(mut self, lsn: u64) -> Self {
        self.lsn = lsn;
        self
    }

    pub fn with_transaction(mut self, tx: TransactionInfo) -> Self {
        self.transaction = Some(tx);
        self
    }
}

// src/cdc/offset.rs

use async_trait::async_trait;
use std::path::Path;

/// Trait for storing CDC offsets
#[async_trait]
pub trait OffsetStore: Send + Sync {
    /// Get the current offset for a source
    async fn get(&self, source_id: &str) -> Result<Option<Offset>, CdcError>;

    /// Save an offset
    async fn save(&self, source_id: &str, offset: &Offset) -> Result<(), CdcError>;

    /// Delete an offset
    async fn delete(&self, source_id: &str) -> Result<(), CdcError>;
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum Offset {
    /// PostgreSQL LSN
    Lsn(u64),
    /// MySQL GTID set
    Gtid(String),
    /// MySQL binlog position
    BinlogPosition { file: String, position: u64 },
    /// MongoDB resume token
    ResumeToken(bson::Document),
    /// Generic sequence number
    Sequence(u64),
}

/// File-based offset store
pub struct FileOffsetStore {
    path: PathBuf,
}

impl FileOffsetStore {
    pub fn new<P: AsRef<Path>>(path: P) -> Result<Self, CdcError> {
        let path = path.as_ref().to_path_buf();
        std::fs::create_dir_all(&path)?;
        Ok(Self { path })
    }

    fn offset_file(&self, source_id: &str) -> PathBuf {
        self.path.join(format!("{}.offset", source_id))
    }
}

#[async_trait]
impl OffsetStore for FileOffsetStore {
    async fn get(&self, source_id: &str) -> Result<Option<Offset>, CdcError> {
        let path = self.offset_file(source_id);
        if !path.exists() {
            return Ok(None);
        }

        let content = tokio::fs::read_to_string(&path).await?;
        let offset: Offset = serde_json::from_str(&content)?;
        Ok(Some(offset))
    }

    async fn save(&self, source_id: &str, offset: &Offset) -> Result<(), CdcError> {
        let path = self.offset_file(source_id);
        let content = serde_json::to_string_pretty(offset)?;

        // Atomic write
        let temp_path = path.with_extension("tmp");
        tokio::fs::write(&temp_path, content).await?;
        tokio::fs::rename(&temp_path, &path).await?;

        Ok(())
    }

    async fn delete(&self, source_id: &str) -> Result<(), CdcError> {
        let path = self.offset_file(source_id);
        if path.exists() {
            tokio::fs::remove_file(&path).await?;
        }
        Ok(())
    }
}

// src/cdc/connectors/postgres/connector.rs

use tokio_postgres::replication::LogicalReplicationStream;
use tokio_postgres::{Client, NoTls};

pub struct PostgresConnector {
    config: PostgresConfig,
    client: Option<Client>,
    offset_store: Arc<dyn OffsetStore>,
    decoder: PgOutputDecoder,
}

#[derive(Debug, Clone)]
pub struct PostgresConfig {
    pub connection_string: String,
    pub slot_name: String,
    pub publication: String,
    pub tables: Vec<TableConfig>,
    pub snapshot_mode: SnapshotMode,
}

#[derive(Debug, Clone)]
pub struct TableConfig {
    pub schema: String,
    pub table: String,
    pub primary_key: Vec<String>,
    pub vector_column: Option<String>,
    pub embed_columns: Option<Vec<String>>,
    pub metadata_columns: Vec<String>,
}

impl PostgresConnector {
    pub async fn new(
        config: PostgresConfig,
        offset_store: Arc<dyn OffsetStore>,
    ) -> Result<Self, CdcError> {
        Ok(Self {
            config,
            client: None,
            offset_store,
            decoder: PgOutputDecoder::new(),
        })
    }

    pub async fn start(
        &mut self,
        sender: mpsc::Sender<ChangeEvent>,
    ) -> Result<(), CdcError> {
        // Create replication connection
        let (client, connection) = tokio_postgres::connect(
            &self.config.connection_string,
            NoTls,
        ).await?;

        // Spawn connection handler
        tokio::spawn(async move {
            if let Err(e) = connection.await {
                tracing::error!("PostgreSQL connection error: {}", e);
            }
        });

        self.client = Some(client);

        // Ensure replication slot exists
        self.ensure_slot().await?;

        // Get starting LSN
        let start_lsn = self.get_start_lsn().await?;

        // Start logical replication
        let query = format!(
            "START_REPLICATION SLOT {} LOGICAL {} (proto_version '1', publication_names '{}')",
            self.config.slot_name,
            start_lsn,
            self.config.publication,
        );

        let client = self.client.as_ref().unwrap();
        let duplex_stream = client.copy_both_simple::<bytes::Bytes>(&query).await?;
        let mut stream = LogicalReplicationStream::new(duplex_stream);

        while let Some(message) = stream.next().await {
            match message? {
                ReplicationMessage::XLogData(data) => {
                    let events = self.decoder.decode(&data.data())?;
                    for event in events {
                        let change_event = self.to_change_event(event, data.wal_end())?;
                        sender.send(change_event).await?;
                    }

                    // Save offset
                    self.offset_store.save(
                        &self.source_id(),
                        &Offset::Lsn(data.wal_end()),
                    ).await?;
                }
                ReplicationMessage::PrimaryKeepAlive(keepalive) => {
                    if keepalive.reply() == 1 {
                        stream.standby_status_update(
                            keepalive.wal_end(),
                            keepalive.wal_end(),
                            keepalive.wal_end(),
                            keepalive.timestamp(),
                            0,
                        ).await?;
                    }
                }
            }
        }

        Ok(())
    }

    async fn ensure_slot(&self) -> Result<(), CdcError> {
        let client = self.client.as_ref().unwrap();

        // Check if slot exists
        let row = client.query_opt(
            "SELECT slot_name FROM pg_replication_slots WHERE slot_name = $1",
            &[&self.config.slot_name],
        ).await?;

        if row.is_none() {
            // Create slot
            client.execute(
                &format!(
                    "SELECT pg_create_logical_replication_slot('{}', 'pgoutput')",
                    self.config.slot_name
                ),
                &[],
            ).await?;
        }

        Ok(())
    }

    async fn get_start_lsn(&self) -> Result<String, CdcError> {
        if let Some(Offset::Lsn(lsn)) = self.offset_store.get(&self.source_id()).await? {
            return Ok(format!("{:X}/{:X}", lsn >> 32, lsn & 0xFFFFFFFF));
        }

        // Start from current position
        let client = self.client.as_ref().unwrap();
        let row = client.query_one("SELECT pg_current_wal_lsn()::text", &[]).await?;
        Ok(row.get(0))
    }

    fn source_id(&self) -> String {
        format!("postgres_{}", self.config.slot_name)
    }
}
