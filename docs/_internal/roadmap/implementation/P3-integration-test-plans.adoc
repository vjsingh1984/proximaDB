"""
Test: Backpressure signal propagation
Validates: Server signals backpressure when buffer fills
"""

import time

def test_backpressure_signaling(client, stream_client):
    client.create_collection("backpressure_test", dimension=1024, engine="sst")

    backpressure_received = []

    with stream_client.stream_insert("backpressure_test",
                                     config={"buffer_size": 100}) as stream:
        # Send vectors rapidly to trigger backpressure
        for i in range(500):
            stream.send({
                "id": f"vec_{i}",
                "vector": [float(i)] * 1024
            })

            # Check for backpressure in acks
            try:
                ack = stream.receive_ack(timeout=0.1)
                if ack.backpressure.level != "none":
                    backpressure_received.append(ack.backpressure)
            except TimeoutError:
                pass

        stream.flush()

    # Should have received some backpressure
    assert len(backpressure_received) > 0
    assert any(bp.level in ["medium", "high", "critical"] for bp in backpressure_received)

    client.delete_collection("backpressure_test")

"""
Test: Live query notifies on new matching insert
Validates: Subscription receives updates when relevant vectors added
"""

import threading

def test_live_query_new_insert_notification(client, stream_client):
    client.create_collection("live_insert", dimension=128, engine="helix")

    updates_received = []

    def listen_for_updates(subscription):
        for update in subscription:
            updates_received.append(update)
            if len(updates_received) >= 2:
                break

    # Start subscription
    subscription = stream_client.subscribe_query(
        collection="live_insert",
        vector=[0.5] * 128,
        top_k=3,
        include_initial=True
    )

    listener = threading.Thread(target=listen_for_updates, args=(subscription,))
    listener.start()

    # Wait for subscription to be active
    time.sleep(0.5)

    # Insert vector that should match
    client.insert("live_insert", [{
        "id": "new_vec",
        "vector": [0.5] * 128  # Exact match to query
    }])

    listener.join(timeout=5)
    subscription.close()

    # Should have initial (empty) + insert update
    assert len(updates_received) >= 1
    insert_updates = [u for u in updates_received if u.type == "insert"]
    assert len(insert_updates) >= 1
    assert insert_updates[0].changed.id == "new_vec"

    client.delete_collection("live_insert")
