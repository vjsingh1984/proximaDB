// src/streaming/ring_buffer.rs

use std::cell::UnsafeCell;
use std::mem::MaybeUninit;
use std::sync::atomic::{AtomicUsize, Ordering};

/// Lock-free ring buffer for stream buffering
pub struct RingBuffer<T> {
    buffer: Box<[UnsafeCell<MaybeUninit<T>>]>,
    capacity: usize,
    mask: usize,
    head: AtomicUsize,
    tail: AtomicUsize,
    watermarks: Watermarks,
}

struct Watermarks {
    low: AtomicUsize,
    high: AtomicUsize,
}

// Safety: T must be Send for cross-thread access
unsafe impl<T: Send> Send for RingBuffer<T> {}
unsafe impl<T: Send> Sync for RingBuffer<T> {}

impl<T> RingBuffer<T> {
    pub fn new(capacity: usize) -> Self {
        assert!(capacity.is_power_of_two(), "Capacity must be power of 2");

        let buffer: Vec<UnsafeCell<MaybeUninit<T>>> = (0..capacity)
            .map(|_| UnsafeCell::new(MaybeUninit::uninit()))
            .collect();

        Self {
            buffer: buffer.into_boxed_slice(),
            capacity,
            mask: capacity - 1,
            head: AtomicUsize::new(0),
            tail: AtomicUsize::new(0),
            watermarks: Watermarks {
                low: AtomicUsize::new(capacity / 4),
                high: AtomicUsize::new(capacity * 3 / 4),
            },
        }
    }

    /// Non-blocking push, returns Err if full
    pub fn try_push(&self, value: T) -> Result<(), T> {
        let tail = self.tail.load(Ordering::Relaxed);
        let next_tail = tail.wrapping_add(1);

        let head = self.head.load(Ordering::Acquire);
        if next_tail.wrapping_sub(head) > self.capacity {
            return Err(value);
        }

        // CAS to claim the slot
        match self.tail.compare_exchange_weak(
            tail,
            next_tail,
            Ordering::Release,
            Ordering::Relaxed,
        ) {
            Ok(_) => {
                let index = tail & self.mask;
                unsafe {
                    (*self.buffer[index].get()).write(value);
                }
                Ok(())
            }
            Err(_) => Err(value),
        }
    }

    /// Check backpressure level
    pub fn backpressure_level(&self) -> BackpressureLevel {
        let len = self.len();
        let high = self.watermarks.high.load(Ordering::Relaxed);
        let low = self.watermarks.low.load(Ordering::Relaxed);

        if len >= self.capacity - 1 {
            BackpressureLevel::Critical
        } else if len >= high {
            BackpressureLevel::High
        } else if len >= (high + low) / 2 {
            BackpressureLevel::Medium
        } else if len >= low {
            BackpressureLevel::Low
        } else {
            BackpressureLevel::None
        }
    }

    pub fn len(&self) -> usize {
        let tail = self.tail.load(Ordering::Relaxed);
        let head = self.head.load(Ordering::Relaxed);
        tail.wrapping_sub(head)
    }

    /// Drain up to n elements
    pub fn drain(&self, max: usize) -> Vec<T> {
        let mut result = Vec::with_capacity(max);
        for _ in 0..max {
            match self.try_pop() {
                Some(v) => result.push(v),
                None => break,
            }
        }
        result
    }

    fn try_pop(&self) -> Option<T> {
        let head = self.head.load(Ordering::Relaxed);
        let tail = self.tail.load(Ordering::Acquire);

        if head == tail {
            return None;
        }

        match self.head.compare_exchange_weak(
            head,
            head.wrapping_add(1),
            Ordering::Release,
            Ordering::Relaxed,
        ) {
            Ok(_) => {
                let index = head & self.mask;
                unsafe {
                    Some((*self.buffer[index].get()).assume_init_read())
                }
            }
            Err(_) => None,
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum BackpressureLevel {
    None,
    Low,
    Medium,
    High,
    Critical,
}

// src/streaming/coordinator.rs

use dashmap::DashMap;
use std::sync::Arc;
use tokio::sync::mpsc;

pub struct StreamCoordinator {
    sessions: DashMap<StreamId, StreamSession>,
    rate_limiter: RateLimiter,
    config: StreamConfig,
    metrics: Arc<StreamMetrics>,
}

pub struct StreamSession {
    pub id: StreamId,
    pub collection: String,
    pub buffer: Arc<RingBuffer<VectorRecord>>,
    pub state: AtomicCell<SessionState>,
    pub created_at: Instant,
    pub last_activity: AtomicCell<Instant>,
    pub ack_sender: mpsc::Sender<AckMessage>,
}

#[derive(Clone)]
pub struct StreamConfig {
    pub max_streams: usize,
    pub default_buffer_size: usize,
    pub global_rate_limit: u64,
    pub flush_interval: Duration,
}

impl Default for StreamConfig {
    fn default() -> Self {
        Self {
            max_streams: 1000,
            default_buffer_size: 10000,
            global_rate_limit: 1_000_000,
            flush_interval: Duration::from_millis(100),
        }
    }
}

impl StreamCoordinator {
    pub fn new(config: StreamConfig) -> Self {
        Self {
            sessions: DashMap::new(),
            rate_limiter: RateLimiter::new(config.global_rate_limit),
            config,
            metrics: Arc::new(StreamMetrics::new()),
        }
    }

    pub async fn create_session(
        &self,
        collection: String,
        config: SessionConfig,
    ) -> Result<StreamId, StreamError> {
        // Check capacity
        if self.sessions.len() >= self.config.max_streams {
            return Err(StreamError::TooManySessions);
        }

        let id = StreamId::new();
        let buffer_size = config.buffer_size.unwrap_or(self.config.default_buffer_size);
        let (ack_tx, _ack_rx) = mpsc::channel(1000);

        let session = StreamSession {
            id: id.clone(),
            collection,
            buffer: Arc::new(RingBuffer::new(buffer_size.next_power_of_two())),
            state: AtomicCell::new(SessionState::Active),
            created_at: Instant::now(),
            last_activity: AtomicCell::new(Instant::now()),
            ack_sender: ack_tx,
        };

        self.sessions.insert(id.clone(), session);
        self.metrics.active_streams.inc();

        Ok(id)
    }

    pub async fn push_records(
        &self,
        session_id: &StreamId,
        records: Vec<VectorRecord>,
    ) -> Result<PushResult, StreamError> {
        // Rate limiting
        if !self.rate_limiter.check(records.len() as u64) {
            return Err(StreamError::RateLimited);
        }

        let session = self.sessions.get(session_id)
            .ok_or(StreamError::SessionNotFound)?;

        session.last_activity.store(Instant::now());

        let mut pushed = 0;
        for record in records {
            match session.buffer.try_push(record) {
                Ok(()) => pushed += 1,
                Err(_) => break,
            }
        }

        self.metrics.records_received.inc_by(pushed as u64);

        Ok(PushResult {
            pushed,
            backpressure: session.buffer.backpressure_level(),
        })
    }

    pub fn close_session(&self, session_id: &StreamId) {
        if self.sessions.remove(session_id).is_some() {
            self.metrics.active_streams.dec();
        }
    }
}

// src/network/grpc/streaming_service.rs

use tonic::{Request, Response, Status, Streaming};
use tokio_stream::wrappers::ReceiverStream;

pub struct StreamingServiceImpl {
    coordinator: Arc<StreamCoordinator>,
    storage: Arc<dyn UnifiedStorageEngine>,
}

#[tonic::async_trait]
impl StreamingService for StreamingServiceImpl {
    type StreamInsertStream = ReceiverStream<Result<StreamInsertResponse, Status>>;

    async fn stream_insert(
        &self,
        request: Request<Streaming<StreamInsertRequest>>,
    ) -> Result<Response<Self::StreamInsertStream>, Status> {
        let mut stream = request.into_inner();
        let (tx, rx) = tokio::sync::mpsc::channel(100);

        let coordinator = self.coordinator.clone();
        let storage = self.storage.clone();

        // Spawn task to handle incoming stream
        tokio::spawn(async move {
            let mut session_id: Option<StreamId> = None;
            let mut pending_sequences: Vec<u64> = Vec::new();

            while let Some(result) = stream.next().await {
                match result {
                    Ok(request) => {
                        // Create session on first message
                        if session_id.is_none() {
                            match coordinator.create_session(
                                request.collection.clone(),
                                SessionConfig::default(),
                            ).await {
                                Ok(id) => session_id = Some(id),
                                Err(e) => {
                                    let _ = tx.send(Err(Status::internal(e.to_string()))).await;
                                    return;
                                }
                            }
                        }

                        // Push records
                        let records: Vec<VectorRecord> = request.vectors
                            .into_iter()
                            .map(|v| v.into())
                            .collect();

                        match coordinator.push_records(
                            session_id.as_ref().unwrap(),
                            records,
                        ).await {
                            Ok(result) => {
                                pending_sequences.push(request.sequence);

                                // Send ack
                                let response = StreamInsertResponse {
                                    acked_sequences: pending_sequences.drain(..).collect(),
                                    backpressure: Some(BackpressureSignal {
                                        level: result.backpressure.into(),
                                        suggested_delay_ms: result.backpressure.delay_ms(),
                                        buffer_percent: result.buffer_percent,
                                    }),
                                    server_timestamp: timestamp_now(),
                                };

                                if tx.send(Ok(response)).await.is_err() {
                                    break;
                                }
                            }
                            Err(e) => {
                                let _ = tx.send(Err(Status::internal(e.to_string()))).await;
                            }
                        }
                    }
                    Err(e) => {
                        let _ = tx.send(Err(e)).await;
                        break;
                    }
                }
            }

            // Cleanup session
            if let Some(id) = session_id {
                coordinator.close_session(&id);
            }
        });

        Ok(Response::new(ReceiverStream::new(rx)))
    }
}
