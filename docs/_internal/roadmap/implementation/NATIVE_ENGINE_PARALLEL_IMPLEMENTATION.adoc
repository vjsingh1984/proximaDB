:toc: left
:toclevels: 3

== Overview

This document defines the parallel workstream implementation plan for the ProximaDB Native Compute Engine, following TDD methodology and SOLID design principles.

== Design Principles

=== SOLID Principles Application

[cols="1,3,3"]
|===
|Principle |Application |Implementation

|**S**ingle Responsibility
|Each module has one clear purpose
|HeaderCache for metadata, SplitPruner for pruning, IoExecutor for I/O

|**O**pen/Closed
|Extend via traits, not modification
|`PruningStrategy` trait, `DecodeStrategy` trait, `IoStrategy` trait

|**L**iskov Substitution
|All implementations interchangeable
|Any `PruningStrategy` works with `SplitPlanner`

|**I**nterface Segregation
|Small, focused traits
|Separate `VectorPruner`, `ScalarPruner`, `SpatialPruner` traits

|**D**ependency Inversion
|Depend on abstractions
|Components take `Arc<dyn Trait>` not concrete types
|===

=== TDD Workflow

[source,text]
----
For each component:
1. Write failing test that defines expected behavior
2. Implement minimum code to pass test
3. Refactor while keeping tests green
4. Add edge case tests
5. Integration test with dependent components
----

/// Decoder strategy trait (O in SOLID - extend via new impls)
pub trait SimdDecoder: Send + Sync {
    /// Decode BitPacked integers
    fn decode_bitpacked_i64(
        &self,
        input: &[u8],
        bits_per_value: u8,
        output: &mut [i64],
    ) -> Result<usize>;

    /// Decode Delta-encoded floats
    fn decode_delta_f32(
        &self,
        input: &[u8],
        base: f32,
        output: &mut [f32],
    ) -> Result<usize>;

    /// CPU features supported
    fn supported_features(&self) -> CpuFeatures;
}

/// Factory for selecting best decoder (D in SOLID)
pub struct DecoderFactory;

impl DecoderFactory {
    pub fn best_available() -> Arc<dyn SimdDecoder> {
        #[cfg(target_arch = "x86_64")]
        {
            if is_x86_feature_detected!("avx2") {
                return Arc::new(Avx2Decoder::new());
            }
        }
        #[cfg(target_arch = "aarch64")]
        {
            return Arc::new(NeonDecoder::new());
        }
        Arc::new(ScalarDecoder::new())
    }
}

/// I/O execution strategy (D in SOLID - depend on abstraction)
#[async_trait]
pub trait IoStrategy: Send + Sync {
    /// Execute optimized read
    async fn execute_read(
        &self,
        file: &str,
        ranges: &[ByteRange],
    ) -> Result<Vec<Vec<u8>>>;

    /// Estimate I/O cost
    fn estimate_cost(&self, ranges: &[ByteRange]) -> IoCostEstimate;
}

/// Range optimization strategy
pub trait RangeOptimizer: Send + Sync {
    /// Coalesce ranges to reduce requests
    fn coalesce(&self, ranges: Vec<ByteRange>, threshold: usize) -> Vec<ByteRange>;

    /// Split large ranges for parallelism
    fn split_for_parallelism(&self, range: ByteRange, target_size: usize) -> Vec<ByteRange>;
}

/// Native operator trait (L in SOLID - all impls substitutable)
#[async_trait]
pub trait NativeOperator: Send + Sync + Debug {
    /// Execute operator
    async fn execute(
        &self,
        input: RecordBatchStream,
        context: &ExecutionContext,
    ) -> Result<RecordBatchStream>;

    /// Schema of output
    fn output_schema(&self) -> SchemaRef;

    /// Estimated cardinality
    fn cardinality(&self) -> CardinalityEstimate;

    /// Memory requirements
    fn memory_estimate(&self) -> MemoryEstimate;

    /// Can this operator be parallelized?
    fn supports_parallelism(&self) -> bool;
}

/// Operator pipeline (S in SOLID - single responsibility: execute pipeline)
pub struct OperatorPipeline {
    operators: Vec<Arc<dyn NativeOperator>>,
    parallelism: usize,
}

impl OperatorPipeline {
    pub async fn execute(&self, input: RecordBatchStream) -> Result<RecordBatchStream> {
        let mut stream = input;
        for op in &self.operators {
            stream = op.execute(stream, &self.context).await?;
        }
        Ok(stream)
    }
}

/// Engine adapter trait (I in SOLID - focused interface)
pub trait EngineAdapter: Send + Sync {
    type Config;
    type Handle;

    fn create_handle(&self, collection: &str, config: Self::Config) -> Result<Self::Handle>;
}

/// Universal adapter (D in SOLID - depends on abstractions)
pub struct UniversalAdapter {
    format_registry: Arc<FormatRegistry>,
    header_cache: Arc<EnhancedHeaderCache>,
    io_layer: Arc<SmartIoLayer>,
    operators: Arc<OperatorRegistry>,
}

impl UniversalAdapter {
    /// Create DataFusion TableProvider
    pub fn to_datafusion(&self, collection: &str) -> Arc<dyn TableProvider> {
        Arc::new(ProximaDBTableProvider::new_with_native(
            collection.to_string(),
            self.header_cache.clone(),
            self.io_layer.clone(),
            self.operators.clone(),
        ))
    }

    /// Execute query natively (bypass external engines)
    pub async fn execute_native(&self, sql: &str) -> Result<RecordBatchStream> {
        let plan = self.plan_query(sql)?;
        self.execute_plan(plan).await
    }
}
