        if dimension <= self.arc_max_dimension && expected_clones <= self.arc_max_clones {
            CloneStrategy::Arc
        } else {
            CloneStrategy::DeepCopy
        }
    }

use lru::LruCache;
use std::sync::{Arc as StdArc, Mutex};
use super::config::{CloneStrategy, MemorySharingConfig};

/// Statistics for clone strategy performance
#[derive(Debug, Default, Clone)]
pub struct CloneStatistics {
    pub arc_clone_count: u64,
    pub deep_clone_count: u64,
    pub total_clone_time_ns: u64,
    pub cache_hits: u64,
    pub cache_misses: u64,
}

/// Manages clone strategy selection with caching and statistics
pub struct CloneStrategySelector {
    config: MemorySharingConfig,
    stats: StdArc<Mutex<CloneStatistics>>,
    // Cache: dimension → (strategy, expected_clones)
    cache: StdArc<Mutex<LruCache<usize, (CloneStrategy, usize)>>>,
}

impl CloneStrategySelector {
    pub fn new(config: MemorySharingConfig) -> Self {
        Self {
            config,
            stats: StdArc::new(Mutex::new(CloneStatistics::default())),
            cache: StdArc::new(Mutex::new(LruCache::new(
                std::num::NonZeroUsize::new(1000).unwrap()
            ))),
        }
    }

    /// Select clone strategy for given dimension and expected clones
    pub fn select_for_dimension(
        &self,
        dimension: usize,
        expected_clones: usize,
    ) -> CloneStrategy {
        // Check cache first
        if let Some(&(strategy, cached_clones)) = self.cache.lock().unwrap().get(&dimension) {
            // Use cached strategy if clone count is similar
            if (cached_clones as i64 - expected_clones as i64).abs() <= 10 {
                self.stats.lock().unwrap().cache_hits += 1;
                return strategy;
            }
        }

        self.stats.lock().unwrap().cache_misses += 1;

        // Select strategy using config
        let strategy = self.config.select_strategy(dimension, expected_clones);

        // Update cache
        self.cache.lock().unwrap().put(dimension, (strategy, expected_clones));

        // Update statistics
        let mut stats = self.stats.lock().unwrap();
        match strategy {
            CloneStrategy::Arc => stats.arc_clone_count += 1,
            CloneStrategy::DeepCopy => stats.deep_clone_count += 1,
        }

        strategy
    }

    /// Get current statistics
    pub fn statistics(&self) -> CloneStatistics {
        self.stats.lock().unwrap().clone()
    }

    /// Reset statistics
    pub fn reset_statistics(&self) {
        *self.stats.lock().unwrap() = CloneStatistics::default();
    }
}

use serde::{Deserialize, Serialize};

/// Aggregation strategy for top-K selection
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum AggregationStrategy {
    /// Full sort then truncate to k (O(n log n))
    Sort,
    /// Partial quicksort to k elements (O(n + k log k))
    PartialSort,
    /// BinaryHeap for top-k tracking (O(n log k))
    Heap,
}

/// Configuration for result aggregation
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AggregationConfig {
    /// Minimum n for heap select consideration
    pub heap_select_threshold: usize,

    /// Maximum k/n ratio for heap select (0.1 = 10%)
    pub heap_select_max_ratio: f64,

    /// Maximum k/n ratio for partial sort (0.3 = 30%)
    pub partial_sort_max_ratio: f64,

    /// Use sort for small result sets (< 500)
    pub sort_fallback_threshold: usize,
}

impl Default for AggregationConfig {
    fn default() -> Self {
        Self {
            heap_select_threshold: 1000,     // Heap optimal above 1000
            heap_select_max_ratio: 0.1,      // k < 10% of n
            partial_sort_max_ratio: 0.3,     // k < 30% of n
            sort_fallback_threshold: 500,    // Sort faster below 500
        }
    }
}

/// Select optimal aggregation strategy
pub fn select_strategy(n: usize, k: usize, config: &AggregationConfig) -> AggregationStrategy {
    // Small result sets: sort is fastest
    if n < config.sort_fallback_threshold {
        return AggregationStrategy::Sort;
    }

    let ratio = k as f64 / n as f64;

    // Large sets with small k: heap dominates
    if n >= config.heap_select_threshold && ratio < config.heap_select_max_ratio {
        return AggregationStrategy::Heap;
    }

    // Medium ratio: partial sort optimal
    if ratio >= config.heap_select_max_ratio && ratio < config.partial_sort_max_ratio {
        return AggregationStrategy::PartialSort;
    }

    // High ratio or other cases: full sort
    AggregationStrategy::Sort
}

use std::cmp::Ordering;

/// Fixed-capacity binary heap for top-K selection
/// Optimized to avoid dynamic allocation
pub struct FixedCapacityHeap<T> {
    data: Vec<T>,
    capacity: usize,
    comparator: fn(&T, &T) -> Ordering,
}

impl<T: Clone> FixedCapacityHeap<T> {
    /// Create heap with fixed capacity
    pub fn new(capacity: usize, comparator: fn(&T, &T) -> Ordering) -> Self {
        Self {
            data: Vec::with_capacity(capacity),
            capacity,
            comparator,
        }
    }

    /// Push element if it's better than the worst element
    pub fn push(&mut self, item: T) {
        if self.data.len() < self.capacity {
            self.data.push(item);
            self.bubble_up(self.data.len() - 1);
        } else if (self.comparator)(&item, &self.data[0]) == Ordering::Greater {
            // Replace root (worst element) with new item
            self.data[0] = item;
            self.bubble_down(0);
        }
    }

    /// Extract elements in sorted order
    pub fn into_sorted_vec(mut self) -> Vec<T> {
        let mut result = Vec::with_capacity(self.data.len());

        while !self.data.is_empty() {
            // Remove root (worst element)
            let last_idx = self.data.len() - 1;
            self.data.swap(0, last_idx);
            let item = self.data.pop().unwrap();

            if !self.data.is_empty() {
                self.bubble_down(0);
            }

            result.push(item);
        }

        // Reverse to get best-first order
        result.reverse();
        result
    }

    fn bubble_up(&mut self, mut idx: usize) {
        while idx > 0 {
            let parent = (idx - 1) / 2;
            if (self.comparator)(&self.data[idx], &self.data[parent]) == Ordering::Less {
                self.data.swap(idx, parent);
                idx = parent;
            } else {
                break;
            }
        }
    }

    fn bubble_down(&mut self, mut idx: usize) {
        loop {
            let left = 2 * idx + 1;
            let right = 2 * idx + 2;
            let mut smallest = idx;

            if left < self.data.len()
                && (self.comparator)(&self.data[left], &self.data[smallest]) == Ordering::Less
            {
                smallest = left;
            }

            if right < self.data.len()
                && (self.comparator)(&self.data[right], &self.data[smallest]) == Ordering::Less
            {
                smallest = right;
            }

            if smallest != idx {
                self.data.swap(idx, smallest);
                idx = smallest;
            } else {
                break;
            }
        }
    }
}

/// Select top-K elements using heap
pub fn heap_select_top_k<T: Clone>(
    items: &[T],
    k: usize,
    comparator: fn(&T, &T) -> Ordering,
) -> Vec<T> {
    if items.len() <= k {
        return items.to_vec();
    }

    // Use min-heap (worst element at root)
    let mut heap = FixedCapacityHeap::new(k, comparator);

    for item in items {
        heap.push(item.clone());
    }

    heap.into_sorted_vec()
}
