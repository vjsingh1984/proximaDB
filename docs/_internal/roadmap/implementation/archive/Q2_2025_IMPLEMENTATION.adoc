== Graph Database Enhancement

=== Overview
Extend ProximaDB with graph capabilities that work within the collection's designated storage engine, leveraging existing infrastructure.

=== Key Architecture Principles
* Graph data stored within collection's bound engine (no cross-engine operations)
* Leverage existing VectorRecord structure for node storage
* Use metadata fields for graph relationships
* Integrate with UnifiedCache and UnifiedMetrics
* Utilize AXIS indexes for graph traversal optimization

use crate::storage::cache::orchestrator::CacheOrchestrator;

/// Graph operations that work within a collection's bound engine
pub struct GraphStorage {
    collection_id: String,
    engine: Arc<dyn UnifiedStorageEngine>,
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    edge_index: Arc<RwLock<HashMap<String, Vec<EdgeRef>>>>,

impl GraphStorage {
    pub async fn new(
        collection_id: String,
        engine: Arc<dyn UnifiedStorageEngine>,
        cache: Arc<CacheOrchestrator>,
        metrics: Arc<UnifiedMetrics>,
    ) -> Result<Self> {
        // Build edge index from existing data
        let edge_index = Self::build_edge_index(&engine, &collection_id).await?;
        
        Ok(Self {
            collection_id,
            engine,
            cache,
            metrics,
            edge_index: Arc::new(RwLock::new(edge_index)),
        })
    }

    /// Create a graph node within the collection's engine
    pub async fn create_node(
        &self,
        id: String,
        labels: Vec<String>,
        properties: HashMap<String, Value>,
        vector: Option<Vec<f32>>,
    ) -> Result<String> {
        self.metrics.record_operation("graph.create_node").await;
        
        // Create VectorRecord with graph extensions
        let record = VectorRecord {
            id: id.clone(),
            values: vector.unwrap_or_default(),
            timestamp: Utc::now().timestamp_millis(),
            graph_node: Some(GraphNode {
                labels,
                properties: properties.clone(),
                node_type: NodeType::Standard as i32,
            }),
            edges_out: vec![],
            edges_in: vec![],
            metadata: self.properties_to_metadata(&properties),
            ..Default::default()
        };

        // Store in the collection's engine
        self.engine.insert(vec![record.clone()]).await?;
        
        // Cache the node for fast traversal
        self.cache.put(
            &format!("graph_node:{}:{}", self.collection_id, id),
            &record,
        ).await?;

        Ok(id)
    }

    /// Create edge between nodes in the same collection
    pub async fn create_edge(
        &self,
        from_node: String,
        to_node: String,
        edge_type: String,
        properties: HashMap<String, Value>,
    ) -> Result<String> {
        self.metrics.record_operation("graph.create_edge").await;
        
        let edge_id = format!("{}_{}_{}_{}", from_node, edge_type, to_node, uuid::Uuid::new_v4());
        
        let edge = GraphEdge {
            edge_id: edge_id.clone(),
            from_node: from_node.clone(),
            to_node: to_node.clone(),
            edge_type: edge_type.clone(),
            properties,
            weight: 1.0,
        };

        // Update both nodes with the edge
        self.update_node_edges(&from_node, &to_node, edge).await?;
        
        // Update edge index
        {
            let mut index = self.edge_index.write().await;
            index.entry(from_node.clone())
                .or_insert_with(Vec::new)
                .push(EdgeRef { 
                    target: to_node.clone(), 
                    edge_type: edge_type.clone(),
                    edge_id: edge_id.clone(),
                });
        }

        Ok(edge_id)
    }

    /// Traverse graph using the collection's data
    pub async fn traverse(
        &self,
        start_node: &str,
        max_depth: usize,
        filter: Option<TraversalFilter>,
    ) -> Result<Vec<TraversalPath>> {
        self.metrics.record_operation("graph.traverse").await;
        
        // Check cache first
        let cache_key = format!("traverse:{}:{}:{}", self.collection_id, start_node, max_depth);
        if let Some(cached) = self.cache.get::<Vec<TraversalPath>>(&cache_key).await? {
            self.metrics.record_cache_hit("graph.traverse").await;
            return Ok(cached);
        }

        // Perform BFS traversal within the engine
        let paths = self.bfs_internal(start_node, max_depth, filter).await?;
        
        // Cache results
        self.cache.put(&cache_key, &paths).await?;
        
        Ok(paths)
    }

    async fn bfs_internal(
        &self,
        start: &str,
        max_depth: usize,
        filter: Option<TraversalFilter>,
    ) -> Result<Vec<TraversalPath>> {
        let mut visited = HashSet::new();
        let mut queue = VecDeque::new();
        let mut paths = Vec::new();

        queue.push_back((start.to_string(), 0, vec![start.to_string()]));
        visited.insert(start.to_string());

        while let Some((node_id, depth, path)) = queue.pop_front() {
            if depth >= max_depth {
                continue;
            }

            // Get edges from index
            let edges = {
                let index = self.edge_index.read().await;
                index.get(&node_id).cloned().unwrap_or_default()
            };

            for edge_ref in edges {
                if let Some(ref f) = filter {
                    if !f.matches_edge(&edge_ref) {
                        continue;
                    }
                }

                if !visited.contains(&edge_ref.target) {
                    visited.insert(edge_ref.target.clone());
                    
                    let mut new_path = path.clone();
                    new_path.push(edge_ref.target.clone());
                    
                    paths.push(TraversalPath {
                        nodes: new_path.clone(),
                        depth: depth + 1,
                        total_weight: self.calculate_path_weight(&new_path).await?,
                    });
                    
                    queue.push_back((edge_ref.target, depth + 1, new_path));
                }
            }
        }

        Ok(paths)
    }

    /// Find shortest path using Dijkstra within collection
    pub async fn shortest_path(
        &self,
        start: &str,
        end: &str,
        use_vector_similarity: bool,
    ) -> Result<Option<TraversalPath>> {
        self.metrics.record_operation("graph.shortest_path").await;
        
        // TODO: Add ML-based path prediction when ready
        // For now, use Dijkstra with optional vector similarity as heuristic
        
        let mut distances = HashMap::new();
        let mut previous = HashMap::new();
        let mut heap = BinaryHeap::new();

        distances.insert(start.to_string(), 0.0);
        heap.push(HeapNode {
            node_id: start.to_string(),
            cost: 0.0,
        });

        // Get end node vector for A* heuristic if requested
        let end_vector = if use_vector_similarity {
            self.get_node_vector(end).await?
        } else {
            None
        };

        while let Some(HeapNode { node_id, cost }) = heap.pop() {
            if node_id == end {
                return Ok(Some(self.reconstruct_path(&previous, end)));
            }

            if cost > *distances.get(&node_id).unwrap_or(&f32::MAX) {
                continue;
            }

            let edges = {
                let index = self.edge_index.read().await;
                index.get(&node_id).cloned().unwrap_or_default()
            };

            for edge_ref in edges {
                let edge_weight = 1.0; // Default weight, could be from edge properties
                let new_cost = cost + edge_weight;

                // A* heuristic using vector similarity if available
                let heuristic = if use_vector_similarity {
                    if let (Some(neighbor_vec), Some(end_vec)) = 
                        (self.get_node_vector(&edge_ref.target).await?, end_vector.as_ref()) {
                        // Use cosine distance as heuristic
                        1.0 - cosine_similarity(neighbor_vec, end_vec)
                    } else {
                        0.0
                    }
                } else {
                    0.0
                };

                let priority = new_cost + heuristic;

                if new_cost < *distances.get(&edge_ref.target).unwrap_or(&f32::MAX) {
                    distances.insert(edge_ref.target.clone(), new_cost);
                    previous.insert(edge_ref.target.clone(), node_id.clone());
                    heap.push(HeapNode {
                        node_id: edge_ref.target,
                        cost: priority,
                    });
                }
            }
        }

        Ok(None)
    }

    async fn get_node_vector(&self, node_id: &str) -> Result<Option<Vec<f32>>> {
        // Try cache first
        let cache_key = format!("node_vector:{}:{}", self.collection_id, node_id);
        if let Some(vector) = self.cache.get::<Vec<f32>>(&cache_key).await? {
            return Ok(Some(vector));
        }

        // Fetch from engine
        let records = self.engine.get_by_ids(vec![node_id.to_string()]).await?;
        if let Some(record) = records.first() {
            if !record.values.is_empty() {
                self.cache.put(&cache_key, &record.values).await?;
                return Ok(Some(record.values.clone()));
            }
        }

        Ok(None)
    }
}

pub struct MultiModalSearchCoordinator {

impl MultiModalSearchCoordinator {
    /// Execute multi-modal search within a collection
    pub async fn search(
        &self,
        collection_config: &CollectionConfig,
        engine: Arc<dyn UnifiedStorageEngine>,
        request: MultiModalSearchRequest,
    ) -> Result<MultiModalSearchResponse> {
        let start = Instant::now();
        self.metrics.record_operation("multimodal.search").await;

        // Determine available search modalities from collection config
        let modalities = self.determine_modalities(&collection_config, &request);
        
        // Execute searches in parallel based on available capabilities
        let search_futures = modalities.iter().map(|modality| {
            self.execute_modality_search(
                modality,
                &collection_config,
                &engine,
                &request,
            )
        });
        
        let results = futures::future::join_all(search_futures).await;
        
        // Collect successful results
        let mut modality_results = ModalityResults::new();
        for (modality, result) in modalities.iter().zip(results) {
            match result {
                Ok(res) => modality_results.add_results(modality, res),
                Err(e) => {
                    self.metrics.record_error("multimodal.search", &e).await;
                    // Continue with other modalities
                }
            }
        }

        // Fuse results using configured strategy
        let fusion_strategy = self.get_fusion_strategy(&request.fusion_strategy)?;
        let fused = fusion_strategy.fuse(
            modality_results,
            &request.fusion_weights,
        ).await?;

        // Apply post-processing if configured
        let final_results = if collection_config.multimodal.reranking_enabled {
            self.rerank_results(fused, &request, &collection_config).await?
        } else {
            fused
        };

        Ok(MultiModalSearchResponse {
            results: final_results.into_iter().take(request.limit).collect(),
            search_time_ms: start.elapsed().as_millis() as u64,
            modalities_used: modalities,
        })
    }

    async fn execute_modality_search(
        &self,
        modality: &SearchModality,
        config: &CollectionConfig,
        engine: &Arc<dyn UnifiedStorageEngine>,
        request: &MultiModalSearchRequest,
    ) -> Result<Vec<ScoredResult>> {
        match modality {
            SearchModality::Vector => {
                self.search_vectors(config, engine, request).await
            }
            SearchModality::Text => {
                self.search_text(config, engine, request).await
            }
            SearchModality::Metadata => {
                self.search_metadata(config, engine, request).await
            }
            SearchModality::Graph => {
                self.search_graph(config, engine, request).await
            }
        }
    }

    async fn search_vectors(
        &self,
        config: &CollectionConfig,
        engine: &Arc<dyn UnifiedStorageEngine>,
        request: &MultiModalSearchRequest,
    ) -> Result<Vec<ScoredResult>> {
        let vector = if let Some(v) = &request.vector {
            v.clone()
        } else if let Some(text) = &request.text {
            // Generate embedding from text
            self.generate_embedding(text, config).await?
        } else {
            return Ok(vec![]);
        };

        // Check for precomputed quantization
        let search_request = if config.quantization.enabled {
            SearchRequest {
                query_vector: vector,
                use_quantization: Some(config.quantization.default_level),
                top_k: request.vector_limit.unwrap_or(100),
                ..Default::default()
            }
        } else {
            SearchRequest {
                query_vector: vector,
                top_k: request.vector_limit.unwrap_or(100),
                ..Default::default()
            }
        };

        let results = engine.search(search_request).await?;
        
        Ok(results.into_iter().map(|r| ScoredResult {
            id: r.id,
            score: r.score,
            source: SearchModality::Vector,
        }).collect())
    }

    async fn search_text(
        &self,
        config: &CollectionConfig,
        engine: &Arc<dyn UnifiedStorageEngine>,
        request: &MultiModalSearchRequest,
    ) -> Result<Vec<ScoredResult>> {
        let Some(text) = &request.text else {
            return Ok(vec![]);
        };

        // Analyze text query
        let analyzed = self.text_analyzer.analyze(text)?;
        
        // Search in metadata fields configured for text search
        let text_fields = &config.text_search.searchable_fields;
        let mut results = Vec::new();
        
        for field in text_fields {
            let field_results = engine.search_metadata(
                field,
                &analyzed.tokens,
                request.text_limit.unwrap_or(100),
            ).await?;
            
            results.extend(field_results.into_iter().map(|r| ScoredResult {
                id: r.id,
                score: r.relevance_score,
                source: SearchModality::Text,
            }));
        }

        // Deduplicate and sort
        self.deduplicate_results(results)
    }

    fn determine_modalities(
        &self,
        config: &CollectionConfig,
        request: &MultiModalSearchRequest,
    ) -> Vec<SearchModality> {
        let mut modalities = Vec::new();
        
        // Vector search if collection has vectors
        if config.dimension > 0 && (request.vector.is_some() || request.text.is_some()) {
            modalities.push(SearchModality::Vector);
        }
        
        // Text search if configured
        if config.text_search.enabled && request.text.is_some() {
            modalities.push(SearchModality::Text);
        }
        
        // Metadata search if filters provided
        if request.metadata_filter.is_some() {
            modalities.push(SearchModality::Metadata);
        }
        
        // Graph search if enabled and requested
        if config.graph_enabled && request.graph_traverse.is_some() {
            modalities.push(SearchModality::Graph);
        }
        
        modalities
    }

        request: &MultiModalSearchRequest,
        config: &CollectionConfig,
    ) -> Result<Vec<FusedResult>> {
        // For now, return as-is

    }

pub trait FusionStrategy: Send + Sync {
    async fn fuse(
        &self,
        results: ModalityResults,
        weights: &FusionWeights,
    ) -> Result<Vec<FusedResult>>;
}

/// Reciprocal Rank Fusion - simple and effective
pub struct RecipocalRankFusion {
    k: f32,  // Typically 60
}

impl FusionStrategy for RecipocalRankFusion {
    async fn fuse(
        &self,
        results: ModalityResults,
        weights: &FusionWeights,
    ) -> Result<Vec<FusedResult>> {
        let mut doc_scores = HashMap::new();
        
        // RRF formula: score = sum(weight * 1 / (k + rank))
        for (modality, modality_results) in results.iter() {
            let weight = weights.get_weight(modality);
            
            for (rank, result) in modality_results.iter().enumerate() {
                let score = weight * (1.0 / (self.k + rank as f32 + 1.0));
                
                let entry = doc_scores.entry(result.id.clone()).or_insert(FusedResult {
                    id: result.id.clone(),
                    score: 0.0,
                    sources: Vec::new(),
                });
                
                entry.score += score;
                entry.sources.push(modality.clone());
            }
        }

        // Sort by fused score
        let mut fused: Vec<_> = doc_scores.into_values().collect();
        fused.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());
        
        Ok(fused)
    }
}

/// Weighted linear combination
pub struct WeightedFusion;

impl FusionStrategy for WeightedFusion {
    async fn fuse(
        &self,
        results: ModalityResults,
        weights: &FusionWeights,
    ) -> Result<Vec<FusedResult>> {
        let mut doc_scores = HashMap::new();
        
        for (modality, modality_results) in results.iter() {
            let weight = weights.get_weight(modality);
            
            // Normalize scores within modality
            let max_score = modality_results
                .iter()
                .map(|r| r.score)
                .max_by(|a, b| a.partial_cmp(b).unwrap())
                .unwrap_or(1.0);
            
            for result in modality_results {
                let normalized_score = result.score / max_score;
                let weighted_score = weight * normalized_score;
                
                let entry = doc_scores.entry(result.id.clone()).or_insert(FusedResult {
                    id: result.id.clone(),
                    score: 0.0,
                    sources: Vec::new(),
                });
                
                entry.score += weighted_score;
                entry.sources.push(modality.clone());
            }
        }

        let mut fused: Vec<_> = doc_scores.into_values().collect();
        fused.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());
        
        Ok(fused)
    }
}

use tantivy::tokenizer::{SimpleTokenizer, TextAnalyzer as TantivyAnalyzer};

pub struct TextAnalyzer {
    tokenizer: SimpleTokenizer,
    stop_words: HashSet<String>,
    synonyms: HashMap<String, Vec<String>>,
    cache: Arc<CacheOrchestrator>,
}

impl TextAnalyzer {
    pub fn analyze(&self, text: &str) -> Result<AnalyzedText> {
        // Check cache
        let cache_key = format!("text_analysis:{}", hash(text));
        if let Some(cached) = self.cache.get::<AnalyzedText>(&cache_key).await? {
            return Ok(cached);
        }

        // Tokenize
        let mut tokens = Vec::new();
        let mut token_stream = self.tokenizer.token_stream(text);
        
        while let Some(token) = token_stream.next() {
            let token_str = token.text.to_lowercase();
            
            // Skip stop words
            if self.stop_words.contains(&token_str) {
                continue;
            }
            
            tokens.push(token_str.clone());
            
            // Add synonyms
            if let Some(syns) = self.synonyms.get(&token_str) {
                tokens.extend(syns.clone());
            }
        }

        let analyzed = AnalyzedText {
            original: text.to_string(),
            tokens,
            stems: self.stem_tokens(&tokens)?,
        };

        // Cache result
        self.cache.put(&cache_key, &analyzed).await?;
        
        Ok(analyzed)
    }

    fn stem_tokens(&self, tokens: &[String]) -> Result<Vec<String>> {
        // Simple stemming - in production, use a proper stemmer
        Ok(tokens.iter().map(|t| {
            if t.ends_with("ing") && t.len() > 4 {
                t[..t.len()-3].to_string()
            } else if t.ends_with("ed") && t.len() > 3 {
                t[..t.len()-2].to_string()
            } else {
                t.clone()
            }
        }).collect())
    }
}

            return Err(anyhow!("Graph not enabled for collection"));
        }

        let engine = self.get_engine(collection_name).await?;
        let storage = Arc::new(GraphStorage::new(
            collection_name.to_string(),
            engine,
            self.cache.clone(),
            self.metrics.clone(),
        ).await?);

        // Cache it
        self.cache.put(&cache_key, &storage).await?;
        
        Ok(storage)
    }

    fn validate_modalities(
        &self,
        config: &CollectionConfig,
        request: &MultiModalSearchRequest,
    ) -> Result<()> {
        if request.vector.is_some() && config.dimension == 0 {
            return Err(anyhow!("Collection does not support vector search"));
        }
        
        if request.text.is_some() && !config.text_search.enabled {
            return Err(anyhow!("Text search not enabled for collection"));
        }
        
        if request.graph_traverse.is_some() && !config.graph_enabled {
            return Err(anyhow!("Graph operations not enabled for collection"));
        }
        
        Ok(())
    }
}
