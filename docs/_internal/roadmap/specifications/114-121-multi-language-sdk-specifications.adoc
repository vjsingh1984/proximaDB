== Overview

This specification defines the implementation requirements for ProximaDB client SDKs in Java (114), C++ (115), Rust (116), Go (117), .NET (118), Node.js (119), Ruby (120), and Swift (121).

=== Design Principles

[source,text]
----
SDK Design Principles:
├── Consistent API surface across languages
├── Idiomatic patterns for each language
├── Async-first with sync wrappers where appropriate
├── Zero-copy where possible
├── Comprehensive error handling
├── Full protocol support (REST, gRPC, Arrow Flight)
└── Minimal dependencies
----

=== Common API Contract

All SDKs must implement this core interface:

[source,text]
----
Core API Contract:

Client
├── connect(config) → Client
├── close()
├── health() → HealthStatus

Collections
├── create_collection(name, config) → Collection
├── get_collection(name) → Collection
├── list_collections() → List<CollectionInfo>
├── delete_collection(name)

Vectors
├── insert(collection, records) → InsertResult
├── upsert(collection, records) → UpsertResult
├── update(collection, id, data) → UpdateResult
├── delete(collection, ids) → DeleteResult
├── get(collection, ids) → List<VectorRecord>

Search
├── search(collection, query) → SearchResult
├── search_batch(collection, queries) → List<SearchResult>

Graph (optional)
├── create_graph(name, config) → Graph
├── create_nodes(graph, nodes) → CreateResult
├── create_edges(graph, edges) → CreateResult
├── query(graph, cypher) → QueryResult
----

package io.proximadb.client;

import java.util.List;
import java.util.concurrent.CompletableFuture;

/**
 * ProximaDB Java client with async-first API.
 */
public class ProximaDBClient implements AutoCloseable {

    private final ClientConfig config;
    private final ProtocolClient protocol;
    private final ConnectionPool connectionPool;

    private ProximaDBClient(ClientConfig config) {
        this.config = config;
        this.protocol = ProtocolFactory.create(config);
        this.connectionPool = new ConnectionPool(config.getPoolConfig());
    }

    public static ProximaDBClientBuilder builder() {
        return new ProximaDBClientBuilder();
    }

    // Async API (primary)
    public CompletableFuture<Collection> createCollectionAsync(
            String name,
            CollectionConfig config) {
        return protocol.createCollection(name, config);
    }

    public CompletableFuture<InsertResult> insertAsync(
            String collection,
            List<VectorRecord> records) {
        return protocol.insert(collection, records);
    }

    public CompletableFuture<SearchResult> searchAsync(
            String collection,
            SearchQuery query) {
        return protocol.search(collection, query);
    }

    // Sync wrappers
    public Collection createCollection(String name, CollectionConfig config) {
        return createCollectionAsync(name, config).join();
    }

    public InsertResult insert(String collection, List<VectorRecord> records) {
        return insertAsync(collection, records).join();
    }

    public SearchResult search(String collection, SearchQuery query) {
        return searchAsync(collection, query).join();
    }

    @Override
    public void close() {
        connectionPool.close();
        protocol.close();
    }
}

// Builder pattern
public class ProximaDBClientBuilder {
    private String url = "http://localhost:5678";
    private Protocol protocol = Protocol.REST;
    private Duration timeout = Duration.ofSeconds(30);
    private int maxConnections = 10;
    private RetryPolicy retryPolicy = RetryPolicy.exponentialBackoff();

    public ProximaDBClientBuilder url(String url) {
        this.url = url;
        return this;
    }

    public ProximaDBClientBuilder protocol(Protocol protocol) {
        this.protocol = protocol;
        return this;
    }

    public ProximaDBClientBuilder timeout(Duration timeout) {
        this.timeout = timeout;
        return this;
    }

    public ProximaDBClient build() {
        return new ProximaDBClient(new ClientConfig(
            url, protocol, timeout, maxConnections, retryPolicy
        ));
    }
}

// Usage example
public class Example {
    public static void main(String[] args) {
        try (ProximaDBClient client = ProximaDBClient.builder()
                .url("http://localhost:5678")
                .protocol(Protocol.GRPC)
                .build()) {

            // Create collection
            client.createCollection("products", CollectionConfig.builder()
                .dimension(768)
                .engine("helix")
                .distanceMetric(DistanceMetric.COSINE)
                .build());

            // Insert vectors
            List<VectorRecord> records = List.of(
                VectorRecord.builder()
                    .id("p1")
                    .vector(new float[768])
                    .metadata(Map.of("category", "electronics"))
                    .build()
            );
            client.insert("products", records);

            // Search
            SearchResult result = client.search("products", SearchQuery.builder()
                .vector(new float[768])
                .topK(10)
                .filter(Filter.eq("category", "electronics"))
                .build());

            result.getMatches().forEach(match ->
                System.out.printf("ID: %s, Score: %.4f%n", match.getId(), match.getScore())
            );
        }
    }
}

#pragma once

#include <string>
#include <vector>
#include <memory>
#include <future>
#include <optional>

namespace proximadb {

// Forward declarations
class Collection;
class SearchResult;
struct VectorRecord;
struct SearchQuery;

/**
 * ProximaDB C++ client with modern C++17/20 patterns.
 */
class Client {
public:
    struct Config {
        std::string url = "http://localhost:5678";
        enum class Protocol { REST, GRPC, ARROW_FLIGHT } protocol = Protocol::REST;
        std::chrono::milliseconds timeout{30000};
        size_t max_connections = 10;
        bool enable_retry = true;
    };

    // Factory method
    static std::unique_ptr<Client> create(const Config& config);

    virtual ~Client() = default;

    // Collection operations
    virtual std::future<Collection> create_collection(
        const std::string& name,
        const CollectionConfig& config) = 0;

    virtual std::future<std::optional<Collection>> get_collection(
        const std::string& name) = 0;

    virtual std::future<std::vector<CollectionInfo>> list_collections() = 0;

    virtual std::future<void> delete_collection(const std::string& name) = 0;

    // Vector operations
    virtual std::future<InsertResult> insert(
        const std::string& collection,
        std::span<const VectorRecord> records) = 0;

    virtual std::future<SearchResult> search(
        const std::string& collection,
        const SearchQuery& query) = 0;

    // Sync wrappers
    Collection create_collection_sync(
        const std::string& name,
        const CollectionConfig& config) {
        return create_collection(name, config).get();
    }

    SearchResult search_sync(
        const std::string& collection,
        const SearchQuery& query) {
        return search(collection, query).get();
    }
};

// Data structures
struct VectorRecord {
    std::string id;
    std::vector<float> vector;
    std::unordered_map<std::string, std::variant<
        std::string, int64_t, double, bool>> metadata;
};

struct SearchQuery {
    std::vector<float> vector;
    size_t top_k = 10;
    std::optional<Filter> filter;
    bool include_metadata = true;
    bool include_vector = false;
};

struct SearchResult {
    struct Match {
        std::string id;
        float score;
        std::optional<std::vector<float>> vector;
        std::optional<Metadata> metadata;
    };

    std::vector<Match> matches;
    std::chrono::microseconds latency;
};

// Error handling
class ProximaDBException : public std::exception {
public:
    enum class Code {
        CONNECTION_ERROR,
        TIMEOUT,
        NOT_FOUND,
        INVALID_ARGUMENT,
        INTERNAL_ERROR
    };

    ProximaDBException(Code code, std::string message)
        : code_(code), message_(std::move(message)) {}

    Code code() const noexcept { return code_; }
    const char* what() const noexcept override { return message_.c_str(); }

private:
    Code code_;
    std::string message_;
};

} // namespace proximadb

// Usage example
int main() {
    using namespace proximadb;

    auto client = Client::create(Client::Config{
        .url = "http://localhost:5678",
        .protocol = Client::Config::Protocol::GRPC
    });

    // Create collection
    auto collection = client->create_collection_sync("products", {
        .dimension = 768,
        .engine = "helix",
        .distance_metric = DistanceMetric::Cosine
    });

    // Insert vectors
    std::vector<VectorRecord> records = {
        {.id = "p1", .vector = std::vector<float>(768, 0.1f)}
    };
    client->insert("products", records).get();

    // Search
    auto result = client->search_sync("products", {
        .vector = std::vector<float>(768, 0.5f),
        .top_k = 10
    });

    for (const auto& match : result.matches) {
        std::cout << "ID: " << match.id << ", Score: " << match.score << "\n";
    }

    return 0;
}

//! ProximaDB Rust SDK
//!
//! Native Rust client with async-first design.

use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::RwLock;

/// Client configuration
#[derive(Debug, Clone)]
pub struct ClientConfig {
    pub url: String,
    pub protocol: Protocol,
    pub timeout: std::time::Duration,
    pub max_connections: usize,
    pub retry_policy: RetryPolicy,
}

impl Default for ClientConfig {
    fn default() -> Self {
        Self {
            url: "http://localhost:5678".to_string(),
            protocol: Protocol::Rest,
            timeout: std::time::Duration::from_secs(30),
            max_connections: 10,
            retry_policy: RetryPolicy::default(),
        }
    }
}

#[derive(Debug, Clone, Copy)]
pub enum Protocol {
    Rest,
    Grpc,
    ArrowFlight,
}

/// ProximaDB client
pub struct ProximaDBClient {
    config: ClientConfig,
    inner: Arc<dyn ProtocolClient + Send + Sync>,
}

impl ProximaDBClient {
    /// Create a new client with default configuration
    pub async fn new(url: &str) -> Result<Self, Error> {
        Self::with_config(ClientConfig {
            url: url.to_string(),
            ..Default::default()
        }).await
    }

    /// Create a new client with custom configuration
    pub async fn with_config(config: ClientConfig) -> Result<Self, Error> {
        let inner = match config.protocol {
            Protocol::Rest => Arc::new(RestClient::new(&config).await?),
            Protocol::Grpc => Arc::new(GrpcClient::new(&config).await?),
            Protocol::ArrowFlight => Arc::new(ArrowFlightClient::new(&config).await?),
        };

        Ok(Self { config, inner })
    }

    /// Create a new collection
    pub async fn create_collection(
        &self,
        name: &str,
        config: CollectionConfig,
    ) -> Result<Collection, Error> {
        self.inner.create_collection(name, config).await
    }

    /// Insert vectors into a collection
    pub async fn insert(
        &self,
        collection: &str,
        records: Vec<VectorRecord>,
    ) -> Result<InsertResult, Error> {
        self.inner.insert(collection, records).await
    }

    /// Search for similar vectors
    pub async fn search(
        &self,
        collection: &str,
        query: SearchQuery,
    ) -> Result<SearchResult, Error> {
        self.inner.search(collection, query).await
    }

    /// Batch search
    pub async fn search_batch(
        &self,
        collection: &str,
        queries: Vec<SearchQuery>,
    ) -> Result<Vec<SearchResult>, Error> {
        self.inner.search_batch(collection, queries).await
    }
}

pub struct VectorRecord {

    pub vector: Vec<f32>,
    pub metadata: Option<HashMap<String, serde_json::Value>>,

    pub top_k: usize,
    pub filter: Option<Filter>,
    pub include_metadata: bool,
    pub include_vector: bool,
}

impl Default for SearchQuery {
    fn default() -> Self {
        Self {
            vector: Vec::new(),
            top_k: 10,
            filter: None,
            include_metadata: true,
            include_vector: false,
        }
    }
}

/// Search result
#[derive(Debug, Clone)]
pub struct SearchResult {
    pub matches: Vec<SearchMatch>,
    pub latency: std::time::Duration,
}

#[derive(Debug, Clone)]
pub struct SearchMatch {
    pub id: String,
    pub score: f32,
    pub vector: Option<Vec<f32>>,
    pub metadata: Option<HashMap<String, serde_json::Value>>,
}

/// Filter expressions
#[derive(Debug, Clone)]
pub enum Filter {
    Eq(String, serde_json::Value),
    Ne(String, serde_json::Value),
    Gt(String, serde_json::Value),
    Gte(String, serde_json::Value),
    Lt(String, serde_json::Value),
    Lte(String, serde_json::Value),
    In(String, Vec<serde_json::Value>),
    And(Vec<Filter>),
    Or(Vec<Filter>),
    Not(Box<Filter>),
}

impl Filter {
    pub fn eq(field: &str, value: impl Into<serde_json::Value>) -> Self {
        Filter::Eq(field.to_string(), value.into())
    }

    pub fn and(filters: Vec<Filter>) -> Self {
        Filter::And(filters)
    }

    pub fn or(filters: Vec<Filter>) -> Self {
        Filter::Or(filters)
    }
}

/// Error types
#[derive(Debug, thiserror::Error)]
pub enum Error {
    #[error("Connection error: {0}")]
    Connection(String),

    #[error("Timeout")]
    Timeout,

    #[error("Not found: {0}")]
    NotFound(String),

    #[error("Invalid argument: {0}")]
    InvalidArgument(String),

    #[error("Internal error: {0}")]
    Internal(String),
}

// Usage example
#[tokio::main]
async fn main() -> Result<(), Error> {
    // Create client
    let client = ProximaDBClient::new("http://localhost:5678").await?;

    // Create collection
    client.create_collection("products", CollectionConfig {
        dimension: 768,
        engine: "helix".to_string(),
        distance_metric: DistanceMetric::Cosine,
        ..Default::default()
    }).await?;

    // Insert vectors
    let records = vec![
        VectorRecord {
            id: "p1".to_string(),
            vector: vec![0.1; 768],
            metadata: Some(HashMap::from([
                ("category".to_string(), "electronics".into())
            ])),
        }
    ];
    client.insert("products", records).await?;

    // Search
    let result = client.search("products", SearchQuery {
        vector: vec![0.5; 768],
        top_k: 10,
        filter: Some(Filter::eq("category", "electronics")),
        ..Default::default()
    }).await?;

    for m in result.matches {
        println!("ID: {}, Score: {:.4}", m.id, m.score);
    }

    Ok(())
}

package proximadb

import (
    "context"
    "time"
)

// Client configuration
type Config struct {
    URL            string
    Protocol       Protocol
    Timeout        time.Duration
    MaxConnections int
    RetryPolicy    *RetryPolicy
}

// DefaultConfig returns default client configuration
func DefaultConfig() *Config {
    return &Config{
        URL:            "http://localhost:5678",
        Protocol:       REST,
        Timeout:        30 * time.Second,
        MaxConnections: 10,
        RetryPolicy:    DefaultRetryPolicy(),
    }
}

// Protocol type
type Protocol int

const (
    REST Protocol = iota
    GRPC
    ArrowFlight
)

// Client is the ProximaDB client
type Client struct {
    config   *Config
    protocol protocolClient
}

// NewClient creates a new ProximaDB client
func NewClient(config *Config) (*Client, error) {
    if config == nil {
        config = DefaultConfig()
    }

    var protocol protocolClient
    var err error

    switch config.Protocol {
    case REST:
        protocol, err = newRestClient(config)
    case GRPC:
        protocol, err = newGrpcClient(config)
    case ArrowFlight:
        protocol, err = newArrowFlightClient(config)
    }

    if err != nil {
        return nil, err
    }

    return &Client{
        config:   config,
        protocol: protocol,
    }, nil
}

// Connect creates a client and verifies connection
func Connect(url string) (*Client, error) {
    client, err := NewClient(&Config{URL: url})
    if err != nil {
        return nil, err
    }

    // Verify connection
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()

    if _, err := client.Health(ctx); err != nil {
        return nil, err
    }

    return client, nil
}

// Close closes the client
func (c *Client) Close() error {
    return c.protocol.Close()
}

// Health checks server health
func (c *Client) Health(ctx context.Context) (*HealthStatus, error) {
    return c.protocol.Health(ctx)
}

// CreateCollection creates a new collection
func (c *Client) CreateCollection(ctx context.Context, name string, config *CollectionConfig) (*Collection, error) {
    return c.protocol.CreateCollection(ctx, name, config)
}

// GetCollection retrieves a collection
func (c *Client) GetCollection(ctx context.Context, name string) (*Collection, error) {
    return c.protocol.GetCollection(ctx, name)
}

// ListCollections lists all collections
func (c *Client) ListCollections(ctx context.Context) ([]*CollectionInfo, error) {
    return c.protocol.ListCollections(ctx)
}

// DeleteCollection deletes a collection
func (c *Client) DeleteCollection(ctx context.Context, name string) error {
    return c.protocol.DeleteCollection(ctx, name)
}

// Insert inserts vectors into a collection
func (c *Client) Insert(ctx context.Context, collection string, records []*VectorRecord) (*InsertResult, error) {
    return c.protocol.Insert(ctx, collection, records)
}

// Search performs a vector similarity search
func (c *Client) Search(ctx context.Context, collection string, query *SearchQuery) (*SearchResult, error) {
    return c.protocol.Search(ctx, collection, query)
}

// SearchBatch performs multiple searches
func (c *Client) SearchBatch(ctx context.Context, collection string, queries []*SearchQuery) ([]*SearchResult, error) {
    return c.protocol.SearchBatch(ctx, collection, queries)
}

// VectorRecord represents a vector with metadata
type VectorRecord struct {
    ID       string
    Vector   []float32
    Metadata map[string]interface{}
}

// SearchQuery represents search parameters
type SearchQuery struct {
    Vector          []float32
    TopK            int
    Filter          *Filter
    IncludeMetadata bool
    IncludeVector   bool
}

// SearchResult represents search results
type SearchResult struct {
    Matches []SearchMatch
    Latency time.Duration
}

// SearchMatch represents a single match
type SearchMatch struct {
    ID       string
    Score    float32
    Vector   []float32
    Metadata map[string]interface{}
}

// Filter represents a filter expression
type Filter struct {
    Field    string
    Operator string
    Value    interface{}
    Children []*Filter
}

// Eq creates an equality filter
func Eq(field string, value interface{}) *Filter {
    return &Filter{Field: field, Operator: "eq", Value: value}
}

// And combines filters with AND
func And(filters ...*Filter) *Filter {
    return &Filter{Operator: "and", Children: filters}
}

// Or combines filters with OR
func Or(filters ...*Filter) *Filter {
    return &Filter{Operator: "or", Children: filters}
}

// Usage example
func Example() {
    ctx := context.Background()

    // Connect
    client, err := Connect("http://localhost:5678")
    if err != nil {
        log.Fatal(err)
    }
    defer client.Close()

    // Create collection
    _, err = client.CreateCollection(ctx, "products", &CollectionConfig{
        Dimension:      768,
        Engine:         "helix",
        DistanceMetric: Cosine,
    })
    if err != nil {
        log.Fatal(err)
    }

    // Insert
    records := []*VectorRecord{
        {
            ID:     "p1",
            Vector: make([]float32, 768),
            Metadata: map[string]interface{}{
                "category": "electronics",
            },
        },
    }
    _, err = client.Insert(ctx, "products", records)
    if err != nil {
        log.Fatal(err)
    }

    // Search
    result, err := client.Search(ctx, "products", &SearchQuery{
        Vector: make([]float32, 768),
        TopK:   10,
        Filter: Eq("category", "electronics"),
    })
    if err != nil {
        log.Fatal(err)
    }

    for _, match := range result.Matches {
        fmt.Printf("ID: %s, Score: %.4f\n", match.ID, match.Score)
    }
}

using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace ProximaDB.Client
{
    /// <summary>
    /// ProximaDB .NET client with async-first design.
    /// </summary>
    public class ProximaDBClient : IDisposable, IAsyncDisposable
    {
        private readonly ClientConfig _config;
        private readonly IProtocolClient _protocol;

        public ProximaDBClient(string url)
            : this(new ClientConfig { Url = url })
        {
        }

        public ProximaDBClient(ClientConfig config)
        {
            _config = config ?? throw new ArgumentNullException(nameof(config));
            _protocol = ProtocolFactory.Create(config);
        }

        /// <summary>
        /// Creates a new collection.
        /// </summary>
        public async Task<Collection> CreateCollectionAsync(
            string name,
            CollectionConfig config,
            CancellationToken cancellationToken = default)
        {
            return await _protocol.CreateCollectionAsync(name, config, cancellationToken);
        }

        /// <summary>
        /// Inserts vectors into a collection.
        /// </summary>
        public async Task<InsertResult> InsertAsync(
            string collection,
            IEnumerable<VectorRecord> records,
            CancellationToken cancellationToken = default)
        {
            return await _protocol.InsertAsync(collection, records, cancellationToken);
        }

        /// <summary>
        /// Searches for similar vectors.
        /// </summary>
        public async Task<SearchResult> SearchAsync(
            string collection,
            SearchQuery query,
            CancellationToken cancellationToken = default)
        {
            return await _protocol.SearchAsync(collection, query, cancellationToken);
        }

        // Sync wrappers for convenience
        public Collection CreateCollection(string name, CollectionConfig config)
            => CreateCollectionAsync(name, config).GetAwaiter().GetResult();

        public InsertResult Insert(string collection, IEnumerable<VectorRecord> records)
            => InsertAsync(collection, records).GetAwaiter().GetResult();

        public SearchResult Search(string collection, SearchQuery query)
            => SearchAsync(collection, query).GetAwaiter().GetResult();

        public void Dispose()
        {
            _protocol?.Dispose();
        }

        public async ValueTask DisposeAsync()

module ProximaDB
  class Client
    attr_reader :config

    def initialize(url_or_config)
      @config = normalize_config(url_or_config)
      @protocol = create_protocol(@config)
    end

    # Create a new collection
    def create_collection(name, **options)
      @protocol.create_collection(name, options)
    end

    # Insert vectors into a collection
    def insert(collection, records)
      @protocol.insert(collection, records)
    end

    # Search for similar vectors
    def search(collection, query)
      @protocol.search(collection, query)
    end

    # Batch search
    def search_batch(collection, queries)
      @protocol.search_batch(collection, queries)
    end

    # Close the client
    def close
      @protocol.close
    end

    private

    def normalize_config(config)
      case config
      when String
        Config.new(url: config)
      when Hash
        Config.new(**config)
      when Config
        config
      else
        raise ArgumentError, "Invalid config: #{config.class}"
      end
    end

    def create_protocol(config)
      case config.protocol
      when :rest
        RestProtocol.new(config)
      when :grpc
        GrpcProtocol.new(config)
      else
        raise ArgumentError, "Unknown protocol: #{config.protocol}"
      end
    end
  end

  class Config
    attr_accessor :url, :protocol, :timeout, :max_connections

    def initialize(
      url: 'http://localhost:5678',
      protocol: :rest,
      timeout: 30,
      max_connections: 10
    )
      @url = url
      @protocol = protocol
      @timeout = timeout
      @max_connections = max_connections
    end
  end

  class VectorRecord
    attr_accessor :id, :vector, :metadata

    def initialize(id:, vector:, metadata: nil)
      @id = id
      @vector = vector
      @metadata = metadata
    end

    def to_h
      { id: @id, vector: @vector, metadata: @metadata }.compact
    end
  end

  class SearchQuery
    attr_accessor :vector, :top_k, :filter, :include_metadata, :include_vector

    def initialize(
      vector:,
      top_k: 10,
      filter: nil,
      include_metadata: true,
      include_vector: false
    )
      @vector = vector
      @top_k = top_k
      @filter = filter
      @include_metadata = include_metadata
      @include_vector = include_vector
    end

    def to_h
      {
        vector: @vector,
        top_k: @top_k,
        filter: @filter&.to_h,
        include_metadata: @include_metadata,
        include_vector: @include_vector
      }.compact
    end
  end

  class Filter
    attr_reader :field, :operator, :value, :children

    def initialize(field: nil, operator:, value: nil, children: nil)
      @field = field
      @operator = operator
      @value = value
      @children = children
    end

    def self.eq(field, value)
      new(field: field, operator: 'eq', value: value)
    end

    def self.and(*filters)
      new(operator: 'and', children: filters)
    end

    def self.or(*filters)
      new(operator: 'or', children: filters)
    end

    def to_h
      result = { operator: @operator }
      result[:field] = @field if @field
      result[:value] = @value if @value
      result[:children] = @children.map(&:to_h) if @children
      result
    end
  end
end
