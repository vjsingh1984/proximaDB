# example_05_filtered_search.py
from proximadb_sdk import ProximaDBClient
import numpy as np

client = ProximaDBClient(url="http://localhost:5678")
query_vector = np.random.randn(384).tolist()

# Filter: exact match
results = client.search(
    collection="products",
    vector=query_vector,
    top_k=10,
    filter={
        "must": [
            {"field": "category", "match": "Electronics"}
        ]
    }
)

# Filter: range query
results = client.search(
    collection="products",
    vector=query_vector,
    top_k=10,
    filter={
        "must": [
            {"field": "price", "range": {"gte": 50, "lte": 200}}
        ]
    }
)

# Filter: multiple conditions (AND)
results = client.search(
    collection="products",
    vector=query_vector,
    top_k=10,
    filter={
        "must": [
            {"field": "category", "match": "Electronics"},
            {"field": "price", "range": {"lte": 100}},
            {"field": "in_stock", "match": True}
        ]
    }
)

# Filter: OR conditions
results = client.search(
    collection="products",
    vector=query_vector,
    top_k=10,
    filter={
        "should": [
            {"field": "category", "match": "Electronics"},
            {"field": "category", "match": "Audio"}
        ],
        "minimum_should_match": 1
    }
)

# Filter: NOT condition
results = client.search(
    collection="products",
    vector=query_vector,
    top_k=10,
    filter={
        "must_not": [
            {"field": "category", "match": "Discontinued"}
        ]
    }
)

# example_06_batch_search.py
from proximadb_sdk import ProximaDBClient
import numpy as np
from concurrent.futures import ThreadPoolExecutor

client = ProximaDBClient(url="http://localhost:5678")

# Multiple query vectors
queries = [np.random.randn(384).tolist() for _ in range(100)]

# Option 1: Built-in batch search
results = client.batch_search(
    collection="products",
    vectors=queries,
    top_k=10
)

for i, result_set in enumerate(results):
    print(f"Query {i}: {len(result_set)} results")

# Option 2: Parallel search with threading
def search_one(query):
    return client.search("products", query, top_k=10)

with ThreadPoolExecutor(max_workers=10) as executor:
    results = list(executor.map(search_one, queries))

# Option 3: Async batch search
import asyncio
from proximadb_sdk import AsyncProximaDBClient

async def async_batch_search():
    async with AsyncProximaDBClient(url="http://localhost:5678") as client:
        tasks = [
            client.search("products", query, top_k=10)
            for query in queries
        ]
        results = await asyncio.gather(*tasks)
        return results

results = asyncio.run(async_batch_search())

from proximadb_sdk import ProximaDBClient

    name="documents",

results = client.hybrid_search(
    collection="documents",
    vector=query_vector,
    text_query="machine learning tutorial",
    top_k=10,
    vector_weight=0.7,
    text_weight=0.3

import numpy as np

client = ProximaDBClient(url="http://localhost:5678")
query_vector = np.random.randn(768).tolist()

# Two-stage search with reranking
# Stage 1: Fast search with quantized vectors

    collection="documents",
    vector=query_vector,
    top_k=100,
    search_params={
        "quantized": True,
        "ef_search": 50
    }
)

# Stage 2: Rerank with full precision
results = client.rerank(
    collection="documents",
    query_vector=query_vector,
    candidate_ids=[c.id for c in candidates],
    top_k=10,
    rerank_method="full_precision"  # or "cross_encoder"
)

# Automatic reranking
results = client.search(
    collection="documents",
    vector=query_vector,
    top_k=10,
    search_params={
        "rerank": True,

    }

=== RAG (Retrieval-Augmented Generation)

[source,python]

# example_10_rag_pipeline.py
from proximadb_sdk import ProximaDBClient
from openai import OpenAI
from typing import List
import numpy as np

class RAGPipeline:
    def __init__(self, proximadb_url: str, openai_api_key: str):
        self.db = ProximaDBClient(url=proximadb_url)
        self.llm = OpenAI(api_key=openai_api_key)
        self.collection = "rag_documents"

    def embed_text(self, text: str) -> List[float]:
        """Generate embedding using OpenAI."""
        response = self.llm.embeddings.create(
            model="text-embedding-3-small",
            input=text
        )
        return response.data[0].embedding

    def index_documents(self, documents: List[dict]):
        """Index documents into ProximaDB."""
        records = []
        for doc in documents:
            embedding = self.embed_text(doc["content"])
            records.append({
                "id": doc["id"],
                "vector": embedding,
                "metadata": {
                    "title": doc.get("title", ""),
                    "content": doc["content"],
                    "source": doc.get("source", "")
                }
            })

        self.db.insert(self.collection, records)

    def retrieve(self, query: str, top_k: int = 5) -> List[dict]:
        """Retrieve relevant documents."""
        query_embedding = self.embed_text(query)

        results = self.db.search(
            collection=self.collection,
            vector=query_embedding,
            top_k=top_k,
            include_metadata=True
        )

        return [
            {
                "content": r.metadata["content"],
                "title": r.metadata.get("title", ""),
                "score": r.score
            }
            for r in results
        ]

    def generate(self, query: str, context: List[dict]) -> str:
        """Generate answer using LLM."""
        context_text = "\n\n".join([
            f"Document: {doc['title']}\n{doc['content']}"
            for doc in context
        ])

        response = self.llm.chat.completions.create(
            model="gpt-4",
            messages=[
                {
                    "role": "system",
                    "content": "Answer the question based on the provided context. "
                               "If the context doesn't contain the answer, say so."
                },
                {
                    "role": "user",
                    "content": f"Context:\n{context_text}\n\nQuestion: {query}"
                }
            ]
        )

        return response.choices[0].message.content

    def query(self, question: str) -> str:
        """Full RAG pipeline."""
        # Retrieve relevant documents
        context = self.retrieve(question, top_k=5)

        # Generate answer
        answer = self.generate(question, context)

        return answer

# Usage
rag = RAGPipeline(
    proximadb_url="http://localhost:5678",
    openai_api_key="your-key"
)

# Index documents
documents = [
    {"id": "doc1", "title": "Python Basics", "content": "Python is..."},
    {"id": "doc2", "title": "Machine Learning", "content": "ML is..."},
]
rag.index_documents(documents)

# Query
answer = rag.query("What is Python?")
print(answer)

# example_11_product_search.py
from proximadb_sdk import ProximaDBClient
from sentence_transformers import SentenceTransformer
from typing import List, Dict
import json

class ProductSearch:
    def __init__(self, db_url: str):
        self.db = ProximaDBClient(url=db_url)
        self.encoder = SentenceTransformer('all-MiniLM-L6-v2')
        self.collection = "products"

    def setup_collection(self):
        """Create collection with appropriate settings."""
        self.db.create_collection(
            name=self.collection,
            dimension=384,
            engine="helix",  # Good for e-commerce
            quantization={"type": "int8"},
            index_config={"type": "hnsw", "ef_construction": 200}
        )

    def index_products(self, products: List[Dict]):
        """Index products with combined text embeddings."""
        records = []
        for product in products:
            # Combine product attributes for embedding
            text = f"{product['name']} {product['description']} {product['category']}"
            embedding = self.encoder.encode(text).tolist()

            records.append({
                "id": product["id"],
                "vector": embedding,
                "metadata": {
                    "name": product["name"],
                    "description": product["description"],
                    "category": product["category"],
                    "price": product["price"],
                    "brand": product.get("brand", ""),
                    "rating": product.get("rating", 0),
                    "in_stock": product.get("in_stock", True)
                }
            })

        self.db.insert(self.collection, records)

    def search(
        self,
        query: str,
        category: str = None,
        price_range: tuple = None,
        min_rating: float = None,
        top_k: int = 10
    ) -> List[Dict]:
        """Search products with optional filters."""
        query_embedding = self.encoder.encode(query).tolist()

        # Build filter
        filter_conditions = []
        if category:
            filter_conditions.append({"field": "category", "match": category})
        if price_range:
            filter_conditions.append({
                "field": "price",
                "range": {"gte": price_range[0], "lte": price_range[1]}
            })
        if min_rating:
            filter_conditions.append({
                "field": "rating",
                "range": {"gte": min_rating}
            })
        filter_conditions.append({"field": "in_stock", "match": True})

        filter_query = {"must": filter_conditions} if filter_conditions else None

        results = self.db.search(
            collection=self.collection,
            vector=query_embedding,
            top_k=top_k,
            filter=filter_query,
            include_metadata=True
        )

        return [
            {
                "id": r.id,
                "name": r.metadata["name"],
                "description": r.metadata["description"],
                "price": r.metadata["price"],
                "score": r.score
            }
            for r in results
        ]

# Usage
search = ProductSearch("http://localhost:5678")
search.setup_collection()

# Index products
products = [
    {
        "id": "prod_001",
        "name": "Wireless Bluetooth Headphones",
        "description": "High-quality noise-canceling headphones",
        "category": "Electronics",
        "price": 149.99,
        "rating": 4.5
    },
    # ... more products
]
search.index_products(products)

# Search
results = search.search(
    query="comfortable headphones for work",
    category="Electronics",
    price_range=(50, 200),
    min_rating=4.0
)

# example_12_recommendations.py
from proximadb_sdk import ProximaDBClient
import numpy as np
from typing import List, Dict

class RecommendationEngine:
    def __init__(self, db_url: str):
        self.db = ProximaDBClient(url=db_url)
        self.items_collection = "items"
        self.users_collection = "users"

    def setup(self, item_dimension: int = 128, user_dimension: int = 128):
        """Setup collections."""
        self.db.create_collection(self.items_collection, dimension=item_dimension)
        self.db.create_collection(self.users_collection, dimension=user_dimension)

    def index_items(self, items: List[Dict]):
        """Index item embeddings."""
        records = [
            {
                "id": item["id"],
                "vector": item["embedding"],
                "metadata": item.get("metadata", {})
            }
            for item in items
        ]
        self.db.insert(self.items_collection, records)

    def update_user_profile(self, user_id: str, interacted_items: List[str]):
        """Update user embedding based on interactions."""
        # Get item embeddings
        items = self.db.get(self.items_collection, ids=interacted_items)

        # Compute user embedding as weighted average
        if items:
            vectors = [item.vector for item in items]
            user_embedding = np.mean(vectors, axis=0).tolist()

            self.db.upsert(self.users_collection, [{
                "id": user_id,
                "vector": user_embedding,
                "metadata": {"interaction_count": len(interacted_items)}
            }])

    def get_recommendations(
        self,
        user_id: str,
        exclude_ids: List[str] = None,
        top_k: int = 10
    ) -> List[Dict]:
        """Get personalized recommendations."""
        # Get user embedding
        users = self.db.get(self.users_collection, ids=[user_id])
        if not users:
            return []

        user_embedding = users[0].vector

        # Find similar items
        filter_query = None
        if exclude_ids:
            filter_query = {
                "must_not": [{"field": "id", "in": exclude_ids}]
            }

        results = self.db.search(
            collection=self.items_collection,
            vector=user_embedding,
            top_k=top_k,
            filter=filter_query,
            include_metadata=True
        )

        return [
            {
                "item_id": r.id,
                "score": r.score,
                "metadata": r.metadata
            }
            for r in results
        ]

    def get_similar_items(self, item_id: str, top_k: int = 10) -> List[Dict]:
        """Find similar items."""
        items = self.db.get(self.items_collection, ids=[item_id])
        if not items:
            return []

        results = self.db.search(
            collection=self.items_collection,
            vector=items[0].vector,
            top_k=top_k + 1,  # Exclude self
            include_metadata=True
        )

        return [
            {"item_id": r.id, "score": r.score, "metadata": r.metadata}
            for r in results if r.id != item_id
        ][:top_k]

# Usage
engine = RecommendationEngine("http://localhost:5678")
engine.setup()

# Index items
items = [
    {"id": "item_001", "embedding": np.random.randn(128).tolist(), "metadata": {"name": "Item 1"}},
    # ...
]
engine.index_items(items)

# Update user profile
engine.update_user_profile("user_001", ["item_001", "item_002", "item_003"])

# Get recommendations
recommendations = engine.get_recommendations("user_001", top_k=10)

# example_13_langchain.py
from langchain.vectorstores import VectorStore
from langchain.embeddings.base import Embeddings
from langchain.schema import Document
from proximadb_sdk import ProximaDBClient
from typing import List, Optional, Tuple

class ProximaDBVectorStore(VectorStore):
    """LangChain VectorStore implementation for ProximaDB."""

    def __init__(
        self,
        collection_name: str,
        embedding: Embeddings,
        client: ProximaDBClient = None,
        url: str = "http://localhost:5678"
    ):
        self.collection_name = collection_name
        self.embedding = embedding
        self.client = client or ProximaDBClient(url=url)

    def add_texts(
        self,
        texts: List[str],
        metadatas: Optional[List[dict]] = None,
        ids: Optional[List[str]] = None,
        **kwargs
    ) -> List[str]:
        """Add texts to the vector store."""
        embeddings = self.embedding.embed_documents(texts)
        metadatas = metadatas or [{} for _ in texts]
        ids = ids or [f"doc_{i}" for i in range(len(texts))]

        records = [
            {
                "id": id_,
                "vector": emb,
                "metadata": {**meta, "text": text}
            }
            for id_, text, emb, meta in zip(ids, texts, embeddings, metadatas)
        ]

        self.client.insert(self.collection_name, records)
        return ids

    def similarity_search(
        self,
        query: str,
        k: int = 4,
        filter: Optional[dict] = None,
        **kwargs
    ) -> List[Document]:
        """Search for similar documents."""
        query_embedding = self.embedding.embed_query(query)

        results = self.client.search(
            collection=self.collection_name,
            vector=query_embedding,
            top_k=k,
            filter=filter,
            include_metadata=True
        )

        return [
            Document(
                page_content=r.metadata.pop("text", ""),
                metadata=r.metadata
            )
            for r in results
        ]

    def similarity_search_with_score(
        self,
        query: str,
        k: int = 4,
        **kwargs
    ) -> List[Tuple[Document, float]]:
        """Search with relevance scores."""
        query_embedding = self.embedding.embed_query(query)

        results = self.client.search(
            collection=self.collection_name,
            vector=query_embedding,
            top_k=k,
            include_metadata=True
        )

        return [
            (
                Document(
                    page_content=r.metadata.pop("text", ""),
                    metadata=r.metadata
                ),
                r.score
            )
            for r in results
        ]

    @classmethod
    def from_texts(
        cls,
        texts: List[str],
        embedding: Embeddings,
        metadatas: Optional[List[dict]] = None,
        collection_name: str = "langchain",
        **kwargs
    ) -> "ProximaDBVectorStore":
        """Create vector store from texts."""
        store = cls(collection_name=collection_name, embedding=embedding, **kwargs)
        store.add_texts(texts, metadatas)
        return store

# Usage
from langchain.embeddings import OpenAIEmbeddings

embeddings = OpenAIEmbeddings()
vectorstore = ProximaDBVectorStore(
    collection_name="langchain_docs",
    embedding=embeddings
)

# Add documents
docs = ["Document 1 content", "Document 2 content"]
vectorstore.add_texts(docs)

# Search
results = vectorstore.similarity_search("query text", k=5)
