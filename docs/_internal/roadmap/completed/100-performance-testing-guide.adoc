== Python Benchmark Framework

=== Basic Benchmark Script

[source,python]
----
import time
import numpy as np
from concurrent.futures import ThreadPoolExecutor, as_completed
from proximadb_sdk import ProximaDBClient, VectorRecord
import statistics

client = ProximaDBClient(url="http://localhost:5678")

def benchmark_search(collection, queries, top_k=10, concurrent=1):
    """Benchmark search performance."""
    latencies = []

    def search_query(query):
        start = time.perf_counter()
        results = client.search(collection, vector=query.tolist(), top_k=top_k)
        return time.perf_counter() - start

    with ThreadPoolExecutor(max_workers=concurrent) as executor:
        start_time = time.perf_counter()
        futures = [executor.submit(search_query, q) for q in queries]
        for future in as_completed(futures):
            latencies.append(future.result())
        total_time = time.perf_counter() - start_time

    return {
        "total_queries": len(queries),
        "total_time_seconds": total_time,
        "qps": len(queries) / total_time,
        "latency_avg_ms": statistics.mean(latencies) * 1000,
        "latency_p50_ms": statistics.median(latencies) * 1000,
        "latency_p95_ms": np.percentile(latencies, 95) * 1000,
        "latency_p99_ms": np.percentile(latencies, 99) * 1000,
    }

def benchmark_insert(collection, vectors, batch_size=1000):
    """Benchmark insert performance."""
    total_vectors = len(vectors)
    latencies = []

    for i in range(0, total_vectors, batch_size):
        batch = vectors[i:i+batch_size]
        records = [
            VectorRecord(id=f"vec_{i+j}", vector=v.tolist())
            for j, v in enumerate(batch)
        ]

        start = time.perf_counter()
        client.insert_vectors(collection, records)
        latencies.append(time.perf_counter() - start)

    total_time = sum(latencies)
    return {
        "total_vectors": total_vectors,
        "total_time_seconds": total_time,
        "vectors_per_second": total_vectors / total_time,
        "batch_latency_avg_ms": statistics.mean(latencies) * 1000,
        "batch_latency_p99_ms": np.percentile(latencies, 99) * 1000,
    }

# Run benchmarks
dimension = 1536
num_vectors = 100000
num_queries = 1000

# Generate test data
vectors = np.random.randn(num_vectors, dimension).astype(np.float32)
queries = np.random.randn(num_queries, dimension).astype(np.float32)

# Create collection
client.create_collection("benchmark_test", dimension=dimension)

# Benchmark inserts
print("Benchmarking inserts...")
insert_results = benchmark_insert("benchmark_test", vectors)
print(f"Insert throughput: {insert_results['vectors_per_second']:.0f} vectors/sec")

# Benchmark searches
print("\nBenchmarking searches...")
for concurrent in [1, 4, 8, 16]:
    results = benchmark_search("benchmark_test", queries, concurrent=concurrent)
    print(f"Concurrent {concurrent}: {results['qps']:.0f} QPS, "
          f"P99: {results['latency_p99_ms']:.1f}ms")

# Cleanup
client.delete_collection("benchmark_test")

=== Comprehensive Benchmark Suite

[source,python]
----
import json
import time
import numpy as np
from dataclasses import dataclass, asdict
from typing import List, Dict, Any
from concurrent.futures import ThreadPoolExecutor, as_completed
from proximadb_sdk import ProximaDBClient, VectorRecord
import psutil

@dataclass
class BenchmarkConfig:
    vectors: int = 100000
    dimension: int = 1536
    queries: int = 1000
    top_k: int = 10
    concurrent_clients: List[int] = None
    batch_size: int = 1000
    warmup_queries: int = 100
    engines: List[str] = None
    quantization_types: List[str] = None

    def __post_init__(self):
        self.concurrent_clients = self.concurrent_clients or [1, 4, 8, 16]
        self.engines = self.engines or ["sst"]
        self.quantization_types = self.quantization_types or ["none", "int8"]

@dataclass
class BenchmarkResult:
    test_name: str
    config: Dict[str, Any]
    metrics: Dict[str, Any]
    system_metrics: Dict[str, Any]
    timestamp: str

class ProximaDBBenchmark:
    def __init__(self, url: str = "http://localhost:5678"):
        self.client = ProximaDBClient(url=url)
        self.results: List[BenchmarkResult] = []

    def generate_data(self, config: BenchmarkConfig):
        """Generate test vectors and queries."""
        np.random.seed(42)
        vectors = np.random.randn(config.vectors, config.dimension).astype(np.float32)
        queries = np.random.randn(config.queries, config.dimension).astype(np.float32)
        return vectors, queries

    def get_system_metrics(self):
        """Collect system metrics."""
        return {
            "cpu_percent": psutil.cpu_percent(interval=1),
            "memory_used_gb": psutil.virtual_memory().used / 1e9,
            "memory_percent": psutil.virtual_memory().percent,
            "disk_read_mb": psutil.disk_io_counters().read_bytes / 1e6,
            "disk_write_mb": psutil.disk_io_counters().write_bytes / 1e6,
        }

    def benchmark_search_throughput(self, collection: str, queries: np.ndarray,
                                     config: BenchmarkConfig) -> Dict:
        """Benchmark search throughput at different concurrency levels."""
        results = {}

        for concurrent in config.concurrent_clients:
            latencies = []

            def search_query(query):
                start = time.perf_counter()
                self.client.search(collection, vector=query.tolist(), top_k=config.top_k)
                return time.perf_counter() - start

            # Warmup
            for q in queries[:config.warmup_queries]:
                search_query(q)

            # Benchmark
            with ThreadPoolExecutor(max_workers=concurrent) as executor:
                start_time = time.perf_counter()
                futures = [executor.submit(search_query, q) for q in queries]
                for future in as_completed(futures):
                    latencies.append(future.result())
                total_time = time.perf_counter() - start_time

            results[f"concurrent_{concurrent}"] = {
                "qps": len(queries) / total_time,
                "latency_avg_ms": np.mean(latencies) * 1000,
                "latency_p50_ms": np.percentile(latencies, 50) * 1000,
                "latency_p95_ms": np.percentile(latencies, 95) * 1000,
                "latency_p99_ms": np.percentile(latencies, 99) * 1000,
                "latency_p999_ms": np.percentile(latencies, 99.9) * 1000,
            }

        return results

    def benchmark_insert_throughput(self, collection: str, vectors: np.ndarray,
                                     config: BenchmarkConfig) -> Dict:
        """Benchmark insert throughput."""
        latencies = []
        total_vectors = len(vectors)

        for i in range(0, total_vectors, config.batch_size):
            batch = vectors[i:i+config.batch_size]
            records = [
                VectorRecord(id=f"vec_{i+j}", vector=v.tolist())
                for j, v in enumerate(batch)
            ]

            start = time.perf_counter()
            self.client.insert_vectors(collection, records)
            latencies.append(time.perf_counter() - start)

        total_time = sum(latencies)
        return {
            "vectors_per_second": total_vectors / total_time,
            "batches_per_second": len(latencies) / total_time,
            "batch_latency_avg_ms": np.mean(latencies) * 1000,
            "batch_latency_p99_ms": np.percentile(latencies, 99) * 1000,
        }

    def benchmark_recall(self, collection: str, queries: np.ndarray,
                         vectors: np.ndarray, config: BenchmarkConfig) -> Dict:
        """Benchmark search recall accuracy."""
        # Compute ground truth using brute force
        def brute_force_search(query, vectors, k):
            distances = np.linalg.norm(vectors - query, axis=1)
            return np.argsort(distances)[:k]

        recalls = []
        for query in queries[:100]:  # Sample for ground truth computation
            # Get ground truth
            gt = set(brute_force_search(query, vectors, config.top_k))

            # Get ProximaDB results
            results = self.client.search(collection, vector=query.tolist(), top_k=config.top_k)
            predicted = set(int(r.id.split("_")[1]) for r in results)

            # Compute recall
            recall = len(gt & predicted) / config.top_k
            recalls.append(recall)

        return {
            "recall_avg": np.mean(recalls),
            "recall_min": np.min(recalls),
            "recall_max": np.max(recalls),
            "recall_std": np.std(recalls),
        }

    def run_full_benchmark(self, config: BenchmarkConfig) -> List[BenchmarkResult]:
        """Run complete benchmark suite."""
        vectors, queries = self.generate_data(config)

        for engine in config.engines:
            for quant in config.quantization_types:
                collection_name = f"bench_{engine}_{quant}"

                # Create collection
                quant_config = {"type": quant} if quant != "none" else None
                self.client.create_collection(
                    collection_name,
                    dimension=config.dimension,
                    engine=engine,
                    quantization=quant_config
                )

                try:
                    # Insert benchmark
                    system_before = self.get_system_metrics()
                    insert_metrics = self.benchmark_insert_throughput(
                        collection_name, vectors, config
                    )
                    system_after = self.get_system_metrics()

                    self.results.append(BenchmarkResult(
                        test_name=f"insert_{engine}_{quant}",
                        config=asdict(config),
                        metrics=insert_metrics,
                        system_metrics={
                            "before": system_before,
                            "after": system_after
                        },
                        timestamp=time.strftime("%Y-%m-%dT%H:%M:%SZ")
                    ))

                    # Search benchmark
                    system_before = self.get_system_metrics()
                    search_metrics = self.benchmark_search_throughput(
                        collection_name, queries, config
                    )
                    system_after = self.get_system_metrics()

                    self.results.append(BenchmarkResult(
                        test_name=f"search_{engine}_{quant}",
                        config=asdict(config),
                        metrics=search_metrics,
                        system_metrics={
                            "before": system_before,
                            "after": system_after
                        },
                        timestamp=time.strftime("%Y-%m-%dT%H:%M:%SZ")
                    ))

                    # Recall benchmark
                    recall_metrics = self.benchmark_recall(
                        collection_name, queries, vectors, config
                    )

                    self.results.append(BenchmarkResult(
                        test_name=f"recall_{engine}_{quant}",
                        config=asdict(config),
                        metrics=recall_metrics,
                        system_metrics={},
                        timestamp=time.strftime("%Y-%m-%dT%H:%M:%SZ")
                    ))

                finally:
                    # Cleanup
                    self.client.delete_collection(collection_name)

        return self.results

    def save_results(self, filepath: str):
        """Save results to JSON file."""
        with open(filepath, 'w') as f:
            json.dump([asdict(r) for r in self.results], f, indent=2)

# Usage
benchmark = ProximaDBBenchmark()
config = BenchmarkConfig(
    vectors=100000,
    dimension=1536,
    queries=1000,
    engines=["sst", "helix"],
    quantization_types=["none", "int8"]
)
results = benchmark.run_full_benchmark(config)
benchmark.save_results("benchmark_results.json")
----

# locustfile.py
from locust import HttpUser, task, between
import numpy as np
import json

class ProximaDBUser(HttpUser):
    wait_time = between(0.1, 0.5)
    host = "http://localhost:5678"

    def on_start(self):
        """Initialize test data."""
        self.dimension = 1536
        self.collection = "load_test"
        self.queries = [
            np.random.randn(self.dimension).tolist()
            for _ in range(100)
        ]
        self.query_idx = 0

    @task(10)
    def search(self):
        """Search query (high frequency)."""
        query = self.queries[self.query_idx % len(self.queries)]
        self.query_idx += 1

        self.client.post(
            f"/api/v1/collections/{self.collection}/search",
            json={
                "vector": query,
                "top_k": 10
            },
            name="/search"
        )

    @task(1)
    def insert(self):
        """Insert vectors (low frequency)."""
        vectors = [
            {
                "id": f"vec_{np.random.randint(1000000)}",
                "vector": np.random.randn(self.dimension).tolist()
            }
            for _ in range(10)
        ]

        self.client.post(
            f"/api/v1/collections/{self.collection}/vectors",
            json={"vectors": vectors},
            name="/insert"
        )

    @task(1)
    def get_collection_info(self):
        """Get collection info."""
        self.client.get(
            f"/api/v1/collections/{self.collection}",
            name="/collection-info"
        )

// k6-test.js
import http from 'k6/http';
import { check, sleep } from 'k6';
import { Rate, Trend } from 'k6/metrics';

const errorRate = new Rate('errors');
const searchLatency = new Trend('search_latency');

export const options = {
  stages: [
    { duration: '1m', target: 50 },   // Ramp up
    { duration: '3m', target: 50 },   // Sustain
    { duration: '1m', target: 100 },  // Peak
    { duration: '1m', target: 0 },    // Ramp down
  ],
  thresholds: {
    http_req_duration: ['p(95)<200'],
    errors: ['rate<0.01'],
  },
};

const BASE_URL = 'http://localhost:5678';
const COLLECTION = 'load_test';
const DIMENSION = 1536;

function randomVector(dim) {
  return Array.from({ length: dim }, () => Math.random() * 2 - 1);
}

export default function () {
  // Search query
  const searchPayload = JSON.stringify({
    vector: randomVector(DIMENSION),
    top_k: 10,
  });

  const searchRes = http.post(
    `${BASE_URL}/api/v1/collections/${COLLECTION}/search`,
    searchPayload,
    { headers: { 'Content-Type': 'application/json' } }
  );

  check(searchRes, {
    'search status is 200': (r) => r.status === 200,
    'search has results': (r) => JSON.parse(r.body).results.length > 0,
  });

  errorRate.add(searchRes.status !== 200);
  searchLatency.add(searchRes.timings.duration);

  sleep(0.1);
}

=== Quick Reference

[source,bash]
----
# Quick benchmark
proximadb-bench --quick --vectors 100000 --queries 1000

# Full benchmark
proximadb-bench --config benchmark.toml

# Load test with Locust
locust -f locustfile.py --users 100 --run-time 5m

# Profile CPU
perf record -g -p $(pgrep proximadb-server) sleep 60

# Monitor performance
curl http://localhost:9090/metrics | grep proximadb_query
----
