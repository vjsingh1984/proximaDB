==== Live Coding Session

[source,python]
----
# Complete working example built during video
import asyncio
from proximadb_sdk import (
    ProximaDBClient,
    VectorRecord,
    SearchQuery,
    Filter
)

async def main():
    # Async client with connection pooling
    client = ProximaDBClient(
        url="http://localhost:5678",
        max_connections=10,
        timeout=30.0
    )

    # Create collection with all options explained
    await client.create_collection(
        name="products",
        dimension=768,
        engine="helix",
        distance_metric="cosine",
        metadata_schema={
            "category": "string",
            "price": "float",
            "in_stock": "bool"
        }
    )

    # Batch insert with progress
    records = [
        VectorRecord(
            id=f"product_{i}",
            vector=[float(i % 100) / 100] * 768,
            metadata={
                "category": "electronics",
                "price": 99.99 + i,
                "in_stock": True
            }
        )
        for i in range(1000)
    ]

    result = await client.insert_batch(
        collection="products",
        records=records,
        batch_size=100
    )
    print(f"Inserted {result.inserted_count} vectors")

    # Advanced search with filters
    results = await client.search(
        collection="products",
        query=SearchQuery(
            vector=[0.5] * 768,
            top_k=10,
            filter=Filter.and_(
                Filter.eq("category", "electronics"),
                Filter.gte("price", 100.0),
                Filter.eq("in_stock", True)
            )
        )
    )

    for r in results:
        print(f"ID: {r.id}, Score: {r.score:.4f}, Metadata: {r.metadata}")

if __name__ == "__main__":
    asyncio.run(main())
----

==== Code Demonstration

from proximadb_sdk import ProximaDBClient, GraphNode, GraphEdge

client = ProximaDBClient("http://localhost:5678")

# Create a knowledge graph
client.create_graph("knowledge_base", engine="orion")

# Add nodes with embeddings
nodes = [
    GraphNode("person_1")
        .with_label("Person")
        .with_property("name", "Alice")
        .with_vector([0.1] * 768),  # Person embedding
    GraphNode("person_2")
        .with_label("Person")
        .with_property("name", "Bob")
        .with_vector([0.2] * 768),
    GraphNode("company_1")
        .with_label("Company")
        .with_property("name", "TechCorp")
        .with_vector([0.3] * 768),
]
client.create_nodes("knowledge_base", nodes)

# Create relationships
edges = [
    GraphEdge("person_1", "company_1", "WORKS_AT")
        .with_property("since", "2020"),
    GraphEdge("person_1", "person_2", "KNOWS")
        .with_weight(0.9),
]
client.create_edges("knowledge_base", edges)

# Hybrid query: Find similar people who work at similar companies
results = client.graph_vector_search(
    graph="knowledge_base",
    start_query=[0.15] * 768,  # Find similar to this embedding
    traversal="MATCH (p:Person)-[:WORKS_AT]->(c:Company) RETURN p, c",
    top_k=5
)

==== Backup Script Walkthrough

[source,bash]
----
#!/bin/bash
# backup_proximadb.sh - Walk through each section

# Configuration
BACKUP_DIR="/backups/proximadb"
S3_BUCKET="s3://mycompany-proximadb-backups"
DATE=$(date +%Y%m%d_%H%M%S)

# Step 1: Create consistent snapshot
echo "Creating backup snapshot..."
curl -X POST http://localhost:5678/api/v1/admin/backup/start \
  -H "Content-Type: application/json" \
  -d '{"type": "full", "destination": "'$BACKUP_DIR/$DATE'"}'

# Step 2: Wait for completion
while true; do
  STATUS=$(curl -s http://localhost:5678/api/v1/admin/backup/status)
  if echo "$STATUS" | grep -q "completed"; then
    break
  fi
  sleep 10
done

# Step 3: Upload to S3
aws s3 sync "$BACKUP_DIR/$DATE" "$S3_BUCKET/$DATE/"

# Step 4: Cleanup old backups (keep 7 days)
find $BACKUP_DIR -type d -mtime +7 -exec rm -rf {} \;

echo "Backup completed: $DATE"
----

==== Complete RAG Implementation

[source,python]
----
# Full RAG system built during video
from proximadb_sdk import ProximaDBClient, VectorRecord
from sentence_transformers import SentenceTransformer
from openai import OpenAI

class RAGSystem:
    def __init__(self):
        self.db = ProximaDBClient("http://localhost:5678")
        self.encoder = SentenceTransformer('all-MiniLM-L6-v2')
        self.llm = OpenAI()

        # Create collection
        self.db.create_collection(
            name="documents",
            dimension=384,
            engine="helix"  # Good for semantic search
        )

    def ingest_document(self, doc_id: str, text: str, metadata: dict):
        """Chunk and embed document"""
        chunks = self._chunk_text(text, chunk_size=500, overlap=50)

        records = []
        for i, chunk in enumerate(chunks):
            embedding = self.encoder.encode(chunk).tolist()
            records.append(VectorRecord(
                id=f"{doc_id}_chunk_{i}",
                vector=embedding,
                metadata={
                    **metadata,
                    "chunk_index": i,
                    "text": chunk
                }
            ))

        self.db.insert("documents", records)

    def retrieve(self, query: str, top_k: int = 5):
        """Retrieve relevant chunks"""
        query_embedding = self.encoder.encode(query).tolist()

        results = self.db.search(
            collection="documents",
            vector=query_embedding,
            top_k=top_k
        )

        return [r.metadata["text"] for r in results]

    def answer(self, question: str) -> str:
        """RAG answer generation"""
        # Retrieve context
        contexts = self.retrieve(question, top_k=5)
        context_text = "\n\n".join(contexts)

        # Generate answer
        response = self.llm.chat.completions.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": "Answer based on the provided context."},
                {"role": "user", "content": f"Context:\n{context_text}\n\nQuestion: {question}"}
            ]
        )

        return response.choices[0].message.content

    def _chunk_text(self, text: str, chunk_size: int, overlap: int):
        """Simple chunking with overlap"""
        words = text.split()
        chunks = []
        for i in range(0, len(words), chunk_size - overlap):
            chunk = " ".join(words[i:i + chunk_size])
            chunks.append(chunk)
        return chunks

# Usage demonstration
rag = RAGSystem()
rag.ingest_document("doc1", "Long document text here...", {"source": "manual"})
answer = rag.answer("What is the main topic?")
print(answer)
----
