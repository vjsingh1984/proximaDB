**Complete guide for migrating from Weaviate to ProximaDB**

== Overview

This guide covers migrating from Weaviate's schema-based vector database to ProximaDB. Both databases support rich metadata, but ProximaDB offers additional storage engines, native SQL, and unified multi-model capabilities.

=== Feature Comparison

[cols="2,2,2", options="header"]
|===
| Feature | Weaviate | ProximaDB
| Schema Definition | Required (GraphQL-like) | Optional (flexible JSON)
| Vector Search | ✅ | ✅
| Hybrid Search | ✅ (BM25 + vectors) | ✅ (Multi-modal fusion)
| Graph Relationships | References (limited) | Full Graph Database
| Modules | Plugin-based | Built-in
| Storage Engines | 1 | 6
| SQL Support | ❌ (GraphQL only) | ✅
| Document Store | ❌ | ✅
| Multi-Tenancy | ✅ | ✅
|===

== Step 1: Export Data from Weaviate

=== Export Using Weaviate Client

[source,python]
----
import weaviate
import json

# Connect to Weaviate
client = weaviate.Client("http://localhost:8080")

def export_weaviate_class(class_name, output_file, batch_size=100):
    """Export all objects from a Weaviate class."""

    # Get class schema for reference
    schema = client.schema.get(class_name)
    print(f"Exporting class: {class_name}")
    print(f"Properties: {[p['name'] for p in schema['properties']]}")

    objects = []
    cursor = None

    while True:
        # Query with cursor-based pagination
        query = (
            client.query
            .get(class_name, [p['name'] for p in schema['properties']])
            .with_additional(['id', 'vector', 'creationTimeUnix'])
            .with_limit(batch_size)
        )

        if cursor:
            query = query.with_after(cursor)

        result = query.do()

        if 'errors' in result:
            print(f"Error: {result['errors']}")
            break

        batch = result.get('data', {}).get('Get', {}).get(class_name, [])

        if not batch:
            break

        for obj in batch:
            additional = obj.pop('_additional', {})
            objects.append({
                'id': additional.get('id'),
                'vector': additional.get('vector'),
                'properties': obj,
                'created_at': additional.get('creationTimeUnix')
            })

        cursor = objects[-1]['id']
        print(f"Exported {len(objects)} objects...")

    # Save to file
    with open(output_file, 'w') as f:
        json.dump({
            'class_name': class_name,
            'schema': schema,
            'objects': objects
        }, f, indent=2)

    print(f"Exported {len(objects)} objects to {output_file}")
    return objects

# Export each class
export_weaviate_class("Article", "weaviate_articles.json")
export_weaviate_class("Author", "weaviate_authors.json")

def export_with_references(class_name, ref_properties, output_file):
    """Export objects including cross-references."""

    schema = client.schema.get(class_name)
    prop_names = [p['name'] for p in schema['properties']
                  if p['dataType'][0] not in ['text[]', 'number[]']]

    # Build reference query
    ref_queries = []
    for ref_prop in ref_properties:
        ref_queries.append(f"{ref_prop} {{ ... on {ref_properties[ref_prop]} {{ _additional {{ id }} }} }}")

    objects = []
    cursor = None

    while True:
        # GraphQL query with references
        gql = f"""
        {{
            Get {{
                {class_name}(limit: 100{f', after: "{cursor}"' if cursor else ''}) {{
                    {' '.join(prop_names)}
                    {' '.join(ref_queries)}
                    _additional {{ id vector }}
                }}
            }}
        }}
        """

        result = client.query.raw(gql)
        batch = result.get('data', {}).get('Get', {}).get(class_name, [])

        if not batch:
            break

        for obj in batch:
            additional = obj.pop('_additional', {})

            # Extract references
            refs = {}
            for ref_prop in ref_properties:
                if ref_prop in obj and obj[ref_prop]:
                    refs[ref_prop] = [
                        r['_additional']['id']
                        for r in obj[ref_prop]
                        if '_additional' in r
                    ]
                    del obj[ref_prop]

            objects.append({
                'id': additional.get('id'),
                'vector': additional.get('vector'),
                'properties': obj,
                'references': refs
            })

        cursor = objects[-1]['id']

    with open(output_file, 'w') as f:
        json.dump(objects, f)

    return objects

# Export with references
export_with_references(
    "Article",
    {"hasAuthor": "Author", "relatedArticles": "Article"},
    "articles_with_refs.json"
)

== Step 2: Create ProximaDB Collections

=== Map Weaviate Schema

[source,python]
----
from proximadb_sdk import ProximaDBClient

client = ProximaDBClient(url="http://localhost:5678")

# Weaviate dataType to ProximaDB (metadata is flexible JSON)
# No schema mapping needed - ProximaDB accepts any JSON metadata

def create_collection_from_weaviate_schema(weaviate_export_file):
    """Create ProximaDB collection from Weaviate export."""

    with open(weaviate_export_file) as f:
        data = json.load(f)

    schema = data['schema']
    class_name = data['class_name']

    # Determine dimension from first vector
    dimension = len(data['objects'][0]['vector']) if data['objects'] else 1536

    # Map Weaviate vectorizer distance to ProximaDB metric
    distance_mapping = {
        'cosine': 'cosine',
        'l2-squared': 'euclidean',
        'dot': 'dot_product'
    }

    vector_config = schema.get('vectorIndexConfig', {})
    distance = vector_config.get('distance', 'cosine')

    collection = client.create_collection(
        name=class_name.lower(),  # Lowercase for consistency
        dimension=dimension,
        distance_metric=distance_mapping.get(distance, 'cosine'),
        engine="sst",
        index_config={
            "type": "hnsw",
            "m": vector_config.get('maxConnections', 16),
            "ef_construction": vector_config.get('efConstruction', 128)
        }
    )

    print(f"Created collection: {collection['name']}")
    return collection

create_collection_from_weaviate_schema("weaviate_articles.json")

== Step 3: Import Data to ProximaDB

=== Basic Import

[source,python]

import json
from proximadb_sdk import ProximaDBClient, VectorRecord

client = ProximaDBClient(url="http://localhost:5678")

def import_weaviate_export(export_file, collection_name):
    """Import Weaviate export to ProximaDB."""

    with open(export_file) as f:
        data = json.load(f)

    objects = data['objects']
    batch_size = 1000
    total = 0

    for i in range(0, len(objects), batch_size):
        batch = objects[i:i+batch_size]

        records = []
        for obj in batch:
            records.append(VectorRecord(
                id=obj['id'],
                vector=obj['vector'],
                metadata=obj['properties']
            ))

        client.insert_vectors(collection_name, records)
        total += len(batch)
        print(f"Imported {total}/{len(objects)}")

    print(f"Import complete: {total} vectors")

import_weaviate_export("weaviate_articles.json", "article")

def import_with_graph_relationships(export_file, vector_collection, graph_name):
    """Import vectors and convert cross-references to graph edges."""

    with open(export_file) as f:
        objects = json.load(f)

    # First, import all vectors
    records = []
    for obj in objects:
        records.append(VectorRecord(
            id=obj['id'],
            vector=obj['vector'],
            metadata=obj['properties']
        ))

    # Batch insert vectors
    client.batch_upsert(vector_collection, records)
    print(f"Imported {len(records)} vectors")

    # Create graph for relationships
    client.create_graph(graph_name)

    # Create nodes (can reference vector IDs)
    nodes = []
    for obj in objects:
        nodes.append({
            'id': obj['id'],
            'labels': [vector_collection.title()],
            'properties': {
                'vector_id': obj['id'],
                **obj['properties']
            }
        })

    client.batch_create_nodes(graph_name, nodes)

    # Create edges from references
    edges = []
    for obj in objects:
        for ref_type, ref_ids in obj.get('references', {}).items():
            for ref_id in ref_ids:
                edges.append({
                    'from_node_id': obj['id'],
                    'to_node_id': ref_id,
                    'edge_type': ref_type.upper()
                })

    if edges:
        client.batch_create_edges(graph_name, edges)
        print(f"Created {len(edges)} graph edges")

import_with_graph_relationships(
    "articles_with_refs.json",
    "article",
    "article_graph"
)

== Step 5: Migrate Weaviate Filters

=== Filter Syntax Mapping

[cols="2,2,2", options="header"]
|===
| Weaviate Operator | ProximaDB Filter | Notes
| `Equal` | `{"field": "x", "match": "val"}` | Equality
| `NotEqual` | `must_not` clause | Negation
| `GreaterThan` | `{"range": {"gt": val}}` | Greater
| `GreaterThanEqual` | `{"range": {"gte": val}}` | Greater or equal
| `LessThan` | `{"range": {"lt": val}}` | Less
| `LessThanEqual` | `{"range": {"lte": val}}` | Less or equal
| `Like` | Text search | Pattern matching
| `ContainsAny` | `{"in": [...]}` | In list
| `And` | `{"must": [...]}` | AND
| `Or` | `{"should": [...]}` | OR
|===

=== Filter Conversion

[source,python]
----
def convert_weaviate_filter(weaviate_filter):
    """Convert Weaviate where clause to ProximaDB filter."""

    if not weaviate_filter:
        return None

    operator = weaviate_filter.get('operator')
    path = weaviate_filter.get('path', [])
    field = path[0] if path else None

    # Compound operators
    if operator == 'And':
        must = [convert_weaviate_filter(op) for op in weaviate_filter.get('operands', [])]
        return {"must": [m for m in must if m]}

    if operator == 'Or':
        should = [convert_weaviate_filter(op) for op in weaviate_filter.get('operands', [])]
        return {"should": [s for s in should if s]}

    # Value extraction
    value = None
    for key in ['valueText', 'valueInt', 'valueNumber', 'valueBoolean', 'valueDate']:
        if key in weaviate_filter:
            value = weaviate_filter[key]
            break

    # Comparison operators
    if operator == 'Equal':
        return {"field": field, "match": value}
    elif operator == 'NotEqual':
        return {"field": field, "match": value, "_not": True}
    elif operator == 'GreaterThan':
        return {"field": field, "range": {"gt": value}}
    elif operator == 'GreaterThanEqual':
        return {"field": field, "range": {"gte": value}}
    elif operator == 'LessThan':
        return {"field": field, "range": {"lt": value}}
    elif operator == 'LessThanEqual':
        return {"field": field, "range": {"lte": value}}
    elif operator == 'ContainsAny':
        values = weaviate_filter.get('valueTextArray', [])
        return {"field": field, "in": values}

    return None

== Step 8: Validation

[source,python]
----
import weaviate
from proximadb_sdk import ProximaDBClient
import numpy as np

weaviate_client = weaviate.Client("http://localhost:8080")
proximadb_client = ProximaDBClient("http://localhost:5678")

def validate_migration(class_name, collection_name, sample_size=100):
    """Validate migration by comparing search results."""

    # Get sample vectors from Weaviate
    result = weaviate_client.query.get(
        class_name, []
    ).with_additional(['id', 'vector']).with_limit(sample_size).do()

    samples = result['data']['Get'][class_name]

    matches = 0
    for sample in samples:
        vector = sample['_additional']['vector']

        # Query Weaviate
        w_result = weaviate_client.query.get(
            class_name, []
        ).with_near_vector({
            "vector": vector
        }).with_limit(10).with_additional(['id']).do()

        w_ids = [o['_additional']['id'] for o in w_result['data']['Get'][class_name]]

        # Query ProximaDB
        p_result = proximadb_client.search(collection_name, vector, top_k=10)
        p_ids = [r.id for r in p_result]

        # Compare
        overlap = len(set(w_ids) & set(p_ids))
        if overlap >= 8:  # 80% match
            matches += 1

    accuracy = matches / len(samples) * 100
    print(f"Migration accuracy: {accuracy:.1f}%")
    return accuracy

validate_migration("Article", "article")

== Summary

Migration from Weaviate to ProximaDB involves:

1. **Export**: Use GraphQL API with pagination
2. **Schema**: Optional in ProximaDB (flexible JSON)
3. **Import**: Batch insert vectors and create graph edges
4. **References**: Convert to ProximaDB graph relationships
5. **Queries**: Migrate GraphQL to ProximaDB Python SDK or SQL
6. **Modules**: Replace with external embedding/generation

=== Key Benefits After Migration

* **SQL Support**: Query vectors with familiar SQL syntax
* **6 Storage Engines**: Optimize for your specific workload
* **Full Graph Database**: More powerful than Weaviate references
* **No Module Dependencies**: All features built-in
* **Self-Hosted Control**: No vendor lock-in
