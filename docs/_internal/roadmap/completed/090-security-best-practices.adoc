== Overview

This guide covers security best practices for ProximaDB deployments, including authentication, authorization, encryption, network security, and compliance considerations.

=== Security Layers

[source,text]
----
Security Architecture:
├── Network Layer
│   ├── TLS/mTLS encryption
│   ├── Firewall rules
│   └── VPC isolation
├── Authentication Layer
│   ├── API keys
│   ├── JWT tokens
│   └── mTLS certificates
├── Authorization Layer
│   ├── RBAC policies
│   ├── Collection permissions
│   └── Row-level security
├── Data Layer
│   ├── Encryption at rest
│   ├── Field-level encryption
│   └── Data masking
└── Audit Layer
    ├── Access logging
    ├── Query auditing
    └── Compliance reporting
----

== Authentication

=== API Key Authentication

[source,toml]
----
[security.auth]
enabled = true
default_method = "api_key"

# API key settings
api_key_header = "X-API-Key"
api_key_min_length = 32
api_key_hash_algorithm = "argon2"

# Key rotation
key_rotation_enabled = true
key_rotation_interval_days = 90
----

==== Generate API Keys

[source,python]
----
from proximadb_sdk import ProximaDBClient

admin_client = ProximaDBClient(
    url="http://localhost:5678",
    api_key="admin_key_here"
)

# Create API key for application
key_response = admin_client.create_api_key(
    name="production-app",
    permissions=["read", "write"],
    collections=["embeddings", "documents"],
    expires_in_days=90
)

print(f"API Key: {key_response['key']}")
print(f"Key ID: {key_response['id']}")
----

    keyfile="/certs/client.key"
)

    url="https://localhost:5678",
    ssl_context=ssl_context

== Authorization

=== Role-Based Access Control (RBAC)

[source,toml]
----
[security.rbac]
enabled = true
default_role = "reader"

# Built-in roles
[[security.rbac.roles]]
name = "admin"
permissions = ["*"]

[[security.rbac.roles]]
name = "writer"
permissions = ["read", "write", "delete"]

[[security.rbac.roles]]
name = "reader"
permissions = ["read"]

[[security.rbac.roles]]
name = "analyst"
permissions = ["read", "query"]
----

==== Define Custom Roles

[source,python]
----
# Create custom role
admin_client.create_role(
    name="ml_engineer",
    permissions=[
        "collections:read",
        "collections:write",
        "vectors:read",
        "vectors:write",
        "search:execute",
        "index:create"
    ],
    deny=[
        "collections:delete",
        "admin:*"
    ]
)

# Assign role to user
admin_client.assign_role(
    user_id="user_123",
    role="ml_engineer"
)
----

=== Collection-Level Permissions

[source,python]
----
# Grant access to specific collection
admin_client.grant_permission(
    principal="user_123",
    resource="collection:embeddings",
    actions=["read", "write"]
)

# Grant read-only access to all collections
admin_client.grant_permission(
    principal="role:analyst",
    resource="collection:*",
    actions=["read"]
)
----

=== Row-Level Security (Planned)

[source,python]

# Define row-level policy
admin_client.create_policy(

=== Field-Level Encryption

[source,python]
----
from cryptography.fernet import Fernet

# Client-side field encryption
def encrypt_metadata(metadata, key):
    fernet = Fernet(key)
    sensitive_fields = ["email", "phone", "ssn"]

    encrypted = metadata.copy()
    for field in sensitive_fields:
        if field in encrypted:
            encrypted[field] = fernet.encrypt(
                str(encrypted[field]).encode()
            ).decode()

    return encrypted

# Insert with encrypted fields
client.insert_vectors("users", [
    VectorRecord(
        id="user_1",
        vector=embedding,
        metadata=encrypt_metadata({"email": "user@example.com"}, key)
    )
])
----

== Network Security

=== Firewall Configuration

[source,bash]
----
# iptables rules
# Allow ProximaDB ports from specific IPs
iptables -A INPUT -p tcp --dport 5678 -s 10.0.0.0/8 -j ACCEPT
iptables -A INPUT -p tcp --dport 5679 -s 10.0.0.0/8 -j ACCEPT
iptables -A INPUT -p tcp --dport 5678 -j DROP
iptables -A INPUT -p tcp --dport 5679 -j DROP

# Allow health checks from load balancer
iptables -A INPUT -p tcp --dport 5678 -s 10.0.1.0/24 -j ACCEPT

=== Audit Log Analysis

[source,python]
----
# Query audit logs
audit_results = admin_client.query_audit_logs(
    start_time="2024-12-01T00:00:00Z",
    end_time="2024-12-28T23:59:59Z",
    filters={
        "event_type": "data_access",
        "user_id": "user_123"
    },
    limit=1000
)

# Detect anomalies
for entry in audit_results:
    if entry['duration_ms'] > 1000:
        print(f"Slow query: {entry['request_id']}")
    if entry['result'] == 'denied':
        print(f"Access denied: {entry['user_id']}")
----

== Compliance

=== GDPR Compliance

[source,python]
----
# Right to erasure (delete user data)
def gdpr_delete_user(user_id):
    # Find all vectors with user metadata
    vectors = client.execute_sql(f"""
        SELECT id FROM all_collections
        WHERE $.user_id = '{user_id}'
    """)

    # Delete vectors
    for v in vectors:
        client.delete_vector(v['collection'], v['id'])

    # Log deletion for compliance
    audit_log.record_gdpr_deletion(user_id)

# Data export
def gdpr_export_user_data(user_id):
    data = client.execute_sql(f"""
        SELECT * FROM all_collections
        WHERE $.user_id = '{user_id}'
    """)
    return json.dumps(data, indent=2)
----

== Summary

=== Quick Security Setup

[source,bash]
----
# 1. Generate TLS certificates
openssl req -x509 -nodes -days 365 \
  -newkey rsa:4096 \
  -keyout server.key \
  -out server.crt

# 2. Generate encryption key
openssl rand -hex 32 > encryption.key

# 3. Generate API key
openssl rand -hex 32 > api.key

# 4. Start with security enabled
proximadb-server --config production-secure.toml
----
