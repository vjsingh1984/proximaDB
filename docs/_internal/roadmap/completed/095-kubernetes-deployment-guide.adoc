**Deploy ProximaDB on Kubernetes for production-grade scalability and reliability**

== Overview

This guide covers deploying ProximaDB on Kubernetes, including single-node, clustered, and high-availability configurations.

=== Deployment Architecture

[source,text]
----
Kubernetes Cluster:
├── Namespace: proximadb
│   ├── Deployment: proximadb (StatefulSet)
│   │   ├── Pod 0: proximadb-0 (primary)
│   │   ├── Pod 1: proximadb-1 (replica)
│   │   └── Pod 2: proximadb-2 (replica)
│   ├── Service: proximadb-headless (internal)
│   ├── Service: proximadb-lb (external)
│   ├── ConfigMap: proximadb-config
│   ├── Secret: proximadb-secrets
│   ├── PersistentVolumeClaim: data-proximadb-*
│   └── HorizontalPodAutoscaler: proximadb-hpa
├── Ingress Controller
└── Monitoring (Prometheus/Grafana)
----

== Quick Start

=== Helm Installation

[source,bash]
----
# Add ProximaDB Helm repository
helm repo add proximadb https://charts.proximadb.io
helm repo update

# Install with default settings
helm install proximadb proximadb/proximadb \
  --namespace proximadb \
  --create-namespace

# Install with custom values
helm install proximadb proximadb/proximadb \
  --namespace proximadb \
  --create-namespace \
  -f values-production.yaml
----

=== Verify Installation

[source,bash]
----
# Check pods
kubectl get pods -n proximadb

# Check services
kubectl get svc -n proximadb

# View logs
kubectl logs -n proximadb proximadb-0

# Test connection
kubectl port-forward -n proximadb svc/proximadb 5678:5678
curl http://localhost:5678/health
----

=== StatefulSet

[source,yaml]
----
# statefulset.yaml
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: proximadb
  namespace: proximadb
spec:
  serviceName: proximadb-headless
  replicas: 3
  podManagementPolicy: OrderedReady
  selector:
    matchLabels:
      app: proximadb
  template:
    metadata:
      labels:
        app: proximadb
      annotations:
        prometheus.io/scrape: "true"
        prometheus.io/port: "9090"
    spec:
      terminationGracePeriodSeconds: 60
      securityContext:
        runAsUser: 1000
        runAsGroup: 1000
        fsGroup: 1000

      initContainers:
        - name: init-permissions
          image: busybox:1.36
          command: ['sh', '-c', 'chown -R 1000:1000 /data']
          volumeMounts:
            - name: data
              mountPath: /data

      containers:
        - name: proximadb
          image: proximadb/proximadb:0.2.0
          imagePullPolicy: IfNotPresent

          ports:
            - name: rest
              containerPort: 5678
              protocol: TCP
            - name: grpc
              containerPort: 5679
              protocol: TCP
            - name: metrics
              containerPort: 9090
              protocol: TCP

          env:
            - name: RUST_LOG
              value: "info"
            - name: POD_NAME
              valueFrom:
                fieldRef:
                  fieldPath: metadata.name
            - name: POD_NAMESPACE
              valueFrom:
                fieldRef:
                  fieldPath: metadata.namespace
            - name: API_KEY
              valueFrom:
                secretKeyRef:
                  name: proximadb-secrets
                  key: api-key

          resources:
            requests:
              memory: "4Gi"
              cpu: "2"
            limits:
              memory: "16Gi"
              cpu: "8"

          volumeMounts:
            - name: config
              mountPath: /etc/proximadb
            - name: data
              mountPath: /data

          livenessProbe:
            httpGet:
              path: /health
              port: rest
            initialDelaySeconds: 30
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 3

          readinessProbe:
            httpGet:
              path: /health/ready
              port: rest
            initialDelaySeconds: 10
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 3

          startupProbe:
            httpGet:
              path: /health
              port: rest
            initialDelaySeconds: 10
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 30

      volumes:
        - name: config
          configMap:
            name: proximadb-config

      affinity:
        podAntiAffinity:
          preferredDuringSchedulingIgnoredDuringExecution:
            - weight: 100
              podAffinityTerm:
                labelSelector:
                  matchLabels:
                    app: proximadb
                topologyKey: kubernetes.io/hostname

      topologySpreadConstraints:
        - maxSkew: 1
          topologyKey: topology.kubernetes.io/zone
          whenUnsatisfiable: ScheduleAnyway
          labelSelector:
            matchLabels:
              app: proximadb

  volumeClaimTemplates:
    - metadata:
        name: data
      spec:
        accessModes: ["ReadWriteOnce"]
        storageClassName: fast-ssd
        resources:
          requests:
            storage: 100Gi
----

== Auto-Scaling

=== Horizontal Pod Autoscaler

[source,yaml]
----
# hpa.yaml
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: proximadb-hpa
  namespace: proximadb
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: StatefulSet
    name: proximadb
  minReplicas: 3
  maxReplicas: 10
  metrics:
    - type: Resource
      resource:
        name: cpu
        target:
          type: Utilization
          averageUtilization: 70
    - type: Resource
      resource:
        name: memory
        target:
          type: Utilization
          averageUtilization: 80
    - type: Pods
      pods:
        metric:
          name: proximadb_query_rate
        target:
          type: AverageValue
          averageValue: "1000"
  behavior:
    scaleUp:
      stabilizationWindowSeconds: 60
      policies:
        - type: Pods
          value: 2
          periodSeconds: 60
    scaleDown:
      stabilizationWindowSeconds: 300
      policies:
        - type: Pods
          value: 1
          periodSeconds: 120
----

=== PrometheusRule

[source,yaml]
----
# prometheusrule.yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: proximadb-alerts
  namespace: proximadb
spec:
  groups:
    - name: proximadb.rules
      rules:
        - alert: ProximaDBDown
          expr: up{job="proximadb"} == 0
          for: 1m
          labels:
            severity: critical
          annotations:
            summary: "ProximaDB instance down"
            description: "ProximaDB instance {{ $labels.instance }} is down"

        - alert: ProximaDBHighMemory
          expr: container_memory_usage_bytes{container="proximadb"} / container_spec_memory_limit_bytes > 0.9
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "ProximaDB high memory usage"
            description: "Memory usage is above 90%"

        - alert: ProximaDBHighLatency
          expr: histogram_quantile(0.99, rate(proximadb_query_duration_seconds_bucket[5m])) > 1
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "ProximaDB high query latency"
            description: "P99 latency is above 1 second"
----

        - namespaceSelector:
            matchLabels:
              name: ingress-nginx
      ports:
        - protocol: TCP
          port: 5678

    - from:
        - namespaceSelector:
            matchLabels:
              name: application

        - protocol: TCP

          port: 5679

              app: proximadb

    defaultType: "int8"

  enabled: true
  minReplicas: 3

  enabled: true

  enabled: true

  enabled: true

nodeSelector:
  node-type: database

== Operations

=== Scaling

[source,bash]
----
# Manual scaling
kubectl scale statefulset proximadb -n proximadb --replicas=5

# Check scaling status
kubectl get pods -n proximadb -w
----

=== Rolling Updates

[source,bash]
----
# Update image
kubectl set image statefulset/proximadb \
  proximadb=proximadb/proximadb:0.1.6 \
  -n proximadb

# Check rollout status
kubectl rollout status statefulset/proximadb -n proximadb

# Rollback if needed
kubectl rollout undo statefulset/proximadb -n proximadb
----

=== Backup with VolumeSnapshots

[source,yaml]
----
# volumesnapshot.yaml
apiVersion: snapshot.storage.k8s.io/v1
kind: VolumeSnapshot
metadata:
  name: proximadb-backup-20241228
  namespace: proximadb
spec:
  volumeSnapshotClassName: csi-snapclass
  source:
    persistentVolumeClaimName: data-proximadb-0
----

=== Disaster Recovery

[source,bash]
----
# Create backup
kubectl exec -n proximadb proximadb-0 -- \
  proximadb-cli backup create /backup/snapshot-$(date +%Y%m%d)

# Copy backup off-cluster
kubectl cp proximadb/proximadb-0:/backup/snapshot-20241228 \
  ./backup-20241228

# Restore from backup
kubectl exec -n proximadb proximadb-0 -- \
  proximadb-cli backup restore /backup/snapshot-20241228
----
