=== Step 1: Creating Meaningful Embeddings

# tutorial_02_step1.py
from sentence_transformers import SentenceTransformer
from proximadb_sdk import ProximaDBClient

# Load embedding model
model = SentenceTransformer('all-MiniLM-L6-v2')

# Sample texts
texts = [
    "The quick brown fox jumps over the lazy dog",
    "A fast auburn fox leaps above a sleepy canine",
    "Python is a great programming language",
    "Machine learning models require lots of data",
    "The weather is beautiful today"
]

# Generate embeddings
embeddings = model.encode(texts)
print(f"Generated {len(embeddings)} embeddings")

# Setup ProximaDB
client = ProximaDBClient(url="http://localhost:5678")
client.create_collection("text_embeddings", dimension=384)

# Insert with text as metadata
records = [
    {"id": f"text_{i}", "vector": emb.tolist(), "metadata": {"text": text}}
    for i, (text, emb) in enumerate(zip(texts, embeddings))
]
client.insert("text_embeddings", records)

=== Step 2: Filter Queries

[source,python]
----
# tutorial_03_step2.py
from proximadb_sdk import ProximaDBClient
import numpy as np

client = ProximaDBClient(url="http://localhost:5678")
query = np.random.randn(128).tolist()

# Filter 1: Exact match
results = client.search(
    collection="products",
    vector=query,
    top_k=5,
    filter={"must": [{"field": "category", "match": "Electronics"}]}
)
print("Electronics only:")
for r in results:
    print(f"  {r.id}: {r.metadata['category']}, ${r.metadata['price']}")

# Filter 2: Range query
results = client.search(
    collection="products",
    vector=query,
    top_k=5,
    filter={"must": [{"field": "price", "range": {"gte": 50, "lte": 100}}]}
)
print("\nPrice $50-$100:")
for r in results:
    print(f"  {r.id}: ${r.metadata['price']}")

# Filter 3: Combined filters
results = client.search(
    collection="products",
    vector=query,
    top_k=5,
    filter={
        "must": [
            {"field": "category", "match": "Electronics"},
            {"field": "rating", "range": {"gte": 4.0}},
            {"field": "in_stock", "match": True}
        ]
    }
)
print("\nElectronics, 4+ stars, in stock:")
for r in results:
    print(f"  {r.id}: {r.metadata['rating']} stars, ${r.metadata['price']}")

=== Step 1: Document Preparation

[source,python]
----
# tutorial_04_step1.py
from proximadb_sdk import ProximaDBClient
from sentence_transformers import SentenceTransformer
import re

# Sample document
document = """
ProximaDB is a high-performance vector database designed for AI applications.
It supports multiple storage engines optimized for different workloads.

The SST engine uses LSM-tree architecture for write-heavy workloads.
It provides excellent write throughput and efficient compaction.

The HELIX engine uses Hilbert curves for spatial organization.
This enables locality-aware search with progressive refinement.

ProximaDB supports various quantization methods including binary,
INT8 scalar, and product quantization for memory efficiency.
"""

# Chunking function
def chunk_text(text, chunk_size=200, overlap=50):
    sentences = re.split(r'(?<=[.!?])\s+', text.strip())
    chunks = []
    current_chunk = []
    current_length = 0

    for sentence in sentences:
        if current_length + len(sentence) > chunk_size and current_chunk:
            chunks.append(' '.join(current_chunk))
            # Keep last sentence for overlap
            current_chunk = current_chunk[-1:] if overlap > 0 else []
            current_length = len(current_chunk[0]) if current_chunk else 0

        current_chunk.append(sentence)
        current_length += len(sentence)

    if current_chunk:
        chunks.append(' '.join(current_chunk))

    return chunks

chunks = chunk_text(document)
print(f"Created {len(chunks)} chunks:")
for i, chunk in enumerate(chunks):
    print(f"\nChunk {i+1}:\n{chunk}")
----

== Final Project: Complete Search Application

Build a complete semantic search application combining all concepts:

[source,python]
----
# final_project.py
"""
Complete Semantic Search Application
=====================================
This project combines:
- Document ingestion with chunking
- Multiple embedding strategies
- Filtered search
- Performance optimization
"""

from proximadb_sdk import ProximaDBClient
from sentence_transformers import SentenceTransformer
import numpy as np
from typing import List, Dict
import time

class SemanticSearchApp:
    def __init__(self, collection_name: str = "search_app"):
        self.client = ProximaDBClient(url="http://localhost:5678")
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
        self.collection = collection_name
        self._setup_collection()

    def _setup_collection(self):
        try:
            self.client.create_collection(
                name=self.collection,
                dimension=384,
                engine="helix",
                quantization={"type": "int8"},
                index_config={"type": "hnsw", "m": 16, "ef_construction": 200}
            )
        except:
            pass  # Collection exists

    def ingest_documents(self, documents: List[Dict]):
        records = []
        for doc in documents:
            embedding = self.model.encode(doc["content"]).tolist()
            records.append({
                "id": doc["id"],
                "vector": embedding,
                "metadata": {
                    "title": doc.get("title", ""),
                    "content": doc["content"],
                    "category": doc.get("category", "general"),
                    "timestamp": doc.get("timestamp", "")
                }
            })
        self.client.insert(self.collection, records)
        return len(records)

    def search(
        self,
        query: str,
        top_k: int = 10,
        category: str = None
    ) -> List[Dict]:
        query_embedding = self.model.encode(query).tolist()

        filter_query = None
        if category:
            filter_query = {"must": [{"field": "category", "match": category}]}

        results = self.client.search(
            collection=self.collection,
            vector=query_embedding,
            top_k=top_k,
            filter=filter_query,
            include_metadata=True
        )

        return [
            {
                "id": r.id,
                "score": r.score,
                "title": r.metadata.get("title", ""),
                "content": r.metadata.get("content", "")[:200] + "..."
            }
            for r in results
        ]

# Usage
app = SemanticSearchApp()

# Ingest documents
documents = [
    {"id": "1", "title": "Python Guide", "content": "...", "category": "tech"},
    {"id": "2", "title": "ML Tutorial", "content": "...", "category": "tech"},
    # Add more documents
]
app.ingest_documents(documents)

# Search
results = app.search("machine learning basics", top_k=5, category="tech")
for r in results:
    print(f"{r['score']:.3f} - {r['title']}")
----
