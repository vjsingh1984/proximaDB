== Overview

This guide covers migrating from Qdrant to ProximaDB. Both are high-performance vector databases with similar APIs, making migration straightforward. ProximaDB adds graph database, document store, and multi-engine flexibility.

=== Feature Comparison

[cols="2,2,2", options="header"]

| Feature | Qdrant | ProximaDB

== Step 1: Export Data from Qdrant

=== Using Qdrant Client

[source,python]
----
from qdrant_client import QdrantClient
from qdrant_client.models import ScrollRequest
import json

client = QdrantClient("localhost", port=6333)

def export_qdrant_collection(collection_name, output_file):
    """Export all points from a Qdrant collection."""

    # Get collection info
    collection_info = client.get_collection(collection_name)
    print(f"Collection: {collection_name}")
    print(f"Points: {collection_info.points_count}")
    print(f"Vector size: {collection_info.config.params.vectors.size}")

    points = []
    offset = None

    while True:
        # Scroll through all points
        response = client.scroll(
            collection_name=collection_name,
            limit=100,
            offset=offset,
            with_payload=True,
            with_vectors=True
        )

        batch, next_offset = response

        if not batch:
            break

        for point in batch:
            # Handle both single and named vectors
            if isinstance(point.vector, dict):
                # Named vectors
                vectors = point.vector
            else:
                # Single vector
                vectors = {"default": point.vector}

            points.append({
                'id': str(point.id),  # Convert to string
                'vectors': vectors,
                'payload': point.payload or {}
            })

        offset = next_offset
        print(f"Exported {len(points)} points...")

        if next_offset is None:
            break

    # Save to file
    with open(output_file, 'w') as f:
        json.dump({
            'collection_name': collection_name,
            'config': {
                'vector_size': collection_info.config.params.vectors.size,
                'distance': collection_info.config.params.vectors.distance.name
            },
            'points': points
        }, f)

    print(f"Exported {len(points)} points to {output_file}")
    return points

export_qdrant_collection("my_collection", "qdrant_export.json")
----

import json
from concurrent.futures import ThreadPoolExecutor

def export_large_collection(collection_name, output_file, batch_size=1000):
    """Export large collection with parallel processing."""

    collection_info = client.get_collection(collection_name)
    total_points = collection_info.points_count

    with open(output_file, 'w') as f:
        offset = None
        exported = 0

        while True:
            response = client.scroll(
                collection_name=collection_name,
                limit=batch_size,
                offset=offset,
                with_payload=True,
                with_vectors=True
            )

            batch, next_offset = response

            if not batch:
                break

            for point in batch:
                record = {
                    'id': str(point.id),
                    'vector': point.vector if not isinstance(point.vector, dict) else point.vector.get('default', list(point.vector.values())[0]),
                    'payload': point.payload or {}
                }
                f.write(json.dumps(record) + '\n')

            exported += len(batch)
            print(f"Exported {exported}/{total_points} ({100*exported/total_points:.1f}%)")

            offset = next_offset
            if next_offset is None:
                break

    print(f"Complete! Exported {exported} points")

== Step 2: Create ProximaDB Collection

=== Map Qdrant Configuration

[source,python]
----
from proximadb_sdk import ProximaDBClient

client = ProximaDBClient(url="http://localhost:5678")

def create_collection_from_qdrant(export_file):
    """Create ProximaDB collection matching Qdrant config."""

    with open(export_file) as f:
        data = json.load(f)

    config = data['config']

    # Map Qdrant distance to ProximaDB
    distance_mapping = {
        'Cosine': 'cosine',
        'Euclid': 'euclidean',
        'Dot': 'dot_product',
        'Manhattan': 'manhattan'
    }

    collection = client.create_collection(
        name=data['collection_name'],
        dimension=config['vector_size'],
        distance_metric=distance_mapping.get(config['distance'], 'cosine'),
        engine="sst",  # Or choose based on workload
        index_config={
            "type": "hnsw",
            "m": 16,
            "ef_construction": 128
        }
    )

    print(f"Created collection: {collection['name']}")
    return collection

create_collection_from_qdrant("qdrant_export.json")

== Step 3: Import Data to ProximaDB

=== Basic Import

[source,python]
----
import json
from proximadb_sdk import ProximaDBClient, VectorRecord

client = ProximaDBClient(url="http://localhost:5678")

def import_qdrant_export(export_file, collection_name):
    """Import Qdrant export to ProximaDB."""

    with open(export_file) as f:
        data = json.load(f)

    points = data['points']
    batch_size = 1000

    for i in range(0, len(points), batch_size):
        batch = points[i:i+batch_size]

        records = []
        for point in batch:
            # Handle single or named vectors
            if 'vectors' in point:
                vector = point['vectors'].get('default', list(point['vectors'].values())[0])
            else:
                vector = point['vector']

            records.append(VectorRecord(
                id=point['id'],
                vector=vector,
                metadata=point.get('payload', {})
            ))

        client.insert_vectors(collection_name, records)
        print(f"Imported {min(i+batch_size, len(points))}/{len(points)}")

    print("Import complete!")

import_qdrant_export("qdrant_export.json", "my_collection")
----

def import_ndjson_stream(file_path, collection_name, batch_size=5000):
    """Stream import from NDJSON file."""

    batch = []
    total = 0

    with open(file_path) as f:
        for line in f:
            record = json.loads(line)
            batch.append({
                'id': record['id'],
                'vector': record['vector'],
                'metadata': record.get('payload', {})
            })

            if len(batch) >= batch_size:
                client.batch_upsert(collection_name, batch)
                total += len(batch)
                print(f"Imported {total}...")
                batch = []

    if batch:
        client.batch_upsert(collection_name, batch)
        total += len(batch)

    print(f"Complete! Imported {total} records")

import_ndjson_stream("qdrant_export.ndjson", "my_collection")

== Step 5: Migrate Filter Syntax

=== Filter Mapping

[cols="2,2,2", options="header"]
|===
| Qdrant Filter | ProximaDB Filter | Notes
| `MatchValue` | `{"match": value}` | Exact match
| `MatchAny` | `{"in": [...]}` | In list
| `MatchExcept` | `must_not` clause | Exclusion
| `Range(gte, lte)` | `{"range": {"gte": x, "lte": y}}` | Range
| `GeoBoundingBox` | Geo filter | Location
| `IsEmpty` | `{"exists": false}` | Null check
| `HasId` | Direct ID query | Use get by ID
| `must` | `{"must": [...]}` | AND
| `should` | `{"should": [...]}` | OR
| `must_not` | `{"must_not": [...]}` | NOT
|===

=== Filter Conversion Function

[source,python]
----
def convert_qdrant_filter(qdrant_filter):
    """Convert Qdrant filter to ProximaDB format."""

    if qdrant_filter is None:
        return None

    result = {}

    # Handle must conditions
    if hasattr(qdrant_filter, 'must') and qdrant_filter.must:
        result['must'] = [convert_condition(c) for c in qdrant_filter.must]

    # Handle should conditions
    if hasattr(qdrant_filter, 'should') and qdrant_filter.should:
        result['should'] = [convert_condition(c) for c in qdrant_filter.should]

    # Handle must_not conditions
    if hasattr(qdrant_filter, 'must_not') and qdrant_filter.must_not:
        result['must_not'] = [convert_condition(c) for c in qdrant_filter.must_not]

    return result if result else None

def convert_condition(condition):
    """Convert single Qdrant condition to ProximaDB."""

    key = condition.key

    if hasattr(condition, 'match'):
        match = condition.match
        if hasattr(match, 'value'):
            return {"field": key, "match": match.value}
        elif hasattr(match, 'any'):
            return {"field": key, "in": match.any}

    if hasattr(condition, 'range'):
        r = condition.range
        range_dict = {}
        if r.gte is not None:
            range_dict['gte'] = r.gte
        if r.gt is not None:
            range_dict['gt'] = r.gt
        if r.lte is not None:
            range_dict['lte'] = r.lte
        if r.lt is not None:
            range_dict['lt'] = r.lt
        return {"field": key, "range": range_dict}

    if hasattr(condition, 'is_empty'):
        return {"field": key, "exists": not condition.is_empty}

    return {"field": key}
----

# ProximaDB: Fetch vectors, average, then search
positive_ids = ["point_id_1", "point_id_2"]
negative_ids = ["point_id_3"]

# Get positive vectors
positive_vectors = []
for pid in positive_ids:
    result = client.get_vector("my_collection", pid)
    positive_vectors.append(result.vector)

# Get negative vectors
negative_vectors = []
for nid in negative_ids:
    result = client.get_vector("my_collection", nid)
    negative_vectors.append(result.vector)

# Compute recommendation vector
import numpy as np
pos_centroid = np.mean(positive_vectors, axis=0)
neg_centroid = np.mean(negative_vectors, axis=0) if negative_vectors else 0
rec_vector = pos_centroid - 0.5 * neg_centroid

# Search with recommendation vector
results = client.search(
    collection="my_collection",
    vector=rec_vector.tolist(),
    top_k=10,
    filter={"must_not": [{"field": "id", "in": positive_ids + negative_ids}]}
)

== Step 9: Validation

[source,python]
----
from qdrant_client import QdrantClient
from proximadb_sdk import ProximaDBClient
import numpy as np

qdrant = QdrantClient("localhost", port=6333)
proximadb = ProximaDBClient("http://localhost:5678")

def validate_migration(collection_name, sample_count=100):
    """Validate migration by comparing search results."""

    # Get sample points from Qdrant
    points, _ = qdrant.scroll(
        collection_name=collection_name,
        limit=sample_count,
        with_vectors=True
    )

    accurate = 0

    for point in points:
        vector = point.vector
        if isinstance(vector, dict):
            vector = vector.get('default', list(vector.values())[0])

        # Search Qdrant
        q_results = qdrant.search(
            collection_name=collection_name,
            query_vector=vector,
            limit=10
        )
        q_ids = [str(r.id) for r in q_results]

        # Search ProximaDB
        p_results = proximadb.search(collection_name, vector, top_k=10)
        p_ids = [r.id for r in p_results]

        # Compare overlap
        overlap = len(set(q_ids) & set(p_ids))
        if overlap >= 8:
            accurate += 1

    accuracy = accurate / len(points) * 100
    print(f"Validation accuracy: {accuracy:.1f}%")
    return accuracy

validate_migration("my_collection")

== Summary

Migration from Qdrant to ProximaDB is straightforward due to similar concepts:

1. **Export**: Use scroll API or snapshots
2. **Create Collection**: Map configuration directly
3. **Import**: Batch insert with progress tracking
4. **Queries**: Very similar filter syntax
5. **Validation**: Compare search results

=== Key Additions After Migration

* **SQL Support**: Query with familiar SQL + vector functions
* **Graph Database**: Add relationship queries
* **Document Store**: MongoDB-like document API
* **6 Storage Engines**: Optimize per workload
* **Multi-Model Fusion**: Combine vector + graph + document queries
