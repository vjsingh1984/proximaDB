**Configure, manage, and analyze ProximaDB logs for debugging, auditing, and operations**

== Overview

ProximaDB uses structured logging with configurable levels, formats, and outputs. This guide covers configuration, best practices, and log analysis techniques.

=== Logging Architecture

[source,text]
----
Log Flow:
├── Application Layer
│   ├── Request/Response logs
│   ├── Query execution logs
│   └── Error logs
├── Storage Layer
│   ├── WAL operations
│   ├── Compaction logs
│   └── I/O statistics
├── System Layer
│   ├── Startup/shutdown
│   ├── Health checks
│   └── Resource usage
└── Output Targets
    ├── Console (stdout/stderr)
    ├── File (rotating)
    ├── Syslog
    └── External systems (Loki, etc.)
----

== Configuration

=== Basic Configuration

[source,toml]
----
# config.toml
[logging]
# Log level: trace, debug, info, warn, error
level = "info"

# Format: json, pretty, compact
format = "json"

# Output target
output = "file"  # console, file, syslog

# File output settings
file_path = "/var/log/proximadb/proximadb.log"
max_size_mb = 100
max_files = 10
compress_rotated = true
----

=== Environment Variable Configuration

[source,bash]
----
# Set log level via RUST_LOG
export RUST_LOG=info

# Component-specific logging
export RUST_LOG=info,proximadb::storage=debug,proximadb::query=trace

# Full debug mode
export RUST_LOG=debug

# Trace specific module
export RUST_LOG=info,proximadb::storage::engines::impls::sst=trace
----

=== Advanced Configuration

[source,toml]
----
# config.toml
[logging]
level = "info"
format = "json"
output = "file"

# File rotation
file_path = "/var/log/proximadb/proximadb.log"
max_size_mb = 100
max_files = 10
compress_rotated = true
rotation_schedule = "daily"  # daily, hourly, size

# Filtering
include_modules = ["proximadb::query", "proximadb::storage"]
exclude_modules = ["proximadb::network::health"]

# Sampling (for high-volume logs)
sample_rate = 0.1  # Log 10% of debug messages
sample_modules = ["proximadb::storage::cache"]

# Request logging
log_requests = true
log_request_body = false
log_response_body = false
log_headers = ["X-Request-ID", "X-API-Key"]

# Performance logging
log_slow_queries = true
slow_query_threshold_ms = 100
----

== Log Levels

=== Level Descriptions

[cols="1,2,3", options="header"]
|===
| Level | When to Use | Example
| `trace` | Detailed debugging | Every function call, variable values
| `debug` | Development debugging | Query plans, intermediate results
| `info` | Normal operations | Request completion, startup events
| `warn` | Potential issues | High memory, slow queries
| `error` | Failures | Query failures, connection errors
|===

=== Level Configuration Examples

[source,bash]
----
# Production (minimal logging)
RUST_LOG=warn cargo run --release

# Development (verbose logging)
RUST_LOG=debug cargo run

# Debugging specific component
RUST_LOG=info,proximadb::storage::engines::impls::helix=trace cargo run

# Performance investigation
RUST_LOG=info,proximadb::query=debug,proximadb::storage::cache=debug cargo run

# Full trace (very verbose)
RUST_LOG=trace cargo run
----

== Log Formats

=== JSON Format (Recommended for Production)

[source,json]
----
{
  "timestamp": "2024-12-28T10:30:45.123456Z",
  "level": "INFO",
  "target": "proximadb::query",
  "message": "Query completed",
  "fields": {
    "collection": "embeddings",
    "query_type": "vector",
    "duration_ms": 12.5,
    "results_count": 10,
    "request_id": "req_abc123"
  },
  "span": {
    "name": "search",
    "collection": "embeddings"
  }
}
----

=== Pretty Format (Development)

[source,text]
----
2024-12-28T10:30:45.123Z  INFO proximadb::query: Query completed
    collection: embeddings
    query_type: vector
    duration_ms: 12.5
    results_count: 10
    request_id: req_abc123
    in search{collection=embeddings}
----

=== Compact Format

[source,text]
----
2024-12-28T10:30:45.123Z INFO  [proximadb::query] Query completed collection=embeddings duration_ms=12.5
----

== Log Rotation

=== Logrotate Configuration

[source,text]
----
# /etc/logrotate.d/proximadb
/var/log/proximadb/*.log {
    daily
    rotate 14
    compress
    delaycompress
    missingok
    notifempty
    create 0640 proximadb proximadb
    sharedscripts
    postrotate
        # Signal ProximaDB to reopen log files
        kill -USR1 $(cat /var/run/proximadb.pid)
    endscript
}
----

=== Built-in Rotation

[source,toml]
----
# config.toml
[logging]

----

== Log Analysis

=== Common Log Queries

[source,bash]
----
# Find errors
grep '"level":"ERROR"' /var/log/proximadb/proximadb.log | jq .

# Find slow queries (>100ms)
cat /var/log/proximadb/proximadb.log | jq 'select(.fields.duration_ms > 100)'

# Count queries by collection
cat /var/log/proximadb/proximadb.log | \
  jq -r 'select(.message == "Query completed") | .fields.collection' | \
  sort | uniq -c | sort -rn

# Find requests by ID
grep '"request_id":"req_abc123"' /var/log/proximadb/proximadb.log | jq .

# Error distribution
cat /var/log/proximadb/proximadb.log | \
  jq -r 'select(.level == "ERROR") | .fields.error_code' | \
  sort | uniq -c | sort -rn

# Query latency percentiles
cat /var/log/proximadb/proximadb.log | \
  jq 'select(.message == "Query completed") | .fields.duration_ms' | \
  sort -n | awk '{a[NR]=$1} END {print "P50:", a[int(NR*0.5)], "P99:", a[int(NR*0.99)]}'
----

=== Loki LogQL Queries

[source,logql]
----
# Error logs
{job="proximadb"} |= "ERROR"

# Slow queries
{job="proximadb"} | json | duration_ms > 100

# Errors by collection
sum by (collection) (count_over_time({job="proximadb"} | json | level="ERROR" [5m]))

# Request rate
rate({job="proximadb"} | json | message="Request completed" [5m])

# P99 latency
quantile_over_time(0.99, {job="proximadb"} | json | unwrap duration_ms [5m])

# Top error messages
topk(10, sum by (error_message) (count_over_time({job="proximadb"} |= "ERROR" [1h])))
----

== Audit Logging

=== Configuration

[source,toml]
----
# config.toml
[logging.audit]
enabled = true
output = "file"
file_path = "/var/log/proximadb/audit.log"
max_size_mb = 500
max_files = 30

# What to audit
log_authentication = true
log_authorization = true
log_data_access = true
log_admin_operations = true
log_schema_changes = true

# Sensitive data handling
mask_api_keys = true
mask_vectors = true
include_request_body = false

== Troubleshooting with Logs

=== Common Debug Scenarios

[source,bash]
----
# Debug slow queries
RUST_LOG=info,proximadb::query=debug,proximadb::storage::cache=debug cargo run

# Debug connection issues
RUST_LOG=info,proximadb::network=debug cargo run

# Debug storage operations
RUST_LOG=info,proximadb::storage=debug cargo run

# Debug WAL recovery
RUST_LOG=info,proximadb::storage::persistence::wal=trace cargo run

# Debug index building
RUST_LOG=info,proximadb::index=debug cargo run

# Full system trace (very verbose)
RUST_LOG=trace cargo run 2>&1 | tee /tmp/proximadb-trace.log
----

=== Log Sampling for High-Volume Debugging

[source,toml]
----
# config.toml - Production debugging without log explosion
[logging]
level = "debug"

# Sample verbose logs
[logging.sampling]
enabled = true
rules = [
    { module = "proximadb::storage::cache", rate = 0.01 },    # 1% of cache logs
    { module = "proximadb::query::planner", rate = 0.1 },     # 10% of planner logs
    { level = "trace", rate = 0.001 }                          # 0.1% of trace logs
]
----
