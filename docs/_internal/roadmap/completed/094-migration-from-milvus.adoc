== Overview

This guide covers migrating from Milvus to ProximaDB. Both are high-performance vector databases supporting multiple index types and distributed deployment. ProximaDB adds unified multi-model capabilities and simpler deployment.

=== Feature Comparison

[cols="2,2,2", options="header"]
|===
| Feature | Milvus | ProximaDB
| Vector Search | ✅ | ✅
| Schema Definition | Required | Optional
| Index Types | IVF, HNSW, ANNOY, etc. | HNSW, IVF, ANNOY, LSH
| Partitions | ✅ | ✅ (Collections)
| Scalar Filtering | ✅ | ✅ (Enhanced)
| Graph Database | ❌ | ✅
| Document Store | ❌ | ✅
| SQL Support | Limited | Full SQL
| Storage Engines | 1 | 6 engines
| Deployment | Complex (etcd, MinIO) | Simple (single binary)
|===

== Step 1: Export Data from Milvus

=== Using PyMilvus Client

[source,python]
----
from pymilvus import connections, Collection, utility
import json

# Connect to Milvus
connections.connect("default", host="localhost", port="19530")

def export_milvus_collection(collection_name, output_file):
    """Export all entities from a Milvus collection."""

    collection = Collection(collection_name)
    collection.load()

    # Get collection schema
    schema = collection.schema
    print(f"Collection: {collection_name}")
    print(f"Fields: {[f.name for f in schema.fields]}")

    # Get primary key field
    pk_field = None
    vector_field = None
    scalar_fields = []

    for field in schema.fields:
        if field.is_primary:
            pk_field = field.name
        if field.dtype in [100, 101]:  # FloatVector, BinaryVector
            vector_field = field.name
        else:
            scalar_fields.append(field.name)

    print(f"Primary key: {pk_field}")
    print(f"Vector field: {vector_field}")

    # Query all entities
    entities = []
    batch_size = 1000
    offset = 0

    while True:
        # Query with pagination
        results = collection.query(
            expr=f"{pk_field} >= 0",  # Get all
            output_fields=["*"],
            limit=batch_size,
            offset=offset
        )

        if not results:
            break

        for entity in results:
            record = {
                'id': str(entity[pk_field]),
                'vector': entity[vector_field],
                'metadata': {
                    k: entity[k] for k in scalar_fields
                    if k in entity and k != pk_field
                }
            }
            entities.append(record)

        offset += len(results)
        print(f"Exported {len(entities)} entities...")

        if len(results) < batch_size:
            break

    # Save to file
    with open(output_file, 'w') as f:
        json.dump({
            'collection_name': collection_name,
            'schema': {
                'vector_field': vector_field,
                'dimension': next(f.params['dim'] for f in schema.fields if f.name == vector_field),
                'pk_field': pk_field,
                'scalar_fields': scalar_fields
            },
            'entities': entities
        }, f)

    print(f"Exported {len(entities)} entities to {output_file}")
    return entities

export_milvus_collection("my_collection", "milvus_export.json")
----

def export_with_partitions(collection_name, output_dir):
    """Export collection with partition information."""

    collection = Collection(collection_name)
    collection.load()

    partitions = collection.partitions
    print(f"Partitions: {[p.name for p in partitions]}")

    all_entities = []

    for partition in partitions:
        partition_name = partition.name

        # Query partition
        results = collection.query(
            expr=f"_partition == '{partition_name}'",
            output_fields=["*"],
            partition_names=[partition_name]
        )

        for entity in results:
            entity['_partition'] = partition_name
            all_entities.append(entity)

        print(f"Exported {len(results)} from partition '{partition_name}'")

    # Save all with partition info
    output_file = f"{output_dir}/{collection_name}_all.json"
    with open(output_file, 'w') as f:
        json.dump(all_entities, f)

    return all_entities

== Step 2: Create ProximaDB Collections

=== Map Milvus Schema

[source,python]

from proximadb_sdk import ProximaDBClient

client = ProximaDBClient(url="http://localhost:5678")

def create_collection_from_milvus(export_file):
    """Create ProximaDB collection from Milvus export."""

    with open(export_file) as f:
        data = json.load(f)

    schema = data['schema']
    collection_name = data['collection_name']

    # Map Milvus metric type to ProximaDB
    # Milvus uses L2, IP (inner product), COSINE
    metric_mapping = {
        'L2': 'euclidean',
        'IP': 'dot_product',
        'COSINE': 'cosine'
    }

    collection = client.create_collection(
        name=collection_name.lower(),
        dimension=schema['dimension'],
        distance_metric='cosine',  # Default, adjust as needed
        engine="sst",
        index_config={
            "type": "hnsw",
            "m": 16,
            "ef_construction": 200
        }
    )

    print(f"Created collection: {collection['name']}")
    return collection

create_collection_from_milvus("milvus_export.json")

== Step 3: Import Data to ProximaDB

=== Basic Import

[source,python]
----
import json
from proximadb_sdk import ProximaDBClient, VectorRecord

client = ProximaDBClient(url="http://localhost:5678")

def import_milvus_export(export_file, collection_name):
    """Import Milvus export to ProximaDB."""

    with open(export_file) as f:
        data = json.load(f)

    entities = data['entities']
    batch_size = 1000

    for i in range(0, len(entities), batch_size):
        batch = entities[i:i+batch_size]

        records = []
        for entity in batch:
            records.append(VectorRecord(
                id=entity['id'],
                vector=entity['vector'],
                metadata=entity.get('metadata', {})
            ))

        client.insert_vectors(collection_name, records)
        print(f"Imported {min(i+batch_size, len(entities))}/{len(entities)}")

    print("Import complete!")

import_milvus_export("milvus_export.json", "my_collection")

def import_with_partitions(export_file, collection_prefix):
    """Import Milvus data, creating collection per partition."""

    with open(export_file) as f:
        entities = json.load(f)

    # Group by partition
    partitions = {}
    for entity in entities:
        partition = entity.get('_partition', 'default')
        if partition not in partitions:
            partitions[partition] = []
        partitions[partition].append(entity)

    for partition_name, partition_entities in partitions.items():
        collection_name = f"{collection_prefix}_{partition_name}"

        # Create collection
        dimension = len(partition_entities[0]['vector'])
        client.create_collection(
            name=collection_name,
            dimension=dimension,
            engine="sst"
        )

        # Import entities
        records = [
            VectorRecord(
                id=e['id'],
                vector=e['vector'],
                metadata={k: v for k, v in e.items()
                         if k not in ['id', 'vector', '_partition']}
            )
            for e in partition_entities
        ]

        client.batch_upsert(collection_name, records)
        print(f"Imported {len(records)} to {collection_name}")

== Step 5: Migrate Expression Filters

=== Expression Syntax Mapping

[cols="2,2,2", options="header"]
|===
| Milvus Expression | ProximaDB Filter/SQL | Notes
| `field == "value"` | `{"field": "x", "match": "val"}` | Equality
| `field != "value"` | `must_not` clause | Not equal
| `field > 10` | `{"range": {"gt": 10}}` | Greater than
| `field >= 10` | `{"range": {"gte": 10}}` | Greater or equal
| `field < 10` | `{"range": {"lt": 10}}` | Less than
| `field <= 10` | `{"range": {"lte": 10}}` | Less or equal
| `field in [1,2,3]` | `{"in": [1,2,3]}` | In list
| `field like "val%"` | Text search | Pattern match
| `expr1 and expr2` | `{"must": [...]}` | AND
| `expr1 or expr2` | `{"should": [...]}` | OR
| `not expr` | `{"must_not": [...]}` | NOT
|===

=== Expression Conversion

[source,python]
----
import re

def convert_milvus_expr(expr):
    """Convert Milvus expression to ProximaDB filter."""

    if not expr:
        return None

    # Handle AND/OR
    if ' and ' in expr.lower():
        parts = re.split(r'\s+and\s+', expr, flags=re.IGNORECASE)
        return {"must": [convert_milvus_expr(p) for p in parts]}

    if ' or ' in expr.lower():
        parts = re.split(r'\s+or\s+', expr, flags=re.IGNORECASE)
        return {"should": [convert_milvus_expr(p) for p in parts]}

    # Handle comparison operators
    patterns = [
        (r"(\w+)\s*==\s*['\"]?([^'\"]+)['\"]?", lambda m: {"field": m.group(1), "match": m.group(2).strip()}),
        (r"(\w+)\s*!=\s*['\"]?([^'\"]+)['\"]?", lambda m: {"field": m.group(1), "match": m.group(2).strip(), "_not": True}),
        (r"(\w+)\s*>=\s*(\d+\.?\d*)", lambda m: {"field": m.group(1), "range": {"gte": float(m.group(2))}}),
        (r"(\w+)\s*<=\s*(\d+\.?\d*)", lambda m: {"field": m.group(1), "range": {"lte": float(m.group(2))}}),
        (r"(\w+)\s*>\s*(\d+\.?\d*)", lambda m: {"field": m.group(1), "range": {"gt": float(m.group(2))}}),
        (r"(\w+)\s*<\s*(\d+\.?\d*)", lambda m: {"field": m.group(1), "range": {"lt": float(m.group(2))}}),
        (r"(\w+)\s+in\s+\[([^\]]+)\]", lambda m: {"field": m.group(1), "in": [x.strip().strip("'\"") for x in m.group(2).split(",")]}),
    ]

    for pattern, handler in patterns:
        match = re.match(pattern, expr.strip())
        if match:
            return handler(match)

    return None

# Example usage
milvus_expr = "category == 'tech' and price < 100"
proximadb_filter = convert_milvus_expr(milvus_expr)
print(proximadb_filter)
# Output: {"must": [{"field": "category", "match": "tech"}, {"field": "price", "range": {"lt": 100}}]}

== Step 8: Validation

[source,python]
----
from pymilvus import Collection, connections
from proximadb_sdk import ProximaDBClient
import numpy as np

connections.connect("default", host="localhost", port="19530")
milvus_collection = Collection("my_collection")
milvus_collection.load()

proximadb = ProximaDBClient("http://localhost:5678")

def validate_migration(collection_name, sample_count=100):
    """Validate by comparing search results."""

    # Get sample vectors from Milvus
    milvus_results = milvus_collection.query(
        expr="id >= 0",
        output_fields=["*"],
        limit=sample_count
    )

    accurate = 0

    for entity in milvus_results:
        vector = entity['embedding']

        # Search Milvus
        m_results = milvus_collection.search(
            data=[vector],
            anns_field="embedding",
            param={"metric_type": "L2", "params": {"nprobe": 10}},
            limit=10
        )
        m_ids = [str(hit.id) for hit in m_results[0]]

        # Search ProximaDB
        p_results = proximadb.search(collection_name, vector, top_k=10)
        p_ids = [r.id for r in p_results]

        # Compare
        overlap = len(set(m_ids) & set(p_ids))
        if overlap >= 8:
            accurate += 1

    accuracy = accurate / len(milvus_results) * 100
    print(f"Validation accuracy: {accuracy:.1f}%")
    return accuracy

validate_migration("my_collection")

== Summary

Migration from Milvus to ProximaDB involves:

1. **Export**: Query API with pagination
2. **Schema**: Optional in ProximaDB (flexible JSON)
3. **Partitions**: Map to collections or metadata
4. **Import**: Batch insert with type conversion
5. **Queries**: Convert expressions to filters or SQL
6. **Index**: Configure at collection creation

=== Key Benefits After Migration

* **Simpler Deployment**: Single binary vs multi-component
* **SQL Support**: Full SQL with vector functions
* **Graph Database**: Native relationship queries
* **Document Store**: MongoDB-like API
* **6 Storage Engines**: Workload-specific optimization
* **Multi-Model Fusion**: Vector + graph + document in one query
