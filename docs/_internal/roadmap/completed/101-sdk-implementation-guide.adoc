== Overview

== API Contracts

=== Core Operations

Every SDK must implement these core operations:

[source,text]
----
Collections:
  - create_collection(name, dimension, config?) -> Collection
  - get_collection(name) -> Collection
  - list_collections() -> List[Collection]
  - delete_collection(name) -> void
  - update_collection(name, config) -> Collection

Vectors:
  - insert(collection, vectors) -> InsertResult
  - upsert(collection, vectors) -> UpsertResult
  - update(collection, id, vector?, metadata?) -> void
  - delete(collection, ids) -> void
  - get(collection, ids) -> List[Vector]

Search:
  - search(collection, vector, top_k, filter?, params?) -> SearchResults
  - batch_search(collection, vectors, top_k, params?) -> List[SearchResults]

SQL:
  - execute_sql(query, params?) -> QueryResult

Graph:
  - create_graph(name, config?) -> Graph
  - create_node(graph, node) -> Node
  - create_edge(graph, edge) -> Edge
  - traverse(graph, start, query) -> TraversalResult
----

# Error hierarchy
class ProximaDBError(Exception):
    """Base exception for all ProximaDB errors."""
    def __init__(self, message: str, code: str = None, details: dict = None):
        super().__init__(message)
        self.code = code
        self.details = details or {}

class ConnectionError(ProximaDBError):
    """Failed to connect to ProximaDB server."""
    pass

class AuthenticationError(ProximaDBError):
    """Authentication failed."""
    pass

class NotFoundError(ProximaDBError):
    """Resource not found."""
    pass

class ValidationError(ProximaDBError):
    """Request validation failed."""
    pass

class RateLimitError(ProximaDBError):
    """Rate limit exceeded."""
    def __init__(self, message: str, retry_after: float = None):
        super().__init__(message, code="RATE_LIMITED")
        self.retry_after = retry_after

class ServerError(ProximaDBError):
    """Internal server error."""
    pass

# Error mapping from HTTP status codes
ERROR_MAP = {
    400: ValidationError,
    401: AuthenticationError,
    403: AuthenticationError,
    404: NotFoundError,
    429: RateLimitError,
    500: ServerError,
    502: ConnectionError,
    503: ServerError,
    504: ConnectionError,
}

def handle_response(response):
    """Convert HTTP response to appropriate exception."""
    if response.status_code >= 400:
        error_class = ERROR_MAP.get(response.status_code, ProximaDBError)
        try:
            body = response.json()
            raise error_class(
                message=body.get("error", response.reason),
                code=body.get("code"),
                details=body.get("details", {})
            )
        except json.JSONDecodeError:
            raise error_class(response.reason)
    return response

=== Retry Logic

[source,python]
----
import time
import random
from functools import wraps

def with_retry(
    max_retries: int = 3,
    base_delay: float = 0.1,
    max_delay: float = 10.0,
    exponential_base: float = 2,
    jitter: bool = True,
    retryable_errors: tuple = (ConnectionError, ServerError, RateLimitError)
):
    """Decorator for retry logic with exponential backoff."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_error = None

            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except retryable_errors as e:
                    last_error = e

                    if attempt == max_retries:
                        raise

                    # Calculate delay
                    if isinstance(e, RateLimitError) and e.retry_after:
                        delay = e.retry_after
                    else:
                        delay = min(
                            base_delay * (exponential_base ** attempt),
                            max_delay
                        )

                    # Add jitter
                    if jitter:
                        delay = delay * (0.5 + random.random())

                    time.sleep(delay)

            raise last_error

        return wrapper
    return decorator

# Usage
class ProximaDBClient:
    @with_retry(max_retries=3)
    def search(self, collection: str, vector: list, top_k: int = 10):
        return self._transport.post(
            f"/api/v1/collections/{collection}/search",
            json={"vector": vector, "top_k": top_k}
        )

=== Async Support

[source,python]
----
import asyncio
import aiohttp
from typing import List, Optional

class AsyncProximaDBClient:
    """Async client for ProximaDB."""

    def __init__(
        self,
        url: str = "http://localhost:5678",
        api_key: Optional[str] = None,
        timeout: float = 30.0
    ):
        self.url = url.rstrip("/")
        self.api_key = api_key
        self.timeout = aiohttp.ClientTimeout(total=timeout)
        self._session: Optional[aiohttp.ClientSession] = None

    async def _get_session(self) -> aiohttp.ClientSession:
        if self._session is None or self._session.closed:
            headers = {}
            if self.api_key:
                headers["X-API-Key"] = self.api_key
            self._session = aiohttp.ClientSession(
                timeout=self.timeout,
                headers=headers
            )
        return self._session

    async def search(
        self,
        collection: str,
        vector: List[float],
        top_k: int = 10,
        **kwargs
    ) -> List[dict]:
        session = await self._get_session()
        async with session.post(
            f"{self.url}/api/v1/collections/{collection}/search",
            json={"vector": vector, "top_k": top_k, **kwargs}
        ) as response:
            response.raise_for_status()
            data = await response.json()
            return data["results"]

    async def batch_search(
        self,
        collection: str,
        vectors: List[List[float]],
        top_k: int = 10
    ) -> List[List[dict]]:
        """Concurrent batch search."""
        tasks = [
            self.search(collection, vector, top_k)
            for vector in vectors
        ]
        return await asyncio.gather(*tasks)

    async def close(self):
        if self._session:
            await self._session.close()

    async def __aenter__(self):
        return self

    async def __aexit__(self, *args):
        await self.close()

# Usage
async def main():
    async with AsyncProximaDBClient(api_key="your-key") as client:
        results = await client.search("my_collection", [0.1, 0.2, ...], top_k=10)
        print(results)

asyncio.run(main())

=== Batch Operations

[source,python]
----
from typing import List, Iterator
import math

def batch_iterator(items: List, batch_size: int) -> Iterator[List]:
    """Split items into batches."""
    for i in range(0, len(items), batch_size):
        yield items[i:i + batch_size]

class BatchOperations:
    """Mixin for batch operations."""

    def insert_batch(
        self,
        collection: str,
        vectors: List[VectorRecord],
        batch_size: int = 1000,
        parallel: bool = False,
        on_progress: callable = None
    ) -> InsertResult:
        """Insert vectors in batches."""
        total = len(vectors)
        inserted = 0
        errors = []

        for i, batch in enumerate(batch_iterator(vectors, batch_size)):
            try:
                result = self.insert(collection, batch)
                inserted += result.inserted_count
            except Exception as e:
                errors.append({"batch": i, "error": str(e)})

            if on_progress:
                on_progress(inserted, total)

        return InsertResult(
            inserted_count=inserted,
            errors=errors
        )

    def search_batch(
        self,
        collection: str,
        vectors: List[List[float]],
        top_k: int = 10,
        batch_size: int = 100
    ) -> List[SearchResults]:
        """Search multiple vectors in batches."""
        all_results = []

        for batch in batch_iterator(vectors, batch_size):
            results = self._transport.post(
                f"/api/v1/collections/{collection}/search/batch",
                json={"vectors": batch, "top_k": top_k}
            )
            all_results.extend(results["results"])

        return all_results

# Iterator protocol for large results
class SearchResultIterator:
    def __init__(self, client, collection, vector, top_k, batch_size=100):
        self.client = client
        self.collection = collection
        self.vector = vector
        self.top_k = top_k
        self.batch_size = batch_size
        self.offset = 0

    def __iter__(self):
        return self

    def __next__(self):
        if self.offset >= self.top_k:
            raise StopIteration

        results = self.client.search(
            self.collection,
            self.vector,
            top_k=min(self.batch_size, self.top_k - self.offset),
            offset=self.offset
        )

        if not results:
            raise StopIteration

        self.offset += len(results)
        return results

// Functional options pattern
type ClientOption func(*Client)

func WithAPIKey(key string) ClientOption {
    return func(c *Client) {
        c.apiKey = key
    }
}

func WithTimeout(timeout time.Duration) ClientOption {
    return func(c *Client) {
        c.timeout = timeout
    }
}

func WithRetries(retries int) ClientOption {
    return func(c *Client) {
        c.maxRetries = retries
    }
}

func NewClient(url string, opts ...ClientOption) *Client {
    c := &Client{
        url:        url,
        timeout:    30 * time.Second,
        maxRetries: 3,
    }

    for _, opt := range opts {
        opt(c)
    }

    return c
}
