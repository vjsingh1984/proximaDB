== Overview

RAPTOR is ProximaDB's **adaptive storage engine** using a unique Matrix Trinity architecture that dynamically switches between row-oriented, column-oriented, and hybrid storage based on workload patterns.

=== Key Characteristics

[cols="2,3", options="header"]
|===
| Metric | Value
| Query Latency | ~9.36ms (adaptive)
| Storage Modes | 3 (Row, Column, Hybrid)
| Workload Detection | Automatic
| Compression | Adaptive per-block
| Best For | Dynamic workloads, mixed access patterns
|===

== Architecture

=== Matrix Trinity Design

[source,text]
----
Matrix Trinity Architecture:
├── Row-Oriented Mode
│   ├── Full record storage
│   ├── Optimal for point lookups

│
├── Column-Oriented Mode
│   ├── Columnar vector storage
│   ├── Optimal for analytics
│   └── SIMD-friendly scans
│
└── Hybrid Mode
    ├── Hot columns in row format

    └── Dynamic column placement

    query_history: RingBuffer<QueryPattern>,

    point_query_ratio: f32,

pub enum RecommendedMode {
    Row,        // >70% point queries
    Column,     // >70% scan queries
    Hybrid,     // Mixed workload
}

impl WorkloadDetector {
    pub fn analyze(&self) -> RecommendedMode {
        if self.point_query_ratio > 0.7 {
            RecommendedMode::Row
        } else if self.scan_query_ratio > 0.7 {
            RecommendedMode::Column
        } else {
            RecommendedMode::Hybrid
        }
    }
}

=== Mode Switching

[source,rust]
----
pub struct ModeSwitcher {
    current_mode: StorageMode,
    transition_threshold: f32,      // 0.3 = 30% workload change
    cooldown_ms: u64,               // Prevent thrashing
    last_switch: Instant,
}

impl ModeSwitcher {
    pub async fn maybe_switch(&mut self, recommended: RecommendedMode) -> Result<()> {
        // Check cooldown
        if self.last_switch.elapsed().as_millis() < self.cooldown_ms as u128 {
            return Ok(());
        }

        // Check if significant change
        if self.should_transition(recommended) {
            self.transition_to(recommended).await?;
        }

        Ok(())
    }
}

== Search Pipeline

=== Adaptive Search

[source,rust]
----
impl RaptorEngine {
    pub async fn search(&self, query: &SearchQuery) -> Result<Vec<SearchResult>> {
        match self.current_mode {
            StorageMode::Row => {
                // Use B-tree index + bloom filters
                self.row_search(query).await
            }
            StorageMode::Column => {
                // Use columnar scan with SIMD
                self.column_search(query).await
            }
            StorageMode::Hybrid => {
                // Parallel search both stores, merge
                let (row_results, col_results) = tokio::join!(
                    self.hot_rows.search(query),
                    self.cold_columns.search(query)
                );
                self.merge_results(row_results?, col_results?)
            }
        }
    }
}

[source,rust]

pub fn select_encoding(column: &ColumnChunk) -> Encoding {
    let unique_ratio = column.unique_values() as f32 / column.len() as f32;
    let sorted_ratio = column.sorted_runs() as f32 / column.len() as f32;

    if unique_ratio < 0.1 {
        Encoding::Dictionary  // Low cardinality
    } else if sorted_ratio > 0.8 {
        Encoding::Delta       // Nearly sorted
    } else if column.has_runs() {
        Encoding::RLE         // Repeated values
    } else {
        Encoding::Plain       // Default
    }
}
