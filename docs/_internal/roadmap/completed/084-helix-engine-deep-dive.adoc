== Overview

HELIX is ProximaDB's **locality-optimized storage engine** using Hilbert space-filling curves to preserve vector proximity in storage layout. Combined with PCA dimensionality reduction, it achieves excellent cache efficiency for spatially clustered queries.

== Hilbert Curve Implementation

=== Hilbert Code Computation

[source,rust]
----
pub struct HilbertCurve {
    dimensions: usize,
    bits_per_dimension: u32,      // Precision
    order: u32,                    // 2^order cells per dimension
}

impl HilbertCurve {
    pub fn point_to_hilbert(&self, point: &[f32]) -> u64 {
        // 1. Normalize to [0, 2^bits) range
        let normalized = self.normalize_point(point);

        // 2. Compute Hilbert index using Gray code
        self.coordinates_to_hilbert(&normalized)
    }

    pub fn hilbert_to_point(&self, hilbert: u64) -> Vec<f32> {
        // Reverse mapping for range queries
        self.hilbert_to_coordinates(hilbert)
    }
}

== PCA Optimization

=== Adaptive Dimensionality Reduction

[source,rust]

pub struct PcaManager {
    // Per-collection PCA models
    models: HashMap<String, PcaModel>,

    // Configuration
    config: PcaConfig,
}

pub struct PcaConfig {
    pub enabled: bool,
    pub target_variance: f32,       // 0.95 = keep 95% variance
    pub min_dimensions: usize,       // Don't reduce below this
    pub max_dimensions: usize,       // Cap reduction
    pub training_samples: usize,     // Samples for model training
}

impl PcaManager {
    pub async fn train(&mut self, vectors: &[Vec<f32>]) -> Result<PcaModel> {
        // 1. Compute covariance matrix
        let cov = self.compute_covariance(vectors);

        // 2. Eigendecomposition
        let (eigenvalues, eigenvectors) = self.eigen_decompose(&cov);

        // 3. Select components for target variance
        let k = self.select_components(&eigenvalues, self.config.target_variance);

        Ok(PcaModel {
            components: eigenvectors[..k].to_vec(),
            mean: self.compute_mean(vectors),
            explained_variance: eigenvalues[..k].to_vec(),
        })
    }
}

== Progressive Search

=== Spatial Search Strategy

[source,rust]
----
pub struct ProgressiveSearch {
    // Search state
    current_hilbert: u64,
    search_radius: u64,
    results: BinaryHeap<SearchResult>,

    // Early termination
    distance_threshold: f32,
    blocks_searched: usize,
}

impl ProgressiveSearch {
    pub async fn search(&mut self, query: &[f32], k: usize) -> Vec<SearchResult> {
        // Start from query's Hilbert position
        self.current_hilbert = self.hilbert.point_to_hilbert(query);

        loop {
            // 1. Search blocks in current Hilbert range
            let blocks = self.get_blocks_in_range(
                self.current_hilbert - self.search_radius,
                self.current_hilbert + self.search_radius,
            );

            for block in blocks {
                self.search_block(block, query);
            }

            // 2. Check early termination
            if self.can_terminate(k) {
                break;
            }

            // 3. Expand search radius
            self.search_radius *= 2;
        }

        self.results.into_sorted_vec()
    }
}

pub struct ClusterManager {
    clusters: Vec<Cluster>,
    centroids: Vec<Vec<f32>>,

pub struct Cluster {
    pub id: usize,
    pub centroid: Vec<f32>,
    pub hilbert_range: (u64, u64),
    pub vector_count: usize,
    pub bounds: BoundingBox,
}

impl ClusterManager {
    pub fn assign_cluster(&self, vector: &[f32]) -> usize {
        // Find nearest centroid
        self.centroids.iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| {
                distance(vector, a).partial_cmp(&distance(vector, b)).unwrap()
            })
            .map(|(i, _)| i)
            .unwrap()
    }
}

== Write Path

=== Flush with Spatial Organization

[source,rust]
----
impl HelixEngine {
    pub async fn flush(&self, vectors: Vec<VectorRecord>) -> Result<()> {
        // 1. Apply PCA transformation
        let transformed = if let Some(pca) = &self.pca_model {
            vectors.iter()
                .map(|v| pca.transform(&v.vector))
                .collect()
        } else {
            vectors.iter().map(|v| v.vector.clone()).collect()
        };

        // 2. Compute Hilbert codes
        let mut with_hilbert: Vec<_> = vectors.into_iter()
            .zip(transformed)
            .map(|(record, transformed)| {
                let hilbert = self.hilbert.point_to_hilbert(&transformed);
                (record, hilbert)
            })
            .collect();

        // 3. Sort by Hilbert code
        with_hilbert.sort_by_key(|(_, h)| *h);

        // 4. Create blocks
        let blocks = with_hilbert.chunks(self.config.block_size)
            .map(|chunk| self.create_block(chunk))
            .collect();

        // 5. Write blocks
        self.write_blocks(blocks).await
    }
}
