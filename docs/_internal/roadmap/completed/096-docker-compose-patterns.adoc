== Overview

This guide provides Docker Compose patterns for various ProximaDB deployment scenarios, from development to production.

=== Pattern Categories

[source,text]
----
Deployment Patterns:
├── Development (single container)
├── Production (multi-replica)
├── High Availability (with monitoring)
├── Multi-tenant (isolated)
└── Hybrid (with external services)
----

== Quick Start

=== Minimal Development Setup

[source,yaml]
----
# docker-compose.yml
version: '3.8'

services:
  proximadb:
    image: proximadb/proximadb:latest
    ports:
      - "5678:5678"  # REST API
      - "5679:5679"  # gRPC
    volumes:
      - proximadb-data:/data
    environment:
      - RUST_LOG=info

volumes:
  proximadb-data:
----

[source,bash]
----
# Start
docker-compose up -d

# Check status
docker-compose ps

# View logs
docker-compose logs -f proximadb

# Stop
docker-compose down
----

== Production Pattern

=== Production with Persistence

[source,yaml]
----
# docker-compose.prod.yml
version: '3.8'

services:
  proximadb:
    image: proximadb/proximadb:0.1.5
    container_name: proximadb-prod
    ports:
      - "5678:5678"
      - "5679:5679"
    volumes:
      - proximadb-data:/data
      - ./config/prod.toml:/etc/proximadb/config.toml:ro
      - ./certs:/certs:ro
    environment:
      - RUST_LOG=info
      - PROXIMADB_CONFIG=/etc/proximadb/config.toml
    deploy:
      resources:
        limits:
          cpus: '8'
          memory: 32G
        reservations:
          cpus: '4'
          memory: 16G
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5678/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 60s
    restart: always
    logging:
      driver: "json-file"
      options:
        max-size: "100m"
        max-file: "5"
    ulimits:
      memlock:
        soft: -1
        hard: -1
      nofile:
        soft: 65535
        hard: 65535

volumes:
  proximadb-data:
    driver: local
    driver_opts:
      type: none
      o: bind
      device: /mnt/nvme/proximadb

networks:
  default:
    name: proximadb-prod
    driver: bridge
    ipam:
      config:
        - subnet: 172.28.0.0/16
----

=== HAProxy Configuration

[source,text]
----
# haproxy/haproxy.cfg
global
    daemon
    maxconn 4096
    log stdout format raw local0

defaults
    mode http
    timeout connect 5s
    timeout client 30s
    timeout server 30s
    option httplog
    log global

frontend stats
    bind *:8404
    stats enable
    stats uri /stats
    stats refresh 10s

frontend rest_frontend
    bind *:5678
    default_backend rest_backend

frontend grpc_frontend
    bind *:5679 proto h2
    default_backend grpc_backend

backend rest_backend
    balance roundrobin
    option httpchk GET /health
    http-check expect status 200
    server proximadb-1 proximadb-1:5678 check inter 5s fall 3 rise 2
    server proximadb-2 proximadb-2:5678 check inter 5s fall 3 rise 2
    server proximadb-3 proximadb-3:5678 check inter 5s fall 3 rise 2

backend grpc_backend
    balance roundrobin
    option httpchk GET /health
    http-check expect status 200
    server proximadb-1 proximadb-1:5679 check inter 5s fall 3 rise 2 proto h2
    server proximadb-2 proximadb-2:5679 check inter 5s fall 3 rise 2 proto h2
    server proximadb-3 proximadb-3:5679 check inter 5s fall 3 rise 2 proto h2

=== Alert Rules

[source,yaml]
----
# prometheus/alerts.yml
groups:
  - name: proximadb
    rules:
      - alert: ProximaDBDown
        expr: up{job="proximadb"} == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "ProximaDB is down"
          description: "ProximaDB has been down for more than 1 minute"

      - alert: ProximaDBHighMemory
        expr: proximadb_memory_used_bytes / proximadb_memory_total_bytes > 0.9
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "ProximaDB high memory usage"
          description: "Memory usage is above 90%"

      - alert: ProximaDBSlowQueries
        expr: histogram_quantile(0.99, rate(proximadb_query_duration_seconds_bucket[5m])) > 1
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "ProximaDB slow queries detected"
          description: "P99 query latency is above 1 second"

      - alert: ProximaDBHighErrorRate
        expr: rate(proximadb_errors_total[5m]) > 10
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "ProximaDB high error rate"
          description: "Error rate is above 10 per second"
----

=== Gateway Configuration

[source,nginx]
----
# nginx/nginx.conf
events {
    worker_connections 1024;
}

http {
    upstream tenant_a {
        server proximadb-tenant-a:5678;
    }

    upstream tenant_b {
        server proximadb-tenant-b:5678;
    }

    map $http_x_tenant_id $backend {
        default tenant_a;
        "tenant-a" tenant_a;
        "tenant-b" tenant_b;
    }

    server {
        listen 80;

        location / {
            proxy_pass http://$backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Tenant-ID $http_x_tenant_id;
        }
    }
}
----

== Utility Scripts

=== docker-compose-utils.sh

[source,bash]
----
#!/bin/bash

# docker-compose-utils.sh

# Start services
start() {
    docker-compose -f docker-compose.${ENV:-dev}.yml up -d
}

# Stop services
stop() {
    docker-compose -f docker-compose.${ENV:-dev}.yml down
}

# Restart services
restart() {
    stop && start
}

# View logs
logs() {
    docker-compose -f docker-compose.${ENV:-dev}.yml logs -f ${1:-proximadb}
}

# Backup data
backup() {
    BACKUP_DIR="./backups/$(date +%Y%m%d_%H%M%S)"
    mkdir -p "$BACKUP_DIR"
    docker-compose -f docker-compose.${ENV:-dev}.yml exec proximadb \
        tar -czf - /data > "$BACKUP_DIR/proximadb-backup.tar.gz"
    echo "Backup saved to $BACKUP_DIR"
}

# Restore data
restore() {
    BACKUP_FILE=$1
    if [ -z "$BACKUP_FILE" ]; then
        echo "Usage: restore <backup-file>"
        exit 1
    fi
    docker-compose -f docker-compose.${ENV:-dev}.yml exec -T proximadb \
        tar -xzf - -C / < "$BACKUP_FILE"
    echo "Restore complete"
}

# Health check
health() {
    curl -s http://localhost:5678/health | jq .
}

# Stats
stats() {
    docker stats proximadb --no-stream
}

# Execute command
case "$1" in
    start|stop|restart|logs|backup|restore|health|stats)
        $1 "${@:2}"
        ;;
    *)
        echo "Usage: $0 {start|stop|restart|logs|backup|restore|health|stats}"
        exit 1
        ;;
esac
----

== Summary

=== Pattern Selection Guide

[cols="2,3,2", options="header"]
|===
| Scenario | Recommended Pattern | Key Features
| Local development | Development | Hot reload, debug logs
| Single server | Production | Resource limits, persistence
| Multi-node | High Availability | Load balancer, replication
| Enterprise | Monitoring Stack | Full observability
| Cloud-native | S3-backed | Object storage, caching
| SaaS | Multi-tenant | Isolation, gateway
| ML workloads | GPU-enabled | CUDA acceleration
|===
