//! Parquet file reader with predicate pushdown and column pruning

use arrow::array::RecordBatch;
use arrow::datatypes::SchemaRef;
use parquet::arrow::arrow_reader::ParquetRecordBatchReaderBuilder;
use parquet::file::properties::ReaderProperties;

use super::{predicate::Predicate, Result, ProximaDBError};
use std::path::Path;
use std::sync::Arc;

/// Parquet reader configuration
#[derive(Debug, Clone)]
pub struct ReaderConfig {
    /// Enable predicate pushdown
    pub enable_predicate_pushdown: bool,
    /// Enable column pruning
    pub enable_column_pruning: bool,
    /// Batch size for record batches
    pub batch_size: usize,
    /// Enable memory mapping
    pub enable_mmap: bool,
}

impl Default for ReaderConfig {
    fn default() -> Self {
        Self {
            enable_predicate_pushdown: true,
            enable_column_pruning: true,
            batch_size: 4096,
            enable_mmap: true,
        }
    }
}

/// Parquet file reader
pub struct ParquetReader {
    config: ReaderConfig,
    file_path: String,
}

impl ParquetReader {
    /// Create a new Parquet reader
    pub fn new<P: AsRef<Path>>(file_path: P, config: ReaderConfig) -> Result<Self> {
        let file_path = file_path.as_ref().to_string_lossy().to_string();

        // Validate file exists
        if !std::path::Path::new(&file_path).exists() {
            return Err(ProximaDBError::StorageError(format!(
                "Parquet file not found: {}",
                file_path
            )));
        }

        Ok(Self { config, file_path })
    }

    /// Execute a query with optional predicate
    pub fn execute(
        &self,
        projection: Option<Vec<usize>>,
        predicate: Option<Predicate>,
    ) -> Result<Vec<RecordBatch>> {
        let mut reader_builder = self.build_reader()?;

        // Apply column pruning
        if let Some(cols) = projection {
            reader_builder = reader_builder.with_projection(cols);
        }

        // Apply predicate pushdown
        if self.config.enable_predicate_pushdown {
            if let Some(pred) = predicate {
                reader_builder = reader_builder.with_predicate(pred.to_arrow_predicate()?);
            }
        }

        let mut reader = reader_builder.build()?;
        let mut batches = Vec::new();

        while let Some(batch) = reader.next() {
            batches.push(batch?);
        }

        Ok(batches)
    }

    /// Get schema for the Parquet file
    pub fn schema(&self) -> Result<SchemaRef> {
        let builder = self.build_reader()?;
        Ok(builder.schema())
    }

    /// Build Parquet reader with configuration
    fn build_reader(&self) -> Result<ParquetRecordBatchReaderBuilder> {
        let file = std::fs::File::open(&self.file_path)?;

        let mut properties = ReaderProperties::builder();
        if self.config.enable_mmap {
            properties = properties.enable_memory_mapping();
        }

        let builder = ParquetRecordBatchReaderBuilder::try_new(file)?
            .with_batch_size(self.config.batch_size)
            .with_reader_properties(properties.build());

        Ok(builder)
    }
}

//! Parquet file statistics for query optimization

use arrow::datatypes::SchemaRef;
use parquet::file::metadata::ParquetMetaData;

/// Statistics for a Parquet file
#[derive(Debug, Clone)]
pub struct ParquetStatistics {
    /// Row group count
    pub row_group_count: usize,
    /// Row count
    pub row_count: usize,
    /// Compressed size in bytes
    pub compressed_size: u64,
    /// Uncompressed size in bytes
    pub uncompressed_size: u64,
}

impl ParquetStatistics {
    /// Extract statistics from Parquet metadata
    pub fn from_metadata(metadata: &ParquetMetaData) -> Self {
        let row_group_count = metadata.row_groups().len();
        let row_count: usize = metadata.row_groups().iter().map(|g| g.num_rows() as usize).sum();
        let compressed_size: u64 = metadata.row_groups().iter().map(|g| g.compressed_size()).sum();
        let uncompressed_size: u64 =
            metadata.row_groups().iter().map(|g| g.uncompressed_size()).sum();

        Self {
            row_group_count,
            row_count,
            compressed_size,
            uncompressed_size,
        }
    }
}

/// Limit operator - limits row count
pub struct LimitOperator {
    input: Arc<dyn ExecutionPlan>,
    limit: usize,
}

impl LimitOperator {
    pub fn new(input: Arc<dyn ExecutionPlan>, limit: usize) -> Self {
        Self { input, limit }
    }
}

impl ExecutionPlan for LimitOperator {
    fn execute(&self) -> Result<Vec<RecordBatch>> {
        let input_batches = self.input.execute()?;
        let mut limited_batches = Vec::new();
        let mut count = 0;

        for batch in input_batches {
            if count + batch.num_rows() <= self.limit {
                limited_batches.push(batch);
                count += batch.num_rows();
            } else {
                // Slice the batch
                let remaining = self.limit - count;
                limited_batches.push(batch.slice(0, remaining));
                break;
            }
        }

        Ok(limited_batches)
    }

    fn schema(&self) -> SchemaRef {
        self.input.schema()
    }
}
