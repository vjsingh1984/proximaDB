== Executive Summary

Transform the Semantic Knowledge Store (SKS) from a split storage model (vector engine + in-memory relations) to a unified graph-first architecture where the graph engine (Orion) serves as the primary storage layer, with vector indices providing efficient broad search capabilities.

**Key Goals:**

* **Performance**: 10-20x faster hybrid queries (similarity + traversal)
* **Data Locality**: Co-locate entities, relations, and embeddings in graph storage
* **Scalability**: Support billions of entities with millions of relationships
* **Query Fusion**: Enable seamless vector-graph hybrid operations

**Pain Points:**

1. **Data Fragmentation**: Embeddings in SST/VIPER, relations in HashMap, metadata in KV store
2. **Performance**: Hybrid queries require multiple system lookups (10-100ms per hop)
3. **Memory**: Duplicate data across storage layers
4. **Complexity**: Three separate storage abstractions to maintain

//! Graph schema definition for SKS entities in Orion

use crate::graph::{Node, Edge, NodeId, EdgeId};
use crate::proto::proximadb_v1::{Entity, Relation, EmbeddingVersion, TypedMetadata};
use anyhow::Result;

/// Convert Entity to Orion Node with properties
pub struct EntityNodeMapper;

impl EntityNodeMapper {
    /// Convert Entity -> Node
    pub fn to_node(entity: &Entity, collection_id: &str) -> Result<Node> {
        let node = Node {
            id: NodeId::from_string(&entity.id),
            labels: vec!["Entity".to_string(), collection_id.to_string()],
            properties: Self::build_properties(entity)?,
        };
        Ok(node)
    }

    /// Convert Node -> Entity
    pub fn from_node(node: &Node, collection_id: &str) -> Result<Entity> {
        let entity = Entity {
            id: node.id.to_string(),
            collection_id: collection_id.to_string(),
            embeddings: Self::extract_embeddings(node)?,
            typed_metadata: Self::extract_typed_metadata(node)?,
            flexible_metadata: Self::extract_flexible_metadata(node)?,
            provenance: Self::extract_provenance(node)?,
            temporal: Self::extract_temporal(node)?,
            relations: vec![], // Loaded separately via edges
        };
        Ok(entity)
    }

    /// Build node properties from entity
    fn build_properties(entity: &Entity) -> Result<HashMap<String, PropertyValue>> {
        let mut props = HashMap::new();

        // Store embeddings as node property
        // Each embedding version is stored separately
        for (idx, embedding) in entity.embeddings.iter().enumerate() {
            let key = format!("embedding_{}", idx);
            props.insert(key, PropertyValue::Vector(embedding.vector.clone()));

            // Store embedding metadata
            props.insert(
                format!("embedding_{}_model", idx),
                PropertyValue::String(embedding.model_id.clone())
            );
            props.insert(
                format!("embedding_{}_version", idx),
                PropertyValue::String(embedding.model_version.clone())
            );
        }

        // Store typed metadata as properties
        if let Some(ref metadata) = entity.typed_metadata {
            for (key, field) in &metadata.fields {
                props.insert(
                    format!("meta_{}", key),
                    Self::typed_field_to_property(field)?
                );
            }
        }

        // Store flexible metadata
        for (key, value) in &entity.flexible_metadata {
            props.insert(
                format!("flex_{}", key),
                Self::sql_value_to_property(value)?
            );
        }

        // Store provenance
        if let Some(ref prov) = entity.provenance {
            props.insert("provenance_source".to_string(),
                        PropertyValue::String(prov.source_id.clone()));
            props.insert("provenance_chunk".to_string(),
                        PropertyValue::String(prov.chunk_id.clone()));
        }

        Ok(props)
    }

    fn extract_embeddings(node: &Node) -> Result<Vec<EmbeddingVersion>> {
        let mut embeddings = Vec::new();
        let mut idx = 0;

        loop {
            let key = format!("embedding_{}", idx);
            if let Some(PropertyValue::Vector(vec)) = node.properties.get(&key) {
                let model = node.properties
                    .get(&format!("embedding_{}_model", idx))
                    .and_then(|v| v.as_string())
                    .unwrap_or("unknown");

                let version = node.properties
                    .get(&format!("embedding_{}_version", idx))
                    .and_then(|v| v.as_string())
                    .unwrap_or("v1");

                embeddings.push(EmbeddingVersion {
                    model_id: model.to_string(),
                    model_version: version.to_string(),
                    vector: vec.clone(),
                    dimension: vec.len() as i32,
                    created_at_ms: 0, // TODO: Extract from properties
                    model_params: Default::default(),
                    modality: 0, // TODO: Extract from properties
                });

                idx += 1;
            } else {
                break;
            }
        }

        Ok(embeddings)
    }

/// Convert Relation to Orion Edge
pub struct RelationEdgeMapper;

impl RelationEdgeMapper {
    pub fn to_edge(relation: &Relation, collection_id: &str) -> Result<Edge> {
        let edge = Edge {
            id: EdgeId::generate(),
            source_node_id: NodeId::from_string(&relation.source_entity_id),
            target_node_id: NodeId::from_string(&relation.target_entity_id),
            label: relation.relation_type.clone(),
            properties: Self::build_properties(relation)?,
        };
        Ok(edge)
    }

    pub fn from_edge(edge: &Edge) -> Result<Relation> {
        let relation = Relation {
            source_entity_id: edge.source_node_id.to_string(),
            target_entity_id: edge.target_node_id.to_string(),
            relation_type: edge.label.clone(),
            weight: Self::extract_weight(edge)?,
            created_at_ms: Self::extract_created_at(edge)?,
            properties: Self::extract_properties(edge)?,
        };
        Ok(relation)
    }

    fn build_properties(relation: &Relation) -> Result<HashMap<String, PropertyValue>> {
        let mut props = HashMap::new();
        props.insert("weight".to_string(), PropertyValue::Float(relation.weight));
        props.insert("created_at".to_string(), PropertyValue::Int(relation.created_at_ms));

        // Convert custom properties
        for (key, value) in &relation.properties {
            props.insert(key.clone(), Self::sql_value_to_property(value)?);
        }

        Ok(props)
    }

    fn extract_weight(edge: &Edge) -> Result<f32> {
        edge.properties
            .get("weight")
            .and_then(|v| v.as_float())
            .ok_or_else(|| anyhow::anyhow!("Missing weight property"))
    }

    fn extract_created_at(edge: &Edge) -> Result<i64> {
        edge.properties
            .get("created_at")
            .and_then(|v| v.as_int())
            .ok_or_else(|| anyhow::anyhow!("Missing created_at property"))
    }

        &self,
        collection_id: &str,

        include_embeddings: bool,
        include_relations: bool,
    ) -> Result<Option<Entity>> {
        // Retrieve node from Orion
        let node_id = NodeId::from_string(entity_id);
        let node = self.graph_engine.get_node(&node_id).await?;

        if node.is_none() {
            return Ok(None);

        // Optionally include relations

            let edges = self.graph_engine.get_outgoing_edges(&node_id).await?;
            for edge in edges {
                let relation = RelationEdgeMapper::from_edge(&edge)?;
                entity.relations.push(relation);
            }
        }

        Ok(Some(entity))

//! Migration utilities for moving from current SKS to graph-first

use crate::storage::entity_store::{ProximaEntityStore, OrionBackedEntityStore};
use crate::proto::proximadb_v1::Entity;
use anyhow::Result;
use std::sync::Arc;

/// Migrate entities from current store to Orion-backed store
pub struct SksStoreMigrator {
    source: Arc<ProximaEntityStore>,
    target: Arc<OrionBackedEntityStore>,
}

impl SksStoreMigrator {
    pub fn new(
        source: Arc<ProximaEntityStore>,
        target: Arc<OrionBackedEntityStore>,
    ) -> Self {
        Self { source, target }
    }

    /// Migrate all entities in a collection
    pub async fn migrate_collection(&self, collection_id: &str) -> Result<MigrationReport> {
        let mut report = MigrationReport::default();

        let mut offset = 0;
        let batch_size = 100;

        loop {
            // Fetch batch from source
            let entities = self.source
                .list_entities(collection_id, offset, batch_size)
                .await?;

            if entities.is_empty() {
                break;
            }

            // Migrate each entity
            for entity in entities {
                match self.migrate_entity(collection_id, &entity).await {
                    Ok(_) => report.migrated_count += 1,
                    Err(e) => {
                        report.failed_count += 1;
                        report.errors.push(format!("Entity {}: {}", entity.id, e));
                    }
                }
            }

            offset += batch_size;

            if offset % 1000 == 0 {
                println!("Migrated {} entities", offset);
            }
        }

        Ok(report)
    }

    async fn migrate_entity(&self, collection_id: &str, entity: &Entity) -> Result<()> {
        // Get full entity with relations
        let full_entity = self.source
            .get_entity(collection_id, &entity.id, true, true)
            .await?
            .ok_or_else(|| anyhow::anyhow!("Entity not found"))?;

        // Insert into target
        self.target.upsert_entity(collection_id, full_entity).await?;

        Ok(())
    }

    /// Verify migration correctness
    pub async fn verify_migration(&self, collection_id: &str) -> Result<VerificationReport> {
        let mut report = VerificationReport::default();

        // Compare entity counts
        let source_entities = self.source.list_entities(collection_id, 0, usize::MAX).await?;
        let target_entities = self.target.list_entities(collection_id, 0, usize::MAX).await?;

        report.source_count = source_entities.len();
        report.target_count = target_entities.len();

        // Verify each entity
        for source_entity in &source_entities {
            let target_entity = self.target
                .get_entity(collection_id, &source_entity.id, true, true)
                .await?;

            if target_entity.is_none() {
                report.missing_entities.push(source_entity.id.clone());
                continue;
            }

            let target_entity = target_entity.unwrap();

            // Compare embeddings
            if source_entity.embeddings.len() != target_entity.embeddings.len() {
                report.mismatched_entities.push(source_entity.id.clone());
            }

            // Compare relations
            if source_entity.relations.len() != target_entity.relations.len() {
                report.mismatched_entities.push(source_entity.id.clone());
            }
        }

        Ok(report)
    }
}

//! Vector index integration for SKS graph-first architecture

use crate::index::axis::AxisIndex;
use crate::proto::proximadb_v1::Entity;
use anyhow::Result;
use std::sync::Arc;

/// Vector index wrapper for SKS
pub struct SksVectorIndex {
    /// AXIS index (HNSW/IVF)
    axis: Arc<AxisIndex>,

    /// Mapping: vector_id -> node_id
    vector_to_node: Arc<RwLock<HashMap<String, String>>>,
}

impl SksVectorIndex {
    pub fn new(axis: Arc<AxisIndex>) -> Self {
        Self {
            axis,
            vector_to_node: Arc::new(RwLock::new(HashMap::new())),
        }
    }

    /// Index entity embeddings
    pub async fn index_entity(&self, entity: &Entity) -> Result<()> {
        for (idx, embedding) in entity.embeddings.iter().enumerate() {
            let vector_id = format!("{}:{}", entity.id, idx);

            // Add to AXIS index
            self.axis.add_vector(
                &vector_id,
                &embedding.vector,
            ).await?;

            // Store mapping
            self.vector_to_node
                .write()
                .unwrap()
                .insert(vector_id, entity.id.clone());
        }

        Ok(())
    }

    /// Remove entity from index
    pub async fn remove_entity(&self, entity_id: &str) -> Result<()> {
        // Find all vector IDs for this entity
        let vector_ids: Vec<String> = self.vector_to_node
            .read()
            .unwrap()
            .iter()
            .filter(|(_, node_id)| *node_id == entity_id)
            .map(|(vec_id, _)| vec_id.clone())
            .collect();

        // Remove from AXIS
        for vector_id in &vector_ids {
            self.axis.remove_vector(vector_id).await?;
        }

        // Remove mappings
        let mut map = self.vector_to_node.write().unwrap();
        for vector_id in vector_ids {
            map.remove(&vector_id);
        }

        Ok(())
    }

    /// Search for similar entities
    pub async fn search_similar(
        &self,
        query_vector: &[f32],
        top_k: usize,
    ) -> Result<Vec<(String, f32)>> {
        // Search AXIS index
        let results = self.axis.search(query_vector, top_k).await?;

        // Map vector IDs to node IDs
        let map = self.vector_to_node.read().unwrap();
        let mut node_results = Vec::new();

        for (vector_id, score) in results {
            if let Some(node_id) = map.get(&vector_id) {
                node_results.push((node_id.clone(), score));
            }
        }

        Ok(node_results)
    }
}

//! Hybrid query engine combining vector similarity and graph traversal

use crate::graph::engines::orion::OrionGraphEngine;
use crate::storage::entity_store::vector_index::SksVectorIndex;
use crate::proto::proximadb_v1::Entity;
use anyhow::Result;
use std::sync::Arc;

/// Hybrid query execution strategies
#[derive(Debug, Clone, Copy)]
pub enum HybridStrategy {
    /// Vector search first, then graph traversal
    VectorFirst,

    /// Graph traversal first, then vector ranking
    GraphFirst,

    /// Parallel execution with result fusion
    Parallel,
}

/// Hybrid query executor
pub struct HybridQueryExecutor {
    graph_engine: Arc<OrionGraphEngine>,
    vector_index: Arc<SksVectorIndex>,
    strategy: HybridStrategy,
}

impl HybridQueryExecutor {
    pub fn new(
        graph_engine: Arc<OrionGraphEngine>,
        vector_index: Arc<SksVectorIndex>,
        strategy: HybridStrategy,
    ) -> Self {
        Self {
            graph_engine,
            vector_index,
            strategy,
        }
    }

    /// Execute hybrid query: similarity + traversal
    pub async fn execute(
        &self,
        query_vector: Vec<f32>,
        top_k: usize,
        max_hops: u32,
        relation_types: Option<Vec<String>>,
    ) -> Result<Vec<(Entity, f32)>> {
        match self.strategy {
            HybridStrategy::VectorFirst => {
                self.execute_vector_first(query_vector, top_k, max_hops, relation_types).await
            }
            HybridStrategy::GraphFirst => {
                self.execute_graph_first(query_vector, top_k, max_hops, relation_types).await
            }
            HybridStrategy::Parallel => {
                self.execute_parallel(query_vector, top_k, max_hops, relation_types).await
            }
        }
    }

    async fn execute_vector_first(
        &self,
        query_vector: Vec<f32>,
        top_k: usize,
        max_hops: u32,
        relation_types: Option<Vec<String>>,
    ) -> Result<Vec<(Entity, f32)>> {
        // Step 1: Vector similarity search
        let candidates = self.vector_index
            .search_similar(&query_vector, top_k)
            .await?;

        // Step 2: Graph traversal from candidates
        let mut all_entities = HashSet::new();

        for (node_id, _score) in candidates {
            let visited = self.graph_engine
                .bfs_traverse(&NodeId::from_string(&node_id), max_hops, relation_types.clone())
                .await?;

            for node in visited {
                all_entities.insert(node);
            }
        }

        // Step 3: Re-rank by similarity
        let mut scored_entities = Vec::new();
        for node in all_entities {
            let entity = EntityNodeMapper::from_node(&node, "collection")?;

            if let Some(embedding) = entity.embeddings.first() {
                let similarity = cosine_similarity(&query_vector, &embedding.vector);
                scored_entities.push((entity, similarity));
            }
        }

        // Sort by similarity
        scored_entities.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
        scored_entities.truncate(top_k);

        Ok(scored_entities)
    }

fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
    let dot_product: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
    let norm_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
    let norm_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
    dot_product / (norm_a * norm_b)
}
