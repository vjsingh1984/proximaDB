== Executive Summary

This document outlines the design for extending ProximaDB to serve as an embedded semantic code knowledge database, combining vector similarity search with graph-based code structure analysis. The primary use case is AST/Tree-sitter based code parsing with call graphs, symbol tracking, and semantic code search at internet scale.

|Code-Specific Data Models

|Generic graph traversal only

|Symbol Resolution
|No symbol tracking

class LanguageParser(ABC):
    """Abstract base for language-specific parsers"""

    @property
    @abstractmethod
    def language(self) -> str:
        pass

    @property
    @abstractmethod
    def file_extensions(self) -> List[str]:
        pass

    @abstractmethod
    def parse(self, content: str, file_path: str) -> ParsedCode:
        pass

    @abstractmethod
    def extract_symbols(self, tree: tree_sitter.Tree, content: str) -> List[CodeSymbol]:
        pass

    @abstractmethod
    def extract_relations(self, tree: tree_sitter.Tree, symbols: List[CodeSymbol]) -> List[CodeRelation]:
        pass

class PythonParser(LanguageParser):
    """Python-specific parser using tree-sitter"""

    def __init__(self):
        self._parser = get_parser("python")
        self._symbol_queries = self._build_queries()

    @property
    def language(self) -> str:
        return "python"

    @property
    def file_extensions(self) -> List[str]:
        return [".py", ".pyi"]

    def _build_queries(self) -> Dict[str, str]:
        return {
            "function": """
                (function_definition
                    name: (identifier) @name
                    parameters: (parameters) @params
                    return_type: (type)? @return_type
                    body: (block) @body) @function
            """,
            "class": """
                (class_definition
                    name: (identifier) @name
                    superclasses: (argument_list)? @bases
                    body: (block) @body) @class
            """,
            "import": """
                [
                    (import_statement) @import
                    (import_from_statement) @import
                ]
            """,
            "call": """
                (call
                    function: [
                        (identifier) @callee
                        (attribute object: (_) @obj attribute: (identifier) @callee)
                    ]
                    arguments: (argument_list) @args) @call
            """
        }

    def parse(self, content: str, file_path: str) -> ParsedCode:
        import hashlib

        tree = self._parser.parse(bytes(content, "utf8"))
        symbols = self.extract_symbols(tree, content, file_path)
        relations = self.extract_relations(tree, symbols, content)
        imports = self._extract_imports(tree, content)

        return ParsedCode(
            file_path=file_path,
            language="python",
            symbols=symbols,
            relations=relations,
            imports=imports,
            content_hash=hashlib.sha256(content.encode()).hexdigest()
        )

    def extract_symbols(self, tree: tree_sitter.Tree, content: str, file_path: str) -> List[CodeSymbol]:
        symbols = []

        # Extract functions
        for node in self._query_nodes(tree, "function"):
            symbol = self._node_to_symbol(node, content, file_path, CodeSymbolType.FUNCTION)
            symbols.append(symbol)

        # Extract classes
        for node in self._query_nodes(tree, "class"):
            symbol = self._node_to_symbol(node, content, file_path, CodeSymbolType.CLASS)
            symbols.append(symbol)

            # Extract methods within class
            for method in self._find_methods_in_class(node, content, file_path):
                method.scope_chain = [symbol.fully_qualified_name]
                symbols.append(method)

        return symbols

    def _node_to_symbol(self, node, content: str, file_path: str,
                        symbol_type: CodeSymbolType) -> CodeSymbol:
        # Get name node
        name_node = node.child_by_field_name("name")
        name = content[name_node.start_byte:name_node.end_byte] if name_node else "anonymous"

        # Get source code
        source = content[node.start_byte:node.end_byte]

        # Get docstring
        doc = self._extract_docstring(node, content)

        # Get parameters for functions
        params = []
        if symbol_type in (CodeSymbolType.FUNCTION, CodeSymbolType.METHOD):
            params = self._extract_parameters(node, content)

        # Get return type
        return_type = None
        return_node = node.child_by_field_name("return_type")
        if return_node:
            return_type = content[return_node.start_byte:return_node.end_byte]

        return CodeSymbol(
            id=self._generate_symbol_id(file_path, name, node.start_point),
            symbol_type=symbol_type,
            fully_qualified_name=f"{file_path}::{name}",
            simple_name=name,
            location=SourceLocation(
                file_path=file_path,
                start_line=node.start_point[0] + 1,
                start_column=node.start_point[1],
                end_line=node.end_point[0] + 1,
                end_column=node.end_point[1],
                byte_offset=node.start_byte,
                byte_length=node.end_byte - node.start_byte
            ),
            source_code=source,
            language="python",
            documentation=doc,
            parameters=params,
            return_type=return_type,
            complexity=self._calculate_complexity(node, content)
        )

    def extract_relations(self, tree: tree_sitter.Tree,
                          symbols: List[CodeSymbol], content: str) -> List[CodeRelation]:
        relations = []
        symbol_map = {s.simple_name: s for s in symbols}

        # Find call relations
        for call_node in self._query_nodes(tree, "call"):
            callee_name = self._get_callee_name(call_node, content)
            if callee_name and callee_name in symbol_map:
                # Find containing function
                caller = self._find_containing_function(call_node, symbols)
                if caller:
                    relations.append(CodeRelation(
                        from_symbol_id=caller.id,
                        to_symbol_id=symbol_map[callee_name].id,
                        relation_type=CodeRelationType.CALLS,
                        call_site=SourceLocation(
                            file_path=caller.location.file_path,
                            start_line=call_node.start_point[0] + 1,
                            start_column=call_node.start_point[1]
                        )
                    ))

        return relations

class CodeChunker:
    """
    AST-aware code chunker that produces semantic code chunks.

    Unlike text chunkers, this produces chunks aligned to code structure:
    - Functions/methods as complete units
    - Classes with methods grouped
    - Proper scope preservation
    """

    PARSERS: Dict[str, LanguageParser] = {
        "python": PythonParser,
        # Add more: "javascript": JavaScriptParser, etc.
    }

    def __init__(self,
                 languages: Optional[List[str]] = None,
                 include_private: bool = True,
                 include_tests: bool = True,
                 max_chunk_size: int = 4096,
                 include_context: bool = True,
                 context_lines: int = 5):
        self.languages = languages or list(self.PARSERS.keys())
        self.include_private = include_private
        self.include_tests = include_tests
        self.max_chunk_size = max_chunk_size
        self.include_context = include_context
        self.context_lines = context_lines
        self._parsers = {
            lang: self.PARSERS[lang]()
            for lang in self.languages
            if lang in self.PARSERS
        }

    def parse_file(self, content: str, file_path: str,
                   language: Optional[str] = None) -> ParsedCode:
        """Parse a single file and extract symbols + relations"""
        lang = language or self._detect_language(file_path)
        if lang not in self._parsers:
            raise ValueError(f"Unsupported language: {lang}")

        return self._parsers[lang].parse(content, file_path)

    def parse_directory(self, directory: str,
                        recursive: bool = True) -> List[ParsedCode]:
        """Parse all supported files in a directory"""
        import os
        results = []

        for root, dirs, files in os.walk(directory):
            if not recursive and root != directory:
                break

            for file in files:
                file_path = os.path.join(root, file)
                lang = self._detect_language(file_path)

                if lang in self._parsers:
                    with open(file_path, 'r') as f:
                        content = f.read()
                    results.append(self.parse_file(content, file_path, lang))

        return results

    def _detect_language(self, file_path: str) -> Optional[str]:
        """Detect language from file extension"""
        ext = os.path.splitext(file_path)[1].lower()
        ext_map = {
            '.py': 'python',
            '.pyi': 'python',
            '.js': 'javascript',
            '.ts': 'typescript',
            '.jsx': 'javascript',
            '.tsx': 'typescript',
            '.java': 'java',
            '.kt': 'kotlin',
            '.rs': 'rust',
            '.go': 'go',
            '.cpp': 'cpp',
            '.c': 'c',
            '.h': 'cpp',
            '.hpp': 'cpp',
            '.cs': 'csharp',
            '.rb': 'ruby',
            '.php': 'php',
        }
        return ext_map.get(ext)

from dataclasses import dataclass
from typing import List, Dict, Optional, Any
from .unified_client import ProximaDBClient
from .chunking_strategies.code import CodeChunker, ParsedCode, CodeSymbol, CodeRelation
from .embedding_interface import EmbeddingProvider

@dataclass
class CodeKnowledgeConfig:
    """Configuration for code knowledge extraction"""
    collection_name: str
    embedding_provider: EmbeddingProvider
    languages: List[str] = None
    include_private: bool = True
    include_tests: bool = True
    extract_call_graph: bool = True
    extract_type_relations: bool = True
    incremental: bool = True  # Only process changed files

class CodeKnowledgeBuilder:
    """
    Unified builder for populating both vector and graph stores
    with code knowledge in a coordinated, atomic manner.
    """

    def __init__(self, client: ProximaDBClient, config: CodeKnowledgeConfig):
        self.client = client
        self.config = config
        self.chunker = CodeChunker(
            languages=config.languages,
            include_private=config.include_private,
            include_tests=config.include_tests
        )
        self._pending_symbols: List[CodeSymbol] = []
        self._pending_relations: List[CodeRelation] = []
        self._file_hashes: Dict[str, str] = {}

    async def index_file(self, content: str, file_path: str,
                         language: Optional[str] = None) -> ParsedCode:
        """
        Parse and index a single file, extracting symbols and relations.
        Returns the parsed result for inspection.
        """
        parsed = self.chunker.parse_file(content, file_path, language)

        # Check if file changed (for incremental updates)
        if self.config.incremental:
            old_hash = self._file_hashes.get(file_path)
            if old_hash == parsed.content_hash:
                return parsed  # No changes
            self._file_hashes[file_path] = parsed.content_hash

        self._pending_symbols.extend(parsed.symbols)
        self._pending_relations.extend(parsed.relations)

        return parsed

    async def index_directory(self, directory: str,
                              recursive: bool = True) -> List[ParsedCode]:
        """Index all supported files in a directory"""
        results = []
        for parsed in self.chunker.parse_directory(directory, recursive):
            await self.index_file(parsed.source_code, parsed.file_path)
            results.append(parsed)
        return results

    async def commit(self) -> Dict[str, Any]:
        """
        Commit all pending symbols and relations atomically.

        This coordinates:
        1. Embedding generation for code symbols
        2. Vector store population
        3. Graph store population (nodes + edges)
        4. Index updates

        Returns statistics about what was committed.
        """
        if not self._pending_symbols:
            return {"symbols": 0, "relations": 0}

        # Generate embeddings for all symbols
        texts = [s.source_code for s in self._pending_symbols]
        embeddings = self.config.embedding_provider.embed_texts(texts)

        # Prepare vector records
        vector_records = []
        for symbol, embedding in zip(self._pending_symbols, embeddings):
            vector_records.append({
                "id": symbol.id,
                "vector": embedding.tolist(),
                "source": symbol.source_code,
                "metadata": {
                    "symbol_type": symbol.symbol_type.name,
                    "fully_qualified_name": symbol.fully_qualified_name,
                    "simple_name": symbol.simple_name,
                    "file_path": symbol.location.file_path,
                    "start_line": symbol.location.start_line,
                    "end_line": symbol.location.end_line,
                    "language": symbol.language,
                    "documentation": symbol.documentation,
                    "signature": symbol.signature,
                    "return_type": symbol.return_type,
                    "modifiers": symbol.modifiers,
                    "scope_chain": symbol.scope_chain,
                    "complexity": symbol.complexity,
                }
            })

        # Prepare graph nodes (same data, different format)
        graph_nodes = []
        for symbol in self._pending_symbols:
            graph_nodes.append({
                "id": symbol.id,
                "labels": [symbol.symbol_type.name, symbol.language.upper()],
                "properties": {
                    "fully_qualified_name": symbol.fully_qualified_name,
                    "simple_name": symbol.simple_name,
                    "file_path": symbol.location.file_path,
                    "start_line": symbol.location.start_line,
                    "source_code": symbol.source_code[:1000],  # Truncate for graph
                    "documentation": symbol.documentation,
                }
            })

        # Prepare graph edges
        graph_edges = []
        for relation in self._pending_relations:
            graph_edges.append({
                "id": f"{relation.from_symbol_id}_{relation.relation_type.name}_{relation.to_symbol_id}",
                "from_node_id": relation.from_symbol_id,
                "to_node_id": relation.to_symbol_id,
                "edge_type": relation.relation_type.name,
                "properties": {
                    "confidence": relation.confidence,
                    "context": relation.context,
                }
            })

        # Atomic commit to both stores
        stats = {"symbols": 0, "relations": 0, "errors": []}

        try:
            # 1. Ensure collection exists
            await self._ensure_collection()

            # 2. Insert vectors (with source for RAG)
            await self.client.insert_vectors(
                self.config.collection_name,
                records=vector_records
            )
            stats["symbols"] = len(vector_records)

            # 3. Insert graph nodes
            for node in graph_nodes:
                await self.client.create_node(
                    self.config.collection_name,
                    node
                )

            # 4. Insert graph edges
            for edge in graph_edges:
                await self.client.create_edge(
                    self.config.collection_name,
                    edge
                )
            stats["relations"] = len(graph_edges)

        except Exception as e:
            stats["errors"].append(str(e))
            raise
        finally:
            # Clear pending
            self._pending_symbols = []
            self._pending_relations = []

        return stats

    async def _ensure_collection(self):
        """Ensure the code knowledge collection exists with proper schema"""
        # Check if collection exists
        collections = await self.client.list_collections()
        if self.config.collection_name not in [c.name for c in collections]:
            # Create with code-optimized configuration
            await self.client.create_collection(
                name=self.config.collection_name,
                dimension=self.config.embedding_provider.dimension,
                storage_engine="SST",  # Fast for incremental updates
                filterable_columns=[
                    {"name": "symbol_type", "data_type": "STRING"},
                    {"name": "language", "data_type": "STRING"},
                    {"name": "file_path", "data_type": "STRING"},
                    {"name": "simple_name", "data_type": "STRING"},
                    {"name": "start_line", "data_type": "INTEGER"},
                ],
                index_config={
                    "type": "HNSW",
                    "m": 16,
                    "ef_construction": 200
                }
            )

class CodeSearchClient:
    """
    High-level client for semantic code search.
    Combines vector similarity with graph traversal.
    """

    def __init__(self, client: ProximaDBClient, collection_name: str):
        self.client = client
        self.collection_name = collection_name

    async def semantic_search(self, query: str,
                              embedding_provider: EmbeddingProvider,
                              top_k: int = 10,
                              filters: Optional[Dict] = None) -> List[Dict]:
        """
        Search for code semantically similar to the query.

        Example:
            results = await client.semantic_search(
                "function that validates email addresses",
                provider,
                filters={"language": "python", "symbol_type": "FUNCTION"}
            )
        """
        query_vector = embedding_provider.embed_text(query).tolist()

        return await self.client.search(
            collection_name=self.collection_name,
            query_vector=query_vector,
            top_k=top_k,
            filters=filters,
            include_source=True
        )

    async def find_callers(self, symbol_id: str,
                           max_depth: int = 3) -> List[Dict]:
        """Find all functions that call the given symbol"""
        return await self.client.graph_traverse(
            graph_id=self.collection_name,
            start_node_id=symbol_id,
            edge_types=["CALLED_BY"],
            max_depth=max_depth,
            algorithm="BFS"
        )

    async def find_callees(self, symbol_id: str,
                           max_depth: int = 3) -> List[Dict]:
        """Find all functions called by the given symbol"""
        return await self.client.graph_traverse(
            graph_id=self.collection_name,
            start_node_id=symbol_id,
            edge_types=["CALLS"],
            max_depth=max_depth,
            algorithm="BFS"
        )

    async def impact_analysis(self, symbol_id: str,
                              max_depth: int = 5) -> Dict[str, Any]:
        """
        Analyze the impact of changing a symbol.
        Returns all callers and symbols that depend on this one.
        """
        callers = await self.find_callers(symbol_id, max_depth)

        # Find tests that exercise this symbol
        tests = [c for c in callers
                 if c.get("labels", []) and "TEST" in c["labels"]]

        return {
            "impacted_symbols": callers,
            "impacted_tests": tests,
            "total_impacted": len(callers),
            "tests_to_run": len(tests)
        }

    async def find_similar_functions(self, symbol_id: str,
                                     embedding_provider: EmbeddingProvider,
                                     top_k: int = 10) -> List[Dict]:
        """
        Find functions similar to the given one.
        Useful for finding duplicate code or refactoring candidates.
        """
        # Get the symbol's embedding
        node = await self.client.get_node(self.collection_name, symbol_id)
        if not node:
            raise ValueError(f"Symbol not found: {symbol_id}")

        # Search for similar
        return await self.client.search(
            collection_name=self.collection_name,
            query_vector=node.get("embedding"),
            top_k=top_k + 1,  # +1 to exclude self
            filters={"symbol_type": "FUNCTION"}
        )

use crate::graph::{Node, Edge, GraphEngine};
use crate::storage::traits::UnifiedStorageEngine;
use crate::proto::proximadb_v1::{CodeSymbol, CodeRelation, CodeFile};
use anyhow::Result;
use std::sync::Arc;

pub struct CodeKnowledgeService {
    vector_engine: Arc<dyn UnifiedStorageEngine>,
    graph_engine: Arc<dyn GraphEngine>,
    embedding_service: Option<Arc<dyn EmbeddingService>>,
}

impl CodeKnowledgeService {
    pub async fn index_file(&self, file: CodeFile) -> Result<IndexingResult> {
        // 1. Parse and extract symbols/relations (done client-side usually)
        // 2. Generate embeddings if embedding service available
        // 3. Insert to vector store
        // 4. Insert to graph store
        // 5. Update indexes

        let mut result = IndexingResult::default();

        for symbol in &file.symbols {
            // Create vector record
            if let Some(ref embedding) = symbol.embedding {
                let record = self.symbol_to_vector_record(symbol);
                self.vector_engine.insert(vec![record]).await?;
                result.vectors_inserted += 1;
            }

            // Create graph node
            let node = self.symbol_to_node(symbol);
            self.graph_engine.insert_node(node).await?;
            result.nodes_created += 1;
        }

        for relation in &file.relations {
            let edge = self.relation_to_edge(relation);
            self.graph_engine.insert_edge(edge).await?;
            result.edges_created += 1;
        }

        Ok(result)
    }

    pub async fn search_code(&self, request: CodeSearchRequest) -> Result<Vec<CodeSearchResult>> {
        // Hybrid search: vector similarity + graph context

        // 1. Vector search for semantic matches
        let vector_results = self.vector_engine.search(/* ... */).await?;

        // 2. For top results, enrich with graph context
        let mut results = Vec::new();
        for vr in vector_results {
            let symbol_id = &vr.id;

            // Get callers/callees from graph
            let callers = self.graph_engine.get_neighbors(
                symbol_id,
                Some("CALLED_BY"),
                1
            ).await?;

            let callees = self.graph_engine.get_neighbors(
                symbol_id,
                Some("CALLS"),
                1
            ).await?;

            results.push(CodeSearchResult {
                symbol: vr,
                callers: callers.len(),
                callees: callees.len(),
                // ... more context
            });
        }

        Ok(results)
    }

    pub async fn call_graph(&self, request: CallGraphRequest) -> Result<CallGraphResponse> {
        let traversal = match request.direction {
            CallGraphDirection::Callers => {
                self.graph_engine.traverse(
                    &request.symbol_id,
                    request.max_depth,
                    Some(vec!["CALLED_BY".to_string()]),
                    TraversalAlgorithm::BFS,
                ).await?
            }
            CallGraphDirection::Callees => {
                self.graph_engine.traverse(
                    &request.symbol_id,
                    request.max_depth,
                    Some(vec!["CALLS".to_string()]),
                    TraversalAlgorithm::BFS,
                ).await?
            }
            CallGraphDirection::Both => {
                // Merge both directions
                todo!()
            }
        };

        Ok(CallGraphResponse {
            nodes: traversal.nodes,
            edges: traversal.edges,
            stats: traversal.stats,
        })
    }

    pub async fn impact_analysis(&self, request: ImpactAnalysisRequest) -> Result<ImpactAnalysisResponse> {
        let mut impacted = Vec::new();
        let mut impacted_tests = Vec::new();

        for symbol_id in &request.changed_symbol_ids {
            // Find all callers recursively
            let callers = self.graph_engine.traverse(
                symbol_id,
                request.max_depth,
                Some(vec!["CALLED_BY".to_string()]),
                TraversalAlgorithm::BFS,
            ).await?;

            for (depth, node) in callers.nodes.iter().enumerate() {
                let severity = self.calculate_impact_severity(depth, node);
                impacted.push(ImpactedSymbol {
                    symbol: node.clone(),
                    depth: depth as u32,
                    severity,
                });

                // Check if it's a test
                if node.labels.iter().any(|l| l.contains("TEST")) {
                    impacted_tests.push(ImpactedTest {
                        test_symbol: node.clone(),
                        tested_symbols: vec![symbol_id.clone()],
                    });
                }
            }
        }

        Ok(ImpactAnalysisResponse {
            impacted,
            impacted_tests,
            summary: self.build_impact_summary(&impacted),
        })
    }
}

=== Sharding Strategy

For code knowledge at scale:

1. **Repository-based sharding**: Each repository gets assigned to a shard
2. **Hot/cold tiering**: Recently accessed files in SWIFT, historical in VIPER
3. **Cross-shard queries**: Symbol resolution across shards via distributed graph

== Conclusion

This design extends ProximaDB to serve as a comprehensive semantic code knowledge store, combining:

1. **Vector similarity** for semantic code search
2. **Graph traversal** for call graphs and impact analysis
3. **AST-aware chunking** for accurate code representation
4. **Coordinated population** for consistent vector + graph state
5. **Incremental updates** for efficient re-indexing

The architecture leverages existing ProximaDB strengths (proto-first design, multi-engine storage, embedded mode) while adding code-specific capabilities that enable powerful developer tooling use cases.
