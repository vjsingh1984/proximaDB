== Executive Summary

This document specifies the design for integrating Apache Arrow IPC (Inter-Process Communication) protocol support into ProximaDB as a high-performance bulk data ingestion pathway on port 5680.

=== Goals

* Add Arrow Flight-based server on port 5680 alongside REST (5678) and gRPC (5679)
* Enable high-throughput bulk vector ingestion for ETL and data migration workloads
* Provide direct engine write path bypassing WAL for maximum performance
* Support all 6 storage engines (SST, HELIX, VIPER, NOVA, SWIFT, RAPTOR)
* Maintain synchronous write acknowledgment for durability guarantees
* Enable explicit flush and compaction triggering via Arrow IPC commands

=== Non-Goals

* Replacing existing REST/gRPC APIs (they remain primary interfaces)
* Real-time streaming updates (Arrow IPC optimized for bulk batch operations)
* Cross-engine queries (single collection = single engine)

pub struct ArrowFlightServiceImpl {
    unified_handlers: Arc<UnifiedHandlers>,
    codec: Arc<ArrowProtoCodec>,
}

#[tonic::async_trait]
impl FlightService for ArrowFlightServiceImpl {
    // Bulk vector insert via DoPut
    async fn do_put(
        &self,
        request: Request<Streaming<FlightData>>,
    ) -> Result<Response<Self::DoPutStream>, Status> {
        let mut stream = request.into_inner();

        // Read FlightDescriptor (contains collection_id, write_mode)
        let descriptor = stream.message().await?
            .ok_or(Status::invalid_argument("Missing descriptor"))?;

        let metadata = self.parse_descriptor(&descriptor)?;
        let collection_id = &metadata.collection_id;
        let write_mode = metadata.write_mode;  // Direct or WAL

        // Read RecordBatch stream
        let mut batches = Vec::new();
        while let Some(data) = stream.message().await? {
            let batch = self.codec.flight_data_to_batch(&data)?;
            batches.push(batch);
        }

        // Convert Arrow → VectorRecord
        let vectors = self.codec.batches_to_vector_records(batches)?;

        // Route based on write mode
        let result = match write_mode {
            WriteMode::Direct => {
                self.direct_engine_write(collection_id, vectors).await?
            }
            WriteMode::WAL => {
                self.unified_handlers
                    .handle_vector_batch_v1(VectorBatchRequest {
                        collection_id: collection_id.clone(),
                        vectors,
                    })
                    .await?
            }
        };

        // Return PutResult stream
        Ok(Response::new(stream::once(async move {
            Ok(self.codec.result_to_flight_data(result))
        })))
    }

    // Vector search via DoGet
    async fn do_get(
        &self,
        request: Request<Ticket>,
    ) -> Result<Response<Self::DoGetStream>, Status> {
        let ticket = request.into_inner();
        let search_request = self.codec.ticket_to_search_request(&ticket)?;

        // Execute search via UnifiedHandlers
        let results = self.unified_handlers
            .handle_vector_search_v1(search_request)
            .await
            .map_err(|e| Status::internal(e.to_string()))?;

        // Convert results to Arrow RecordBatch stream
        let batches = self.codec.search_results_to_batches(results)?;

        Ok(Response::new(stream::iter(batches.into_iter().map(Ok))))
    }

    // Flush collection via DoAction
    async fn do_action(
        &self,
        request: Request<Action>,
    ) -> Result<Response<Self::DoActionStream>, Status> {
        let action = request.into_inner();

        match action.r#type.as_str() {
            "flush_collection" => {
                let collection_id = String::from_utf8(action.body)?;
                self.flush_collection(&collection_id).await?;
            }
            "compact_collection" => {
                let collection_id = String::from_utf8(action.body)?;
                self.compact_collection(&collection_id).await?;
            }
            "flush_and_compact" => {
                let collection_id = String::from_utf8(action.body)?;
                self.flush_collection(&collection_id).await?;
                self.compact_collection(&collection_id).await?;
            }
            _ => return Err(Status::unimplemented("Unknown action")),
        }

        Ok(Response::new(stream::empty()))
    }
}

pub struct ArrowProtoCodec;

impl ArrowProtoCodec {
    /// Convert Arrow RecordBatch to VectorRecord protos
    pub fn batches_to_vector_records(
        &self,
        batches: Vec<RecordBatch>,
    ) -> Result<Vec<VectorRecord>> {
        let mut records = Vec::new();

        for batch in batches {
            // Extract schema-validated columns
            let id_array = batch.column(0)
                .as_any()
                .downcast_ref::<StringArray>()
                .ok_or(ArrowError::InvalidColumn("id"))?;

            let vector_array = batch.column(1)
                .as_any()
                .downcast_ref::<FixedSizeListArray>()
                .ok_or(ArrowError::InvalidColumn("vector"))?;

            let metadata_array = batch.column(2)
                .as_any()
                .downcast_ref::<StructArray>()
                .ok_or(ArrowError::InvalidColumn("metadata"))?;

            // Convert each row
            for row_idx in 0..batch.num_rows() {
                let id = id_array.value(row_idx).to_string();
                let vector = self.extract_vector(vector_array, row_idx)?;
                let metadata = self.extract_metadata(metadata_array, row_idx)?;

                records.push(VectorRecord {
                    id,
                    vector,
                    metadata,
                    timestamp: Some(chrono::Utc::now().timestamp()),
                    ..Default::default()
                });
            }
        }

        Ok(records)
    }

    /// Convert VectorRecord protos to Arrow RecordBatch
    pub fn vector_records_to_batches(
        &self,
        records: Vec<VectorRecord>,
        dimension: usize,
    ) -> Result<Vec<RecordBatch>> {
        let schema = self.create_vector_schema(dimension)?;

        // Build arrays
        let id_array = StringArray::from_iter_values(
            records.iter().map(|r| r.id.as_str())
        );

        let vector_array = self.build_vector_array(&records, dimension)?;
        let metadata_array = self.build_metadata_struct_array(&records)?;

        let batch = RecordBatch::try_new(
            Arc::new(schema),
            vec![
                Arc::new(id_array),
                Arc::new(vector_array),
                Arc::new(metadata_array),
            ],
        )?;

        Ok(vec![batch])
    }

    /// Standard schema for vector data
    fn create_vector_schema(&self, dimension: usize) -> Result<Schema> {
        Ok(Schema::new(vec![
            Field::new("id", DataType::Utf8, false),
            Field::new(
                "vector",
                DataType::FixedSizeList(
                    Arc::new(Field::new("item", DataType::Float32, false)),
                    dimension as i32,
                ),
                false,
            ),
            Field::new(
                "metadata",
                DataType::Struct(Fields::from(vec![
                    Field::new("key", DataType::Utf8, false),
                    Field::new("value", DataType::Utf8, true),
                ])),
                true,
            ),
        ]))
    }
}

impl ArrowFlightServiceImpl {
    /// Direct write bypassing WAL for maximum throughput
    async fn direct_engine_write(
        &self,
        collection_id: &str,
        vectors: Vec<VectorRecord>,
    ) -> Result<VectorOperationResponse> {
        // Get collection metadata
        let collection = self.unified_handlers
            .collection_service
            .get_collection(collection_id)
            .await?
            .ok_or(ProximaError::CollectionNotFound)?;

        // Get storage engine instance
        let storage_engine = self.get_storage_engine_for_collection(&collection)?;

        // Convert VectorRecord → Engine-specific format
        let engine_data = match collection.storage_engine {
            StorageEngine::Sst => {
                self.convert_to_sst_format(vectors)?
            }
            StorageEngine::Viper => {
                self.convert_to_viper_format(vectors)?
            }
            StorageEngine::Nova => {
                self.convert_to_nova_format(vectors)?
            }
            StorageEngine::Swift => {
                self.convert_to_swift_format(vectors)?
            }
            StorageEngine::Helix => {
                self.convert_to_helix_format(vectors)?
            }
            StorageEngine::Raptor => {
                self.convert_to_raptor_format(vectors)?
            }
        };

        // Direct write with flush
        let flush_result = storage_engine.flush(FlushParameters {
            collection_id: Some(collection_id.to_string()),
            force: true,
            synchronous: true,  // Wait for disk sync
            vector_records: engine_data,
            trigger_compaction: false,  // Explicit via DoAction
            ..Default::default()
        }).await?;

        // Return acknowledgment
        Ok(VectorOperationResponse {
            vector_ids: flush_result.vector_ids,
            operation_time_ms: flush_result.elapsed_ms as f64,
            vectors_processed: flush_result.vectors_written,
            ..Default::default()
        })
    }

    /// Convert to SST ProximaBlock format
    fn convert_to_sst_format(
        &self,
        vectors: Vec<VectorRecord>,
    ) -> Result<Vec<VectorRecord>> {
        // SST accepts VectorRecord directly
        Ok(vectors)
    }

    /// Convert to VIPER Parquet RowGroup format
    fn convert_to_viper_format(
        &self,
        vectors: Vec<VectorRecord>,
    ) -> Result<Vec<VectorRecord>> {
        // VIPER uses columnar Arrow internally
        // Already converted via Arrow RecordBatch
        Ok(vectors)
    }

    // Similar conversions for NOVA, SWIFT, HELIX, RAPTOR...
}

=== Write Modes

==== Mode 1: WAL-Backed Write (Default)

[mermaid, target=wal-write-flow, format=svg]
....
%%{init: {"theme": "neutral"}}%%
sequenceDiagram
    participant Client
    participant Flight as Arrow Flight Server
    participant UH as UnifiedHandlers
    participant WAL as WriteAheadLog
    participant Engine as Storage Engine

    Client->>Flight: DoPut(RecordBatch)
    Flight->>Flight: Convert Arrow → VectorRecord
    Flight->>UH: handle_vector_batch_v1()
    UH->>WAL: write_vector_batch()
    WAL->>WAL: Add to memtable
    WAL->>WAL: Sync to disk
    WAL-->>UH: Sequence numbers
    UH-->>Flight: VectorOperationResponse
    Flight-->>Client: PutResult

    Note over WAL,Engine: Async flush when threshold reached
    WAL->>Engine: flush(vectors)
    Engine->>Engine: Write engine format
....

**Characteristics**:

* Durability guaranteed (WAL disk sync)
* Immediate searchability (WAL memtable)
* Async flush to storage engine
* Throughput: ~30K-50K vectors/sec

==== Mode 2: Direct Engine Write (High Throughput)

[mermaid, target=direct-write-flow, format=svg]
....
%%{init: {"theme": "neutral"}}%%
sequenceDiagram
    participant Client
    participant Flight as Arrow Flight Server
    participant Engine as Storage Engine

    Client->>Flight: DoPut(RecordBatch, mode=Direct)
    Flight->>Flight: Convert Arrow → VectorRecord
    Flight->>Flight: Convert to engine format
    Flight->>Engine: flush(vectors, synchronous=true)

* No WAL overhead
* Synchronous disk write
* Not searchable until flush completes
* Throughput: ~100K-200K vectors/sec
* Use case: Bulk ETL, initial load

// src/storage/engines/impls/sst/arrow_ingest.rs

impl UnifiedStorageEngine for SstEngine {
    async fn ingest_arrow_batch(
        &self,
        collection_id: &str,
        batch: RecordBatch,
        params: IngestParameters,
    ) -> Result<IngestResult> {
        // 1. Convert RecordBatch → VectorRecord (reuse existing codec)
        let codec = ArrowProtoCodec;
        let vectors = codec.batches_to_vector_records(vec![batch])?;

        // 2. Create SST ProximaBlocks
        let blocks = self.create_proxima_blocks(&vectors)?;

        // 3. Write blocks to disk
        let file_path = self.allocate_sst_file(collection_id)?;
        let mut writer = self.create_block_writer(&file_path)?;

        for block in blocks {
            writer.write_block(block)?;
        }

        if params.synchronous {
            writer.sync_all()?;
        }

        writer.finalize()?;

        // 4. Update metadata
        self.register_sst_file(collection_id, &file_path).await?;

        Ok(IngestResult {
            vectors_written: vectors.len(),
            bytes_written: writer.bytes_written(),
            elapsed_ms: writer.elapsed().as_millis() as u64,
            files_created: vec![file_path],
        })
    }
}

// src/storage/engines/impls/viper/arrow_ingest.rs

impl UnifiedStorageEngine for ViperEngine {
    async fn ingest_arrow_batch(
        &self,
        collection_id: &str,
        batch: RecordBatch,
        params: IngestParameters,
    ) -> Result<IngestResult> {
        // VIPER natively uses Arrow → Direct Parquet write

        // 1. Create Parquet schema from Arrow schema
        let parquet_schema = self.arrow_to_parquet_schema(batch.schema())?;

        // 2. Allocate row group file
        let file_path = self.allocate_parquet_file(collection_id)?;
        let props = WriterProperties::builder()
            .set_compression(Compression::ZSTD(ZstdLevel::default()))
            .build();

        // 3. Write directly (Arrow → Parquet, no intermediate conversion)
        let mut writer = ArrowWriter::try_new(
            File::create(&file_path)?,
            batch.schema(),
            Some(props),
        )?;

        writer.write(&batch)?;
        writer.close()?;

        // 4. Build metadata index
        self.index_parquet_file(collection_id, &file_path).await?;

        Ok(IngestResult {
            vectors_written: batch.num_rows(),
            bytes_written: std::fs::metadata(&file_path)?.len() as usize,
            elapsed_ms: 0,  // TODO: track
            files_created: vec![file_path],
        })
    }
}

=== C. Client SDK Examples

==== Python

[source,python]
----
import pyarrow as pa
import pyarrow.flight as flight
import numpy as np

# Connect
client = flight.FlightClient("grpc://localhost:5680")

# Create schema
schema = pa.schema([
    pa.field("id", pa.string()),
    pa.field("vector", pa.list_(pa.float32(), 384)),
    pa.field("metadata", pa.struct([("key", pa.string()), ("value", pa.string())])),
])

# Create batch
ids = pa.array([f"vec_{i}" for i in range(10000)])
vectors = pa.array([np.random.randn(384).tolist() for _ in range(10000)])
metadata = pa.array([{"key": "category", "value": "tech"} for _ in range(10000)])

batch = pa.RecordBatch.from_arrays([ids, vectors, metadata], schema=schema)

# Upload
descriptor = flight.FlightDescriptor.for_path("my_collection")
writer, reader = client.do_put(descriptor, schema)
writer.write_batch(batch)
writer.close()

# Read results
for result in reader:
    print(f"Uploaded {result.body.to_pybytes()} vectors")
----

The implementation builds on ProximaDB's existing Arrow infrastructure and follows established patterns from REST/gRPC integration.
