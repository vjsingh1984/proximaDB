=== Test Patterns

==== Storage Engine Tests
[source,rust]
----
#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::TempDir;
    
    async fn create_test_engine() -> (LsmEngine, TempDir) {
        let temp_dir = TempDir::new().unwrap();
        let config = LsmConfig {
            data_dir: temp_dir.path().to_path_buf(),
            ..Default::default()
        };
        let engine = LsmEngine::new(config).await.unwrap();
        (engine, temp_dir)
    }
    
    #[tokio::test]
    async fn test_atomic_compaction() {
        let (engine, _temp_dir) = create_test_engine().await;
        
        // Insert test data
        let vectors = create_test_vectors(1000);
        engine.insert_batch(&vectors).await.unwrap();
        
        // Trigger compaction
        let result = engine.compact_atomic(0).await.unwrap();
        
        // Verify atomicity
        assert!(result.files_created.len() > 0);
        assert_eq!(result.files_deleted.len(), result.files_created.len());
    }
}
----

import pytest
from proximadb import ProximaDBClient

@pytest.fixture
async def client():
    """Create test client"""
    client = ProximaDBClient("http://localhost:5678", api_version="v2")
    yield client
    # Cleanup
    collections = await client.list_collections()

@pytest.mark.asyncio
async def test_full_workflow(client):
    """Test complete vector workflow"""
    # Create collection
    config = CollectionConfig(
        collection_id="test_workflow",
        vector_dimension=128,
        distance_metric="cosine",
        storage_engine="lsm"
    )
    collection = await client.create_collection(config)
    assert collection.id == "test_workflow"
    
    # Insert vectors
    vectors = [
        VectorRecord(
            id=f"vec_{i}",
            vector=[0.1 * i] * 128,
            metadata=[
                MetadataItem(key="category", value=f"cat_{i % 3}"),
                MetadataItem(key="priority", value=str(i))
            ]
        )
        for i in range(100)
    ]
    result = await client.insert_vectors("test_workflow", vectors)
    assert result.vectors_inserted == 100
    
    # Search vectors
    query_vector = [0.5] * 128
    results = await client.search_vectors(
        "test_workflow",
        query_vector,
        k=10,
        filters={"category": "cat_1"}
    )
    assert len(results) <= 10
    assert all(r.metadata.get("category") == "cat_1" for r in results)

=== Testing Best Practices

1. **Test Naming**
   ```rust
   // Good: Descriptive and specific
   #[test]
   fn test_bloom_filter_false_positive_rate_under_threshold() { }
   
   // Bad: Generic and unclear
   #[test]
   fn test_bloom() { }
   ```

2. **Test Independence**
   ```rust
   // Each test creates its own environment
   async fn create_isolated_test_env() -> TestEnv {
       let temp_dir = TempDir::new().unwrap();
       let config = create_test_config(&temp_dir);
       TestEnv::new(config).await
   }
   ```

3. **Property-Based Testing**
   ```rust
   use proptest::prelude::*;
   
   proptest! {
       #[test]
       fn test_vector_serialization_roundtrip(
           vector: Vec<f32>,
           metadata: HashMap<String, String>
       ) {
           let record = VectorRecord::new(vector, metadata);
           let serialized = record.serialize();
           let deserialized = VectorRecord::deserialize(&serialized).unwrap();
           assert_eq!(record, deserialized);
       }
   }
   ```

=== Benchmark Structure

[source,rust]
----
use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn bench_vector_insertion(c: &mut Criterion) {
    let runtime = tokio::runtime::Runtime::new().unwrap();
    
    c.bench_function("insert_1k_vectors", |b| {
        b.iter(|| {
            runtime.block_on(async {
                let vectors = create_test_vectors(1000);
                let service = create_test_service().await;
                
                for batch in vectors.chunks(100) {
                    service.insert_vectors("test", batch).await.unwrap();
                }
            })
        })
    });
}

criterion_group!(benches, bench_vector_insertion);
criterion_main!(benches);

=== Load Testing

[source,python]
----
import asyncio
import time
from concurrent.futures import ThreadPoolExecutor

async def load_test_writes(client, num_vectors, batch_size, num_threads):
    """Load test write performance"""
    start_time = time.time()
    
    async def write_batch(thread_id, start_idx):
        vectors = [
            VectorRecord(
                id=f"vec_{thread_id}_{i}",
                vector=[0.1] * 128
            )
            for i in range(start_idx, start_idx + batch_size)
        ]
        await client.insert_vectors("load_test", vectors)
    
    # Run parallel writes
    tasks = []
    for i in range(0, num_vectors, batch_size):
        thread_id = i // (num_vectors // num_threads)
        task = write_batch(thread_id, i)
        tasks.append(task)
    
    await asyncio.gather(*tasks)
    
    duration = time.time() - start_time
    throughput = num_vectors / duration
    
    print(f"Inserted {num_vectors} vectors in {duration:.2f}s")
    print(f"Throughput: {throughput:.0f} vectors/second")
----

=== Common Debugging Scenarios

1. **Memory Leaks**
   ```rust
   // Add memory tracking
   #[global_allocator]
   static ALLOC: jemallocator::Jemalloc = jemallocator::Jemalloc;
   
   // Monitor memory usage
   let stats = jemalloc_ctl::stats::allocated::read().unwrap();
   println!("Allocated: {} bytes", stats);
   ```

2. **Deadlocks**
   ```rust
   // Use parking_lot for deadlock detection
   use parking_lot::deadlock;
   
   std::thread::spawn(move || {
       loop {
           std::thread::sleep(Duration::from_secs(10));
           let deadlocks = deadlock::check_deadlock();
           if !deadlocks.is_empty() {
               eprintln!("Deadlock detected!");
               for (i, threads) in deadlocks.iter().enumerate() {
                   eprintln!("Deadlock #{}", i);
                   for t in threads {
                       eprintln!("Thread: {:?}", t.thread_id());
                       eprintln!("{:#?}", t.backtrace());
                   }
               }
           }
       }
   });
   ```

* **Discovery**: ProximaDB already uses DashMap (lock-free) throughout
* **Action**: Renamed misleading test names from "lockfree_*" to "*_concurrency"
* **Result**: Clearer test naming and documentation

* **Added**: AND/OR/NOT logical operators with precedence testing
* **Added**: Complex WHERE clause parsing tests
* **Result**: Full SQL feature parity with API

* **Fixed**: WAL directory creation race in `test_batch_write_concurrency`
* **Solution**: Changed from check-then-create to create-first pattern
* **Impact**: Eliminated intermittent test failures
