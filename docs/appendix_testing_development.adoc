=== Test Types

==== Unit Tests
[source,bash]
----
# Run all unit tests
cargo test --lib

# Run specific module
cargo test storage::engines::sst

# Run with output
cargo test -- --nocapture
----

==== Integration Tests
[source,bash]
----
# Run integration tests
cargo test --test '*'

# Run specific integration test
cargo test --test vector_operations

# Run with release mode (faster)
cargo test --release
----

==== Python SDK Tests
[source,bash]
----
cd clients/python
pytest tests/ -v

# With coverage
pytest tests/ --cov=proximadb --cov-report=html
----

== Development Workflow

=== Setup Development Environment

[source,bash]
----
# Clone repository
git clone https://github.com/vjsingh1984/proximadb.git
cd proximadb

# Install dependencies
rustup update stable
cargo install cargo-watch cargo-tarpaulin

# Setup Python environment
cd clients/python
pip install -e ".[dev]"
----

=== Common Tasks

==== Running Development Server
[source,bash]
----
# Watch mode (auto-restart on changes)
cargo watch -x 'run --bin proximadb-server'

# With debug logging
RUST_LOG=debug cargo run --bin proximadb-server

# With custom config
cargo run --bin proximadb-server -- --config dev.toml
----

==== Code Quality Checks
[source,bash]
----
# Format code
cargo fmt

# Run linter
cargo clippy -- -D warnings

# Check dependencies
cargo audit
----

== Debugging Guide

=== Enable Debug Logging

[source,toml]
----
# In config.toml
[server]
log_level = "debug"
log_format = "json"

# Or via environment
RUST_LOG=proximadb=debug,tower_http=debug
----

=== Common Issues

==== Memory Leaks
[source,bash]
----
# Run with memory profiler
cargo build --release
valgrind --leak-check=full ./target/release/proximadb-server
----

==== Performance Issues
[source,bash]
----
# CPU profiling
cargo build --release
perf record -F 99 ./target/release/proximadb-server
perf report

# Flame graphs
cargo flamegraph --bin proximadb-server
----

==== Test Failures
[source,bash]
----
# Run single test with backtrace
RUST_BACKTRACE=1 cargo test test_name -- --exact

# Run tests sequentially
cargo test -- --test-threads=1
----

== Benchmarking

=== Running Benchmarks

[source,bash]
----
# Run all benchmarks
cargo bench

# Run specific benchmark
cargo bench vector_insert

# Compare with baseline
cargo bench -- --save-baseline main
cargo bench -- --baseline main
----

=== Writing Benchmarks

[source,rust]
----
use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn vector_insert_benchmark(c: &mut Criterion) {
    let vectors = create_test_vectors(1000);
    
    c.bench_function("insert 1000 vectors", |b| {
        b.iter(|| {
            insert_vectors(black_box(&vectors))
        });
    });
}

criterion_group!(benches, vector_insert_benchmark);
criterion_main!(benches);
----

== CI/CD Pipeline

=== GitHub Actions

[source,yaml]
----
# .github/workflows/test.yml
name: Tests
on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions-rs/toolchain@v1
      - run: cargo test --all
      - run: cargo fmt --check
      - run: cargo clippy -- -D warnings
----

=== Pre-commit Hooks

[source,bash]
----
# .git/hooks/pre-commit
#!/bin/bash
cargo fmt --check
cargo clippy -- -D warnings
cargo test --lib
----

== Release Process

=== Version Bumping
[source,bash]
----
# Update version in Cargo.toml
cargo set-version 1.0.1

# Update Python SDK version
cd clients/python
poetry version 1.0.1
----

=== Building Release
[source,bash]
----
# Build optimized binary
cargo build --release

# Build Docker image
docker build -t proximadb:latest .

# Build Python wheel
cd clients/python
poetry build
----

=== Testing Release
[source,bash]
----
# Test release binary
./target/release/proximadb-server --version

# Test Docker image
docker run proximadb:latest --version

# Test Python package
pip install dist/proximadb-*.whl
python -c "import proximadb; print(proximadb.__version__)"
----
