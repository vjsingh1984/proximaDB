=== Test Patterns

==== Storage Engine Test Template
[source,rust]
----
#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::TempDir;
    
    async fn create_test_engine() -> (SstEngine, TempDir) {
        let temp_dir = TempDir::new().unwrap();
        let config = SstConfig {
            data_dir: temp_dir.path().to_path_buf(),
            ..Default::default()
        };
        let engine = SstEngine::new(config).await.unwrap();
        (engine, temp_dir)
    }
    
    #[tokio::test]
    async fn test_atomic_compaction() {
        let (engine, _temp_dir) = create_test_engine().await;
        let vectors = create_test_vectors(1000);
        engine.insert_batch(&vectors).await.unwrap();
        
        let result = engine.compact_atomic(0).await.unwrap();
        assert!(result.files_created.len() > 0);
    }
}
----

== Performance Testing

=== Benchmark Structure

[source,rust]
----
use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn bench_vector_insertion(c: &mut Criterion) {
    let runtime = tokio::runtime::Runtime::new().unwrap();
    
    c.bench_function("insert_1k_vectors", |b| {
        b.iter(|| {
            runtime.block_on(async {
                let vectors = create_test_vectors(1000);
                let service = create_test_service().await;
                service.insert_vectors("test", &vectors).await.unwrap();
            })
        })
    });
}

criterion_group!(benches, bench_vector_insertion);
criterion_main!(benches);
----

=== Performance Targets
- **gRPC Insert**: 10-49K vectors/sec (batch size dependent)
- **Search Latency**: gRPC < 2ms, REST 2-5ms, SQL 3-30ms
- **SIMD Distance**: 671M elements/sec (Cosine, 128D)
- **Memory Usage**: <8MB per collection (Bloom filters)

=== Test Scenarios

==== Version Gap Example
[source]
----
Input:  [id="A" v=1], [id="A" v=2], [id="A" v=4]
Result: Keep only v=2 (stop at gap between v=2 and v=4)
----

==== Duplicate Version Resolution
[source]
----
Input:  [id="B" v=1 ts=300], [id="B" v=1 ts=100]
Result: Keep v=1 ts=100 (earliest timestamp wins)
----

== Continuous Integration

=== GitHub Actions Workflow

[source,yaml]
----
name: CI
on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    
    - name: Install Rust
      uses: actions-rs/toolchain@v1
      with:
        toolchain: stable
        components: rustfmt, clippy
    
    - name: Check formatting
      run: cargo fmt -- --check
    
    - name: Run clippy
      run: cargo clippy --all-targets --all-features -- -D warnings
    
    - name: Run tests
      run: cargo test --all-features
----
