=== Test Patterns

==== Storage Engine Tests
[source,rust]
----
#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::TempDir;
    
    async fn create_test_engine() -> (LsmEngine, TempDir) {
        let temp_dir = TempDir::new().unwrap();
        let config = LsmConfig {
            data_dir: temp_dir.path().to_path_buf(),
            ..Default::default()
        };
        let engine = LsmEngine::new(config).await.unwrap();
        (engine, temp_dir)
    }
    
    #[tokio::test]
    async fn test_atomic_compaction() {
        let (engine, _temp_dir) = create_test_engine().await;
        
        // Insert test data
        let vectors = create_test_vectors(1000);
        engine.insert_batch(&vectors).await.unwrap();
        
        // Trigger compaction
        let result = engine.compact_atomic(0).await.unwrap();
        
        // Verify atomicity
        assert!(result.files_created.len() > 0);
        assert_eq!(result.files_deleted.len(), result.files_created.len());
    }
}
----

=== Testing Best Practices

1. **Test Naming**
   ```rust
   // Good: Descriptive and specific
   #[test]
   fn test_bloom_filter_false_positive_rate_under_threshold() { }
   
   // Bad: Generic and unclear
   #[test]
   fn test_bloom() { }
   ```

2. **Test Independence**
   ```rust
   // Each test creates its own environment
   async fn create_isolated_test_env() -> TestEnv {
       let temp_dir = TempDir::new().unwrap();
       let config = create_test_config(&temp_dir);
       TestEnv::new(config).await
   }
   ```

3. **Property-Based Testing**
   ```rust
   use proptest::prelude::*;
   
   proptest! {
       #[test]
       fn test_vector_serialization_roundtrip(
           vector: Vec<f32>,
           metadata: HashMap<String, String>
       ) {
           let record = VectorRecord::new(vector, metadata);
           let serialized = record.serialize();
           let deserialized = VectorRecord::deserialize(&serialized).unwrap();
           assert_eq!(record, deserialized);
       }
   }
   ```

=== Load Testing

[source,python]
----
import asyncio
import time
from concurrent.futures import ThreadPoolExecutor

async def load_test_writes(client, num_vectors, batch_size, num_threads):
    """Load test write performance"""
    start_time = time.time()
    
    async def write_batch(thread_id, start_idx):
        vectors = [
            VectorRecord(
                id=f"vec_{thread_id}_{i}",
                vector=[0.1] * 128
            )
            for i in range(start_idx, start_idx + batch_size)
        ]
        await client.insert_vectors("load_test", vectors)
    
    # Run parallel writes
    tasks = []
    for i in range(0, num_vectors, batch_size):
        thread_id = i // (num_vectors // num_threads)
        task = write_batch(thread_id, i)
        tasks.append(task)
    
    await asyncio.gather(*tasks)
    
    duration = time.time() - start_time
    throughput = num_vectors / duration
    
    print(f"Inserted {num_vectors} vectors in {duration:.2f}s")
    print(f"Throughput: {throughput:.0f} vectors/second")

   std::thread::spawn(move || {
       loop {
           std::thread::sleep(Duration::from_secs(10));
           let deadlocks = deadlock::check_deadlock();
           if !deadlocks.is_empty() {
               eprintln!("Deadlock detected!");
               for (i, threads) in deadlocks.iter().enumerate() {
                   eprintln!("Deadlock #{}", i);
                   for t in threads {
                       eprintln!("Thread: {:?}", t.thread_id());
                       eprintln!("{:#?}", t.backtrace());
                   }
               }
           }
       }
   });

==== Known Intermittent Failures

===== 1. Storage Assignment Weighted Distribution Test
**Test**: `integration::isolated_storage_assignment_test::test_isolated_weighted_assignment`

**Issue**: Statistical distribution variance in weighted assignment algorithm
```
Location 2 count 11 too far from expected 20 (tolerance: 8)
```

**Root Cause**: Hash-based distribution can have variance that occasionally exceeds test tolerance
**Status**: Non-critical - algorithm is functionally correct
**Recommendation**: Increase tolerance or use larger sample size

===== 2. SST Core Operations Test  
**Test**: `unit_tests::storage::sst_core_tests::test_lsm_basic_operations`

**Issue**: Race condition in file system operations during concurrent SSTable creation
```
File not found: /tmp/.tmpXXX/test_collection_0/data/test_collection_0_level0_XXX.sst
```

**Root Cause**: Timing issue between file creation and search operation
**Status**: Race condition in test setup - not production code
**Recommendation**: Add proper synchronization barriers in test
