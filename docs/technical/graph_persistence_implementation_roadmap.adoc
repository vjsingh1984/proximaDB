== Executive Summary

This roadmap provides **actionable implementation steps** to restore and enhance ProximaDB's graph functionality following the successful GraphService migration to multi-graph architecture. The work is organized into **three critical phases** with specific deliverables and time estimates.

[CRITICAL]
====
**Current Status**: GraphService migration is complete, but **ALL graph APIs are non-functional** due to missing graph_id parameter propagation. This is the **highest priority** restoration task.
====

// New hybrid coordinator
pub struct HybridQueryCoordinator {
    vector_service: Arc<VectorOperationsService>,
    graph_service: Arc<GraphService>,
    mapping_registry: Arc<DashMap<String, GraphCollectionMapping>>,
}

#[derive(Clone)]
pub struct GraphCollectionMapping {
    pub collection_name: String,
    pub graph_id: String,
    pub node_id_field: String,        // Which field maps to graph nodes
    pub embedding_field: String,      // Which field contains vectors
}

impl HybridQueryCoordinator {
    /// Execute hybrid query combining vector similarity and graph traversal
    pub async fn execute_hybrid_query(
        &self,
        collection: &str,
        query: HybridQuery,
    ) -> Result<Vec<HybridResult>> {
        // 1. Get collection ↔ graph mapping
        let mapping = self.mapping_registry.get(collection)
            .ok_or_else(|| anyhow!("No graph mapping for collection: {}", collection))?;

        // 2. Execute vector similarity
        let vector_results = self.vector_service
            .similarity_search(&query.vector_query)
            .await?;

        // 3. Execute graph traversal for each result
        let mut hybrid_results = Vec::new();
        for vector_result in vector_results {
            if let Some(node_id) = vector_result.metadata.get(&mapping.node_id_field) {
                let graph_context = self.graph_service
                    .traverse(&mapping.graph_id, create_context_query(node_id))
                    .await?;

                hybrid_results.push(HybridResult {
                    vector_result,
                    graph_context,
                    combined_score: calculate_hybrid_score(&vector_result, &graph_context),
                });
            }
        }

        Ok(hybrid_results)
    }
}

// Fix WAL writer integration in OrionPersistence
pub struct OrionPersistence {
    pub graph_id: String,
    pub base_url: String,
    pub filesystem_factory: Arc<FilesystemFactory>,
    pub filesystem: Arc<dyn FileSystem>,
    pub wal_writer: Option<Arc<Mutex<UnifiedWALWriter>>>,  // ← FIX: Add missing field
    pub wal_path: String,
    pub snapshot_path: String,
    pub metadata_path: String,
    pub compression_enabled: bool,
}

impl OrionPersistence {
    pub async fn new(graph_id: String, base_url: String) -> Result<Self> {
        let filesystem_factory = Arc::new(FilesystemFactory::new());
        let filesystem = filesystem_factory.create_filesystem(&base_url).await?;

        // Create WAL writer
        let wal_path = format!("{}/wal", base_url);
        let wal_writer = Some(Arc::new(Mutex::new(
            UnifiedWALWriter::new(&wal_path, &*filesystem).await?
        )));

        Ok(Self {
            graph_id,
            base_url,
            filesystem_factory,
            filesystem,
            wal_writer,
            wal_path,
            snapshot_path: format!("{}/snapshots", base_url),
            metadata_path: format!("{}/metadata", base_url),
            compression_enabled: true,
        })
    }

    pub async fn write_node_operation(&self, node: Node) -> Result<()> {
        if let Some(ref wal_writer) = self.wal_writer {
            // Use the correct GraphOperation from memtable module
            let graph_op = crate::storage::memtable::implementations::graph_memtable::GraphOperation::CreateNode(node);
            let unified_op = UnifiedWALOperation::GraphOp(graph_op);
            wal_writer.lock().await.append(unified_op).await?;
        }
        Ok(())
    }
}
