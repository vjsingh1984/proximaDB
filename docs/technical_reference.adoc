    }

async fn select_search_engine(&self, query: &SearchQuery) -> Vec<Box<dyn SearchEngine>> {
    let mut engines = Vec::new();
    
    // WriteBuffer search for unflushed data
    if self.write_buffer_has_unflushed_data(query.collection_id) {
        engines.push(Box::new(WriteBufferSearchEngine::new(self.write_buffer.clone())));
    }
    
    // VIPER search with two-stage strategy for quantized data
    if self.has_quantized_data(query.collection_id) {
        engines.push(Box::new(TwoStageSearchEngine::new(
            self.distance_compute.clone(),
            self.quantization_engine.clone(),

    // Storage engine selection based on data presence
    if self.viper_has_data(query.collection_id) {
        engines.push(Box::new(ViperSearchEngine::new(self.viper.clone())));
    }
    
    if self.sst_has_data(query.collection_id) {
        engines.push(Box::new(SstSearchEngine::new(self.sst.clone())));
    }
    
    engines
}

=== Index-Based Filtering Framework

ProximaDB implements a sophisticated index-based filtering framework that separates filter evaluation from data reading:

[source,rust]
----
// Core framework in /src/core/search/index_based_filter.rs
pub trait IndexBasedFilter {
    /// Evaluate filter and return qualifying row indices
    async fn evaluate_filter(&self, filter: &FilterExpression, metadata: &MetadataView) -> Result<Vec<usize>>;
    
    /// Apply indices to extract matching records
    async fn apply_indices(&self, indices: &[usize], data: &DataView) -> Result<Vec<VectorRecord>>;
}

// SST Implementation: Three-stage filtering pipeline
impl IndexBasedFilter for SstIndexReader {
    async fn evaluate_filter(&self, filter: &FilterExpression, metadata: &MetadataView) -> Result<Vec<usize>> {
        let bloom_candidates = self.bloom_filter_check(filter)?;
        let block_candidates = self.index_entry_filtering(filter, bloom_candidates)?;
        let row_indices = self.in_memory_row_filtering(filter, block_candidates)?;
        Ok(row_indices)
    }
}

// VIPER Implementation: Column-level predicate pushdown
impl IndexBasedFilter for ViperIndexReader {
    async fn evaluate_filter(&self, filter: &FilterExpression, metadata: &MetadataView) -> Result<Vec<usize>> {
        let filter_columns = self.extract_filter_columns(filter);
        let column_data = self.load_metadata_columns(&filter_columns).await?;
        let indices = self.parallel_column_evaluation(filter, column_data).await?;
        Ok(indices)
    }
}
----

**Engine Strategies**:
- **SST Strategy**: Read entire blocks, filter using indices
- **VIPER Strategy**: Use indices to selectively read parquet rows/columns (60-90% I/O reduction)

**Parallel Filter Execution**:
Complex filter expressions can be executed in parallel across multiple threads with set operations (union/intersection) on indices.

=== MVCC Implementation Details

**Core Rules**:
1. **Version Continuity**: Versions must be sequential (1,2,3...) without gaps
2. **Gap Handling**: Stop processing at first version gap
3. **Duplicate Resolution**: For same ID and version, keep record with earliest timestamp
4. **Tombstone Handling**: Records marked as deleted are removed during compaction
5. **Append-Only Support**: Vectors without IDs bypass all version checks

**Example Behaviors**:
- Version Gap: [id="A" v=1], [id="A" v=2], [id="A" v=4] → Keep only v=2
- Duplicate Version: [id="B" v=1 ts=300], [id="B" v=1 ts=100] → Keep v=1 ts=100
- Append-Only: [id="" v=1], [id="" v=1] → Keep both (no deduplication)

== Multi-Version Concurrency Control (MVCC)

=== Implementation

**VIPER Engine MVCC**:
[source,rust]
----
async fn compact_records(&self, records: Vec<RecordData>) -> Vec<RecordData> {
    let mut records_by_id: HashMap<String, Vec<RecordData>> = HashMap::new();
    
    // Group by ID (skip append-only vectors)
    for record in records {
        if record.id.is_empty() { continue; }
        records_by_id.entry(record.id.clone()).or_default().push(record);
    }
    
    // Process each ID with MVCC rules
    for (id, mut versions) in records_by_id {
        versions.sort_by(|a, b| a.version.cmp(&b.version).then(a.timestamp.cmp(&b.timestamp)));
        
        // Validate version continuity
        let mut expected_version = 1;
        for record in versions {
            if record.version == expected_version {
                // Handle duplicate versions - keep earliest timestamp
                expected_version += 1;
            } else if record.version > expected_version {
                break; // Version gap detected
            }
        }
    }
}
----

=== Metadata-Based File Analysis

**File Size Estimation**:
The compaction system analyzes parquet file metadata to create optimally-sized output files without loading entire datasets:

[source,rust]
----
async fn analyze_files_and_plan_compaction(&self, input_files: &[String], target_file_size_mb: u64) -> Result<CompactionPlan> {
    let mut total_size = 0u64;
    let mut total_rows = 0u64;
    
    // Analyze each file's metadata (fast operation)
    for file_path in input_files {
        let metadata = self.read_parquet_metadata(file_path).await?;
        total_size += metadata.file_size;
        total_rows += metadata.num_rows;
    }
    
    // Calculate optimal output file count
    let target_size_bytes = target_file_size_mb * 1024 * 1024;
    let target_file_count = (total_size / target_size_bytes).max(1);
    let rows_per_file = total_rows / target_file_count;
    
    Ok(CompactionPlan { input_files: input_files.to_vec(), target_file_count, rows_per_file, estimated_output_size: total_size })
}

== Configuration for Cost Optimization

[source,toml]
----
[storage.compaction]
max_files_before_compaction = 5
min_compaction_interval_secs = 60
target_file_size_mb = 128

[storage.write_buffer_config]
memory_flush_size_bytes = 16777216  # 16MB per collection
write_buffer_size_mb = 8192         # 8GB global limit
vector_count_threshold = 10000      # 10K vectors per collection

[performance.optimization]
enable_simd = true
enable_gpu_acceleration = true
batch_size_optimization = true
cache_enabled = true
compression_level = 6  # ZSTD compression level
----

== Operational Guide

=== Deployment

ProximaDB supports multiple deployment models:

* **Single node**: Development and testing
* **Distributed**: Production workloads
* **Cloud native**: Kubernetes operators
* **Multi-region**: Global deployments

See link:architecture.adoc#deployment-architecture[Deployment Architecture] for detailed guidance.

=== Monitoring

Built-in monitoring includes:

* Prometheus metrics endpoint at `/metrics`
* OpenTelemetry tracing integration
* Health check endpoints (`/health`, `/ready`)
* Performance dashboards

=== Maintenance

Regular maintenance tasks:

1. **Compaction**: Automatic via CompactionCoordinator or manual triggers
2. **Backup**: Incremental snapshots of collections and metadata
3. **Upgrades**: Rolling updates with zero-downtime deployment
4. **Monitoring**: Alert configuration for performance thresholds
