**Critical Issues Identified**:
- **Tenant Isolation**: Global store mixes data from different tenants
- **RBAC Violations**: No access control at entity/relationship level
- **Compliance Risk**: Audit trails and provenance mixed across tenants
- **Performance Bottleneck**: Single global lock for all operations
- **Domain Confusion**: Entities from different domains mixed together

== 💼 Implementation Strategy

=== Phase 1: Core Architecture Implementation

```rust
/// Multi-tenant knowledge graph coordinator
pub struct TenantKnowledgeGraph {
    tenant_id: String,
    domain_graphs: Arc<DashMap<String, DomainKnowledgeGraph>>,
    rbac_policy: Arc<TenantRBACPolicy>,
    security_context: SecurityContext,
    audit_logger: Arc<TenantAuditLogger>,
}

impl TenantKnowledgeGraph {
    /// Create or get domain-specific knowledge graph
    pub async fn get_or_create_domain(&self, domain_name: &str) -> Result<Arc<DomainKnowledgeGraph>> {
        if let Some(domain) = self.domain_graphs.get(domain_name) {
            Ok(domain.clone())
        } else {
            let domain_graph = DomainKnowledgeGraph::new(
                format!("{}::{}", self.tenant_id, domain_name),
                self.rbac_policy.clone(),
                self.audit_logger.clone(),
            ).await?;
            
            let domain_arc = Arc::new(domain_graph);
            self.domain_graphs.insert(domain_name.to_string(), domain_arc.clone());
            Ok(domain_arc)
        }
    }
    
    /// Cross-domain query with RBAC validation
    pub async fn cross_domain_query(
        &self,
        query: CrossDomainQuery,
        user_context: &UserContext,
    ) -> Result<CrossDomainResult> {
        // Validate user has access to all requested domains
        for domain_name in &query.domains {
            self.rbac_policy.validate_domain_access(user_context, domain_name)?;
        }
        
        // Execute query across authorized domains
        let mut results = Vec::new();
        for domain_name in query.domains {
            if let Some(domain) = self.domain_graphs.get(&domain_name) {
                let domain_result = domain.execute_query(&query, user_context).await?;
                results.push(domain_result);
            }
        }
        
        // Merge results with provenance tracking
        self.merge_cross_domain_results(results, &query, user_context).await
    }
}

/// Domain-specific knowledge graph
pub struct DomainKnowledgeGraph {
    domain_id: String,
    entity_store: Arc<DomainEntityStore>,
    relationship_store: Arc<DomainRelationshipStore>,
    provenance_store: Arc<DomainProvenanceStore>,
    collection_bridges: Arc<DashMap<String, CollectionEntityBridge>>,
    domain_schema: DomainSchema,
    rbac_filter: Arc<DomainRBACFilter>,
}

impl DomainKnowledgeGraph {
    /// Link a collection to this domain's knowledge graph
    pub async fn link_collection(
        &self,
        collection_id: &str,
        link_policy: CollectionLinkPolicy,
        user_context: &UserContext,
    ) -> Result<()> {
        // Validate user can link collections to this domain
        self.rbac_filter.validate_collection_link(user_context, collection_id)?;
        
        let bridge = CollectionEntityBridge::new(
            collection_id.to_string(),
            self.domain_id.clone(),
            link_policy,
            self.rbac_filter.clone(),
        ).await?;
        
        self.collection_bridges.insert(collection_id.to_string(), bridge);
        
        // Log the link operation for audit
        self.provenance_store.log_collection_link(
            collection_id,
            &self.domain_id,
            user_context,
        ).await?;
        
        Ok(())
    }
    
    /// Query entities with RBAC filtering
    pub async fn query_entities(
        &self,
        query: EntityQuery,
        user_context: &UserContext,
    ) -> Result<Vec<Entity>> {
        // Apply RBAC filtering to query
        let filtered_query = self.rbac_filter.filter_entity_query(&query, user_context)?;
        
        // Execute query on domain entity store
        let results = self.entity_store.query_entities(&filtered_query).await?;
        
        // Apply field-level filtering based on user permissions
        let filtered_results = self.rbac_filter.filter_entity_results(results, user_context)?;
        
        // Log access for audit
        self.provenance_store.log_entity_access(&query, user_context, filtered_results.len()).await?;
        
        Ok(filtered_results)
    }
}
```

== 🔄 Migration Strategy

=== Phase 1: Tenant-Aware Architecture

```rust
// New tenant-aware entity store factory
pub struct EntityStoreFactory;

impl EntityStoreFactory {
    /// Create tenant-specific knowledge graph
    pub async fn create_tenant_knowledge_graph(
        tenant_id: &str,
        tenant_config: TenantConfig,
        rbac_policy: TenantRBACPolicy,
    ) -> Result<TenantKnowledgeGraph> {
        let knowledge_graph = TenantKnowledgeGraph {
            tenant_id: tenant_id.to_string(),
            domain_graphs: Arc::new(DashMap::new()),
            rbac_policy: Arc::new(rbac_policy),
            security_context: SecurityContext::for_tenant(tenant_id),
            audit_logger: Arc::new(TenantAuditLogger::new(tenant_id)),
        };
        
        // Create default domains based on tenant configuration
        for domain_config in tenant_config.default_domains {
            knowledge_graph.get_or_create_domain(&domain_config.name).await?;
        }
        
        Ok(knowledge_graph)
    }
    
    /// Migrate from global entity store
    pub async fn migrate_from_global_store(
        global_store: &ProximaEntityStore,
        tenant_mapping: TenantMigrationMapping,
    ) -> Result<Vec<TenantKnowledgeGraph>> {
        let mut tenant_graphs = Vec::new();
        
        for (tenant_id, domain_mapping) in tenant_mapping.tenant_domains {
            let tenant_graph = Self::create_tenant_knowledge_graph(
                &tenant_id,
                domain_mapping.config,
                domain_mapping.rbac_policy,
            ).await?;
            
            // Migrate entities to appropriate domains
            for (entity_id, domain_assignment) in domain_mapping.entity_assignments {
                if let Some(entity) = global_store.get_entity_raw(&entity_id).await? {
                    let domain = tenant_graph.get_or_create_domain(&domain_assignment.domain_name).await?;
                    domain.entity_store.migrate_entity(entity, domain_assignment.permissions).await?;
                }
            }
            
            tenant_graphs.push(tenant_graph);
        }
        
        Ok(tenant_graphs)
    }
}
```

=== Phase 2: Domain Composition

```rust
/// Complex cross-domain knowledge composition
pub struct KnowledgeComposer {
    tenant_graph: Arc<TenantKnowledgeGraph>,
    composition_rules: CompositionRules,
}

impl KnowledgeComposer {
    /// Compose knowledge across multiple domains
    pub async fn compose_knowledge(
        &self,
        composition_query: CompositionQuery,
        user_context: &UserContext,
    ) -> Result<ComposedKnowledge> {
        let mut composed_knowledge = ComposedKnowledge::new();
        
        // Validate access to all required domains
        for domain_name in &composition_query.required_domains {
            self.tenant_graph.rbac_policy.validate_domain_access(user_context, domain_name)?;
        }
        
        // Execute domain-specific queries
        for domain_query in composition_query.domain_queries {
            let domain = self.tenant_graph.get_domain(&domain_query.domain_name)?;
            let domain_result = domain.execute_contextualized_query(&domain_query, user_context).await?;
            composed_knowledge.add_domain_result(domain_result);
        }
        
        // Apply composition rules to create unified knowledge
        self.composition_rules.compose(&mut composed_knowledge)?;
        
        // Track provenance across domains
        self.track_cross_domain_provenance(&composed_knowledge, user_context).await?;
        
        Ok(composed_knowledge)
    }
}
```
