== Prerequisites

* **Docker**: 20.10+ with Docker Compose v2
* **Memory**: 4GB minimum (8GB recommended)
* **Ports**: 5678 (REST), 5679 (gRPC)

== Installation

=== Docker (Recommended)

[source,bash]
----
# Clone and start
git clone https://github.com/vjsingh1984/proximadb.git
cd proximadb
docker compose up -d

# Verify
curl http://localhost:5678/health
# Expected: {"status": "healthy", "version": "1.0.0"}
----

=== From Source

[source,bash]
----
# Install Rust
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh

# Build and run
git clone https://github.com/vjsingh1984/proximadb.git
cd proximadb
cargo build --release
cargo run --release --bin proximadb-server
----

== Quick Start Example

=== 1. Install Python Client

[source,bash]
----
pip install proximadb-client
----

=== 2. Basic Connection

[source,python]
----
from proximadb import ProximaDBClient

# Connect to ProximaDB (default ports: REST 5678, gRPC 5679)
client = ProximaDBClient("http://localhost:5678")

# Or use gRPC for better performance
client = ProximaDBClient("grpc://localhost:5679")

# Verify connection
health = client.health_check()
print(f"ProximaDB status: {health['status']}")
----

=== 3. Create a Collection

[source,python]
----
from proximadb import ProximaDBClient

# Connect to ProximaDB
client = ProximaDBClient("http://localhost:5678")

# Create a collection optimized for your use case
collection = client.create_collection(
    name="products",
    dimension=1536,        # OpenAI ada-002 embeddings dimension
    storage_engine="viper" # Columnar for analytics workloads
)

print(f"Collection created: {collection.id}")

# Alternative: Create with SST engine for write-heavy workloads
collection_sst = client.create_collection(
    name="user_sessions",
    dimension=768,         # BERT embeddings dimension
    storage_engine="sst"   # Row-based for high-throughput writes
)
----

=== 5. Search Vectors

[source,python]
----
# Basic semantic search
query_vector = [0.2, 0.3, 0.1, ...]  # Your query embedding

results = client.search_vectors(
    collection_id="products",
    query_vector=query_vector,
    top_k=10
)

# Display results
for result in results['results']:
    print(f"ID: {result['id']}")
    print(f"Score: {result['score']:.4f}")
    print(f"Metadata: {result['metadata']}")
    print("---")
----

=== 6. Advanced Search with Filtering

[source,python]
----
# Search with metadata filtering
results = client.search_vectors(
    collection_id="products",
    query_vector=query_vector,
    metadata_filter={
        "category": "Electronics",
        "price": {"$lt": 200},
        "in_stock": True
    },
    top_k=5
)

# Search with multiple filters
complex_filter = {
    "$and": [
        {"category": {"$in": ["Electronics", "Accessories"]}},
        {"price": {"$gte": 50, "$lte": 500}},
        {"rating": {"$gt": 4.0}}
    ]
}

filtered_results = client.search_vectors(
    collection_id="products",
    query_vector=query_vector,
    metadata_filter=complex_filter,
    top_k=10
)
----

== Production Examples

=== E-commerce Recommendation System

[source,python]
----
import numpy as np
from proximadb import ProximaDBClient
from sentence_transformers import SentenceTransformer

# Initialize embedding model
model = SentenceTransformer('all-MiniLM-L6-v2')
client = ProximaDBClient("http://localhost:5678")

# Create product catalog collection
client.create_collection(
    name="product_catalog",
    dimension=384,
    storage_engine="viper",  # Optimized for analytics
    index_type="hnsw",       # Fast approximate search
    compression={             # Enable compression
        "enabled": True,
        "type": "quantization",
        "level": "INT8"
    }
)

# Index products
products = [
    {"id": "SKU001", "name": "Wireless Mouse", "desc": "Ergonomic wireless mouse with RGB lighting"},
    {"id": "SKU002", "name": "Mechanical Keyboard", "desc": "Cherry MX switches, RGB backlit"},
    # ... more products
]

for product in products:
    # Generate embedding from product description
    embedding = model.encode(product['desc']).tolist()
    
    client.insert_vector(
        collection_id="product_catalog",
        vector_id=product['id'],
        vector=embedding,
        metadata={
            "name": product['name'],
            "description": product['desc']
        }
    )

# Find similar products
query = "gaming accessories with lights"
query_embedding = model.encode(query).tolist()

recommendations = client.search_vectors(
    collection_id="product_catalog",
    query_vector=query_embedding,
    top_k=5
)

print("Recommended products:")
for rec in recommendations['results']:
    print(f"- {rec['metadata']['name']}: {rec['score']:.2f}")
----

=== RAG (Retrieval-Augmented Generation) System

[source,python]
----
from proximadb import ProximaDBClient
import openai

client = ProximaDBClient("grpc://localhost:5679")  # Use gRPC for speed

# Create knowledge base collection
client.create_collection(
    name="knowledge_base",
    dimension=1536,           # OpenAI ada-002 dimension
    storage_engine="sst",     # Optimized for writes
    index_type="ivf",         # Good for large datasets
    compression={
        "enabled": True,
        "type": "pq",         # Product quantization
        "segments": 8         # 8-bit PQ
    }
)

# Function to chunk and index documents
def index_document(doc_id, content, chunk_size=500):
    # Split into chunks
    chunks = [content[i:i+chunk_size] 
              for i in range(0, len(content), chunk_size)]
    
    vectors = []
    for i, chunk in enumerate(chunks):
        # Generate embedding
        response = openai.Embedding.create(
            model="text-embedding-ada-002",
            input=chunk
        )
        embedding = response['data'][0]['embedding']
        
        vectors.append({
            "id": f"{doc_id}_chunk_{i}",
            "vector": embedding,
            "metadata": {
                "doc_id": doc_id,
                "chunk_index": i,
                "text": chunk
            }
        })
    
    # Bulk insert for efficiency
    client.insert_vectors("knowledge_base", vectors)

# RAG query function
def rag_query(question):
    # Get question embedding
    response = openai.Embedding.create(
        model="text-embedding-ada-002",
        input=question
    )
    query_embedding = response['data'][0]['embedding']
    
    # Retrieve relevant context
    results = client.search_vectors(
        collection_id="knowledge_base",
        query_vector=query_embedding,
        top_k=5
    )
    
    # Build context from retrieved chunks
    context = "\n".join([
        r['metadata']['text'] 
        for r in results['results']
    ])
    
    # Generate answer with context
    completion = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=[
            {"role": "system", "content": "Answer based on the context provided."},
            {"role": "user", "content": f"Context: {context}\n\nQuestion: {question}"}
        ]
    )
    
    return completion.choices[0].message.content
----

== Configuration

=== Basic config.toml

[source,toml]
----
[server]
node_id = "proximadb-node-1"
bind_address = "0.0.0.0"
port = 5678
data_dir = "/data/proximadb"

[api]
rest_port = 5678
grpc_port = 5679
max_request_size_mb = 64
timeout_seconds = 30

[storage]
# Metadata storage location
metadata_url = "file:///data/proximadb/metadata"

# Storage location for collections
[[storage.storage_locations]]
url = "file:///data/proximadb/storage"
weight = 1
tags = ["local", "primary"]

# Storage configuration
mmap_enabled = true
cache_size_mb = 1024
bloom_filter_bits = 10

[storage.wal_config]
write_buffer_size_mb = 2048
memory_flush_size_bytes = 16777216  # 16MB
sync_mode = "sync"
enable_wal = true

[monitoring]
metrics_enabled = true
log_level = "info"
----

=== Batch Operations for Scale

[source,python]
----
# Efficient batch processing
from proximadb import ProximaDBClient
import numpy as np

client = ProximaDBClient("grpc://localhost:5679")

# Process in batches of 1000 for optimal performance
batch_size = 1000
vectors = []

for i in range(10000):
    vectors.append({
        "id": f"vec_{i}",
        "vector": np.random.rand(384).tolist(),
        "metadata": {"batch": i // batch_size}
    })
    
    # Insert when batch is full
    if len(vectors) >= batch_size:
        client.insert_vectors("my_collection", vectors)
        vectors = []  # Reset batch

# Insert remaining vectors
if vectors:
    client.insert_vectors("my_collection", vectors)
----

=== Connection Pooling

[source,python]
----
# Use connection pooling for concurrent operations
from proximadb import ProximaDBClient
from concurrent.futures import ThreadPoolExecutor

# Create a client pool
clients = [ProximaDBClient("grpc://localhost:5679") for _ in range(10)]

def search_task(client_id, query_id):
    client = clients[client_id % len(clients)]
    query_vector = np.random.rand(384).tolist()
    return client.search_vectors(
        collection_id="my_collection",
        query_vector=query_vector,
        top_k=10
    )

# Parallel searches
with ThreadPoolExecutor(max_workers=10) as executor:
    futures = [executor.submit(search_task, i, i) for i in range(100)]
    results = [f.result() for f in futures]
----

== Monitoring & Observability

=== Health Checks

[source,python]
----
# Monitor ProximaDB health
import time
import requests

def monitor_health():
    while True:
        try:
            response = requests.get("http://localhost:5678/health")
            health = response.json()
            
            if health['status'] != 'healthy':
                print(f"WARNING: ProximaDB unhealthy - {health}")
                # Trigger alert
            
        except Exception as e:
            print(f"ERROR: Cannot reach ProximaDB - {e}")
        
        time.sleep(30)  # Check every 30 seconds
----
