=== Docker Quick Start

[source,bash]
----
# 1. Run ProximaDB with default configuration
docker run -d \
  --name proximadb \
  -p 5678:5678 \
  -p 5679:5679 \
  -v proximadb-data:/data \
  proximadb/proximadb:latest

# 2. Verify service is running
curl http://localhost:5678/health

# 3. Create your first collection
curl -X POST http://localhost:5678/api/v1/collections \
  -H "Content-Type: application/json" \
  -d '{"name": "my_vectors", "dimension": 384}'

# 4. Insert sample vectors
curl -X POST http://localhost:5678/api/v1/collections/my_vectors/vectors \
  -H "Content-Type: application/json" \
  -d '{
    "vectors": [
      {
        "id": "doc1",
        "vector": [0.1, 0.2, 0.3, ...],
        "metadata": {"title": "Sample Document"}
      }
    ]
  }'

# 5. Search vectors
curl -X POST http://localhost:5678/api/v1/collections/my_vectors/search \
  -H "Content-Type: application/json" \
  -d '{
    "vector": [0.1, 0.2, 0.3, ...],
    "k": 10
  }'
----

**Step-by-step:**

[source,bash]
----
# 1. Download and extract
wget https://github.com/vjsingh1984/proximadb/releases/latest/proximadb-linux.tar.gz
tar -xzf proximadb-linux.tar.gz
cd proximadb

# 2. Create basic configuration
cat > config.toml << EOF
[server]
rest_port = 5678
grpc_port = 5679

[storage]
data_path = "./data"
default_engine = "sst"

[cache]
total_memory_mb = 4096
EOF

# 3. Start ProximaDB
./proximadb-server --config config.toml

# 4. Verify in another terminal
curl http://localhost:5678/health
----

== Client Integration

=== Python SDK

[source,python]
----
import proximadb
import numpy as np

# Connect to ProximaDB
client = proximadb.Client(
    host="localhost",
    rest_port=5678,
    grpc_port=5679
)

# Create collection
collection = await client.create_collection(
    name="documents",
    dimension=384,
    distance_metric="cosine"
)

# Insert vectors
vectors = [
    {
        "id": "doc1",
        "vector": np.random.random(384).tolist(),
        "metadata": {"title": "Document 1", "category": "research"}
    },
    {
        "id": "doc2", 
        "vector": np.random.random(384).tolist(),
        "metadata": {"title": "Document 2", "category": "tutorial"}
    }
]

await client.insert(collection="documents", vectors=vectors)

# Search similar vectors
results = await client.search(
    collection="documents",
    query_vector=np.random.random(384).tolist(),
    k=5,
    filter={"category": "research"}
)

print(f"Found {len(results)} similar documents")
for result in results:
    print(f"- {result['id']}: {result['score']:.3f}")
----

=== gRPC Integration

[source,rust]
----
use proximadb::proto::proximadb_v1::{
    vector_service_client::VectorServiceClient,
    VectorSearchRequest, VectorQuery
};
use tonic::transport::Channel;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // Connect to gRPC service
    let channel = Channel::from_static("http://localhost:5679").connect().await?;
    let mut client = VectorServiceClient::new(channel);
    
    // Create search request
    let request = tonic::Request::new(VectorSearchRequest {
        collection_id: "documents".to_string(),
        query: Some(VectorQuery {
            vector: vec![0.1, 0.2, 0.3], // Your query vector
            k: 5,
            distance_metric: "cosine".to_string(),
            ..Default::default()
        }),
        ..Default::default()
    });
    
    // Execute search
    let response = client.vector_search(request).await?;
    let results = response.into_inner();
    
    println!("Search completed: {} results", results.results.len());
    
    Ok(())
}
----
