== API Usage

=== REST API

==== Create Collection
[source,http]
----
POST /collections
{
  "name": "product_vectors",  # Min 8 characters
  "dimension": 1536,
  "distance_metric": "cosine",
  "storage_engine": "viper",   # viper or sst
  "compression": {
    "algorithm": "zstd",
    "level": 6,
    "enable_quantization": true
  }
}
----

==== Insert Vectors
[source,http]
----
POST /collections/product_vectors/vectors
{
  "vectors": [
    {
      "id": "prod_001",
      "vector": [0.1, 0.2, ...],
      "metadata": {
        "category": "electronics",
        "price": 299.99
      }
    }
  ]
}
----

==== Search Vectors
[source,http]
----
POST /collections/product_vectors/search
{
  "vector": [0.1, 0.2, ...],
  "top_k": 10,
  "metadata_filter": {
    "category": "electronics",
    "price": {"$lt": 500}
  },
  "include_metadata": true
}
----

==== Python Client
[source,python]
----
from proximadb import ProximaDBClient, Protocol

# Use gRPC for best performance
client = ProximaDBClient(
    "localhost:5679",
    protocol=Protocol.GRPC
)

# Bulk insert with batching
vectors = [{"id": f"vec_{i}", "vector": [...]} 
           for i in range(10000)]
client.insert_vectors("product_vectors", vectors, batch_size=1000)

# Search with filters
results = client.search_vectors(
    "product_vectors",
    query_vector=[...],
    metadata_filter={"category": "electronics"},
    top_k=10
)
----

=== SQL Interface

[source,sql]
----
-- Create collection
CREATE COLLECTION customer_embeddings (
  dimension=384,
  engine='viper',
  distance_metric='cosine'
);

-- Insert with metadata
INSERT INTO customer_embeddings (id, vector, metadata)
VALUES ('cust_123', [0.1, 0.2, ...], '{"tier": "premium"}');

-- Vector similarity search
SELECT 
  id,
  metadata->>'tier' as tier,
  VECTOR_SIMILARITY(vector, [0.1, 0.2, ...], 'cosine') as score
FROM customer_embeddings
WHERE metadata->>'tier' = 'premium'
ORDER BY score DESC
LIMIT 10;
----

== Storage Engine Selection

=== Decision Matrix
[cols="2,2,2,2", options="header"]
|===
| Workload | Engine | Configuration | Reason

| **High Write Volume**
| SST
| `compression: lz4, level: 1`
| Row-based, optimized for writes

| **Analytics/BI**
| VIPER
| `compression: zstd, quantization: pq8`
| Columnar, high compression

| **Real-time Search**
| SST
| `compression: none`
| Lowest latency

| **Cost-Optimized**
| VIPER
| `compression: zstd, level: 9`
| Maximum compression
|===

=== Engine Implementation
[source,rust]
----
// SST Engine - Row-based
pub struct SstEngine {
    three_stage_filter: FilterPipeline,  // Bloom → Index → Row
    cache: VectorStore,                   // Hot data cache
}

// VIPER Engine - Columnar  
pub struct ViperEngine {
    parquet_reader: ParquetReader,       // Predicate pushdown
    quantization: QuantizationEngine,    // Two-stage search
}
----

== Distance Metrics

=== Metric Selection Guide
[cols="2,3,2", options="header"]
|===
| Metric | Use Case | Formula

| **cosine**
| Text embeddings, normalized vectors
| `1 - (A·B)/(‖A‖‖B‖)`

| **euclidean**
| Spatial data, absolute distances
| `√Σ(Ai - Bi)²`

| **dot_product**
| Recommendation scores
| `Σ(Ai × Bi)`

| **manhattan**
| Sparse data, outlier-robust
| `Σ│Ai - Bi│`

| **hamming**
| Binary vectors, feature hashes
| `Σ(Ai ≠ Bi)`
|===

== Performance Optimization

=== Hardware Acceleration
[source,rust]
----
// Automatic SIMD/GPU detection
let caps = get_hardware_capabilities();

if caps.has_gpu() {
    // 10x faster for large batches
    distance::compute_gpu(&vectors, &query)
} else if caps.has_avx512() {
    // 4x faster with SIMD
    distance::compute_simd(&vectors, &query)
} else {
    // Fallback to scalar
    distance::compute_scalar(&vectors, &query)
}
----

=== Batch Processing
[source,rust]
----
// Optimal batch sizes
const SIMD_BATCH_SIZE: usize = 256;      // AVX-512 optimal
const GPU_BATCH_SIZE: usize = 10_000;    // GPU efficiency threshold
const DISK_BATCH_SIZE: usize = 1_000;    // I/O optimization

// Adaptive batching
impl BatchProcessor {
    fn select_batch_size(&self) -> usize {
        match self.hardware {
            Hardware::GPU => GPU_BATCH_SIZE,
            Hardware::SIMD => SIMD_BATCH_SIZE,
            _ => DISK_BATCH_SIZE,
        }
    }
}
----

== Error Handling

=== Common Errors
[source,rust]
----
use proximadb::error::ProximaError;

match operation_result {
    Err(ProximaError::CollectionNotFound(name)) => {
        // Auto-create collection
        create_collection(&name, default_config())?;
    },
    Err(ProximaError::DimensionMismatch { expected, actual }) => {
        // Validate before retry
        validate_dimensions(expected, actual)?;
    },
    Err(ProximaError::StorageError(e)) => {
        // Check storage health
        if e.is_disk_full() {
            trigger_compaction()?;
        }
    },
    Ok(result) => process_result(result),
}
----
