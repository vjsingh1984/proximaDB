== Executive Summary

This document presents the comprehensive, unified AXIS tiering architecture that optimizes for both performance and cost across diverse deployment scenarios, leveraging storage-specific characteristics (NVMe vs network paths, EBS vs instance store bandwidth) with configurable durability guarantees.

pub struct UnifiedTierManager {
    /// FileSystem manager handles multi-cloud abstraction
    fs_manager: Arc<FileSystemManager>,
    
    /// Tier configurations map to FileSystem instances
    tier_filesystems: HashMap<TierLevel, Arc<dyn FileSystemAPI>>,
    
    /// Rest of the tier management logic
    // ...

impl UnifiedTierManager {
    pub async fn from_config(config: &AxisConfig) -> Result<Self> {
        let fs_manager = FileSystemManager::from_config(&config.filesystem)?;
        
        // Map tiers to appropriate FileSystem implementations
        let mut tier_filesystems = HashMap::new();
        
        // Memory tier doesn't use FileSystem
        // L0: Memory (in-process)
        
        // L1: Fast local storage (NVMe/SSD)
        tier_filesystems.insert(
            TierLevel::L1_NVMe,
            fs_manager.get_filesystem("local:///mnt/nvme/indexes")?,
        );
        
        // L2: Network-attached storage (EBS/Persistent Disk/Managed Disk)
        tier_filesystems.insert(
            TierLevel::L2_NetworkDisk,
            fs_manager.get_filesystem("local:///mnt/persistent/indexes")?,
        );
        
        // L3: Hot cloud storage (S3 Express/GCS Standard/Blob Hot)
        let l3_url = match config.cloud_provider {
            CloudProvider::AWS => "s3://indexes-hot/",
            CloudProvider::GCP => "gs://indexes-standard/",
            CloudProvider::Azure => "az://indexes-hot/",
        };
        tier_filesystems.insert(
            TierLevel::L3_CloudHot,
            fs_manager.get_filesystem(l3_url)?,
        );
        
        // L4: Cool cloud storage (S3 Standard/GCS Nearline/Blob Cool)
        let l4_url = match config.cloud_provider {
            CloudProvider::AWS => "s3://indexes-standard/",
            CloudProvider::GCP => "gs://indexes-nearline/",
            CloudProvider::Azure => "az://indexes-cool/",
        };
        tier_filesystems.insert(
            TierLevel::L4_CloudCool,
            fs_manager.get_filesystem(l4_url)?,
        );
        
        // L5: Archive (S3 Glacier/GCS Archive/Blob Archive)
        let l5_url = match config.cloud_provider {
            CloudProvider::AWS => "s3://indexes-glacier/",
            CloudProvider::GCP => "gs://indexes-archive/",
            CloudProvider::Azure => "az://indexes-archive/",
        };
        tier_filesystems.insert(
            TierLevel::L5_Archive,
            fs_manager.get_filesystem(l5_url)?,
        );
        
        Ok(Self {
            fs_manager,
            tier_filesystems,
            // ...
        })
    }
}

=== Bandwidth-Aware I/O Scheduling

[source,rust]
----
pub struct BandwidthAwareScheduler {
    /// Track bandwidth per storage tier
    bandwidth_trackers: HashMap<TierLevel, BandwidthTracker>,
    
    /// Network vs local path optimization
    path_optimizer: PathOptimizer,
}

impl BandwidthAwareScheduler {
    /// Schedule I/O based on available bandwidth and path
    pub async fn schedule_io(
        &self,
        operation: IoOperation,
        tier: &StorageTier,
    ) -> Result<IoHandle> {
        // Check if we should use local path vs network
        let use_local = self.should_use_local_path(&operation, tier).await;
        
        if use_local && tier.network_path {
            // Try to cache locally first
            if let Some(local_tier) = self.find_local_tier() {
                self.stage_to_local(operation.clone(), local_tier).await?;
                return self.execute_from_local(operation).await;
            }
        }
        
        // Calculate optimal I/O size based on bandwidth
        let optimal_io_size = self.calculate_optimal_io_size(tier);
        
        // For network paths (EBS, S3), batch operations
        if tier.network_path {
            self.batch_network_io(operation, optimal_io_size).await
        } else {
            // Direct local I/O
            self.execute_local_io(operation, optimal_io_size).await
        }
    }
    
    fn calculate_optimal_io_size(&self, tier: &StorageTier) -> usize {
        match tier.storage_type {
            StorageType::Memory => 8 * 1024 * 1024,      // 8MB chunks
            StorageType::NVMe => 1024 * 1024,            // 1MB chunks
            StorageType::EbsGp3 => 256 * 1024,           // 256KB (network optimal)
            StorageType::S3Express => 5 * 1024 * 1024,   // 5MB (S3 multipart)
            _ => 64 * 1024,                              // 64KB default
        }
    }
    
    async fn should_use_local_path(
        &self,
        operation: &IoOperation,
        tier: &StorageTier,
    ) -> bool {
        // Prefer local path for small, frequent operations
        operation.size_bytes < 1024 * 1024 &&  // <1MB
        operation.frequency > 100 &&            // >100 ops/sec
        tier.network_path                       // Currently using network
    }
}

=== Performance-Optimized Promotion/Demotion

[source,rust]
----
impl UnifiedTierManager {
    /// Promote index with bandwidth optimization
    pub async fn promote_index(
        &self,
        collection_id: &str,
        index_type: IndexType,
        target_tier: TierLevel,
    ) -> Result<()> {
        let current_state = self.get_collection_state(collection_id).await?;
        let source_tier = current_state.get_tier_for_index(&index_type);
        
        // Determine optimal promotion path
        let promotion_path = self.optimizer.find_optimal_path(
            source_tier,
            target_tier,
            &self.tier_config.tiers,
        )?;
        
        info!("📈 Promoting {}/{} via path: {:?}", 
              collection_id, index_type, promotion_path);
        
        // Execute promotion along optimal path
        for (from_tier, to_tier) in promotion_path.windows(2).map(|w| (&w[0], &w[1])) {
            self.move_between_tiers(
                collection_id,
                &index_type,
                from_tier,
                to_tier,
            ).await?;
        }
        
        Ok(())
    }
    
    /// Move index between specific tiers
    async fn move_between_tiers(
        &self,
        collection_id: &str,
        index_type: &IndexType,
        from_tier: &StorageTier,
        to_tier: &StorageTier,
    ) -> Result<()> {
        match (from_tier.storage_type, to_tier.storage_type) {
            // NVMe to Memory (fast local path)
            (StorageType::NVMe, StorageType::Memory) => {
                let data = self.read_from_nvme(collection_id, index_type).await?;
                let index = self.deserialize_index(&data, index_type)?;
                self.load_to_memory(collection_id, index).await?;
            }
            
            // EBS to Memory (network path - optimize)
            (StorageType::EbsGp3, StorageType::Memory) => {
                // Use parallel reads to maximize EBS bandwidth
                let data = self.parallel_read_from_ebs(
                    collection_id,
                    index_type,
                    4,  // 4 parallel streams for gp3
                ).await?;
                let index = self.deserialize_index(&data, index_type)?;
                self.load_to_memory(collection_id, index).await?;
            }
            
            // S3 to NVMe (network to local - stage)
            (StorageType::S3Express, StorageType::NVMe) => {
                // Use S3 multipart download
                let data = self.multipart_download_from_s3(
                    collection_id,
                    index_type,
                    10,  // 10 parallel parts
                ).await?;
                self.write_to_nvme(collection_id, index_type, &data).await?;
            }
            
            // Memory to S3 (local to network - optimize)
            (StorageType::Memory, StorageType::S3Express) => {
                let index = self.get_memory_index(collection_id, index_type)?;
                let data = self.serialize_index(&index)?;
                
                // Use multipart upload for large indexes
                if data.len() > 100 * 1024 * 1024 {  // >100MB
                    self.multipart_upload_to_s3(
                        collection_id,
                        index_type,
                        &data,
                        10,  // 10 parallel parts
                    ).await?;
                } else {
                    self.upload_to_s3(collection_id, index_type, &data).await?;
                }
            }
            
            _ => {
                // Generic path
                let data = self.read_from_tier(from_tier, collection_id, index_type).await?;
                self.write_to_tier(to_tier, collection_id, index_type, &data).await?;
            }
        }
        
        Ok(())
    }
}

=== Cost-Optimized Configuration

[source,rust]
----
pub struct CostOptimizer {
    /// Cost model for each tier
    cost_model: CostModel,
    
    /// Access pattern analyzer
    access_analyzer: AccessPatternAnalyzer,
}

impl CostOptimizer {
    /// Find optimal tier placement for cost
    pub async fn optimize_placement(
        &self,
        collection_id: &str,
        constraints: &CostConstraints,
    ) -> Result<TierPlacement> {
        let access_pattern = self.access_analyzer.analyze(collection_id).await?;
        
        // Calculate cost for each tier
        let tier_costs: Vec<(TierLevel, f32)> = self.cost_model.tiers
            .iter()
            .map(|tier| {
                let storage_cost = self.calculate_storage_cost(tier, collection_id);
                let access_cost = self.calculate_access_cost(tier, &access_pattern);
                let transfer_cost = self.calculate_transfer_cost(tier, &access_pattern);
                
                (tier.level, storage_cost + access_cost + transfer_cost)
            })
            .collect();
        
        // Find tier that meets latency constraint at minimum cost
        let optimal_tier = tier_costs
            .into_iter()
            .filter(|(tier, _)| self.meets_latency_constraint(tier, constraints))
            .min_by(|(_, cost1), (_, cost2)| cost1.partial_cmp(cost2).unwrap())
            .map(|(tier, _)| tier)
            .ok_or_else(|| anyhow!("No tier meets constraints"))?;
        
        Ok(TierPlacement {
            collection_id: collection_id.to_string(),
            recommended_tier: optimal_tier,
            estimated_monthly_cost: self.calculate_monthly_cost(&optimal_tier, collection_id),
            expected_latency_ms: self.estimate_latency(&optimal_tier),
        })
    }
    
    fn calculate_storage_cost(&self, tier: &StorageTier, collection_id: &str) -> f32 {
        let size_gb = self.get_index_size_gb(collection_id);
        size_gb * tier.cost_per_gb_month
    }
    
    fn calculate_access_cost(&self, tier: &StorageTier, pattern: &AccessPattern) -> f32 {
        match tier.storage_type {
            StorageType::S3Express => {
                // $0.0004 per 1000 requests
                let monthly_requests = pattern.requests_per_hour * 24 * 30;
                (monthly_requests / 1000.0) * 0.0004
            }
            StorageType::S3Standard => {
                // $0.0004 per 1000 GET requests
                let monthly_gets = pattern.reads_per_hour * 24 * 30;
                (monthly_gets / 1000.0) * 0.0004
            }
            _ => 0.0,  // No per-request cost for other tiers
        }
    }
    
    fn calculate_transfer_cost(&self, tier: &StorageTier, pattern: &AccessPattern) -> f32 {
        if !tier.network_path {
            return 0.0;  // No transfer cost for local storage
        }
        
        // S3 data transfer: $0.09/GB after 1GB free
        let monthly_transfer_gb = pattern.data_transfer_gb_per_hour * 24 * 30;
        if monthly_transfer_gb > 1.0 {
            (monthly_transfer_gb - 1.0) * 0.09
        } else {
            0.0
        }
    }
}

impl UnifiedTierManager {
    /// HNSW-specific tiering
    pub async fn optimize_hnsw_placement(
        &self,
        collection_id: &str,
    ) -> Result<()> {
        // Layer 0 (entry point) - always memory
        self.ensure_in_memory(collection_id, "hnsw/layers/layer_0.idx").await?;
        
        // Layers 1-2 (frequently traversed) - NVMe
        for layer in 1..=2 {
            let path = format!("hnsw/layers/layer_{}.idx", layer);
            self.ensure_on_nvme(collection_id, &path).await?;
        }
        
        // Layers 3+ (rarely accessed) - HDD or S3
        for layer in 3..self.get_hnsw_levels(collection_id) {
            let path = format!("hnsw/layers/layer_{}.idx", layer);
            let access_freq = self.get_access_frequency(&path).await?;
            
            if access_freq < 10 {  // <10 accesses/hour
                self.move_to_s3_standard(collection_id, &path).await?;
            } else if access_freq < 100 {  // <100 accesses/hour
                self.move_to_hdd(collection_id, &path).await?;
            }
        }
        
        Ok(())
    }
    
    /// IVF-specific tiering
    pub async fn optimize_ivf_placement(
        &self,
        collection_id: &str,
    ) -> Result<()> {
        // Centroids - always memory (small, critical)
        self.ensure_in_memory(collection_id, "ivf/centroids.idx").await?;
        
        // Posting lists - tier based on cluster heat
        let cluster_heats = self.analyze_cluster_access_patterns(collection_id).await?;
        
        for (cluster_id, heat_score) in cluster_heats {
            let path = format!("ivf/postings/cluster_{:04}.idx", cluster_id);
            
            match heat_score {
                90..=100 => self.ensure_in_memory(collection_id, &path).await?,
                70..=89 => self.ensure_on_nvme(collection_id, &path).await?,
                30..=69 => self.ensure_on_ebs(collection_id, &path).await?,
                10..=29 => self.move_to_s3_express(collection_id, &path).await?,
                _ => self.move_to_s3_standard(collection_id, &path).await?,
            }
        }
        
        Ok(())
    }
    
    /// PQ-specific tiering
    pub async fn optimize_pq_placement(
        &self,
        collection_id: &str,
    ) -> Result<()> {
        // Codebooks - always memory (small, frequently accessed)
        self.ensure_in_memory(collection_id, "pq/codebooks.idx").await?;
        
        // Quantized codes - NVMe or memory based on size
        let codes_size = self.get_file_size(collection_id, "pq/codes.idx").await?;
        
        if codes_size < 1_000_000_000 {  // <1GB
            self.ensure_in_memory(collection_id, "pq/codes.idx").await?;
        } else {
            self.ensure_on_nvme(collection_id, "pq/codes.idx").await?;
        }
        
        // Residuals - tier based on refinement usage
        if self.is_refinement_enabled(collection_id) {
            self.ensure_on_nvme(collection_id, "pq/residuals.idx").await?;
        } else {
            self.move_to_s3_standard(collection_id, "pq/residuals.idx").await?;
        }
        
        Ok(())
    }
}

=== Tier-Aware Query Router

----
impl UnifiedTierManager {
    pub async fn execute_query(
        &self,
        query: VectorQuery,
    ) -> Result<QueryResult> {
        let collection_id = &query.collection_id;
        let index_type = self.determine_index_type(collection_id)?;
        
        // Get current tier placement
        let tier_state = self.get_collection_state(collection_id).await?;
        
        // Build query plan based on tier locations
        let query_plan = self.build_query_plan(&query, &tier_state, &index_type)?;
        
        // Execute based on optimization strategy
        match self.tier_config.optimization {
            OptimizationStrategy::PerformanceOptimized { .. } => {
                self.execute_performance_query(query_plan).await
            }
            OptimizationStrategy::CostOptimized { .. } => {
                self.execute_cost_aware_query(query_plan).await
            }
            OptimizationStrategy::Balanced { .. } => {
                self.execute_balanced_query(query_plan).await
            }
            OptimizationStrategy::Custom { ref rules } => {
                self.execute_custom_query(query_plan, rules).await
            }
        }
    }
    
    async fn execute_performance_query(&self, plan: QueryPlan) -> Result<QueryResult> {
        // Aggressively promote to memory if not already there
        for required_component in &plan.required_components {
            if !self.is_in_memory(&required_component).await {
                // Promote immediately for this query
                self.promote_index(
                    &required_component.collection_id,
                    required_component.index_type.clone(),
                    TierLevel::Memory,
                ).await?;
            }
        }
        
        // Execute from memory
        self.execute_from_memory(plan).await
    }
    
    async fn execute_cost_aware_query(&self, plan: QueryPlan) -> Result<QueryResult> {
        // Check if worth promoting based on access pattern
        let worth_promoting = self.analyze_promotion_value(&plan).await?;
        
        if worth_promoting {
            // Promote to cheapest acceptable tier
            let target_tier = self.find_cheapest_acceptable_tier(&plan)?;
            self.promote_to_tier(&plan, target_tier).await?;
            self.execute_from_tier(plan, target_tier).await
        } else {
            // Execute from current location
            self.execute_from_current_tier(plan).await
        }
    }
}

== Conclusion

This unified design provides:

1. **Bandwidth-aware I/O**: Optimizes for NVMe vs network paths
2. **Flexible optimization**: Performance vs cost with configurable balance
3. **Index-specific tiering**: HNSW layers, IVF clusters, PQ components
4. **Deployment adaptability**: Works across on-prem, cloud-managed, cloud-native
5. **Durable storage**: S3 as configurable durability layer
6. **Comprehensive monitoring**: Full observability of tier operations

The architecture maximizes performance when resources are available while gracefully degrading to cost-effective operation under constraints.
