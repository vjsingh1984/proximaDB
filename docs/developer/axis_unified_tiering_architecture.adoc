/// AXIS Tiering Integration Manager - Lightweight integration layer
pub struct AxisTieringIntegration {
    /// Reference to existing unified access pattern tracker
    access_tracker: Arc<AccessPatternTracker>,
    
    /// Reference to existing global tier manager  
    global_tier_manager: Arc<GlobalTierManager>,
    
    /// AXIS-specific index type preferences for tier mapping
    index_type_preferences: AxisIndexPreferences,
}

/// Index-specific tier preferences (extends existing infrastructure)
pub struct AxisIndexPreferences {
    /// HNSW: Memory-preferred (tier level 1-2)
    /// IVF: NVMe-optimized (tier level 2-3) 
    /// LSH: Disk-tolerant (tier level 3-4)
    preferences: HashMap<AxisIndexType, IndexTierPreference>,
}

impl AxisTieringIntegration {
    /// Track index access using existing unified infrastructure
    pub async fn track_index_access(&self, collection_id: &str, index_type: AxisIndexType) {
        // Uses existing AccessPatternTracker - no duplicate tracking
        self.access_tracker.track_access_async(
            collection_id.to_string(),
            CacheType::IndexStructure,
        );
    }
    
    /// Get tier recommendation using existing infrastructure
    pub async fn recommend_tier(&self, ...) -> Result<Option<AxisTierRecommendation>> {
        // Uses existing is_frequently_accessed() - no custom heat scoring
        let is_hot = self.access_tracker.is_frequently_accessed(collection_id, 10).await;
        
        // Uses existing GlobalTierManager rules - no duplicate policy engine
        let rule_policy = self.global_tier_manager.rule_based_policy();
        let tier_level = rule_policy.determine_tier(&workload_pattern, freq, age);
        
        // AXIS-specific: Apply index type preferences to existing tier decision
        let recommended_tier = self.apply_index_preferences(tier_level, index_type);
    }
}

=== Integration Design Principles

The AXIS tiering system is built on these key integration principles:

1. **No Redundancy**: Reuse existing proven components instead of creating duplicates
2. **Existing Infrastructure First**: Leverage GlobalTierManager, AccessPatternTracker, AdaptiveStore
3. **AXIS-Specific Enhancement**: Add index-specific logic without replacing core systems
4. **Collection Granularity**: Move entire collections to maintain consistency
5. **Format Strategy**: Optimize serialization format based on storage tier characteristics

}

impl BandwidthAwareScheduler {
    /// Schedule I/O based on available bandwidth and path
    pub async fn schedule_io(
        &self,
        operation: IoOperation,
        tier: &StorageTier,
    ) -> Result<IoHandle> {
        // Check if we should use local path vs network
        let use_local = self.should_use_local_path(&operation, tier).await;
        
        if use_local && tier.network_path {
            // Try to cache locally first
            if let Some(local_tier) = self.find_local_tier() {
                self.stage_to_local(operation.clone(), local_tier).await?;
                return self.execute_from_local(operation).await;
            }
        }
        
        // Calculate optimal I/O size based on bandwidth
        let optimal_io_size = self.calculate_optimal_io_size(tier);
        
        // For network paths (EBS, S3), batch operations
        if tier.network_path {
            self.batch_network_io(operation, optimal_io_size).await
        } else {
            // Direct local I/O
            self.execute_local_io(operation, optimal_io_size).await
        }
    }
    
    fn calculate_optimal_io_size(&self, tier: &StorageTier) -> usize {
        match tier.storage_type {
            StorageType::Memory => 8 * 1024 * 1024,      // 8MB chunks
            StorageType::NVMe => 1024 * 1024,            // 1MB chunks
            StorageType::EbsGp3 => 256 * 1024,           // 256KB (network optimal)
            StorageType::S3Express => 5 * 1024 * 1024,   // 5MB (S3 multipart)
            _ => 64 * 1024,                              // 64KB default
        }
    }
    
    async fn should_use_local_path(
        &self,
        operation: &IoOperation,
        tier: &StorageTier,
    ) -> bool {
        // Prefer local path for small, frequent operations
        operation.size_bytes < 1024 * 1024 &&  // <1MB
        operation.frequency > 100 &&            // >100 ops/sec
        tier.network_path                       // Currently using network
    }
}

impl UnifiedTierManager {
    /// Promote index with bandwidth optimization
    pub async fn promote_index(
        &self,
        collection_id: &str,
        index_type: IndexType,
        target_tier: TierLevel,
    ) -> Result<()> {
        let current_state = self.get_collection_state(collection_id).await?;
        let source_tier = current_state.get_tier_for_index(&index_type);
        
        // Determine optimal promotion path
        let promotion_path = self.optimizer.find_optimal_path(

            target_tier,
            &self.tier_config.tiers,
        )?;
        
        info!("📈 Promoting {}/{} via path: {:?}", 
              collection_id, index_type, promotion_path);
        
        // Execute promotion along optimal path
        for (from_tier, to_tier) in promotion_path.windows(2).map(|w| (&w[0], &w[1])) {
            self.move_between_tiers(
                collection_id,
                &index_type,
                from_tier,
                to_tier,
            ).await?;
        }
        
        Ok(())
    }
    
    /// Move index between specific tiers
    async fn move_between_tiers(
        &self,
        collection_id: &str,
        index_type: &IndexType,
        from_tier: &StorageTier,
        to_tier: &StorageTier,
    ) -> Result<()> {
        match (from_tier.storage_type, to_tier.storage_type) {
            // NVMe to Memory (fast local path)
            (StorageType::NVMe, StorageType::Memory) => {
                let data = self.read_from_nvme(collection_id, index_type).await?;
                let index = self.deserialize_index(&data, index_type)?;
                self.load_to_memory(collection_id, index).await?;
            }
            
            // EBS to Memory (network path - optimize)
            (StorageType::EbsGp3, StorageType::Memory) => {
                // Use parallel reads to maximize EBS bandwidth
                let data = self.parallel_read_from_ebs(
                    collection_id,
                    index_type,
                    4,  // 4 parallel streams for gp3
                ).await?;
                let index = self.deserialize_index(&data, index_type)?;
                self.load_to_memory(collection_id, index).await?;
            }
            
            // S3 to NVMe (network to local - stage)
            (StorageType::S3Express, StorageType::NVMe) => {
                // Use S3 multipart download
                let data = self.multipart_download_from_s3(
                    collection_id,
                    index_type,
                    10,  // 10 parallel parts
                ).await?;
                self.write_to_nvme(collection_id, index_type, &data).await?;
            }
            
            // Memory to S3 (local to network - optimize)
            (StorageType::Memory, StorageType::S3Express) => {
                let index = self.get_memory_index(collection_id, index_type)?;
                let data = self.serialize_index(&index)?;
                
                // Use multipart upload for large indexes
                if data.len() > 100 * 1024 * 1024 {  // >100MB
                    self.multipart_upload_to_s3(
                        collection_id,
                        index_type,
                        &data,
                        10,  // 10 parallel parts
                    ).await?;
                } else {
                    self.upload_to_s3(collection_id, index_type, &data).await?;
                }
            }
            
            _ => {
                // Generic path
                let data = self.read_from_tier(from_tier, collection_id, index_type).await?;
                self.write_to_tier(to_tier, collection_id, index_type, &data).await?;
            }
        }
        
        Ok(())
    }

pub struct CostOptimizer {
    /// Cost model for each tier
    cost_model: CostModel,
    
    /// Access pattern analyzer
    access_analyzer: AccessPatternAnalyzer,
}

impl CostOptimizer {
    /// Find optimal tier placement for cost
    pub async fn optimize_placement(
        &self,
        collection_id: &str,
        constraints: &CostConstraints,
    ) -> Result<TierPlacement> {
        let access_pattern = self.access_analyzer.analyze(collection_id).await?;
        
        // Calculate cost for each tier
        let tier_costs: Vec<(TierLevel, f32)> = self.cost_model.tiers
            .iter()
            .map(|tier| {
                let storage_cost = self.calculate_storage_cost(tier, collection_id);
                let access_cost = self.calculate_access_cost(tier, &access_pattern);
                let transfer_cost = self.calculate_transfer_cost(tier, &access_pattern);
                
                (tier.level, storage_cost + access_cost + transfer_cost)
            })
            .collect();
        
        // Find tier that meets latency constraint at minimum cost
        let optimal_tier = tier_costs
            .into_iter()
            .filter(|(tier, _)| self.meets_latency_constraint(tier, constraints))
            .min_by(|(_, cost1), (_, cost2)| cost1.partial_cmp(cost2).unwrap())
            .map(|(tier, _)| tier)
            .ok_or_else(|| anyhow!("No tier meets constraints"))?;
        
        Ok(TierPlacement {
            collection_id: collection_id.to_string(),
            recommended_tier: optimal_tier,
            estimated_monthly_cost: self.calculate_monthly_cost(&optimal_tier, collection_id),
            expected_latency_ms: self.estimate_latency(&optimal_tier),
        })
    }
    
    fn calculate_storage_cost(&self, tier: &StorageTier, collection_id: &str) -> f32 {
        let size_gb = self.get_index_size_gb(collection_id);
        size_gb * tier.cost_per_gb_month
    }
    
    fn calculate_access_cost(&self, tier: &StorageTier, pattern: &AccessPattern) -> f32 {
        match tier.storage_type {
            StorageType::S3Express => {
                // $0.0004 per 1000 requests
                let monthly_requests = pattern.requests_per_hour * 24 * 30;
                (monthly_requests / 1000.0) * 0.0004
            }
            StorageType::S3Standard => {
                // $0.0004 per 1000 GET requests
                let monthly_gets = pattern.reads_per_hour * 24 * 30;
                (monthly_gets / 1000.0) * 0.0004
            }
            _ => 0.0,  // No per-request cost for other tiers
        }
    }
    
    fn calculate_transfer_cost(&self, tier: &StorageTier, pattern: &AccessPattern) -> f32 {
        if !tier.network_path {
            return 0.0;  // No transfer cost for local storage
        }
        
        // S3 data transfer: $0.09/GB after 1GB free
        let monthly_transfer_gb = pattern.data_transfer_gb_per_hour * 24 * 30;
        if monthly_transfer_gb > 1.0 {
            (monthly_transfer_gb - 1.0) * 0.09
        } else {
            0.0
        }
    }
}

            }
            OptimizationStrategy::Custom { ref rules } => {
                self.execute_custom_query(query_plan, rules).await
            }
        }
    }
    
    async fn execute_performance_query(&self, plan: QueryPlan) -> Result<QueryResult> {
        // Aggressively promote to memory if not already there
        for required_component in &plan.required_components {
            if !self.is_in_memory(&required_component).await {
                // Promote immediately for this query
                self.promote_index(
                    &required_component.collection_id,
                    required_component.index_type.clone(),
                    TierLevel::Memory,
                ).await?;
            }
        }
        
        // Execute from memory
        self.execute_from_memory(plan).await
    }
    
    async fn execute_cost_aware_query(&self, plan: QueryPlan) -> Result<QueryResult> {
        // Check if worth promoting based on access pattern
        let worth_promoting = self.analyze_promotion_value(&plan).await?;
        
        if worth_promoting {
            // Promote to cheapest acceptable tier
            let target_tier = self.find_cheapest_acceptable_tier(&plan)?;
            self.promote_to_tier(&plan, target_tier).await?;
            self.execute_from_tier(plan, target_tier).await
        } else {
            // Execute from current location
            self.execute_from_current_tier(plan).await
        }
    }
}

impl MultiZoneTierManager {
    /// Place index with zone affinity
    pub async fn place_index_with_affinity(
        &self,
        collection_id: &str,
        index: &dyn AxisVectorIndex,
    ) -> Result<IndexPlacement> {
        // Check collection affinity
        let preferred_zone = self.get_preferred_zone(collection_id)?;
        
        // Determine storage location based on heat and affinity
        let heat_score = self.calculate_heat_score(collection_id);
        
        let placement = match heat_score {
            90..=100 => {
                // Very hot: replicate across zones in memory
                IndexPlacement::MultiZone {
                    primary: preferred_zone.clone(),
                    replicas: self.get_replica_zones(&preferred_zone),
                    tier: TierLevel::Memory,
                }
            }
            70..=89 => {
                // Hot: zone-local EBS/PD
                IndexPlacement::ZoneLocal {
                    zone: preferred_zone,
                    storage: StorageLocation::ZoneLocal {
                        volume_id: self.get_zone_volume(&preferred_zone)?,
                        mount_path: self.get_mount_path(&preferred_zone)?,
                    },
                    tier: TierLevel::L2_NetworkDisk,
                }
            }
            30..=69 => {
                // Warm: regional S3/GCS
                IndexPlacement::Regional {
                    region: self.get_region(&preferred_zone)?,
                    bucket: self.get_regional_bucket()?,
                    tier: TierLevel::L3_CloudHot,
                }
            }
            _ => {
                // Cold: cross-region archive
                IndexPlacement::Archive {
                    primary_region: self.get_primary_region()?,
                    tier: TierLevel::L5_Archive,
                }
            }
        };
        
        // Update metadata with placement
        self.regional_metadata.update_placement(
            collection_id,
            &placement,
        ).await?;
        
        Ok(placement)
    }
    
    /// Handle zone failure
    pub async fn handle_zone_failure(
        &self,
        failed_zone: &AvailabilityZone,
    ) -> Result<()> {
        warn!("⚠️ Zone failure detected: {:?}", failed_zone);
        
        // Get affected collections
        let affected = self.regional_metadata
            .get_collections_in_zone(failed_zone)
            .await?;
        
        // Failover based on strategy
        match self.affinity_config.failover_mode {
            FailoverMode::Automatic => {
                // Immediate failover to replicas
                for collection_id in affected {
                    self.failover_collection(&collection_id, failed_zone).await?;
                }
            }
            FailoverMode::Manual => {
                // Alert and wait for manual intervention
                self.alert_zone_failure(failed_zone, &affected).await?;
            }
            FailoverMode::ReadOnly => {
                // Switch to read-only from replicas
                for collection_id in affected {
                    self.enable_read_only_mode(&collection_id).await?;
                }
            }
        }
        
        Ok(())
    }
}

        let metadata = CollectionMetadata {
            collection_id: collection_id.to_string(),

            version: self.increment_version(collection_id).await?,

        for cache in self.zone_caches.values() {

        collection_id: &str,
        requester_zone: &str,

        // Check for zone-local replica
        if let Some(local_replica) = self.find_zone_local_replica(
            &metadata.storage_location,
            requester_zone,
        ) {
            return Ok(local_replica);
        }
        
        // Check for same-region replica
        if let Some(regional_replica) = self.find_regional_replica(
            &metadata.storage_location,
            requester_zone,
        ) {
            return Ok(regional_replica);
        }
        
        // Return primary location
        Ok(metadata.storage_location)

impl MultiZoneTierManager {
    /// Route query to optimal zone
    pub async fn route_query(
        &self,
        query: VectorQuery,
        client_zone: Option<String>,
    ) -> Result<QueryResult> {
        let collection_id = &query.collection_id;
        
        // Determine query zone preference
        let target_zone = if let Some(zone) = client_zone {
            // Client-specified zone preference
            zone
        } else {
            // Auto-detect optimal zone
            self.find_optimal_query_zone(collection_id).await?
        };
        
        // Check if index is available in target zone
        if let Some(zone_manager) = self.zone_managers.get(&target_zone) {
            if zone_manager.has_index(collection_id).await {
                // Execute locally in zone
                return zone_manager.execute_query(query).await;
            }
        }
        
        // Fallback to cross-zone query
        self.execute_cross_zone_query(query, &target_zone).await
    }
    
    async fn execute_cross_zone_query(
        &self,
        query: VectorQuery,
        preferred_zone: &str,
    ) -> Result<QueryResult> {
        // Find zones with the index
        let available_zones = self.find_zones_with_index(&query.collection_id).await?;
        
        // Sort by network distance from preferred zone
        let sorted_zones = self.sort_by_network_distance(available_zones, preferred_zone);
        
        // Try zones in order of proximity
        for zone in sorted_zones {
            if let Ok(result) = self.zone_managers[&zone]
                .execute_query(query.clone())
                .await
            {
                return Ok(result);
            }
        }
        
        Err(anyhow!("No zone could execute query"))
    }
}
