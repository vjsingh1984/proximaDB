== Executive Summary

This document presents a unified, configuration-driven tiering design that adapts to deployment context: on-premise (persistent disk), cloud-managed (EBS/persistent disks), or cloud-native (ephemeral instances).

=== Adaptive Tier Manager

[source,rust]
----
pub struct AdaptiveTierManager {
    deployment_mode: DeploymentMode,
    tier_strategy: Box<dyn TierStrategy>,
    persistence_layer: Box<dyn PersistenceLayer>,
}

#[derive(Debug, Clone, PartialEq)]
pub enum DeploymentMode {
    OnPremise {
        disk_persistence: bool,
        raid_config: Option<RaidConfig>,
    },
    CloudManaged {
        volume_type: VolumeType,  // EBS, GCE PD, Azure Disk
        snapshot_enabled: bool,
        multi_attach: bool,
    },
    CloudNative {
        instance_type: InstanceType,  // Spot, OnDemand, Reserved
        s3_tier: S3Tier,
        ephemeral_aware: bool,
    },
}

impl AdaptiveTierManager {
    /// Factory method that creates appropriate manager based on config
    pub async fn from_config(config: &AxisConfig) -> Result<Self> {
        let deployment_mode = Self::detect_deployment_mode(config)?;
        
        let tier_strategy: Box<dyn TierStrategy> = match &deployment_mode {
            DeploymentMode::OnPremise { .. } => {
                Box::new(OnPremiseTierStrategy::new(config))
            }
            DeploymentMode::CloudManaged { .. } => {
                Box::new(CloudManagedTierStrategy::new(config))
            }
            DeploymentMode::CloudNative { .. } => {
                Box::new(CloudNativeTierStrategy::new(config))
            }
        };
        
        let persistence_layer: Box<dyn PersistenceLayer> = match &deployment_mode {
            DeploymentMode::OnPremise { .. } => {
                Box::new(LocalDiskPersistence::new(config))
            }
            DeploymentMode::CloudManaged { volume_type, .. } => {
                Box::new(PersistentVolumePersistence::new(volume_type.clone()))
            }
            DeploymentMode::CloudNative { s3_tier, .. } => {
                Box::new(S3Persistence::new(s3_tier.clone()))
            }
        };
        
        Ok(Self {
            deployment_mode,
            tier_strategy,
            persistence_layer,
        })
    }
    
    /// Detect deployment environment
    fn detect_deployment_mode(config: &AxisConfig) -> Result<DeploymentMode> {
        // Check explicit configuration first
        if let Some(mode) = &config.deployment_mode {
            return Ok(mode.clone());
        }
        
        // Auto-detect based on environment
        if Self::is_kubernetes() {
            if Self::has_persistent_volume_claims() {
                Ok(DeploymentMode::CloudManaged {
                    volume_type: VolumeType::K8sPVC,
                    snapshot_enabled: true,
                    multi_attach: false,
                })
            } else {
                Ok(DeploymentMode::CloudNative {
                    instance_type: InstanceType::K8sPod,
                    s3_tier: S3Tier::Express,
                    ephemeral_aware: true,
                })
            }
        } else if Self::is_ec2_spot() {
            Ok(DeploymentMode::CloudNative {
                instance_type: InstanceType::EC2Spot,
                s3_tier: S3Tier::Express,
                ephemeral_aware: true,
            })
        } else if Self::has_local_nvme() {
            Ok(DeploymentMode::OnPremise {
                disk_persistence: true,
                raid_config: Self::detect_raid_config(),
            })
        } else {
            // Default to cloud-native for safety
            Ok(DeploymentMode::CloudNative {
                instance_type: InstanceType::Unknown,
                s3_tier: S3Tier::Standard,
                ephemeral_aware: true,
            })
        }
    }
}

=== On-Premise Strategy (Persistent Disk)

[source,rust]
----
pub struct OnPremiseTierStrategy {
    disk_manager: DiskManager,
    promotion_engine: DiskPromotionEngine,
}

impl TierStrategy for OnPremiseTierStrategy {
    async fn initialize(&self) -> Result<()> {
        info!("🏢 On-Premise Mode: Persistent disk storage");
        
        // Scan disk for existing indexes
        let existing_indexes = self.disk_manager.scan_indexes().await?;
        info!("Found {} existing indexes on disk", existing_indexes.len());
        
        // No rebuild needed - indexes persist across restarts
        for (collection_id, index_path) in existing_indexes {
            self.register_disk_index(&collection_id, &index_path).await?;
        }
        
        Ok(())
    }
    
    async fn promote_to_memory(&self, collection_id: &str) -> Result<()> {
        // Load from local disk (fast)
        let index_path = self.disk_manager.get_index_path(collection_id)?;
        let index = self.load_from_disk(&index_path).await?;
        
        self.promotion_engine.load_into_memory(collection_id, index).await?;
        
        // Keep disk copy for fast recovery
        debug!("Keeping disk copy at: {}", index_path.display());
        
        Ok(())
    }
    
    async fn demote_from_memory(&self, collection_id: &str) -> Result<()> {
        // Save to local disk
        let index = self.get_memory_index(collection_id)?;
        let path = self.disk_manager.get_index_path(collection_id)?;
        
        self.save_to_disk(&path, &index).await?;
        self.remove_from_memory(collection_id).await?;
        
        info!("💾 Demoted {} to disk: {}", collection_id, path.display());
        
        Ok(())
    }
    
    async fn handle_restart(&self) -> Result<()> {
        // Indexes persist on disk - just register them
        info!("✅ On-premise restart: Indexes available on disk");
        self.scan_and_register_indexes().await?;
        
        // Optionally preload hot indexes based on historical access
        self.preload_hot_indexes().await?;
        
        Ok(())
    }
}

=== Cloud-Managed Strategy (Persistent Volumes)

[source,rust]
----
pub struct CloudManagedTierStrategy {
    volume_manager: PersistentVolumeManager,
    backup_manager: S3BackupManager,
}

impl TierStrategy for CloudManagedTierStrategy {
    async fn initialize(&self) -> Result<()> {
        info!("☁️ Cloud-Managed Mode: Persistent volumes + S3 backup");
        
        // Check if volumes are attached and mounted
        self.volume_manager.verify_volumes().await?;
        
        // Scan persistent volumes for indexes
        let volume_indexes = self.volume_manager.scan_indexes().await?;
        
        if volume_indexes.is_empty() {
            // First time or volume loss - restore from S3 backup
            warn!("No indexes on volumes, restoring from S3 backup");
            self.restore_from_backup().await?;
        } else {
            info!("Found {} indexes on persistent volumes", volume_indexes.len());
        }
        
        // Start background backup scheduler
        self.backup_manager.start_periodic_backup().await?;
        
        Ok(())
    }
    
    async fn promote_to_memory(&self, collection_id: &str) -> Result<()> {
        // Try persistent volume first
        if let Ok(index) = self.volume_manager.load_index(collection_id).await {
            return self.load_into_memory(collection_id, index).await;
        }
        
        // Fallback to S3 backup
        warn!("Index not on volume, restoring from S3");
        let index = self.backup_manager.restore_index(collection_id).await?;
        
        // Save to volume for next time
        self.volume_manager.save_index(collection_id, &index).await?;
        
        self.load_into_memory(collection_id, index).await
    }
    
    async fn handle_volume_failure(&self) -> Result<()> {
        error!("⚠️ Persistent volume failure detected");
        
        // Immediate failover to S3
        self.backup_manager.enable_s3_primary_mode().await?;
        
        // Attempt to reattach/remount volume
        if self.volume_manager.attempt_recovery().await.is_ok() {
            info!("✅ Volume recovered");
            self.backup_manager.disable_s3_primary_mode().await?;
        }
        
        Ok(())
    }
}

=== Cloud-Native Strategy (Ephemeral Instances)

[source,rust]
----
pub struct CloudNativeTierStrategy {
    s3_manager: S3IndexManager,
    rebuild_engine: IndexRebuildEngine,
    spot_handler: Option<SpotInstanceHandler>,
}

impl TierStrategy for CloudNativeTierStrategy {
    async fn initialize(&self) -> Result<()> {
        info!("🚀 Cloud-Native Mode: Ephemeral instance, S3 primary storage");
        
        // Register spot instance handler if applicable
        if let Some(handler) = &self.spot_handler {
            handler.start_monitoring().await?;
        }
        
        // List available indexes in S3
        let s3_indexes = self.s3_manager.list_indexes().await?;
        
        if !s3_indexes.is_empty() {
            info!("Found {} indexes in S3, rebuilding on startup", s3_indexes.len());
            
            // Rebuild indexes from S3 (parallel for speed)
            self.rebuild_from_s3(s3_indexes).await?;
        } else {
            info!("No existing indexes in S3, starting fresh");
        }
        
        Ok(())
    }
    
    async fn rebuild_from_s3(&self, indexes: Vec<S3IndexMetadata>) -> Result<()> {
        let start = Instant::now();
        
        // Prioritize rebuilding based on access patterns
        let prioritized = self.prioritize_rebuild_order(indexes).await?;
        
        // Rebuild in parallel batches
        let batch_size = self.calculate_optimal_batch_size();
        
        for batch in prioritized.chunks(batch_size) {
            let mut tasks = Vec::new();
            
            for index_meta in batch {
                let s3_manager = self.s3_manager.clone();
                let task = tokio::spawn(async move {
                    s3_manager.download_and_load_index(&index_meta).await
                });
                tasks.push(task);
            }
            
            futures::future::join_all(tasks).await;
        }
        
        info!("✅ Rebuilt {} indexes in {:?}", prioritized.len(), start.elapsed());
        
        Ok(())
    }
    
    async fn promote_to_memory(&self, collection_id: &str) -> Result<()> {
        // Always load from S3 (no local disk to trust)
        let index = self.s3_manager.download_index(collection_id).await?;
        
        // Apply any pending deltas since last S3 save
        let deltas = self.s3_manager.get_pending_deltas(collection_id).await?;
        let index = self.apply_deltas(index, deltas).await?;
        
        self.load_into_memory(collection_id, index).await?;
        
        // Cache locally for performance (best-effort only)
        if let Some(local_cache) = &self.local_cache {
            let _ = local_cache.store(collection_id, &index).await;
        }
        
        Ok(())
    }
    
    async fn demote_from_memory(&self, collection_id: &str) -> Result<()> {
        let index = self.get_memory_index(collection_id)?;
        
        // Save directly to S3 (primary storage)
        self.s3_manager.upload_index(collection_id, &index).await?;
        
        self.remove_from_memory(collection_id).await?;
        
        info!("☁️ Demoted {} to S3", collection_id);
        
        Ok(())
    }
    
    async fn handle_termination_warning(&self) -> Result<()> {
        warn!("⚠️ Instance termination in 2 minutes!");
        
        // Persist all dirty indexes to S3
        let dirty_indexes = self.get_dirty_indexes().await?;
        
        let mut tasks = Vec::new();
        for (collection_id, index) in dirty_indexes {
            let s3_manager = self.s3_manager.clone();
            tasks.push(tokio::spawn(async move {
                s3_manager.emergency_upload(&collection_id, &index).await
            }));
        }
        
        tokio::time::timeout(
            Duration::from_secs(90),
            futures::future::join_all(tasks),
        ).await?;
        
        info!("✅ All indexes persisted before termination");
        
        Ok(())
    }
}

== Unified API with Deployment Adaptation

=== Query Execution Across Deployments

[source,rust]
----
impl AdaptiveTierManager {
    /// Unified query interface that adapts to deployment
    pub async fn execute_query(
        &self,
        query: VectorQuery,
    ) -> Result<QueryResult> {
        let collection_id = &query.collection_id;
        
        // Check if index is in memory (same for all deployments)
        if let Some(index) = self.get_memory_index(collection_id).await {
            return self.query_from_memory(index, query).await;
        }
        
        // Deployment-specific fallback
        match &self.deployment_mode {
            DeploymentMode::OnPremise { .. } => {
                // Load from disk (fast, persistent)
                self.tier_strategy.promote_to_memory(collection_id).await?;
                self.retry_query(query).await
            }
            
            DeploymentMode::CloudManaged { .. } => {
                // Load from persistent volume or S3 backup
                self.tier_strategy.promote_to_memory(collection_id).await?;
                self.retry_query(query).await
            }
            
            DeploymentMode::CloudNative { .. } => {
                // Check local cache first (if available)
                if let Some(cached) = self.check_local_cache(collection_id).await {
                    return self.query_from_cache(cached, query).await;
                }
                
                // Load from S3 or fallback to storage scan
                if self.is_promotion_worthwhile(collection_id).await {
                    self.tier_strategy.promote_to_memory(collection_id).await?;
                    self.retry_query(query).await
                } else {
                    self.fallback_to_storage_scan(query).await
                }
            }
        }
    }
    
    /// Handle server startup based on deployment
    pub async fn handle_startup(&self) -> Result<()> {
        info!("🚀 Starting AXIS with deployment mode: {:?}", self.deployment_mode);
        
        match &self.deployment_mode {
            DeploymentMode::OnPremise { .. } => {
                // Fast startup - indexes on disk
                self.tier_strategy.initialize().await?;
                info!("✅ On-premise startup complete (indexes ready)");
            }
            
            DeploymentMode::CloudManaged { .. } => {
                // Medium startup - check volumes, maybe restore from S3
                self.tier_strategy.initialize().await?;
                info!("✅ Cloud-managed startup complete");
            }
            
            DeploymentMode::CloudNative { ephemeral_aware, .. } => {
                if *ephemeral_aware {
                    // Slow startup - rebuild from S3
                    info!("⏳ Rebuilding indexes from S3 (ephemeral instance)");
                    self.tier_strategy.initialize().await?;
                    info!("✅ Cloud-native startup complete (indexes rebuilt)");
                } else {
                    // Unexpected - log warning
                    warn!("Cloud-native mode but not ephemeral-aware");
                    self.tier_strategy.initialize().await?;
                }
            }
        }
        
        Ok(())
    }
}

=== Deployment-Aware Persistence

impl AdaptiveFlushCoordinator {
    pub async fn flush_with_adaptive_persistence(
        &self,
        collection_id: &str,
        vectors: Vec<VectorRecord>,
    ) -> Result<()> {
        // Update in-memory index if available
        if let Some(index) = self.get_memory_index(collection_id).await {
            self.update_index(&index, &vectors).await?;
        }
        
        // Persist based on deployment mode
        match &self.deployment_mode {
            DeploymentMode::OnPremise { .. } => {
                // Save to local disk immediately
                if let Some(index) = self.get_memory_index(collection_id).await {
                    self.save_to_disk(collection_id, &index).await?;
                } else {
                    // Append delta to disk
                    self.append_delta_to_disk(collection_id, &vectors).await?;
                }
            }
            
            DeploymentMode::CloudManaged { .. } => {
                // Save to persistent volume, async backup to S3
                if let Some(index) = self.get_memory_index(collection_id).await {
                    self.save_to_volume(collection_id, &index).await?;
                    self.schedule_s3_backup(collection_id).await?;
                } else {
                    self.append_delta_to_volume(collection_id, &vectors).await?;
                }
            }
            
            DeploymentMode::CloudNative { .. } => {
                // Save to S3 immediately (no local persistence)
                if let Some(index) = self.get_memory_index(collection_id).await {
                    self.upload_to_s3(collection_id, &index).await?;
                } else {
                    // Store delta in S3 for later application
                    self.upload_delta_to_s3(collection_id, &vectors).await?;
                }
            }
        }
        
        Ok(())
    }
}

== Conclusion

This unified design provides:

1. **Configuration-driven behavior**: Single codebase adapts to deployment context
2. **On-premise optimization**: Fast disk-based promotion/demotion
3. **Cloud-managed resilience**: Persistent volumes with S3 backup
4. **Cloud-native flexibility**: Ephemeral-aware with S3 as primary storage
5. **Automatic detection**: Can detect and adapt to environment

The key insight: **Same API, different durability assumptions** based on deployment context, with graceful degradation when moving from on-premise to cloud-native.
