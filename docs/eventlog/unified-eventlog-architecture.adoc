== Executive Summary

The EventLog system provides a lightweight, Kafka-like commit log that decouples storage operations from index building. It eliminates write amplification and synchronous blocking by enabling asynchronous AXIS indexing while maintaining consistency through acknowledgment-based compaction control.

[NOTE]
====

====

=== Producer Code

[source,rust]

pub struct EventLogProducer {
    log_writer: Arc<CommitLogWriter>,
    buffer: Vec<EventLogEntry>,
    buffer_size: usize,
    flush_interval: Duration,
}

impl EventLogProducer {
    /// Produce event with synchronous acknowledgment
    pub async fn produce(&mut self, event: EventLogEntry) -> Result<EventOffset> {
        // Add to buffer
        self.buffer.push(event);
        
        // Flush if buffer full or timeout
        if self.should_flush() {
            return self.flush_buffer().await;
        }
        
        Ok(EventOffset::pending())
    }
    
    /// Flush buffer to commit log
    async fn flush_buffer(&mut self) -> Result<EventOffset> {
        if self.buffer.is_empty() {
            return Ok(EventOffset::none());
        }
        
        // Serialize events
        let batch = self.serialize_batch(&self.buffer)?;
        
        // Write to active segment
        let offset = self.log_writer.append(batch).await?;
        
        // Wait for fsync (synchronous durability)
        self.log_writer.sync().await?;
        
        self.buffer.clear();
        Ok(offset)
    }
}

[source,rust]

pub struct AxisEventConsumer {
    consumer_id: String,
    log_reader: Arc<CommitLogReader>,
    axis_manager: Arc<AxisManager>,
    last_offset: AtomicU64,
    checkpoint_interval: Duration,
}

impl AxisEventConsumer {
    /// Main consumer loop
    pub async fn run(&self) -> Result<()> {
        let mut last_checkpoint = Instant::now();
        
        loop {
            // Poll for new events
            let events = self.poll_events().await?;
            
            if events.is_empty() {
                tokio::time::sleep(Duration::from_millis(100)).await;
                continue;
            }
            
            // Process batch of events
            for event in events {
                match event {
                    FlushEvent { .. } => {
                        self.process_flush_event(event).await?;
                    }
                    CompactionEvent { .. } => {
                        self.process_compaction_event(event).await?;
                    }
                    DeleteEvent { .. } => {
                        self.process_delete_event(event).await?;
                    }
                }
                
                // Update offset
                self.last_offset.store(
                    event.offset(),
                    Ordering::Release
                );
            }
            
            // Checkpoint periodically
            if last_checkpoint.elapsed() > self.checkpoint_interval {
                self.checkpoint().await?;
                last_checkpoint = Instant::now();
            }
        }
    }
    
    /// Process flush event by updating AXIS indexes
    async fn process_flush_event(&self, event: FlushEvent) -> Result<()> {
        // Read vectors from flushed files
        let vectors = self.read_vectors_from_files(
            &event.files_created
        ).await?;
        
        // Update appropriate index based on collection config
        let index_type = self.get_index_type(&event.collection_id)?;
        
        match index_type {
            IndexType::HNSW => {
                self.axis_manager
                    .update_hnsw(&event.collection_id, vectors)
                    .await?;
            }
            IndexType::IVF => {
                self.axis_manager
                    .update_ivf(&event.collection_id, vectors)
                    .await?;
            }
            // ... other index types
        }
        
        Ok(())
    }
}

== Queue-Aware Compaction

=== Protection Mechanism

....
File Lifecycle with EventLog:

1. File Created (Flush)
   └─> Added to EventLog
       └─> Status: PENDING_INDEX

2. AXIS Processes Event
   └─> Indexes built/updated
       └─> Status: INDEXED

3. AXIS Acknowledges
   └─> EventLog marks complete
       └─> Status: SAFE_TO_COMPACT

4. Compaction Checks EventLog
   └─> Only compacts SAFE_TO_COMPACT files
       └─> Creates new file
           └─> Goto step 1

Protected Files Example:
┌─────────────────────────────────────┐
│ SST Files         │ EventLog Status │
├───────────────────┼─────────────────┤
│ sst_001.sst       │ SAFE_TO_COMPACT │ ← Can compact
│ sst_002.sst       │ SAFE_TO_COMPACT │ ← Can compact
│ sst_003.sst       │ INDEXED         │ ← Protected
│ sst_004.sst       │ PENDING_INDEX   │ ← Protected
└─────────────────────────────────────┘
....

=== Compaction Integration

[source,rust]
----
pub struct CompactionManager {
    event_log: Arc<EventLogService>,
    // ... other fields
}

impl CompactionManager {
    /// Check if files can be compacted
    async fn can_compact_files(&self, files: &[String]) -> Result<bool> {
        for file in files {
            let status = self.event_log
                .get_file_status(file)
                .await?;
            
            match status {
                FileStatus::SafeToCompact => continue,
                FileStatus::PendingIndex | 
                FileStatus::Indexed => {
                    tracing::debug!(
                        "File {} not safe to compact, status: {:?}",
                        file, status
                    );
                    return Ok(false);
                }
            }
        }
        
        Ok(true)
    }
    
    /// Perform compaction with EventLog integration
    async fn compact_files(&self, input_files: Vec<String>) -> Result<String> {
        // Check if safe to compact
        if !self.can_compact_files(&input_files).await? {
            return Err(anyhow!("Files not ready for compaction"));
        }
        
        // Perform actual compaction
        let output_file = self.do_compact(&input_files).await?;
        
        // Notify EventLog of compaction
        self.event_log.produce(
            CompactionEvent {
                input_files: input_files.clone(),
                output_file: output_file.clone(),
                timestamp: SystemTime::now(),
            }
        ).await?;
        
        Ok(output_file)
    }
}

[source,rust]

pub struct EventLogRecovery {
    log_reader: Arc<CommitLogReader>,
    consumer_offset_store: Arc<OffsetStore>,
}

impl EventLogRecovery {
    /// Recover from last known good state
    pub async fn recover(&self) -> Result<RecoveryState> {
        // Read last committed offset
        let last_offset = self.consumer_offset_store
            .get_last_committed_offset()
            .await?;
        
        // Find unprocessed events
        let pending_events = self.log_reader
            .read_from(last_offset + 1)
            .await?;
        
        tracing::info!(
            "Recovery: Found {} unprocessed events from offset {}",
            pending_events.len(),
            last_offset + 1
        );
        
        Ok(RecoveryState {
            last_processed_offset: last_offset,
            pending_events,
            corrupted_segments: vec![],
        })
    }
    
    /// Validate segment integrity
    async fn validate_segments(&self) -> Result<Vec<SegmentId>> {
        let mut corrupted = Vec::new();
        
        for segment in self.log_reader.list_segments().await? {
            if !self.verify_segment_crc(&segment).await? {
                corrupted.push(segment.id);
            }
        }
        
        Ok(corrupted)
    }
}

impl EventLogMetrics {
    pub fn record_produce(&self, event_type: &str) {
        self.produce_counter
            .with_label_values(&[event_type])
            .inc();
        
        self.queue_depth.inc();
    }
    
    pub fn record_consume(&self, event_type: &str, duration: Duration) {
        self.consume_counter
            .with_label_values(&[event_type])
            .inc();
        
        self.consume_latency
            .with_label_values(&[event_type])
            .observe(duration.as_secs_f64());
        
        self.queue_depth.dec();
    }
    
    pub fn record_lag(&self, lag: Duration) {
        self.consumer_lag
            .set(lag.as_secs_f64());
    }
