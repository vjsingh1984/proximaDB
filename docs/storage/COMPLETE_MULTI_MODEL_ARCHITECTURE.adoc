== StoreType Enum (unified routing)

[source,rust]
----
pub enum StoreType {
    Vector,        // SST/HELIX/VIPER/NOVA → OptimizedSearchRecord
    Document,      // CEDAR                → DocumentRecord
    Graph,         // ORION/PULSAR         → Node/Edge/Path
    Observability, // CHRONO               → MetricSample/LogEntry/TraceData
    Relational,    // SEQUOIA              → TypedRow
}
----

== SQL Lowering Per StoreType

[cols="1,2,2,2"]
|===
| SQL | StoreType Detection | Lowering Target | Return Type

| `CREATE TABLE t (id INT, name VARCHAR)`
| Column types are standard SQL → Relational
| `CatalogService.create_table(TableSchema)`
| DDL success

| `CREATE TABLE docs (id TEXT, data JSONB) USING DOCUMENT`
| Explicit USING DOCUMENT
| `DocumentService.create_collection(config)`
| DDL success

| `CREATE TABLE vecs (id TEXT, vec VECTOR(384))`
| VECTOR column type → Vector
| `CollectionService.create_collection(config)`
| DDL success

| `INSERT INTO t (id, name) VALUES (1, 'Alice')`
| Catalog lookup → StoreType::Relational
| `RelationalService.insert_rows(table, rows)`
| INSERT 0 1

| `INSERT INTO docs (data) VALUES ('{"x":1}')`
| Catalog lookup → StoreType::Document
| `DocumentService.insert_document(col, doc)`
| INSERT 0 1

| `SELECT * FROM t WHERE id > 5`
| Catalog lookup → StoreType::Relational
| `RelationalService.query(table, filter, projection)`
| Typed PG rows

| `SELECT * FROM docs WHERE $.name = 'Alice'`
| JSON path in WHERE → StoreType::Document
| `DocumentService.query_documents(col, filter)`
| (id, document JSONB, version)

| `SELECT * FROM vecs ORDER BY vec <-> '[0.1,0.2]' LIMIT 10`
| `<->` operator → StoreType::Vector
| `VectorService.search(col, query_vec, k)`
| (id, distance, metadata)

| `SELECT * FROM graph_social WHERE label = 'Person'`
| `graph_` prefix → StoreType::Graph
| `GraphService.query_nodes(filter)`
| (id, labels, properties)

| `SELECT * FROM log_app WHERE severity >= 'ERROR'`
| `log_` prefix → StoreType::Observability
| `ObservabilityService.query_logs(filter)`
| (timestamp, severity, message)
|===

== RelationalStorageEngine Trait (SEQUOIA)

[source,rust]
----
#[async_trait]
pub trait RelationalStorageEngine: Send + Sync {
    fn engine_name(&self) -> &'static str;

    // Schema-aware operations
    async fn create_table(&self, schema: CatalogTableSchema) -> Result<()>;
    async fn drop_table(&self, table: &str) -> Result<()>;
    async fn get_schema(&self, table: &str) -> Result<Option<CatalogTableSchema>>;

    // Row CRUD
    async fn insert_rows(&self, table: &str, rows: Vec<TypedRow>) -> Result<u64>;
    async fn query_rows(&self, table: &str, params: RelationalQueryParams) -> Result<RelationalQueryResult>;
    async fn update_rows(&self, table: &str, updates: Vec<(String, TypedValue)>,
                         filter: Option<RowFilter>) -> Result<u64>;
    async fn delete_rows(&self, table: &str, filter: Option<RowFilter>) -> Result<u64>;

    // Index management
    async fn create_index(&self, table: &str, columns: Vec<String>, unique: bool) -> Result<()>;

    // Aggregation
    async fn aggregate(&self, table: &str, params: AggregateParams) -> Result<Vec<TypedRow>>;

    // Persistence
    async fn flush(&self, table: &str) -> Result<u64>;
    async fn compact(&self, table: &str) -> Result<u64>;

    async fn row_count(&self, table: &str) -> Result<u64>;
    async fn collect_metrics(&self) -> Result<HashMap<String, serde_json::Value>>;
}
----

== TypedRow and TypedValue (relational result type)

[source,rust]
----
/// A row with typed column values, validated against CatalogTableSchema
pub struct TypedRow {
    pub values: Vec<TypedValue>,
}

/// A typed column value matching CatalogDataType
pub enum TypedValue {
    Null,
    Boolean(bool),
    Int8(i8),
    Int16(i16),
    Int32(i32),
    Int64(i64),
    Float32(f32),
    Float64(f64),
    String(String),
    Binary(Vec<u8>),
    Date(i32),          // days since epoch
    Time(i64),          // microseconds since midnight
    Timestamp(i64),     // microseconds since epoch
    TimestampTz(i64),   // microseconds since epoch (UTC)
    Decimal(i128),      // scaled decimal
    Uuid(uuid::Uuid),
    Json(serde_json::Value),
    Vector(Vec<f32>),
}
