== Progressive Columnar Architecture

=== What Makes NOVA "Progressive"?

NOVA implements a **progressive refinement strategy** across three dimensions:

1. **Progressive Spatial Pruning**: Hierarchical zone maps (SuperBlock → Row Group → Page)
2. **Progressive Quantization**: Multi-level quantization (Binary → INT8 → PQ8 → FP32)
3. **Progressive I/O**: Lazy loading with streaming for large datasets

This differs from:

- **VIPER**: Pure columnar, optimized for batch analytics
- **SST**: Hybrid columnar, optimized for real-time OLTP

NOVA balances between these extremes for **mixed workloads**.

==== Zone Map Pruning Algorithm

[source,rust]
----
/// Check if zone map can be pruned for given query
pub fn can_prune_zone(
    zone: &ZoneMap,
    query_vector: &[f32],
    threshold: f32,
    metric: DistanceMetric,
) -> bool {
    match metric {
        DistanceMetric::Euclidean => {
            // Compute minimum possible distance to zone
            let mut min_dist_sq = 0.0;
            for i in 0..query_vector.len() {
                let q = query_vector[i];
                let min = zone.min_values[i];
                let max = zone.max_values[i];

                // If query point is outside zone bounds
                if q < min {
                    min_dist_sq += (q - min).powi(2);
                } else if q > max {
                    min_dist_sq += (q - max).powi(2);
                }
                // If inside zone bounds, contributes 0 to distance
            }

            let min_dist = min_dist_sq.sqrt();
            min_dist > threshold  // Prune if min possible distance > threshold
        }
        DistanceMetric::Cosine => {
            // For cosine, use sphere-based pruning with centroid
            if let (Some(centroid), Some(radius)) = (&zone.centroid, zone.radius) {
                let centroid_similarity = cosine_similarity(query_vector, centroid);
                // Conservative pruning: if centroid is too far considering radius
                centroid_similarity < threshold - radius
            } else {
                false  // Can't prune without centroid
            }
        }
        _ => false,
    }
}
----

// From progressive_search.rs
pub async fn stage1_zone_map_pruning(
    &self,
    query: &SearchQuery,
    superblocks: &[SuperBlock],
) -> Result<Vec<String>> {
    let mut selected_row_groups = Vec::new();

    for superblock in superblocks {
        // Check SuperBlock zone map
        if !can_prune_zone(&superblock.zone_map, &query.vector, query.threshold, query.metric) {
            // SuperBlock passed, check constituent row groups
            for rg_id in &superblock.row_groups {
                let rg_metadata = self.load_row_group_metadata(rg_id).await?;
                if !can_prune_zone(&rg_metadata.zone_map, &query.vector, query.threshold, query.metric) {
                    selected_row_groups.push(rg_id.clone());
                }
            }
        }
    }

    info!(
        total_row_groups = superblocks.iter().map(|sb| sb.row_groups.len()).sum::<usize>(),
        selected_row_groups = selected_row_groups.len(),
        pruning_ratio = 1.0 - (selected_row_groups.len() as f64 / superblocks.iter().map(|sb| sb.row_groups.len()).sum::<usize>() as f64),
        "Zone map pruning complete"
    );

    Ok(selected_row_groups)
}

==== Stage 2: Binary Filter

[source,rust]

pub async fn stage2_binary_filter(
    &self,
    query: &SearchQuery,
    row_groups: Vec<String>,
) -> Result<Vec<CandidateVector>> {
    let mut candidates = Vec::new();
    let binary_query = quantize_to_binary(&query.vector);

    for rg_id in row_groups {
        // Read only q_binary column (1 bit/dim)
        let binary_vectors = self.read_binary_column(&rg_id).await?;

        for (idx, binary_vec) in binary_vectors.into_iter().enumerate() {
            let hamming_dist = hamming_distance(&binary_query, &binary_vec);

            // Keep if hamming distance is within threshold
            if hamming_dist < query.binary_threshold {
                candidates.push(CandidateVector {
                    row_group: rg_id.clone(),
                    index: idx,
                    binary_distance: hamming_dist,
                });
            }
        }
    }

    // Sort by hamming distance and keep top candidates
    candidates.sort_by_key(|c| c.binary_distance);
    candidates.truncate(query.top_k * query.candidate_multiplier);

    Ok(candidates)
}

**Performance**: ~50ms for 100K candidates, 95% elimination

pub async fn stage3_int8_ranking(
    &self,
    query: &SearchQuery,
    candidates: Vec<CandidateVector>,
) -> Result<Vec<CandidateVector>> {
    let int8_query = quantize_to_int8(&query.vector, &query.int8_params);
    let mut ranked_candidates = Vec::new();

    for candidate in candidates {
        // Read q_int8 column (8 bits/dim)
        let int8_vec = self.read_int8_vector(&candidate.row_group, candidate.index).await?;

        let int8_distance = compute_int8_distance(&int8_query, &int8_vec, query.metric);

        ranked_candidates.push(CandidateVector {
            int8_distance: Some(int8_distance),
            ..candidate
        });
    }

    // Sort by INT8 distance
    ranked_candidates.sort_by(|a, b| {
        a.int8_distance.unwrap().partial_cmp(&b.int8_distance.unwrap()).unwrap()
    });
    ranked_candidates.truncate(query.top_k * 10);  // Keep 10x final k

    Ok(ranked_candidates)
}

pub async fn stage4_pq8_scoring(
    &self,
    query: &SearchQuery,
    candidates: Vec<CandidateVector>,

    let pq_codebook = self.load_pq_codebook(&query.collection_id).await?;

    scored_candidates.sort_by(|a, b| {
        a.pq8_distance.unwrap().partial_cmp(&b.pq8_distance.unwrap()).unwrap()

    scored_candidates.truncate(query.top_k * 5);  // Keep 5x final k

pub async fn stage5_fp32_reranking(
    &self,
    query: &SearchQuery,
    candidates: Vec<CandidateVector>,
) -> Result<Vec<SearchResult>> {
    let mut results = Vec::new();

    for candidate in candidates {
        // Read vector_fp32 column (full precision)
        let fp32_vec = self.read_fp32_vector(&candidate.row_group, candidate.index).await?;

        let exact_distance = compute_exact_distance(&query.vector, &fp32_vec, query.metric);

        results.push(SearchResult {
            id: self.get_vector_id(&candidate.row_group, candidate.index).await?,
            distance: exact_distance,
            vector: fp32_vec,
            metadata: self.get_metadata(&candidate.row_group, candidate.index).await?,
        });
    }

    // Final sort by exact distance
    results.sort_by(|a, b| a.distance.partial_cmp(&b.distance).unwrap());
    results.truncate(query.top_k);

    Ok(results)
}

    pub async fn progressive_search(
        &self,
        query: &SearchQuery,

        let superblocks = self.load_superblocks(&query.collection_id).await?;

        let int8_candidates = self.stage3_int8_ranking(query, binary_candidates).await?;

        let final_results = self.stage5_fp32_reranking(query, pq8_candidates).await?;

        Ok(ProgressiveSearchResult {
            results: final_results,
            stages_executed: refinement_stages,
            pruning_stats: self.get_pruning_statistics(),
        })
    }
}

=== Selective Column Reading

NOVA optimizes I/O by reading only necessary columns at each stage:

/// Read specific quantization level columns
pub async fn read_quantized_column(
    &self,
    file_path: &str,
    column_name: &str,
    row_group_indices: &[usize],
) -> Result<Vec<Vec<u8>>> {
    let parquet_reader = self.filesystem.open_parquet(file_path).await?;

    // Project to specific column
    let schema = parquet_reader.schema();
    let projection = schema.project_columns(&[column_name])?;

    // Read only selected row groups
    let mut column_data = Vec::new();
    for rg_idx in row_group_indices {
        let row_group = parquet_reader.row_group(*rg_idx)?;
        let column = row_group.column_by_name(column_name)?;
        column_data.push(column.read_bytes()?);
    }

    Ok(column_data)
}

**I/O Savings**:

- Binary stage: Read 1/32 of FP32 data (1 bit vs 32 bits per dimension)
- INT8 stage: Read 1/4 of FP32 data
- PQ8 stage: Read 1/32 of FP32 data (32 subquantizers)
- FP32 stage: Read only top candidates (0.01% of data)

**Total I/O Reduction**: 99%+ for typical queries

pub struct StreamingProcessor {
    /// Chunk size for streaming
    chunk_size: usize,

    /// Buffer pool for memory reuse
    buffer_pool: Arc<BufferPool>,

    /// Prefetch queue for async I/O
    prefetch_queue: Arc<Mutex<VecDeque<PrefetchRequest>>>,
}

impl StreamingProcessor {
    /// Stream large dataset in chunks
    pub async fn stream_search(
        &self,
        query: &SearchQuery,
        collection_id: &str,
    ) -> Result<impl Stream<Item = Result<SearchResult>>> {
        // Create stream that yields results incrementally
        let (tx, rx) = tokio::sync::mpsc::channel(self.chunk_size);

        tokio::spawn(async move {
            let superblocks = self.load_superblocks(collection_id).await?;

            for superblock in superblocks {
                // Process SuperBlock in chunks
                let row_groups = self.stage1_zone_map_pruning(query, &[superblock]).await?;

                for chunk in row_groups.chunks(self.chunk_size) {
                    // Process chunk
                    let results = self.process_chunk(query, chunk).await?;

                    // Send results to stream
                    for result in results {
                        tx.send(Ok(result)).await?;
                    }
                }
            }

            Ok(())
        });

        Ok(tokio_stream::wrappers::ReceiverStream::new(rx))
    }
}

**Benefits**:

- Constant memory usage regardless of result size
- Early termination for top-k queries
- Async prefetching for better latency

pub struct BatchOperations {
    /// Batch size for optimal I/O
    batch_size: usize,

    /// Parallel workers
    worker_count: usize,
}

impl BatchOperations {
    /// Batch insert with hierarchical organization
    pub async fn batch_insert(
        &self,
        vectors: Vec<VectorRecord>,
        collection_id: &str,
    ) -> Result<BatchInsertResult> {
        // 1. Partition into batches
        let batches: Vec<_> = vectors
            .chunks(self.batch_size)
            .map(|chunk| chunk.to_vec())
            .collect();

        // 2. Process batches in parallel
        let mut tasks = Vec::new();
        for batch in batches {
            let task = tokio::spawn(async move {
                self.process_insert_batch(batch, collection_id).await
            });
            tasks.push(task);
        }

        // 3. Await all batches
        let mut results = Vec::new();
        for task in tasks {
            results.push(task.await??);
        }

        // 4. Merge results and create SuperBlocks
        let merged_result = self.merge_batch_results(results).await?;

        Ok(merged_result)
    }
}

pub struct HierarchicalCache {
    /// SuperBlock metadata cache (hot data)
    superblock_cache: Arc<DashMap<String, SuperBlock>>,

    /// Zone map cache (warm data)
    zone_map_cache: Arc<DashMap<String, ZoneMap>>,

    /// Row group metadata cache (cold data)
    row_group_cache: Arc<LruCache<String, RowGroupMetadata>>,

    /// Cache statistics
    stats: Arc<RwLock<CacheStats>>,
}

impl HierarchicalCache {
    /// Get SuperBlock with caching
    pub async fn get_superblock(&self, superblock_id: &str) -> Result<Arc<SuperBlock>> {
        // Check cache first
        if let Some(cached) = self.superblock_cache.get(superblock_id) {
            self.stats.write().await.superblock_hits += 1;
            return Ok(cached.value().clone());
        }

        // Cache miss - load from storage
        self.stats.write().await.superblock_misses += 1;
        let superblock = self.load_superblock_from_storage(superblock_id).await?;

        // Insert into cache
        let superblock_arc = Arc::new(superblock.clone());
        self.superblock_cache.insert(superblock_id.to_string(), superblock_arc.clone());

        Ok(superblock_arc)
    }
}

✅ **Use NOVA when**:

- Mixed workload (reads + writes)
- Queries benefit from spatial locality
- Hierarchical pruning effective
- Progressive refinement valuable
- Moderate update frequency
