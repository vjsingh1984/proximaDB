== Overview

This guide covers various deployment options for ProximaDB, from single-node installations to distributed clusters.

== Deployment Options

=== Single Node Deployment

==== Binary Installation

[source,bash]
----
# Download latest release
wget https://github.com/proximadb/releases/latest/proximadb-server
chmod +x proximadb-server

# Create configuration
cat > config.toml <<EOF
[server]
rest_port = 5678
grpc_port = 5679

[storage]
data_dir = "/var/lib/proximadb/data"
default_engine = "VIPER"

[performance]
enable_simd = true
enable_gpu = false
EOF

# Start server
./proximadb-server --config config.toml
----

==== Systemd Service

[source,ini]
----
# /etc/systemd/system/proximadb.service
[Unit]
Description=ProximaDB Vector Database
After=network.target

[Service]
Type=simple
User=proximadb
Group=proximadb
ExecStart=/usr/local/bin/proximadb-server --config /etc/proximadb/config.toml
Restart=always
RestartSec=10
StandardOutput=journal
StandardError=journal
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
----

[source,bash]
----
# Enable and start service
sudo systemctl daemon-reload
sudo systemctl enable proximadb
sudo systemctl start proximadb
sudo systemctl status proximadb
----

=== Docker Deployment

==== Single Container

[source,bash]
----
# Run with Docker
docker run -d \
  --name proximadb \
  -p 5678:5678 \
  -p 5679:5679 \
  -v /data/proximadb:/data \
  -v /config/proximadb:/config \
  proximadb/proximadb:latest

# With custom configuration
docker run -d \
  --name proximadb \
  -p 5678:5678 \
  -p 5679:5679 \
  -v $(pwd)/config.toml:/config/config.toml \
  -v /data/proximadb:/data \
  proximadb/proximadb:latest \
  --config /config/config.toml
----

==== Docker Compose

[source,yaml]
----
# docker-compose.yml
version: '3.8'

services:
  proximadb:
    image: proximadb/proximadb:latest
    container_name: proximadb
    restart: unless-stopped
    ports:
      - "5678:5678"  # REST API
      - "5679:5679"  # gRPC API
      - "9090:9090"  # Metrics
    volumes:
      - proximadb_data:/data
      - ./config:/config
    environment:
      - PROXIMADB_LOG_LEVEL=info
      - PROXIMADB_STORAGE_ENGINE=VIPER
      - PROXIMADB_PERFORMANCE_ENABLE_GPU=false
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5678/health"]
      interval: 30s
      timeout: 10s
      retries: 3
    deploy:
      resources:
        limits:
          cpus: '4'
          memory: 8G
        reservations:
          cpus: '2'
          memory: 4G

volumes:
  proximadb_data:
    driver: local
----

=== Kubernetes Deployment

==== Basic Deployment

[source,yaml]
----
# proximadb-deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: proximadb
  namespace: proximadb
spec:
  replicas: 3
  selector:
    matchLabels:
      app: proximadb
  template:
    metadata:
      labels:
        app: proximadb
    spec:
      containers:
      - name: proximadb
        image: proximadb/proximadb:latest
        ports:
        - containerPort: 5678
          name: rest
        - containerPort: 5679
          name: grpc
        - containerPort: 9090
          name: metrics
        env:
        - name: PROXIMADB_STORAGE_DATA_DIR
          value: "/data"
        volumeMounts:
        - name: data
          mountPath: /data
        - name: config
          mountPath: /config
        resources:
          requests:
            memory: "4Gi"
            cpu: "2"
          limits:
            memory: "8Gi"
            cpu: "4"
        livenessProbe:
          httpGet:
            path: /health
            port: 5678
          initialDelaySeconds: 30
          periodSeconds: 10
        readinessProbe:
          httpGet:
            path: /ready
            port: 5678
          initialDelaySeconds: 10
          periodSeconds: 5
      volumes:
      - name: data
        persistentVolumeClaim:
          claimName: proximadb-data-pvc
      - name: config
        configMap:
          name: proximadb-config
---
apiVersion: v1
kind: Service
metadata:
  name: proximadb
  namespace: proximadb
spec:
  selector:
    app: proximadb
  ports:
  - name: rest
    port: 5678
    targetPort: 5678
  - name: grpc
    port: 5679
    targetPort: 5679
  - name: metrics
    port: 9090
    targetPort: 9090
  type: LoadBalancer
----

== Network Configuration

=== Load Balancing

==== HAProxy Configuration

[source,conf]
----
# /etc/haproxy/haproxy.cfg
global
    maxconn 4096
    
defaults
    mode tcp
    timeout connect 5s
    timeout client 30s
    timeout server 30s

frontend proximadb_rest
    bind *:5678
    default_backend proximadb_rest_nodes

backend proximadb_rest_nodes
    balance roundrobin
    server node1 10.0.1.10:5678 check
    server node2 10.0.1.11:5678 check
    server node3 10.0.1.12:5678 check

frontend proximadb_grpc
    bind *:5679
    default_backend proximadb_grpc_nodes

backend proximadb_grpc_nodes
    balance leastconn
    server node1 10.0.1.10:5679 check
    server node2 10.0.1.11:5679 check
    server node3 10.0.1.12:5679 check
----

==== NGINX Configuration

[source,nginx]
----
# /etc/nginx/nginx.conf
upstream proximadb_rest {
    least_conn;
    server 10.0.1.10:5678;
    server 10.0.1.11:5678;
    server 10.0.1.12:5678;
}

upstream proximadb_grpc {
    least_conn;
    server 10.0.1.10:5679;
    server 10.0.1.11:5679;
    server 10.0.1.12:5679;
}

server {
    listen 80;
    server_name api.proximadb.example.com;
    
    location / {
        proxy_pass http://proximadb_rest;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

server {
    listen 5679 http2;
    
    location / {
        grpc_pass grpc://proximadb_grpc;
    }
}
----

== Storage Configuration

=== Local Storage

[source,bash]
----
# Prepare local storage
sudo mkdir -p /var/lib/proximadb/data
sudo chown -R proximadb:proximadb /var/lib/proximadb

# Mount dedicated disk
sudo mkfs.ext4 /dev/nvme1n1
sudo mount /dev/nvme1n1 /var/lib/proximadb/data
echo "/dev/nvme1n1 /var/lib/proximadb/data ext4 defaults 0 0" | sudo tee -a /etc/fstab
----

=== Network Storage

==== NFS

[source,bash]
----
# Mount NFS share
sudo apt-get install nfs-common
sudo mkdir -p /mnt/proximadb
sudo mount -t nfs nfs-server:/export/proximadb /mnt/proximadb
echo "nfs-server:/export/proximadb /mnt/proximadb nfs defaults 0 0" | sudo tee -a /etc/fstab
----

==== AWS EBS

[source,bash]
----
# Attach and mount EBS volume
aws ec2 attach-volume \
  --volume-id vol-xxxxx \
  --instance-id i-xxxxx \
  --device /dev/xvdf

sudo mkfs.ext4 /dev/xvdf
sudo mount /dev/xvdf /var/lib/proximadb/data
----

== Security Hardening

=== TLS Configuration

[source,bash]
----
# Generate certificates
openssl req -x509 -newkey rsa:4096 \
  -keyout server.key \
  -out server.crt \
  -days 365 \
  -nodes \
  -subj "/CN=proximadb.example.com"

# Configure TLS in config.toml
cat >> config.toml <<EOF
[server]
enable_tls = true
tls_cert_path = "/certs/server.crt"
tls_key_path = "/certs/server.key"
EOF
----

=== Firewall Rules

[source,bash]
----
# UFW configuration
sudo ufw allow 22/tcp  # SSH
sudo ufw allow 5678/tcp  # REST API
sudo ufw allow 5679/tcp  # gRPC API
sudo ufw allow 9090/tcp  # Metrics (internal only)
sudo ufw enable
----

=== Smoke Tests

[source,python]
----
# smoke_test.py
from proximadb import Client
import numpy as np

client = Client("localhost:5678")

# Create test collection
collection = client.create_collection(
    name="smoke_test",
    dimension=128
)

# Insert test vector
vector = np.random.rand(128).astype('float32')
collection.insert(id="test", vector=vector)

# Search
results = collection.search(vector=vector, k=1)
assert len(results) == 1
assert results[0].id == "test"

print("✓ Smoke test passed")

# Cleanup
client.delete_collection("smoke_test")
----
