**Implementation Pattern:**
[source,rust]
----
pub struct EmbeddingVersion {
    pub vector: Vec<f32>,
    pub model: String,
    pub version: i32,
    pub created_at: SystemTime,
    pub accuracy_metrics: Option<AccuracyMetrics>,
}

impl EntityStore {
    pub async fn upsert_entity(&self, entity: &Entity) -> Result<(), ProximaDBError> {
        // Handle embedding versioning
        if let Some(new_embedding) = &entity.embedding {
            self.handle_embedding_versioning(entity, new_embedding).await?;
        }
        
        // Process flexible metadata
        let processed_metadata = self.process_flexible_metadata(&entity.metadata)?;
        
        // Store in appropriate engine
        self.store_entity_optimally(entity, processed_metadata).await
    }
}
----

== 📊 SQL Query Extensions

=== SIMILAR Operator

[source,mermaid]
----
flowchart LR
    A[SQL Query with SIMILAR] --> B[Parse Query]
    B --> C[Extract Vector Component] 
    C --> D[Execute Vector Search]
    D --> E[Apply Similarity Threshold]
    E --> F[Return Results]
    
    style A fill:#e3f2fd
    style F fill:#e8f5e8
----

**Syntax Examples:**
[source,sql]
----
-- Find similar documents
SELECT * FROM documents 
WHERE embedding SIMILAR TO [0.1, 0.2, 0.3, ...] 
WITH THRESHOLD 0.8;

-- Find documents similar to a specific document
SELECT * FROM documents d1, documents d2
WHERE d1.embedding SIMILAR TO d2.embedding
AND d2.id = 'doc_123'
WITH THRESHOLD 0.75;
----

=== FOLLOW Operator  

[source,mermaid]
----
graph LR
    A[Start Entity] --> B[Follow Relations]
    B --> C[Traverse Graph]
    C --> D[Apply Filters]
    D --> E[Collect Results]
    
    style A fill:#e3f2fd
    style E fill:#e8f5e8
----

**Syntax Examples:**
[source,sql]
----
-- Follow citation relationships
SELECT * FROM papers p
FOLLOW citations FROM 'paper_123'
WITH MAX_DEPTH 3;

-- Follow with relationship filtering
SELECT * FROM users u
FOLLOW connections FROM 'user_123' 
WHERE connection_type = 'FRIEND'
WITH MAX_DEPTH 2;
----

**Implementation Example:**
[source,python]
----
import proximadb

# Initialize SKS client
sks_client = proximadb.SKSClient(
    grpc_url="localhost:5679",
    collection_id="research_papers"
)

# Create entity with embedding
entity = {
    "id": "paper_123",
    "entity_type": "research_paper",
    "metadata": {
        "title": "Advanced Vector Search Techniques",
        "authors": ["Smith, J.", "Doe, A."],
        "year": 2025,
        "venue": "VLDB 2025"
    },
    "embedding": {
        "vector": paper_embedding,  # Generated from title + abstract
        "model": "text-embedding-ada-002",
        "version": 1
    },
    "content": "Full paper text content...",
    "source": {
        "url": "https://example.com/paper_123.pdf",
        "extracted_at": "2025-09-11T10:00:00Z"
    }
}

# Store entity
await sks_client.upsert_entity(entity)

# Create relationships
citation_relation = {
    "from_entity": "paper_123",
    "to_entity": "paper_456", 
    "relation_type": "CITES",
    "properties": {
        "citation_context": "Building on previous work...",
        "strength": 0.8
    }
}

await sks_client.create_relation(citation_relation)

# Hybrid search: similar papers + citation network
results = await sks_client.hybrid_search(
    query_text="vector database optimization",
    max_similar=10,
    follow_relations=["CITES", "CITED_BY"],
    max_depth=2
)
----

== 🔍 Advanced Query Capabilities

=== Hybrid Query Execution

[source,mermaid]
----
sequenceDiagram
    participant QueryParser
    participant VectorEngine
    participant GraphEngine
    participant FusionEngine
    participant ResultAssembler
    
    QueryParser->>VectorEngine: Execute vector component
    QueryParser->>GraphEngine: Execute graph component
    
    VectorEngine-->>FusionEngine: Vector candidates
    GraphEngine-->>FusionEngine: Graph results
    
    FusionEngine->>ResultAssembler: Fused results
    ResultAssembler->>QueryParser: Final response
    
    Note over VectorEngine,GraphEngine: Parallel execution
    Note over FusionEngine: Unified ranking algorithm
----

**Complex Query Example:**
[source,sql]
----
-- Find research papers similar to a query, 
-- following citation networks,
-- and assembling full context
WITH similar_papers AS (
    SELECT * FROM research_papers
    WHERE embedding SIMILAR TO $query_embedding
    WITH THRESHOLD 0.75
    LIMIT 20
),
citation_network AS (
    SELECT p.* FROM research_papers p
    FOLLOW citations FROM similar_papers.id
    WITH MAX_DEPTH 2
    WHERE p.year >= 2020
),
assembled_context AS (
    SELECT * FROM citation_network
    ASSEMBLE CONTEXT FROM citations, references
    WITH PROVENANCE_TRACKING true
)
SELECT DISTINCT id, title, authors, similarity_score, graph_distance
FROM assembled_context
ORDER BY (similarity_score * 0.7 + (1.0 / graph_distance) * 0.3) DESC
LIMIT 10;
----
