== Proto-First Advantages

**Zero Double Serialization**:: Protocol Buffers flow natively through the entire system without conversions

**85% Code Reduction**:: Unified handlers eliminate REST/gRPC duplication

**Storage-Aware Search**:: Automatic engine selection delivers optimized performance improvement

**Unified Python SDK**:: Single client (87% smaller) supports both REST and gRPC protocols

**Hardware Optimization**:: SIMD/GPU acceleration with intelligent quantization selection

== Core Concepts

**Proto-Native Collections**:: Containers using native protobuf types with .proto extensions

**Unified Vector Operations**:: Single API surface for insert, search, update, delete across REST/gRPC

**Storage-Aware Intelligence**:: Automatic WriteBuffer → VIPER → SST engine selection based on data location

**Two-Stage Search**:: Binary quantization for candidate selection + FP32 refinement for accuracy

**Multi-Engine Architecture**:: VIPER (Parquet columnar) and SST (row-based SSTable) storage engines

async def optimized_batch_insert(collection_id, documents):
    batch_size = 100
    batches = [documents[i:i + batch_size] for i in range(0, len(documents), batch_size)]
    
    total_inserted = 0
    start_time = time.time()

from transformers import AutoTokenizer, AutoModel
import torch
import numpy as np
from proximadb import UnifiedClient

class BERTEmbeddingService:
    def __init__(self, model_name="bert-base-uncased"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name)
        self.model.eval()
        
    def encode_text(self, text: str) -> list[float]:
        """Generate BERT embedding for text."""
        inputs = self.tokenizer(
            text, 
            return_tensors="pt", 
            truncation=True, 
            max_length=512,
            padding=True
        )
        
        with torch.no_grad():
            outputs = self.model(**inputs)
            # Use [CLS] token representation
            embedding = outputs.last_hidden_state[:, 0, :].squeeze()
            
        return embedding.numpy().tolist()

async def bert_workflow_example():
    # Initialize services
    bert = BERTEmbeddingService("bert-base-uncased")
    client = UnifiedClient("http://localhost:5678")
    
    # Create BERT collection
    collection_config = {
        "collection_id": "bert_embeddings",
        "dimension": 768,  # BERT base dimension
        "distance_metric": "cosine",
        "description": "BERT document embeddings"
    }
    collection_id = await client.create_collection(collection_config)
    
    # Sample documents
    documents = [
        {
            "id": "article_1",
            "text": "Machine learning is transforming how we process and understand data.",
            "category": "technology",
            "source": "tech_blog"
        },
        {
            "id": "article_2", 
            "text": "Artificial intelligence will revolutionize healthcare diagnostics.",
            "category": "healthcare",
            "source": "medical_journal"
        }
    ]
    
    # Generate embeddings and insert
    for doc in documents:
        embedding = bert.encode_text(doc["text"])
        await client.insert_vector(
            collection_id=collection_id,
            vector_id=doc["id"],
            vector=embedding,
            metadata={
                "text": doc["text"],
                "category": doc["category"],
                "source": doc["source"]
            }
        )
    
    # Search with natural language query
    query_text = "AI and machine learning applications"
    query_embedding = bert.encode_text(query_text)
    
    results = await client.search_vectors(
        collection_id=collection_id,
        query_vector=query_embedding,
        k=3,
        return_metadata=True
    )
    
    print("🔍 BERT Search Results:")
    for i, result in enumerate(results, 1):
        print(f"{i}. ID: {result.vector_id}")
        print(f"   Similarity: {result.similarity_score:.4f}")

import asyncio

async def parallel_insert_example(collection_id, documents):
    client = UnifiedClient("http://localhost:5678")
    
    async def insert_single_vector(doc):
        try:
            await client.insert_vector(
                collection_id=collection_id,
                vector_id=doc["id"],
                vector=doc["embedding"],
                metadata=doc["metadata"]
            )
            return f"✅ {doc['id']}"
        except Exception as e:
            return f"❌ {doc['id']}: {e}"
    
    # Process up to 10 vectors concurrently
    semaphore = asyncio.Semaphore(10)
    
    async def bounded_insert(doc):
        async with semaphore:
            return await insert_single_vector(doc)
    
    # Run all insertions concurrently
    tasks = [bounded_insert(doc) for doc in documents]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    success_count = sum(1 for r in results if "✅" in str(r))
    print(f"📊 Inserted {success_count}/{len(documents)} vectors successfully")

=== Retry Logic Implementation

[source,python]
----
import asyncio
from typing import Callable, Any

async def retry_with_backoff(
    func: Callable,
    max_retries: int = 3,
    backoff_factor: float = 1.5,
    max_backoff: float = 60.0,
    *args,
    **kwargs
) -> Any:
    """Retry function with exponential backoff."""
    
    for attempt in range(max_retries + 1):
        try:
            return await func(*args, **kwargs)
            
        except (ConnectionError, RateLimitError) as e:
            if attempt == max_retries:
                raise e
                
            wait_time = min(backoff_factor ** attempt, max_backoff)
            print(f"⏳ Attempt {attempt + 1} failed, retrying in {wait_time:.1f}s...")
            await asyncio.sleep(wait_time)
            
        except Exception as e:
            # Don't retry non-retryable errors
            raise e

# Usage example
async def robust_search(collection_id, query_vector):
    client = UnifiedClient("http://localhost:5678")
    
    return await retry_with_backoff(
        client.search_vectors,
        max_retries=3,
        collection_id=collection_id,
        query_vector=query_vector,
        k=10
    )

=== Performance Guidelines

**Protocol Selection:**
* **Use gRPC** for high-throughput batch processing (3.4x faster than REST)
* **Use REST** for simple integrations and web applications
* **Use AUTO** (default) for automatic selection based on workload

**Batch Size Optimization:**
* **gRPC optimal**: 3,000 vectors per batch (77,872 vectors/second)
* **REST optimal**: 1,500 vectors per batch (21,273 vectors/second)
* **Small batches**: Use 100-500 vectors for low latency
