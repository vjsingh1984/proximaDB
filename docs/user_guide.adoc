== Overview

ProximaDB is a high-performance vector database with unified REST/gRPC APIs and optimized storage engines.

== Collections

=== Create Collection
[source,python]
----
from proximadb import ProximaDBClient

client = ProximaDBClient("http://localhost:5678")

collection = client.create_collection({
    "name": "documents",
    "dimension": 768,
    "distance_metric": "cosine"
})
----

=== Manage Collections
[source,python]
----
# List collections
collections = client.list_collections()

# Get collection
collection = client.get_collection("documents")

# Delete collection
client.delete_collection("documents")
----

== Vector Operations

=== Insert Vectors
[source,python]
----
# Single vector
client.insert_vector(
    collection_id="documents",
    vector_id="doc_1",
    vector=[0.1, 0.2, 0.3],
    metadata={"title": "My Document"}
)

# Batch insert
vectors = [
    {"id": "doc_1", "vector": [0.1, 0.2], "metadata": {"title": "Doc 1"}},
    {"id": "doc_2", "vector": [0.3, 0.4], "metadata": {"title": "Doc 2"}}
]
client.batch_insert("documents", vectors)
----

=== Search Vectors
[source,python]
----
# Basic search
results = client.search_vectors(
    collection_id="documents",
    query_vector=[0.1, 0.2],
    k=10
)

# Search with metadata filter
results = client.search_vectors(
    collection_id="documents", 
    query_vector=[0.1, 0.2],
    k=10,
    metadata_filter={"category": "tech"}
)
----

== BERT Integration

=== Basic Setup
[source,python]
----
from transformers import AutoTokenizer, AutoModel
from proximadb import ProximaDBClient

# BERT embedding service
class BERTEmbeddings:
    def __init__(self, model="bert-base-uncased"):
        self.tokenizer = AutoTokenizer.from_pretrained(model)
        self.model = AutoModel.from_pretrained(model)
    
    def encode(self, text):
        inputs = self.tokenizer(text, return_tensors="pt", truncation=True)
        outputs = self.model(**inputs)
        return outputs.last_hidden_state[:, 0, :].squeeze().tolist()

# Usage
bert = BERTEmbeddings()
client = ProximaDBClient("http://localhost:5678")

# Create collection for BERT
collection = client.create_collection({
    "name": "bert_docs",
    "dimension": 768,
    "distance_metric": "cosine"
})

# Generate and insert embeddings
text = "Machine learning is transforming data processing"
embedding = bert.encode(text)
client.insert_vector("bert_docs", "doc_1", embedding, {"text": text})
----

== Error Handling

=== Basic Error Handling
[source,python]
----
from proximadb.exceptions import ProximaDBException, CollectionNotFoundError

try:
    collection = client.get_collection("my_collection")
except CollectionNotFoundError:
    print("Collection not found")
except ProximaDBException as e:
    print(f"Error: {e}")
----

== Best Practices

* Use batch operations for better performance
* Normalize vectors when using cosine similarity  
* Add meaningful metadata for filtering
* Implement retry logic for production use
* Use appropriate distance metrics for your data type
