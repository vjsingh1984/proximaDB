== Executive Summary

ProximaDB's unified compression architecture provides centralized, context-aware compression across all system components. The system supports 14 compression algorithms including a unique "Mixed" strategy that applies optimal compression per data type, achieving 50-80% storage reduction while maintaining query performance.

== Implementation Architecture

=== Core Compression Engine

[source,rust]
----
pub struct StandardCompression {
    algorithm: CompressionAlgorithm,
    level: Option<u32>,
}

#[derive(Debug, Clone)]
pub enum CompressionAlgorithm {
    None,
    Lz4,
    Snappy,
    Gzip,
    Brotli,
    Bzip2,
    Deflate,
    Xz,
    Zlib,
    Lzo,
    Lz4Hc,
    Lzma,
    Zstd,
    Mixed(MixedCompressionStrategy),
}

impl StandardCompression {
    /// Context-aware compression
    pub fn compress(
        &self,
        data: &[u8],
        context: CompressionContext,
    ) -> Result<Vec<u8>> {
        match self.algorithm {
            Mixed(ref strategy) => {
                self.compress_mixed(data, context, strategy)
            }
            _ => self.compress_single(data)
        }
    }
    
    /// Mixed strategy implementation
    fn compress_mixed(
        &self,
        data: &[u8],
        context: CompressionContext,
        strategy: &MixedCompressionStrategy,
    ) -> Result<Vec<u8>> {
        // Determine data type from context
        let data_type = self.detect_data_type(data, context)?;
        
        // Select optimal algorithm
        let algorithm = strategy.select_algorithm(data_type);
        
        // Apply selected compression
        self.compress_with_algorithm(data, algorithm)
    }
}

SST Block Structure with Compression:

┌──────────────────────────────────────┐
│ Block Header (16 bytes)              │
├──────────────────────────────────────┤
│ • Magic: 0xBLK (4 bytes)             │
│ • Compressed size (4 bytes)          │
│ • Uncompressed size (4 bytes)        │
│ • Compression type (1 byte)          │
│ • CRC32 (4 bytes)                    │
├──────────────────────────────────────┤
│ Compressed Data                      │
│ ┌──────────────────────────────────┐ │
│ │ VectorRecord 1                   │ │
│ │ VectorRecord 2                   │ │
│ │ ...                              │ │
│ │ VectorRecord N                   │ │
│ └──────────────────────────────────┘ │
└──────────────────────────────────────┘

Compression Flow:
1. Collect records into block
2. Serialize to bytes
3. Apply compression algorithm
4. Write header + compressed data
5. Update index with offsets
....

[source,rust]
----
impl SstBlockWriter {
    pub fn write_compressed_block(
        &mut self,
        records: &[VectorRecord],
    ) -> Result<BlockMetadata> {
        // Serialize records
        let uncompressed = self.serialize_records(records)?;
        
        // Apply compression based on config
        let compressed = self.compression.compress(
            &uncompressed,
            CompressionContext::SstBlock,
        )?;
        
        // Calculate compression ratio
        let ratio = compressed.len() as f32 / uncompressed.len() as f32;
        
        // Write block with header
        let offset = self.writer.position();
        self.write_block_header(
            compressed.len(),
            uncompressed.len(),
            self.compression.algorithm(),
        )?;
        self.writer.write_all(&compressed)?;
        
        Ok(BlockMetadata {
            offset,
            compressed_size: compressed.len(),
            uncompressed_size: uncompressed.len(),
            compression_ratio: ratio,
            record_count: records.len(),
        })
    }
}

impl ViperWriter {
    /// Configure Parquet writer with mixed compression
    fn create_writer_properties(&self) -> WriterProperties {
        let mut builder = WriterProperties::builder()
            .set_compression(Compression::SNAPPY); // Default
        
        if let CompressionAlgorithm::Mixed(ref strategy) = self.compression {
            // Apply per-column compression
            builder = builder
                .set_column_compression(
                    "vector_fp32".into(),
                    Compression::LZ4,
                )
                .set_column_compression(
                    "vector_quantized".into(),
                    Compression::SNAPPY,
                )
                .set_column_compression(
                    "metadata".into(),
                    Compression::BROTLI(BrotliLevel::default()),
                )
                .set_column_compression(
                    "id".into(),
                    Compression::GZIP(GzipLevel::default()),
                );
        }
        
        builder.build()
    }
}

│ • Reuse for batch operations        │
└─────────────────────────────────────┘

Allocation Strategy:
1. Try to get from pool
2. If pool empty, allocate new
3. Return to pool after use
4. Periodic cleanup of unused buffers
....

[source,rust]
----
pub struct CompressionBufferPool {
    small: Arc<Mutex<Vec<Vec<u8>>>>,   // <1MB
    medium: Arc<Mutex<Vec<Vec<u8>>>>,  // 1-10MB  
    large: Arc<Mutex<Vec<Vec<u8>>>>,   // >10MB
}

impl CompressionBufferPool {
    pub fn acquire(&self, size: usize) -> PooledBuffer {
        let buffer = if size < 1024 * 1024 {
            self.small.lock().unwrap().pop()
        } else if size < 10 * 1024 * 1024 {
            self.medium.lock().unwrap().pop()
        } else {
            self.large.lock().unwrap().pop()
        };
        
        PooledBuffer {
            buffer: buffer.unwrap_or_else(|| vec![0; size]),
            pool: Arc::clone(self),
            size_category: self.categorize_size(size),
        }
    }
}

mod tests {
    #[test]
    fn test_mixed_strategy_selection() {
        let strategy = MixedCompressionStrategy::default();
        
        // Test algorithm selection per data type
        assert_eq!(
            strategy.select_algorithm(DataType::BinaryVector),
