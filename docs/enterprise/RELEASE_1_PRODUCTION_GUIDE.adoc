=== Deployment Options

==== Option 1: One-Click Enterprise Deployment

```bash

./scripts/deploy_enterprise_release_1.sh \
  --compliance financial \
  --tenant-count 1000 \
```

==== Option 2: Kubernetes Enterprise Deployment

```bash
# Deploy with Kubernetes manifests
kubectl apply -f k8s/release-1/

# Configure enterprise settings
kubectl patch configmap proximadb-enterprise-config \
  -n proximadb-enterprise \
  --patch '{"data":{"max_tenants":"1000","compliance_mode":"strict"}}'

# Scale for enterprise workload
kubectl scale deployment proximadb-enterprise \
  --replicas=10 \
  -n proximadb-enterprise
```

== 🔐 Enterprise Security Configuration

=== SSO Integration Setup

==== AWS IAM Integration

```bash
# Configure AWS IAM integration
kubectl create secret generic aws-iam-config \
  --from-literal=region=us-east-1 \
  --from-literal=account-id=123456789012 \
  --from-literal=role-arn=arn:aws:iam::123456789012:role/ProximaDBEnterpriseRole \
  --namespace=proximadb-enterprise

# Apply AWS IAM role mapping
cat > aws-role-mapping.json << EOF
{
  "role_mappings": [
    {
      "aws_role_arn": "arn:aws:iam::123456789012:role/ProximaDBAdmin",
      "proximadb_role": "tenant_admin",
      "tenant_id": "global_enterprise"
    },
    {
      "aws_role_arn": "arn:aws:iam::123456789012:role/ProximaDBAnalyst", 
      "proximadb_role": "analyst",
      "tenant_id": "global_enterprise"
    }
  ]
}
EOF

kubectl create configmap aws-role-mapping \
  --from-file=aws-role-mapping.json \
  --namespace=proximadb-enterprise
```

== 🎯 Enterprise Customer Onboarding

=== Financial Services Enterprise Setup

```bash
# Step 1: Create financial services tenant
curl -X POST http://proximadb.enterprise.com:5678/api/v2/tenants \
  -H "Authorization: Bearer $AWS_IAM_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "tenant_id": "global_investment_bank",
    "organization_name": "Global Investment Bank Corp",
    "industry": "financial_services",
    "compliance_requirements": ["SOC2", "SOX", "Basel_III"],
    "resource_limits": {
      "max_memory_mb": 32768,
      "max_storage_mb": 2097152,
      "max_operations_per_minute": 100000,
      "max_concurrent_users": 2000,
      "max_collections": 500,
      "max_domains": 50
    }
  }'

# Step 2: Create financial services domains
curl -X POST http://proximadb.enterprise.com:5678/api/v2/tenants/global_investment_bank/domains \
  -H "Authorization: Bearer $AWS_IAM_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "domain_name": "risk_management",
    "business_context": {
      "primary_function": "enterprise_risk_assessment",
      "data_sensitivity": "confidential",
      "performance_requirements": {
        "latency_requirement_ms": 50,
        "throughput_requirement_qps": 10000,
        "availability_requirement": 0.9999
      }
    }
  }'

# Step 3: Link collections to domains
curl -X POST http://proximadb.enterprise.com:5678/api/v2/tenants/global_investment_bank/domains/risk_management/collections/portfolio_risk_vectors/link \
  -H "Authorization: Bearer $AWS_IAM_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "bridge_config": {
      "bridge_type": "contextual",
      "sync_policy": "realtime",
      "auto_entity_creation": true,
      "business_mapping": {
        "vector_to_entity_rules": ["portfolio_risk_mapping"],
        "metadata_extraction": ["risk_score", "asset_class", "geographic_region"]
      }
    }
  }'
```
