==== Phase 1: Metadata Recovery (11.8ms avg)

**Objective**: Restore collection configurations and engine assignments

[source,rust]
----
pub async fn recover_metadata(&self) -> Result<HashMap<String, Collection>> {
    let start = Instant::now();
    
    // Load all collection metadata from persistent backend
    let collections = self.metadata_backend.get_all_collections().await?;
    
    // Initialize storage assignments for discovered collections
    for (collection_id, collection) in &collections {
        let assignment = self.assignment_service
            .assign_collection(collection_id, &self.storage_locations, "hash")
            .await?;
        tracing::info!("Restored assignment for collection {}", collection_id);
    }
    
    let duration = start.elapsed();
    tracing::info!("Metadata recovery completed in {}ms", duration.as_millis());
    Ok(collections)
}
----

**Performance**: 11.8ms for 1 collection, linear scaling up to ~50ms for 50 collections

pub async fn recover_assignments(&self) -> Result<RecoveryStats> {
    let start = Instant::now();

==== Phase 3: Write Buffer Recovery (Direct-to-Storage)

**Objective**: Replay unflushed operations directly to storage engines

[source,rust]
----
pub async fn recover_write_buffer_direct(&self) -> Result<WriteBufferRecoveryStats> {
    let start = Instant::now();
    let mut total_records = 0;
    let mut total_collections = 0;
    
    // Get all WriteBuffer assignments 
    let assignments = self.assignment_service.get_all_assignments().await;
    
    for (collection_id, assignment) in assignments {
        let write_buffer_files = self.discover_write_buffer_files(&assignment.write_buffer_url).await?;
        
        if !write_buffer_files.is_empty() {
            total_collections += 1;
            
            // Read and replay WriteBuffer files directly to storage engines
            for file_path in write_buffer_files {
                let records = self.read_write_buffer_file(&file_path).await?;
                total_records += records.len();
                
                // Direct write to storage engine (bypassing memtable)
                self.storage_engine.write_direct(&collection_id, records).await?;
            }
            
            // Clean up processed WriteBuffer files
            self.cleanup_write_buffer_files(&assignment.write_buffer_url).await?;
        }
    }
    
    let duration = start.elapsed();
    tracing::info!(
        "WriteBuffer recovery: {} records from {} collections in {}ms",
        total_records, total_collections, duration.as_millis()
    );
    
    Ok(WriteBufferRecoveryStats {
        total_records,
        total_collections, 
        recovery_time: duration,
    })
}
----

**Design Principles**:
* **Direct-to-Storage**: Bypasses memtable to avoid memory pressure during recovery
* **Parallel Processing**: Collections recovered concurrently using CPU-based worker pools
* **Automatic Cleanup**: Successfully recovered WriteBuffer files are removed
* **Progress Tracking**: Real-time metrics on recovery progress

==== Parallel Collection Loading Optimization

[source,rust]
----
// Optimized parallel loading based on CPU count
pub async fn load_collections_parallel(&self, collection_ids: Vec<String>) -> Result<()> {
    let num_cpus = num_cpus::get();
    let chunk_size = (collection_ids.len() + num_cpus - 1) / num_cpus;
    let chunk_size = chunk_size.max(1).min(10); // Between 1 and 10 per task
    
    let chunks: Vec<_> = collection_ids.chunks(chunk_size).collect();
    let tasks = chunks.into_iter().map(|chunk| {
        let engine = self.storage_engine.clone();
        let chunk = chunk.to_vec();
        
        tokio::spawn(async move {
            for collection_id in chunk {
                engine.ensure_collection_initialized(&collection_id).await?;
            }
            Ok::<(), Error>(())
        })
    });
    
    // Wait for all parallel tasks to complete
    let results = futures::future::try_join_all(tasks).await?;
    for result in results {
        result??; // Propagate any errors
    }
    
    Ok(())
}
----

**Performance Impact**: Reduced startup time from hanging (sequential) to ~200ms (parallel)

=== Recovery Error Handling

==== Corruption Detection and Recovery

[source,rust]
----
pub async fn handle_corrupted_write_buffer_file(&self, file_path: &str) -> Result<RecoveryAction> {
    match self.attempt_recovery(&file_path).await {
        Ok(records) => {
            tracing::warn!("Recovered {} records from corrupted file {}", records.len(), file_path);
            Ok(RecoveryAction::Recovered(records))
        }
        Err(CorruptionError::HeaderCorruption) => {
            tracing::error!("Header corruption in {}, attempting partial recovery", file_path);
            self.attempt_partial_recovery(&file_path).await
        }
        Err(CorruptionError::ChecksumMismatch) => {
            tracing::error!("Checksum mismatch in {}, skipping file", file_path);
            Ok(RecoveryAction::Skipped)
        }
        Err(CorruptionError::TruncatedFile) => {
            tracing::error!("File {} truncated, recovering valid prefix", file_path);
            self.recover_valid_prefix(&file_path).await
        }
    }
}
----

**Recovery Strategies**:
* **Header Corruption**: Attempt to find valid data blocks
* **Checksum Mismatch**: Skip corrupted blocks, recover valid ones  
* **Truncated Files**: Recover complete records from valid prefix
* **Total Corruption**: Log error and continue with other files
