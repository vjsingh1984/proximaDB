/// Global intelligent routing for multi-region ProximaDB deployment
#[derive(Debug, Clone)]
pub struct GlobalRouter {

    pub primary_strategy: RoutingStrategy,
    pub fallback_strategy: RoutingStrategy,

    pub health_check_interval_seconds: u32,
    pub enable_intelligent_routing: bool,
    pub enable_data_residency_enforcement: bool,

/// Routing strategies for global deployment
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum RoutingStrategy {
    LatencyOptimized,      // Route to lowest latency region
    DataResidency,         // Route based on data residency requirements
    LoadBalanced,          // Distribute load across healthy regions
    GeographicProximity,   // Route to geographically closest region
    CapacityOptimized,     // Route to region with most available capacity
}

impl GlobalRouter {
    /// Create new global router
    pub async fn new(routing_policy: RoutingPolicy) -> Result<Self, anyhow::Error> {
        let regional_endpoints = Arc::new(RwLock::new(HashMap::new()));
        let latency_monitor = Arc::new(LatencyMonitor::new());
        let health_monitor = Arc::new(RegionalHealthMonitor::new());

        // Initialize default regional endpoints
        let router = Self {
            regional_endpoints,
            routing_policy: Arc::new(routing_policy),
            latency_monitor,
            health_monitor,
        };

        // Setup default regions
        router.initialize_default_regions().await?;

        // Start monitoring
        router.start_monitoring().await?;

        Ok(router)
    }

    /// Route request to optimal region based on policy
    pub async fn route_request(&self, request: &GlobalRequest) -> Result<RoutingDecision, anyhow::Error> {
        tracing::debug!("🌍 Routing global request from {} for tenant {}",
                       request.source_region.as_ref().unwrap_or(&"unknown".to_string()),
                       request.tenant_id.as_ref().unwrap_or(&"unknown".to_string()));

        // Step 1: Apply data residency constraints
        let compliant_regions = self.filter_by_data_residency(request).await?;

        if compliant_regions.is_empty() {
            return Err(anyhow::anyhow!("No regions available that meet data residency requirements"));
        }

        // Step 2: Filter by health status
        let healthy_regions = self.filter_by_health_status(&compliant_regions).await?;

        if healthy_regions.is_empty() {
            tracing::warn!("No healthy regions available, using degraded regions for continuity");
            // Use compliant regions even if degraded for business continuity
        }

        let available_regions = if healthy_regions.is_empty() { compliant_regions } else { healthy_regions };

        // Step 3: Apply routing strategy
        let selected_region = match self.routing_policy.primary_strategy {
            RoutingStrategy::LatencyOptimized => {
                self.select_lowest_latency_region(&available_regions, request).await?
            }
            RoutingStrategy::DataResidency => {
                self.select_data_residency_optimal_region(&available_regions, request).await?
            }
            RoutingStrategy::LoadBalanced => {
                self.select_load_balanced_region(&available_regions).await?
            }
            RoutingStrategy::GeographicProximity => {
                self.select_geographically_closest_region(&available_regions, request).await?
            }
            RoutingStrategy::CapacityOptimized => {
                self.select_highest_capacity_region(&available_regions).await?
            }
        };

        // Step 4: Get endpoint details
        let endpoints = self.regional_endpoints.read().await;
        let endpoint = endpoints.get(&selected_region)
            .ok_or_else(|| anyhow::anyhow!("Endpoint not found for region: {:?}", selected_region))?;

        let routing_decision = RoutingDecision {
            selected_region: selected_region.clone(),
            endpoint: endpoint.primary_endpoint.clone(),
            backup_endpoints: endpoint.backup_endpoints.clone(),
            estimated_latency_ms: endpoint.current_latency_ms,
            routing_reason: format!("Selected based on {:?} strategy", self.routing_policy.primary_strategy),
            data_residency_compliant: true,
            expected_performance: self.estimate_performance(&selected_region, request).await?,
        };

        tracing::info!("✅ Routed request to region {:?}: {} (latency: {:.1}ms)",
                      selected_region, endpoint.primary_endpoint, endpoint.current_latency_ms);

        Ok(routing_decision)
    }

        let latency_monitor = self.latency_monitor.clone();
        let endpoints = self.regional_endpoints.clone();

        tokio::spawn(async move {
            let mut interval = tokio::time::interval(
                std::time::Duration::from_secs(30) // Check every 30 seconds

            loop {
                interval.tick().await;

                // Update health status for all regions
                let regions: Vec<Region> = {
                    let endpoints_read = endpoints.read().await;
                    endpoints_read.keys().cloned().collect()

                    }
                }
            }
        });

        tracing::info!("✅ Global monitoring started for all regions");
        Ok(())
    }

use serde::{Deserialize, Serialize};
use std::collections::HashMap;

/// Data residency enforcement for global compliance
#[derive(Debug, Clone)]
pub struct DataResidencyEngine {
    residency_rules: Arc<RwLock<HashMap<String, ResidencyRule>>>,
    compliance_monitor: Arc<ComplianceMonitor>,
    violation_handler: Arc<ViolationHandler>,
}

/// Data residency rule for specific jurisdictions
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ResidencyRule {
    pub rule_id: String,
    pub jurisdiction: String,              // "EU", "US", "APAC", "Canada", etc.
    pub allowed_regions: Vec<Region>,
    pub prohibited_regions: Vec<Region>,
    pub data_types_covered: Vec<DataType>,
    pub legal_basis: String,
    pub enforcement_level: EnforcementLevel,
    pub audit_requirements: Vec<AuditRequirement>,
}

/// Types of data subject to residency rules
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum DataType {
    PersonalData,          // GDPR personal data
    FinancialData,         // Financial records
    HealthData,            // HIPAA-covered health information
    GovernmentData,        // Government or defense data
    IntellectualProperty,  // IP and trade secrets
    CorporateData,         // General corporate information
}

/// Enforcement levels for compliance
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum EnforcementLevel {
    Strict,     // No exceptions, immediate blocking
    Monitored,  // Allow but log for audit
    Advisory,   // Warn but allow
}

impl DataResidencyEngine {
    /// Validate data operation against residency rules
    pub async fn validate_data_operation(&self, operation: &DataOperation) -> Result<ResidencyValidation, anyhow::Error> {
        tracing::debug!("🌍 Validating data residency for operation: {:?}", operation.operation_type);

        // Step 1: Identify applicable residency rules
        let applicable_rules = self.get_applicable_rules(operation).await?;

        if applicable_rules.is_empty() {
            return Ok(ResidencyValidation {
                allowed: true,
                applicable_rules: vec![],
                violations: vec![],
                required_actions: vec![],
                compliance_status: ComplianceStatus::Compliant,
            });
        }

        // Step 2: Check each rule for violations
        let mut violations = Vec::new();
        let mut required_actions = Vec::new();

        for rule in &applicable_rules {
            if let Some(violation) = self.check_rule_violation(rule, operation).await? {
                violations.push(violation.clone());

                match rule.enforcement_level {
                    EnforcementLevel::Strict => {
                        required_actions.push(RequiredAction {
                            action_type: ActionType::BlockOperation,
                            description: format!("Operation blocked due to strict residency rule: {}", rule.rule_id),
                            deadline: Some(Utc::now()), // Immediate
                        });
                    }
                    EnforcementLevel::Monitored => {
                        required_actions.push(RequiredAction {
                            action_type: ActionType::LogViolation,
                            description: format!("Residency violation logged for monitoring: {}", violation.description),
                            deadline: None,
                        });
                    }
                    EnforcementLevel::Advisory => {
                        required_actions.push(RequiredAction {
                            action_type: ActionType::WarnUser,
                            description: format!("Advisory: Consider data residency implications: {}", violation.description),
                            deadline: None,
                        });
                    }
                }
            }
        }

        // Step 3: Determine overall compliance status
        let compliance_status = if violations.iter().any(|v| v.severity == ViolationSeverity::Critical) {
            ComplianceStatus::NonCompliant
        } else if violations.iter().any(|v| v.severity == ViolationSeverity::High) {
            ComplianceStatus::PartiallyCompliant
        } else if violations.is_empty() {
            ComplianceStatus::Compliant
        } else {
            ComplianceStatus::PartiallyCompliant
        };

        let validation = ResidencyValidation {
            allowed: compliance_status != ComplianceStatus::NonCompliant,
            applicable_rules: applicable_rules.clone(),
            violations,
            required_actions,
            compliance_status,
        };

        if validation.allowed {
            tracing::debug!("✅ Data residency validation passed");
        } else {
            tracing::warn!("🚨 Data residency validation failed: {} violations", validation.violations.len());
        }

        Ok(validation)
    }

    /// Get residency rules applicable to data operation
    async fn get_applicable_rules(&self, operation: &DataOperation) -> Result<Vec<ResidencyRule>, anyhow::Error> {
        let rules = self.residency_rules.read().await;
        let mut applicable_rules = Vec::new();

        for rule in rules.values() {
            // Check if rule applies to this data type
            if rule.data_types_covered.contains(&operation.data_type) {
                // Check if rule applies to this jurisdiction
                if self.operation_in_jurisdiction(operation, &rule.jurisdiction).await? {
                    applicable_rules.push(rule.clone());
                }
            }
        }

        Ok(applicable_rules)
    }

    /// Check if operation violates specific residency rule
    async fn check_rule_violation(&self, rule: &ResidencyRule, operation: &DataOperation) -> Result<Option<ResidencyViolation>, anyhow::Error> {
        // Check if target region is prohibited
        if rule.prohibited_regions.contains(&operation.target_region) {
            return Ok(Some(ResidencyViolation {
                rule_id: rule.rule_id.clone(),
                violation_type: ViolationType::ProhibitedRegion,
                description: format!("Operation targets prohibited region {:?} for jurisdiction {}", operation.target_region, rule.jurisdiction),
                severity: ViolationSeverity::Critical,
                detected_at: Utc::now(),
            }));
        }

        // Check if target region is allowed (if allowlist exists)
        if !rule.allowed_regions.is_empty() && !rule.allowed_regions.contains(&operation.target_region) {
            return Ok(Some(ResidencyViolation {
                rule_id: rule.rule_id.clone(),
                violation_type: ViolationType::UnauthorizedRegion,
                description: format!("Operation targets region {:?} not in allowlist for jurisdiction {}", operation.target_region, rule.jurisdiction),
                severity: ViolationSeverity::High,
                detected_at: Utc::now(),
            }));
        }

        // No violations found
        Ok(None)
    }
}

// File: src/global/replication/cross_region_sync.rs
use crate::storage::engines::factory::StorageEngineFactory;

/// Cross-region data synchronization for global deployment
#[derive(Debug, Clone)]
pub struct CrossRegionSynchronizer {
    replication_config: Arc<ReplicationConfig>,
    region_coordinators: Arc<RwLock<HashMap<Region, RegionCoordinator>>>,
    conflict_resolver: Arc<ConflictResolver>,
    sync_monitor: Arc<SyncMonitor>,
}

/// Configuration for cross-region replication
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ReplicationConfig {
    pub replication_strategy: ReplicationStrategy,
    pub consistency_level: ConsistencyLevel,
    pub sync_interval_seconds: u32,
    pub conflict_resolution_policy: ConflictResolutionPolicy,
    pub enable_cross_region_backup: bool,
    pub max_replication_lag_seconds: u32,
}

/// Replication strategies for global deployment
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ReplicationStrategy {
    MasterSlave,          // Single master, multiple read replicas
    MultiMaster,          // Multiple masters with conflict resolution
    RegionalMaster,       // Master per region with cross-region sync
    EventualConsistency,  // Async replication with eventual consistency
}

/// Consistency levels for global operations
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ConsistencyLevel {
    Strong,              // Synchronous cross-region confirmation
    Eventual,            // Asynchronous replication
    SessionConsistency,  // Consistent within user session
    MonotonicRead,       // Read-after-write consistency
}

impl CrossRegionSynchronizer {
    /// Synchronize data across all configured regions
    pub async fn synchronize_tenant_data(&self, tenant_id: &str, sync_request: SyncRequest) -> Result<SyncResult, anyhow::Error> {
        tracing::info!("🔄 Starting cross-region synchronization for tenant: {}", tenant_id);

        let start_time = std::time::Instant::now();

        // Step 1: Identify source and target regions
        let source_region = sync_request.source_region;
        let target_regions = sync_request.target_regions.clone();

        // Step 2: Extract data to synchronize
        let sync_data = self.extract_tenant_data(tenant_id, &source_region).await?;

        // Step 3: Apply data residency validation
        let residency_validation = self.validate_cross_region_sync(tenant_id, &sync_data, &target_regions).await?;

        if !residency_validation.allowed {
            return Err(anyhow::anyhow!("Cross-region sync blocked by data residency rules"));
        }

        // Step 4: Execute synchronization to each target region
        let mut region_results = HashMap::new();

        for target_region in &target_regions {
            match self.sync_to_region(tenant_id, &sync_data, target_region).await {
                Ok(region_result) => {
                    region_results.insert(target_region.clone(), region_result);
                    tracing::debug!("✅ Sync completed to region: {:?}", target_region);
                }
                Err(e) => {
                    tracing::error!("❌ Sync failed to region {:?}: {}", target_region, e);
                    region_results.insert(target_region.clone(), RegionSyncResult {
                        region: target_region.clone(),
                        success: false,
                        error: Some(e.to_string()),
                        sync_time_ms: 0,
                        records_synced: 0,
                    });
                }
            }
        }

        // Step 5: Verify synchronization consistency
        let consistency_check = self.verify_cross_region_consistency(tenant_id, &target_regions).await?;

        let total_sync_time = start_time.elapsed().as_millis() as u64;

        let sync_result = SyncResult {
            tenant_id: tenant_id.to_string(),
            source_region,
            target_regions,
            region_results,
            consistency_verified: consistency_check.consistent,
            total_sync_time_ms: total_sync_time,
            records_synced: sync_data.total_records,
            compliance_status: residency_validation.compliance_status,
        };

        tracing::info!("✅ Cross-region sync complete for tenant {}: {} regions in {}ms",
                      tenant_id, target_regions.len(), total_sync_time);

        Ok(sync_result)
    }

    /// Extract tenant data for synchronization
    async fn extract_tenant_data(&self, tenant_id: &str, source_region: &Region) -> Result<TenantSyncData, anyhow::Error> {
        // Get region coordinator for source region
        let coordinators = self.region_coordinators.read().await;
        let coordinator = coordinators.get(source_region)
            .ok_or_else(|| anyhow::anyhow!("No coordinator found for region: {:?}", source_region))?;

        // Extract all tenant data
        let collections = coordinator.get_tenant_collections(tenant_id).await?;
        let vectors = coordinator.get_tenant_vectors(tenant_id).await?;
        let metadata = coordinator.get_tenant_metadata(tenant_id).await?;

        Ok(TenantSyncData {
            tenant_id: tenant_id.to_string(),
            collections,
            vectors,
            metadata,
            total_records: vectors.len(),
            extracted_at: Utc::now(),
        })
    }

    /// Synchronize data to specific target region
    async fn sync_to_region(&self, tenant_id: &str, sync_data: &TenantSyncData, target_region: &Region) -> Result<RegionSyncResult, anyhow::Error> {
        let sync_start = std::time::Instant::now();

        // Get target region coordinator
        let coordinators = self.region_coordinators.read().await;
        let target_coordinator = coordinators.get(target_region)
            .ok_or_else(|| anyhow::anyhow!("No coordinator found for target region: {:?}", target_region))?;

        // Sync collections
        target_coordinator.sync_collections(tenant_id, &sync_data.collections).await?;

        // Sync vectors with batching for large datasets
        let batch_size = 1000;
        let mut synced_vectors = 0;

        for vector_batch in sync_data.vectors.chunks(batch_size) {
            target_coordinator.sync_vector_batch(tenant_id, vector_batch).await?;
            synced_vectors += vector_batch.len();

            // Progress logging for large syncs
            if synced_vectors % 10000 == 0 {
                tracing::debug!("📊 Synced {} vectors to region {:?}", synced_vectors, target_region);
            }
        }

        // Sync metadata
        target_coordinator.sync_metadata(tenant_id, &sync_data.metadata).await?;

        let sync_time_ms = sync_start.elapsed().as_millis() as u64;

        Ok(RegionSyncResult {
            region: target_region.clone(),
            success: true,
            error: None,
            sync_time_ms,
            records_synced: sync_data.total_records,
        })
    }
}

DEPLOYMENT_ID="global_$(date +%Y%m%d_%H%M%S)"

# Function to deploy to specific region
deploy_to_region() {
    local region=$1
    local region_config=$2

    echo ""
    echo "🌐 Deploying to region: $region"
    echo "================================"

    # Step 1: Generate region-specific configuration
    echo "⚙️ Generating region-specific configuration..."
    python3 scripts/global_deployment/generate_region_config.py \
        --template "$CONFIG_TEMPLATE" \
        --region "$region" \
        --output "config/generated/proximadb-${region}.toml"

    if [ $? -ne 0 ]; then
        echo "❌ Failed to generate configuration for region: $region"
        return 1
    fi

    # Step 2: Deploy ProximaDB to region
    echo "🚀 Deploying ProximaDB to $region..."

    case $region in
        "us-east"|"us-west")
            # Deploy to AWS
            ./scripts/cloud_deployment/deploy_aws.sh \
                --region "$region" \
                --config "config/generated/proximadb-${region}.toml" \
                --deployment-id "$DEPLOYMENT_ID"
            ;;
        "eu-west"|"eu-central")
            # Deploy to Azure (GDPR compliance)
            ./scripts/cloud_deployment/deploy_azure.sh \
                --region "$region" \
                --config "config/generated/proximadb-${region}.toml" \
                --deployment-id "$DEPLOYMENT_ID" \
                --enable-gdpr-compliance
            ;;
        "apac-"*)
            # Deploy to GCP (APAC optimization)
            ./scripts/cloud_deployment/deploy_gcp.sh \
                --region "$region" \
                --config "config/generated/proximadb-${region}.toml" \
                --deployment-id "$DEPLOYMENT_ID"
            ;;
    esac

    if [ $? -eq 0 ]; then
        echo "✅ ProximaDB deployed successfully to region: $region"
    else
        echo "❌ Deployment failed for region: $region"
        return 1
    fi

    # Step 3: Validate regional deployment
    echo "🔍 Validating deployment in region: $region..."
    ./scripts/enterprise_deployment/validate_deployment.sh "$DEPLOYMENT_ID" "$region"

    if [ $? -eq 0 ]; then
        echo "✅ Deployment validation passed for region: $region"
    else
        echo "❌ Deployment validation failed for region: $region"
        return 1
    fi

    # Step 4: Configure regional monitoring
    echo "📊 Setting up monitoring for region: $region..."
    ./scripts/monitoring/setup_regional_monitoring.sh "$region" "$DEPLOYMENT_ID"

    echo "🎉 Region $region deployment complete!"
    return 0
}

    else
        echo "❌ Global routing setup failed"
    fi
else
    echo "❌ No successful regional deployments - cannot setup global routing"
    exit 1
fi

# Setup cross-region replication
echo ""
echo "🔄 Setting up cross-region data replication..."
echo "=============================================="

./scripts/global_deployment/setup_cross_region_replication.sh \

if [ $? -eq 0 ]; then

else

# Final global deployment summary
echo ""
echo "🎉 GLOBAL PROXIMADB DEPLOYMENT COMPLETE"
echo "========================================"
echo "Deployment ID: $DEPLOYMENT_ID"
echo "Successful Regions: ${SUCCESSFUL_DEPLOYMENTS[*]}"
echo "Failed Regions: ${FAILED_DEPLOYMENTS[*]}"
echo ""
echo "Global Endpoints:"
echo "- Primary: https://global.proximadb.com"

echo "- APAC: https://apac.proximadb.com"
echo ""
echo "✅ ProximaDB is now globally deployed and operational!"
