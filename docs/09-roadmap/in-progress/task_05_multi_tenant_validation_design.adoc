async fn validate_collection_isolation(
    test_env: &MultiTenantTestEnv,
    tenant_a: &TestTenant,
    tenant_b: &TestTenant,
) {
    // Create collections with sensitive data for each tenant
    let financial_collection = test_env.create_collection(
        &tenant_a,
        "sensitive_financial_data",
        create_financial_test_data(),
    ).await.expect("Failed to create financial collection");

    let healthcare_collection = test_env.create_collection(
        &tenant_b,
        "sensitive_healthcare_data",
        create_healthcare_test_data(),
    ).await.expect("Failed to create healthcare collection");

    // SECURITY TEST: Attempt cross-tenant collection access
    println!("  🔍 Testing cross-tenant collection access prevention...");

    let cross_access_result = test_env.get_collection(
        &tenant_b, // Healthcare tenant
        &financial_collection.id, // Financial collection
    ).await;

    assert!(cross_access_result.is_err() || cross_access_result.unwrap().is_none(),
           "🚨 SECURITY VIOLATION: Cross-tenant collection access should be blocked!");

    println!("    ✅ Cross-tenant collection access properly blocked");

    // SECURITY TEST: Validate tenant ownership verification
    let legitimate_access = test_env.get_collection(
        &tenant_a, // Financial tenant
        &financial_collection.id, // Their own collection
    ).await;

    assert!(legitimate_access.is_ok() && legitimate_access.unwrap().is_some(),
           "❌ FUNCTIONALITY ISSUE: Legitimate collection access should work");

    println!("    ✅ Legitimate collection access works correctly");
}

async fn validate_search_isolation(
    test_env: &MultiTenantTestEnv,
    tenant_a: &TestTenant,
    tenant_b: &TestTenant,
) {
    println!("  🔍 Testing vector search isolation...");

    // Insert sensitive data for tenant A
    let sensitive_vectors = create_sensitive_healthcare_vectors(&tenant_a.id);
    test_env.insert_vectors(&tenant_a, "healthcare_data", sensitive_vectors).await
        .expect("Failed to insert healthcare vectors");

    // SECURITY TEST: Cross-tenant search attempt
    let malicious_search = test_env.vector_search(VectorSearchRequest {
        tenant_id: tenant_b.id.clone(), // Wrong tenant
        collection_id: "healthcare_data".to_string(),
        query_vector: vec![0.1; 384],
        k: 10,
        user_context: create_user_context(&tenant_b.id),
    }).await;

    // Should either fail or return empty results
    match malicious_search {
        Ok(results) => {
            assert!(results.is_empty(),
                   "🚨 SECURITY VIOLATION: Cross-tenant search returned data!");
            println!("    ✅ Cross-tenant search properly returns empty results");
        }
        Err(_) => {
            println!("    ✅ Cross-tenant search properly rejected");
        }
    }

    // SECURITY TEST: Metadata filter injection
    let injection_search = test_env.vector_search(VectorSearchRequest {
        tenant_id: tenant_b.id.clone(),
        collection_id: "healthcare_data".to_string(),
        query_vector: vec![0.1; 384],
        k: 10,
        metadata_filters: Some(json!({
            "tenant_id": tenant_a.id // Attempting to access other tenant's data
        })),
        user_context: create_user_context(&tenant_b.id),
    }).await;

    // Should not return tenant A's data
    if let Ok(results) = injection_search {
        for result in results {
            let result_tenant = result.metadata.get("tenant_id").unwrap();
            assert_eq!(result_tenant, &tenant_b.id,
                      "🚨 SECURITY VIOLATION: Metadata filter injection successful!");
        }
        println!("    ✅ Metadata filter injection properly blocked");
    }
}

async fn validate_performance_isolation(
    test_env: &MultiTenantTestEnv,
    tenants: Vec<&TestTenant>,
) {
    println!("  🔍 Testing performance isolation...");

    let mut tenant_handles = Vec::new();

    // Start concurrent load for each tenant
    for tenant in tenants {
        let test_env_clone = test_env.clone();
        let tenant_clone = tenant.clone();

        let handle = tokio::spawn(async move {
            let start_time = std::time::Instant::now();
            let mut successful_queries = 0;

            // Execute 100 queries per tenant
            for _ in 0..100 {
                let query = create_tenant_query(&tenant_clone.id);
                match test_env_clone.vector_search(query).await {
                    Ok(_) => successful_queries += 1,
                    Err(e) => println!("    ⚠️ Query failed for tenant {}: {}", tenant_clone.id, e),
                }

                // Small delay to simulate realistic usage
                tokio::time::sleep(std::time::Duration::from_millis(10)).await;
            }

            let duration = start_time.elapsed();
            let qps = successful_queries as f64 / duration.as_secs_f64();

            TenantPerformanceResult {
                tenant_id: tenant_clone.id.clone(),
                qps,
                successful_queries,
                duration,
            }
        });

        tenant_handles.push(handle);
    }

    // Wait for all tenants and collect results
    let tenant_results: Vec<TenantPerformanceResult> = futures::future::join_all(tenant_handles)
        .await
        .into_iter()
        .map(|r| r.unwrap())
        .collect();

    // Validate performance isolation
    let qps_values: Vec<f64> = tenant_results.iter().map(|r| r.qps).collect();
    let avg_qps = qps_values.iter().sum::<f64>() / qps_values.len() as f64;
    let min_qps = qps_values.iter().fold(f64::INFINITY, |a, &b| a.min(b));
    let max_qps = qps_values.iter().fold(0.0, |a, &b| a.max(b));

    // Check isolation quality (no tenant should get <50% of average)
    let isolation_threshold = avg_qps * 0.5;
    let isolation_maintained = qps_values.iter().all(|&qps| qps >= isolation_threshold);

    println!("    📊 Performance Isolation Results:");
    println!("       Average QPS: {:.1}", avg_qps);
    println!("       QPS Range: {:.1} - {:.1}", min_qps, max_qps);
    println!("       Isolation Maintained: {}", if isolation_maintained { "✅ YES" } else { "❌ NO" });

    assert!(isolation_maintained,
           "🚨 PERFORMANCE ISOLATION VIOLATION: Some tenants received <50% of average performance");

    println!("    ✅ Performance isolation validation passed");
}

#[derive(Debug, Clone)]
struct TenantPerformanceResult {
    tenant_id: String,
    qps: f64,
    successful_queries: u32,
    duration: std::time::Duration,
}
