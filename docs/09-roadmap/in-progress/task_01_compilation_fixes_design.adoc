    pub async fn build_context(&self, accessible_tables: &[String]) -> Result<String, anyhow::Error> {
        // Generate schema description for LLM prompts
        let schema_desc = accessible_tables.iter()
            .map(|table| format!("TABLE: {}", table))
            .collect::<Vec<_>>()
            .join("\n");

        Ok(schema_desc)
    }

// File: src/ai/natural_language/sql_validator.rs (MISSING - CREATE)
//! SQL validation for security and safety

pub struct SQLValidator {
    config: ValidatorConfig,
}

#[derive(Debug, Clone)]
pub struct ValidatorConfig {
    pub enable_strict_validation: bool,
}

impl SQLValidator {
    pub async fn new() -> Result<Self, anyhow::Error> {
        Ok(Self {
            config: ValidatorConfig {
                enable_strict_validation: true,
            },
        })
    }

    pub async fn validate_and_sanitize(
        &self,
        sql: &str,
        user_context: &super::translator::UserContext,
    ) -> Result<String, anyhow::Error> {
        // Basic SQL validation - prevent obviously dangerous queries
        let sql_upper = sql.to_uppercase();

        if sql_upper.contains("DROP") || sql_upper.contains("DELETE") || sql_upper.contains("INSERT") {
            return Err(anyhow::anyhow!("Dangerous SQL operations not allowed"));
        }

        // Basic sanitization - ensure SELECT only
        if !sql_upper.trim().starts_with("SELECT") {
            return Err(anyhow::anyhow!("Only SELECT statements allowed"));
        }

        Ok(sql.to_string())
    }
}
