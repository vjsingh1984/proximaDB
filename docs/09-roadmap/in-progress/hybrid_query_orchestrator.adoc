pub async fn execute_frontend(&self, query: Query) -> Result<QueryResult> {
    let plan = self.planner.create_plan(&query)?;
    
    match plan.execution_strategy {
        ExecutionStrategy::VectorOnly => self.executor.execute_vector_plan(plan).await,
        ExecutionStrategy::GraphOnly => self.executor.execute_graph_plan(plan).await,
        ExecutionStrategy::Hybrid => self.executor.execute_hybrid_plan(plan).await,
        ExecutionStrategy::Relational => Err(anyhow!("Not yet implemented")),
    }
}

pub async fn execute_hybrid_plan(&self, plan: ExecutionPlan) -> Result<QueryResult> {
    // 1. Partition operations by type
    let mut vector_ops = Vec::new();
    let mut graph_ops = Vec::new();
    
    // 2. Concurrent execution with tokio::join!
    let (vector_results, graph_results) = tokio::join!(vector_fut, graph_fut);
    
    // 3. Bidirectional seeding
    if graph_needs_seeds && !graph_ops.is_empty() {
        // Vector → Graph seeding (up to 100 nodes)
        let seeds = vector_results.iter().map(|r| extract_id(r)).take(100).collect();
        graph_results.extend(execute_graph_traversal(&seeds).await?);
    }
    
    // 4. Graph → Vector seeding with strategies
    match plan.seeding_strategy {
        SeedingStrategy::Average => /* average 32 embeddings */,
        SeedingStrategy::PerSeed => /* individual SIMILAR queries */,
        SeedingStrategy::None => /* catalog-only derivation */,
    }
    
    // 5. Advanced fusion algorithms
    let fused = apply_fusion_algorithm(&vector_results, &graph_results, &strategy, &weights)?;
}

// src/query/execution/executor.rs:907-954 (RRF Algorithm)
fn apply_reciprocal_rank_fusion(&self, vector_results: &[QueryRow], graph_results: &[QueryRow], k: f64) -> Result<Vec<QueryRow>> {
    let mut fused_results = std::collections::HashMap::new();
    
    // RRF Formula: score = 1 / (k + rank_in_list)
    for (rank, result) in vector_results.iter().enumerate() {
        let rrf_score = 1.0 / (k + rank as f64 + 1.0);
        fused_results.entry(result_id).or_insert(result.clone()).similarity_score = Some(rrf_score);
    }
    
    // Merge graph scores with RRF
    for (rank, result) in graph_results.iter().enumerate() {
        let rrf_score = 1.0 / (k + rank as f64 + 1.0);
        if let Some(existing) = fused_results.get_mut(&result_id) {
            existing.similarity_score = Some(existing.similarity_score.unwrap_or(0.0) + rrf_score);
        }
    }
    
    // Sort by combined RRF score
    sorted_results.sort_by(|a, b| b.similarity_score.partial_cmp(&a.similarity_score).unwrap_or(Equal));
}

match seeding_strategy {
    SeedingStrategy::Average => {
        // Average up to 32 seed embeddings into single vector
        let mut acc = Vec::new();
        for entity_id in seeds.iter().take(32) {
            if let Some(embedding) = store.get_embedding(entity_id) {
                if acc.is_empty() { acc = embedding.clone(); }
                else { for i in 0..acc.len() { acc[i] += embedding[i]; } }
                count += 1.0;
            }
        }
        if count > 0.0 { for i in 0..acc.len() { acc[i] /= count; } }
        
        // Execute SIMILAR search with averaged embedding
        let similar_results = execute_vector_search(&collection, &acc, 50, "cosine").await?;
    },
    
    SeedingStrategy::PerSeed => {
        // Individual SIMILAR query per seed
        for entity_id in seeds {
            if let Some(embedding) = store.get_embedding(&entity_id) {
                let results = execute_vector_search(&collection, &embedding, 10, "cosine").await?;
                vector_results.extend(results);
            }
        }
    },
    
    SeedingStrategy::None => {
        // Catalog-only: derive minimal rows from entity IDs

        vector_results.extend(derived);
    }
