== Implementation Status (Reviewed: 2025-09-09)

- Implemented: Core entity/source fields in proto and REST; storage backends support large payloads; examples include retrieval pipelines.
- In Progress: Standardized size limits/streaming semantics and SDK helpers; end-to-end tests across binary types (images/audio) pending.

== Executive Summary

This document specifies the design for storing source content alongside vector embeddings in ProximaDB. Since embeddings are one-way transformations, the original content (text, images, audio, video) must be stored for retrieval after similarity search. This enhancement introduces a dedicated `source` field separate from metadata, with mandatory IDs for correlation.

== Problem Statement

=== Current Limitations
* Embeddings cannot be reversed to recover original content
* Metadata field is inappropriate for large content storage
* No standardized way to store source documents, images, or multimedia
* Missing correlation between vectors and their source content
* Different content types require different storage strategies

== Proposed Solution

=== Core Design Principles

1. **Mandatory IDs**: Every vector MUST have an ID for source correlation
2. **Separation of Concerns**: Source content ≠ metadata ≠ vectors
3. **Flexible Storage**: Support inline, compressed, and external storage
4. **Engine Optimization**: Each engine optimizes differently
5. **Backward Compatibility**: Optional field, gradual migration

// Storage decision based on content size
impl SstEngine {
    fn store_source(&mut self, id: &str, source: SourceContent) -> Result<()> {
        match source.size_bytes {
            0..=1_000_000 => {
                // < 1MB: Inline in source block with compression
                self.add_to_source_block(id, source, CompressionType::Zstd)
            }
            1_000_001..=10_000_000 => {
                // 1-10MB: Separate file with reference
                let path = self.write_source_file(id, &source)?;
                self.add_source_reference(id, path)
            }
            _ => {
                // > 10MB: External storage
                let uri = self.upload_to_external(id, &source)?;
                self.add_external_reference(id, uri)
            }
        }
    }
}

// NOVA's progressive loading strategy
impl NovaEngine {
    async fn load_source_progressive(&self, id: &str) -> Result<SourceContent> {
        // 1. Check page-level cache
        if let Some(cached) = self.page_cache.get(id) {
            return Ok(cached);
        }
        
        // 2. Check row group summary
        let row_group = self.locate_row_group(id)?;
        if row_group.stats.source_bloom_filter.might_contain(id) {
            // 3. Load only necessary column chunks
            let source = self.load_column_chunk(row_group, "source", id)?;
            
            // 4. Cache for future access
            self.page_cache.insert(id, source.clone());
            return Ok(source);
        }
        
        Err(anyhow!("Source not found"))
    }
}

impl SwiftEngine {
    async fn stream_source(&self, id: &str) -> Result<SourceStream> {
        let location = self.locate_source(id)?;
        
        match location {
            SourceLocation::HotBlock(block_id, offset) => {
                // Direct memory access for hot data
                let block = &self.hot_text_blocks[block_id];
                Ok(SourceStream::Memory(block.zero_copy_slice(offset)))
            }
            SourceLocation::StandardBlock(block_id) => {
                // Memory-mapped access for standard blocks
                let mmap = self.mmap_pool.get(block_id)?;
                Ok(SourceStream::Mmap(mmap))
            }
            SourceLocation::External(uri) => {
                // Streaming from external storage
                let stream = self.external_client.stream(uri).await?;
                Ok(SourceStream::External(stream))
            }
        }
    }
}

    fn store_document(&mut self, id: &str, text: &str) -> Result<Vec<String>> {

        let chunk_ids = Vec::new();

// PRISM's progressive search with source
impl PrismEngine {
    async fn search_with_source(
        &self,
        query_vector: &[f32],
        top_k: usize,
        include_source: bool,
        expand_chunks: bool, // Include surrounding chunks
    ) -> Result<Vec<SearchResultWithSource>> {
        // 1. Vector search
        let vector_results = self.vector_search(query_vector, top_k).await?;
        
        // 2. Enrich with source
        let mut results = Vec::new();
        for vr in vector_results {
            let source = if include_source {
                Some(self.load_source(&vr.id).await?)
            } else {
                None
            };
            
            // 3. Expand to include surrounding chunks if requested
            let expanded_context = if expand_chunks {
                if let Some(ref src) = source {
                    if let Some(chunk_ctx) = src.get_chunk_context() {
                        self.load_surrounding_chunks(
                            &chunk_ctx.document_id,
                            chunk_ctx.chunk_index,
                            2, // Load 2 chunks before/after
                        ).await?
                    } else {
                        None
                    }
                } else {
                    None
                }
            } else {
                None
            };
            
            results.push(SearchResultWithSource {
                id: vr.id,
                score: vr.score,
                source,
                expanded_context,
            });
        }
        
        Ok(results)
    }
}

pub struct RaptorRecordBatch {
    // Arrow arrays
    id_array: StringArray,              // Mandatory IDs
    vector_array: FixedSizeListArray,   // Vectors
    source_text_array: StringArray,     // Text sources
    source_binary_array: BinaryArray,   // Binary sources
    source_ref_array: StringArray,      // External refs

    adjacency_list: ListArray,          // HNSW connections

// Efficient batch operations for RAPTOR
impl RaptorEngine {
    async fn insert_batch_with_sources(
        &mut self,
        records: Vec<(String, Vec<f32>, SourceContent)>,
    ) -> Result<()> {
        // Build Arrow arrays
        let mut id_builder = StringBuilder::new(records.len());
        let mut vector_builder = FixedSizeListBuilder::new(
            Float32Builder::new(records.len() * self.dimension),
            self.dimension as i32,
        );
        let mut text_builder = StringBuilder::new(records.len());
        let mut binary_builder = BinaryBuilder::new(records.len());
        
        for (id, vector, source) in records {
            // Mandatory ID
            id_builder.append_value(&id)?;
            
            // Vector
            for v in vector {
                vector_builder.values().append_value(v)?;
            }
            vector_builder.append(true)?;
            
            // Source based on type
            match source.data {
                Some(Data::Text(text)) => {
                    text_builder.append_value(&text.content)?;
                    binary_builder.append_null()?;
                }
                Some(Data::Binary(bin)) => {
                    text_builder.append_null()?;
                    binary_builder.append_value(&bin.data)?;
                }
                _ => {
                    text_builder.append_null()?;
                    binary_builder.append_null()?;
                }
            }
            
            // Update HNSW graph
            self.hnsw_index.insert(&id, &vector)?;
        }
        
        // Create RecordBatch
        let batch = RecordBatch::try_new(
            self.schema.clone(),
            vec![
                Arc::new(id_builder.finish()),
                Arc::new(vector_builder.finish()),
                Arc::new(text_builder.finish()),
                Arc::new(binary_builder.finish()),
            ],
        )?;
        
        // Write to storage
        self.write_batch(batch).await
    }
    
    // Graph traversal with source retrieval
    async fn graph_search_with_sources(
        &self,
        query: &[f32],
        top_k: usize,
        hop_limit: usize,
    ) -> Result<Vec<GraphSearchResult>> {
        // 1. Initial HNSW search
        let initial = self.hnsw_index.search(query, top_k)?;
        
        // 2. Graph expansion
        let mut visited = HashSet::new();
        let mut results = Vec::new();
        
        for (id, score) in initial {
            visited.insert(id.clone());
            
            // Load source
            let source = self.load_source(&id).await?;
            
            // Explore neighbors
            let neighbors = self.explore_neighbors(&id, hop_limit, &mut visited)?;
            
            results.push(GraphSearchResult {
                id,
                score,
                source,
                graph_context: neighbors,
            });
        }
        
        Ok(results)
    }
}

pub struct StorageTieringPolicy {
    inline_threshold: usize,      // Default: 100KB
    block_threshold: usize,       // Default: 1MB
    file_threshold: usize,        // Default: 10MB
    
    // Access pattern based promotion/demotion
    access_count_threshold: u32,  // Promote after N accesses
    age_threshold: Duration,      // Demote after N days
    
    // Cost optimization
    storage_budget: StorageBudget,
    optimize_for: OptimizationGoal, // Latency vs Cost
}

impl StorageTieringPolicy {
    fn determine_tier(&self, content: &SourceContent) -> StorageTier {
        let size = content.size_bytes as usize;
        
        match size {
            0..=100_000 => StorageTier::Hot,
            100_001..=1_000_000 => StorageTier::Warm,
            1_000_001..=10_000_000 => StorageTier::Cool,
            _ => StorageTier::Cold,
        }
    }
    
    fn should_promote(&self, stats: &AccessStats) -> bool {
        stats.access_count > self.access_count_threshold
            || stats.last_access.elapsed() < Duration::from_secs(3600)
    }
    
    fn should_demote(&self, stats: &AccessStats) -> bool {
        stats.last_access.elapsed() > self.age_threshold
            && stats.access_count < 2
    }
}

    rpc SearchVectors(SearchVectorsRequest) returns (SearchVectorsResponse);
    
    // Get vector with source
    rpc GetVector(GetVectorRequest) returns (VectorRecord);
    
    // Update source content only
    rpc UpdateSource(UpdateSourceRequest) returns (UpdateSourceResponse);
}
