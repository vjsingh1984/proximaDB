/// Complete vector loading implementation for customer data ingestion
impl VectorOperations {
    /// Load vectors in batches for efficient ingestion
    pub async fn load_vectors_batch(
        &self,
        collection_id: &str,
        vectors: &[VectorRecord],
        batch_size: usize,
    ) -> Result<LoadingResult> {
        tracing::info!("📥 Loading {} vectors in batches of {}", vectors.len(), batch_size);

        let mut loaded_count = 0;
        let mut failed_count = 0;
        let mut loading_errors = Vec::new();

        // Process vectors in batches
        for (batch_idx, batch) in vectors.chunks(batch_size).enumerate() {
            tracing::debug!("📦 Processing batch {}: {} vectors", batch_idx, batch.len());

            match self.insert_vector_batch(collection_id, batch).await {
                Ok(batch_result) => {
                    loaded_count += batch_result.successful_insertions;
                    failed_count += batch_result.failed_insertions;
                }
                Err(e) => {
                    tracing::error!("❌ Batch {} failed: {}", batch_idx, e);
                    failed_count += batch.len();
                    loading_errors.push(LoadingError {
                        batch_index: batch_idx,
                        error_message: e.to_string(),
                        vector_count: batch.len(),
                    });
                }
            }

            // Progress reporting for large datasets
            if batch_idx % 100 == 0 && batch_idx > 0 {
                tracing::info!("📊 Progress: {} batches processed, {} vectors loaded", batch_idx, loaded_count);
            }
        }

        let loading_result = LoadingResult {
            total_vectors: vectors.len(),
            loaded_successfully: loaded_count,
            failed_insertions: failed_count,
            loading_time_ms: 0, // Would measure actual time
            errors: loading_errors,
        };

        tracing::info!("✅ Vector loading complete: {}/{} successful", loaded_count, vectors.len());
        Ok(loading_result)
    }

    /// Insert single batch of vectors
    async fn insert_vector_batch(
        &self,

    ) -> Result<BatchInsertResult> {

        // Current: todo!("Implement batch vector insertion")

        let mut successful_insertions = 0;
        let mut failed_insertions = 0;

use std::collections::{HashMap, HashSet, VecDeque};

/// Complete graph traversal algorithms for customer demonstrations
pub struct GraphTraversalEngine {
    graph_storage: Arc<dyn GraphStorage>,
    traversal_cache: HashMap<String, TraversalResult>,
}

impl GraphTraversalEngine {
    /// Breadth-First Search implementation
    pub async fn breadth_first_search(
        &self,
        start_node: &str,
        target_node: Option<&str>,
        max_depth: usize,
    ) -> Result<TraversalResult> {
        tracing::info!("🔍 BFS traversal from {} to {:?}", start_node, target_node);

        let mut visited = HashSet::new();
        let mut queue = VecDeque::new();
        let mut path = HashMap::new();
        let mut found_nodes = Vec::new();

        // Initialize BFS
        queue.push_back((start_node.to_string(), 0));
        visited.insert(start_node.to_string());

        while let Some((current_node, depth)) = queue.pop_front() {
            if depth >= max_depth {
                continue;
            }

            found_nodes.push(current_node.clone());

            // Check if we found target
            if let Some(target) = target_node {
                if current_node == target {
                    let path_to_target = self.reconstruct_path(&path, start_node, target);
                    return Ok(TraversalResult {
                        algorithm: TraversalAlgorithm::BFS,
                        start_node: start_node.to_string(),
                        target_node: target_node.map(|s| s.to_string()),
                        found_nodes,
                        path_to_target: Some(path_to_target),
                        traversal_time_ms: 0, // Would measure actual time
                        nodes_visited: visited.len(),
                    });
                }
            }

            // Get neighbors and add to queue
            let neighbors = self.graph_storage.get_neighbors(&current_node).await?;
            for neighbor in neighbors {
                if !visited.contains(&neighbor) {
                    visited.insert(neighbor.clone());
                    queue.push_back((neighbor.clone(), depth + 1));
                    path.insert(neighbor, current_node.clone());
                }
            }
        }

        // Return traversal result
        Ok(TraversalResult {
            algorithm: TraversalAlgorithm::BFS,
            start_node: start_node.to_string(),
            target_node: target_node.map(|s| s.to_string()),
            found_nodes,
            path_to_target: None,
            traversal_time_ms: 0,
            nodes_visited: visited.len(),
        })
    }

    /// Depth-First Search implementation
    pub async fn depth_first_search(
        &self,
        start_node: &str,
        target_node: Option<&str>,
        max_depth: usize,
    ) -> Result<TraversalResult> {
        tracing::info!("🔍 DFS traversal from {} to {:?}", start_node, target_node);

        let mut visited = HashSet::new();
        let mut path = Vec::new();
        let mut found_nodes = Vec::new();

        // Execute DFS recursively
        let result = self.dfs_recursive(
            start_node,
            target_node,
            &mut visited,
            &mut path,
            &mut found_nodes,
            0,
            max_depth,
        ).await?;

        Ok(TraversalResult {
            algorithm: TraversalAlgorithm::DFS,
            start_node: start_node.to_string(),
            target_node: target_node.map(|s| s.to_string()),
            found_nodes,
            path_to_target: if result.found { Some(path) } else { None },
            traversal_time_ms: 0,
            nodes_visited: visited.len(),
        })
    }

    /// Recursive DFS implementation
    async fn dfs_recursive(
        &self,
        current_node: &str,
        target_node: Option<&str>,
        visited: &mut HashSet<String>,
        path: &mut Vec<String>,
        found_nodes: &mut Vec<String>,
        depth: usize,
        max_depth: usize,
    ) -> Result<DFSResult> {
        if depth >= max_depth {
            return Ok(DFSResult { found: false });
        }

        visited.insert(current_node.to_string());
        path.push(current_node.to_string());
        found_nodes.push(current_node.to_string());

        // Check if target found
        if let Some(target) = target_node {
            if current_node == target {
                return Ok(DFSResult { found: true });
            }
        }

        // Visit neighbors
        let neighbors = self.graph_storage.get_neighbors(current_node).await?;
        for neighbor in neighbors {
            if !visited.contains(&neighbor) {
                let result = self.dfs_recursive(
                    &neighbor,
                    target_node,
                    visited,
                    path,
                    found_nodes,
                    depth + 1,
                    max_depth,
                ).await?;

                if result.found {
                    return Ok(result);
                }
            }
        }

        path.pop(); // Backtrack
        Ok(DFSResult { found: false })
    }
}

                self.authenticate_jwt_token(&token).await?

                session_token: Some(session.session_token),
