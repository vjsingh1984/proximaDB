use serde::{Deserialize, Serialize};
use std::collections::HashMap;

/// Basic configuration management for customer deployments
#[derive(Debug, Clone)]
pub struct DeploymentConfigManager {
    config_templates: HashMap<DeploymentType, ConfigTemplate>,
    environment_configs: HashMap<String, EnvironmentConfig>,
}

/// Deployment types for different customer scenarios
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum DeploymentType {
    Development,
    Staging,
    Production,
    CustomerDemo,
    Enterprise,
}

/// Configuration template for deployment types
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ConfigTemplate {
    pub deployment_type: DeploymentType,
    pub default_settings: ProximaDBConfig,
    pub required_resources: ResourceRequirements,
    pub security_settings: SecurityConfig,
    pub monitoring_config: MonitoringConfig,
}

/// ProximaDB configuration structure
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ProximaDBConfig {
    pub server: ServerConfig,
    pub storage: StorageConfig,
    pub ai: AIConfig,
    pub multi_tenant: MultiTenantConfig,
    pub performance: PerformanceConfig,
}

impl DeploymentConfigManager {
    /// Generate configuration for customer deployment
    pub async fn generate_customer_config(
        &self,
        customer_name: &str,
        deployment_type: DeploymentType,
        customizations: Option<CustomerCustomizations>,
    ) -> Result<GeneratedConfig> {
        tracing::info!("⚙️ Generating configuration for customer: {} ({:?})", customer_name, deployment_type);

        // Get base template
        let template = self.config_templates.get(&deployment_type)
            .ok_or_else(|| anyhow::anyhow!("No template for deployment type: {:?}", deployment_type))?;

        // Start with default configuration
        let mut config = template.default_settings.clone();

        // Apply customer customizations
        if let Some(customizations) = customizations {
            self.apply_customizations(&mut config, &customizations)?;
        }

        // Generate customer-specific values
        config.server.node_id = format!("proximadb-{}", customer_name.to_lowercase().replace(' ', "-"));
        config.multi_tenant.default_tenant_id = Some(format!("tenant_{}", customer_name.to_lowercase().replace(' ', "_")));

        // Set appropriate defaults based on deployment type
        match deployment_type {
            DeploymentType::CustomerDemo => {
                config.performance.enable_demo_optimizations = true;
                config.ai.enable_all_providers = false; // Reduce complexity for demos
                config.storage.default_storage_engine = "NOVA".to_string(); // Balanced performance
            }
            DeploymentType::Production => {
                config.performance.enable_production_optimizations = true;
                config.security.enable_comprehensive_audit = true;
                config.multi_tenant.enable_strict_isolation = true;
            }
            DeploymentType::Enterprise => {
                config.performance.enable_enterprise_optimizations = true;
                config.security.enable_enterprise_security = true;
                config.ai.enable_all_providers = true;
                config.monitoring.enable_comprehensive_monitoring = true;
            }
            _ => {
                // Development/Staging defaults
                config.performance.enable_debug_logging = true;
            }
        }

        // Generate deployment artifacts
        let toml_config = self.generate_toml_config(&config)?;
        let docker_compose = self.generate_docker_compose(&config, customer_name)?;
        let kubernetes_manifest = self.generate_kubernetes_manifest(&config, customer_name)?;

        Ok(GeneratedConfig {
            customer_name: customer_name.to_string(),
            deployment_type,
            proximadb_config: config,
            toml_configuration: toml_config,
            docker_compose_yaml: docker_compose,
            kubernetes_manifest,
            deployment_instructions: self.generate_deployment_instructions(&deployment_type),
        })
    }

    /// Generate TOML configuration file
    fn generate_toml_config(&self, config: &ProximaDBConfig) -> Result<String> {
        let toml_content = format!(
            r#"# ProximaDB Configuration
# Generated: {}

[server]
node_id = "{}"
bind_address = "0.0.0.0"
port = 5678
grpc_port = 5679
data_dir = "/data/proximadb"

[storage]
metadata_url = "file:///data/metadata"
default_storage_engine = "{}"

[[storage.storage_locations]]
url = "file:///data/collections"
weight = 1
tags = ["primary"]

[ai]
enable_natural_language = {}
enable_business_intelligence = {}

[multi_tenant]
enable_tenant_isolation = {}
enable_resource_limits = {}

[performance]
enable_caching = true
cache_size_mb = 512

[monitoring]
enable_metrics = true
enable_health_checks = true
"#,
            chrono::Utc::now().format("%Y-%m-%d %H:%M:%S UTC"),
            config.server.node_id,
            config.storage.default_storage_engine,
            config.ai.enable_natural_language,
            config.ai.enable_business_intelligence,
            config.multi_tenant.enable_tenant_isolation,
            config.multi_tenant.enable_resource_limits
        );

        Ok(toml_content)
    }

    /// Generate Docker Compose for easy deployment
    fn generate_docker_compose(&self, config: &ProximaDBConfig, customer_name: &str) -> Result<String> {
        let compose_content = format!(
            r#"# ProximaDB Docker Compose - {}
# Generated: {}

version: '3.8'

services:
  proximadb:
    image: proximadb/proximadb:latest
    container_name: proximadb-{}
    ports:
      - "5678:5678"  # REST API
      - "5679:5679"  # gRPC API
    environment:
      - PROXIMADB_NODE_ID={}
      - PROXIMADB_DATA_DIR=/data
      - PROXIMADB_ENABLE_AI={}
      - PROXIMADB_ENABLE_MULTI_TENANT={}
    volumes:
      - proximadb-data:/data
      - ./config.toml:/config/config.toml:ro
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5678/health"]
      interval: 30s
      timeout: 10s
      retries: 3

volumes:
  proximadb-data:
    driver: local

networks:
  default:
    name: proximadb-network
"#,
            customer_name,
            chrono::Utc::now().format("%Y-%m-%d %H:%M:%S UTC"),
            customer_name.to_lowercase().replace(' ', "-"),
            config.server.node_id,
            config.ai.enable_natural_language,
            config.multi_tenant.enable_tenant_isolation
        );

        Ok(compose_content)
    }
}

/// Overall system health status
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SystemHealth {
    pub overall_status: HealthStatus,
    pub component_health: Vec<ComponentHealth>,
    pub last_check: chrono::DateTime<chrono::Utc>,
    pub uptime_seconds: u64,
    pub performance_summary: PerformanceSummary,
}

/// Component health details
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ComponentHealth {
    pub component_name: String,
    pub status: HealthStatus,
    pub details: String,
    pub last_success: Option<chrono::DateTime<chrono::Utc>>,
    pub failure_count: u32,
}

/// Health status levels
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum HealthStatus {
    Healthy,
    Warning,
    Unhealthy,
    Unknown,
}

impl HealthMonitor {
    /// Perform comprehensive health check
    pub async fn check_system_health(&self) -> Result<SystemHealth> {
        tracing::debug!("🏥 Performing system health check");

        let mut component_health = Vec::new();
        let mut overall_healthy = true;

        // Check each component
        for checker in &self.health_checkers {
            match checker.check_health().await {
                Ok(health) => {
                    if !matches!(health.status, HealthStatus::Healthy) {
                        overall_healthy = false;
                    }
                    component_health.push(health);
                }
                Err(e) => {
                    tracing::error!("❌ Health check failed: {}", e);
                    overall_healthy = false;
                    component_health.push(ComponentHealth {
                        component_name: checker.component_name(),
                        status: HealthStatus::Unhealthy,
                        details: format!("Health check failed: {}", e),
                        last_success: None,
                        failure_count: 1,
                    });
                }
            }
        }

        let system_health = SystemHealth {
            overall_status: if overall_healthy { HealthStatus::Healthy } else { HealthStatus::Warning },
            component_health,
            last_check: chrono::Utc::now(),
            uptime_seconds: self.get_uptime_seconds(),
            performance_summary: self.get_performance_summary().await?,
        };

        tracing::info!("🏥 Health check complete: {:?} ({} components checked)",
                      system_health.overall_status, system_health.component_health.len());

        Ok(system_health)
    }
}
