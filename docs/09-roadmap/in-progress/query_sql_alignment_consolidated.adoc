/// SKS function types for unified intelligence queries

    /// SIMILAR(field, vector, metric) → semantic similarity search
    Similar {
        field: String,
        query_vector: Vec<f32>,
        metric: String,
        threshold: Option<f32>,
    },
    
    /// FOLLOW(node, edge_type, depth) → graph traversal
    Follow {
        start_node: String,
        edge_type: String,
        max_depth: u32,
        direction: TraversalDirection,
    },
    
    /// ASSEMBLE(items, context) → knowledge assembly
    Assemble {
        context_items: Vec<String>,
        assembly_strategy: AssemblyStrategy,
        max_context_size: Option<usize>,
    },
}

impl SksFunction {
    /// Parse SKS function from SQL function call
    pub fn parse_from_sql(func: &FuncCall) -> Result<Self> {
        match func.name.as_str() {
            "SIMILAR" => Self::parse_similar(func),
            "FOLLOW" => Self::parse_follow(func),
            "ASSEMBLE" => Self::parse_assemble(func),

#[cfg(test)]
mod metadata_performance {
    use criterion::{black_box, Criterion};
    use std::collections::HashMap;
    
    #[test]
    fn validate_hashmap_performance_improvement() {
        let metadata_vec = create_test_metadata_vec(10);
        let metadata_map = create_test_metadata_hashmap(10);
        
        // Benchmark Vec<MetadataItem> linear scan (O(n))
        let vec_start = std::time::Instant::now();
        for _ in 0..10000 {
            let _found = metadata_vec.iter().find(|item| item.key == "category");
        }
        let vec_duration = vec_start.elapsed();
        
        // Benchmark HashMap lookup (O(1))
        let map_start = std::time::Instant::now();
        for _ in 0..10000 {
            let _found = metadata_map.get("category");
        }
        let map_duration = map_start.elapsed();
        
        // Verify HashMap is at least 5x faster (conservative target)
        let improvement_ratio = vec_duration.as_nanos() / map_duration.as_nanos();
        assert!(improvement_ratio >= 5, "HashMap should be at least 5x faster, got {}x", improvement_ratio);
        
        println!("🚀 Performance improvement: {}x faster with HashMap", improvement_ratio);
    }
}
