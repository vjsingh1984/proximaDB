    /// Mapping of collection IDs to tenant-specific collection configurations

    /// Isolation level for the bridge

    pub collection_id: String,
    pub tenant_id: String,
    pub access_permissions: Vec<CollectionPermission>,
    pub resource_quota: ResourceQuota,

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum IsolationLevel {
    /// Complete isolation - no cross-tenant access
    Complete,
    /// Controlled sharing with explicit permissions
    Controlled,
    /// Shared access within tenant group
    Shared,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum CollectionPermission {
    Read,
    Write,
    Delete,
    Admin,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ResourceQuota {
    pub max_vectors: Option<u64>,
    pub max_storage_bytes: Option<u64>,
    pub max_queries_per_second: Option<u32>,
}

impl Default for CollectionBridgeConfig {
    fn default() -> Self {
        Self {
            tenant_id: String::new(),
            collection_mapping: HashMap::new(),
            isolation_level: IsolationLevel::Complete,
            resource_limits: TenantResourceLimits::default(),
            created_at: chrono::Utc::now(),
            updated_at: chrono::Utc::now(),
        }
    }
}

impl CollectionBridgeConfig {
    pub fn new(tenant_id: String) -> Self {
        Self {
            tenant_id,
            ..Default::default()
        }
    }

    pub fn add_collection_mapping(&mut self, collection_id: String, config: TenantCollectionConfig) {
        self.collection_mapping.insert(collection_id, config);
        self.updated_at = chrono::Utc::now();
    }

    pub fn remove_collection_mapping(&mut self, collection_id: &str) -> Option<TenantCollectionConfig> {
        self.updated_at = chrono::Utc::now();
        self.collection_mapping.remove(collection_id)
    }

    pub fn get_collection_config(&self, collection_id: &str) -> Option<&TenantCollectionConfig> {
        self.collection_mapping.get(collection_id)
    }
}

#!/bin/bash
# File: scripts/validate_module_structure.sh

echo "=== ProximaDB Module Structure Validation ==="

# Function to check if a module declaration has corresponding files
check_module_declarations() {
    local file="$1"
    echo "Checking module declarations in: $file"

    # Extract module declarations
    grep -n "^pub mod\|^mod" "$file" | while read -r line_num declaration; do
        module_name=$(echo "$declaration" | sed -n 's/.*mod \([a-zA-Z_][a-zA-Z0-9_]*\).*/\1/p')

        if [ -n "$module_name" ]; then
            file_dir=$(dirname "$file")

            # Check if module file exists
            if [ -f "$file_dir/$module_name.rs" ]; then
                echo "  ✅ $module_name.rs exists"
            elif [ -d "$file_dir/$module_name" ] && [ -f "$file_dir/$module_name/mod.rs" ]; then
                echo "  ✅ $module_name/mod.rs exists"
            else
                echo "  ❌ Missing module: $module_name (declared at line $line_num)"
                echo "     Expected: $file_dir/$module_name.rs OR $file_dir/$module_name/mod.rs"
            fi
        fi
    done
}

# Check all mod.rs files
echo "=== Checking mod.rs files ==="
find src -name "mod.rs" | while read -r mod_file; do
    check_module_declarations "$mod_file"
done

echo ""
echo "=== Checking lib.rs ==="
if [ -f "src/lib.rs" ]; then
    check_module_declarations "src/lib.rs"
fi

echo ""
echo "=== Checking main.rs files ==="
find src -name "main.rs" | while read -r main_file; do
    check_module_declarations "$main_file"
done

echo ""
echo "=== Checking for orphaned modules ==="
# Find .rs files that might not be properly declared
find src -name "*.rs" -not -name "mod.rs" -not -name "main.rs" -not -name "lib.rs" | while read -r rs_file; do
    module_name=$(basename "$rs_file" .rs)
    parent_dir=$(dirname "$rs_file")

    # Check if this module is declared in the parent mod.rs or lib.rs
    if [ -f "$parent_dir/mod.rs" ]; then
        if ! grep -q "mod $module_name" "$parent_dir/mod.rs"; then
            echo "  ⚠️  Potential orphaned module: $rs_file"
            echo "     Not declared in: $parent_dir/mod.rs"
        fi
    elif [ "$parent_dir" = "src" ] && [ -f "src/lib.rs" ]; then
        if ! grep -q "mod $module_name" "src/lib.rs"; then
            echo "  ⚠️  Potential orphaned module: $rs_file"
            echo "     Not declared in: src/lib.rs"
        fi
    fi
done

echo ""
echo "=== Module Structure Validation Complete ==="

#!/bin/bash
# File: scripts/validate_build_system.sh

set -e

echo "🔧 ProximaDB Build System Validation"
echo "==================================="

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Function to print status
print_status() {
    if [ $1 -eq 0 ]; then
        echo -e "${GREEN}✅ $2${NC}"
    else
        echo -e "${RED}❌ $2${NC}"
    fi
}

# Function to print warning
print_warning() {
    echo -e "${YELLOW}⚠️  $1${NC}"
}

# Function to print info
print_info() {
    echo -e "${BLUE}ℹ️  $1${NC}"
}

# Create results directory
RESULTS_DIR="build_validation_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$RESULTS_DIR"

echo ""
echo "📋 Step 1: Clean Build Environment"
echo "=================================="

print_info "Cleaning previous build artifacts..."
cargo clean > "$RESULTS_DIR/clean.log" 2>&1
print_status $? "Build environment cleaned"

echo ""
echo "🔍 Step 2: Compilation Validation"
echo "================================="

print_info "Checking basic compilation..."
if cargo check > "$RESULTS_DIR/check.log" 2>&1; then
    print_status 0 "Basic compilation check passed"
else
    print_status 1 "Basic compilation check failed"
    echo "First 20 compilation errors:"
    head -20 "$RESULTS_DIR/check.log"

    print_info "Attempting automatic fixes..."
    if [ -f "scripts/fix_module_structure.sh" ]; then
        ./scripts/fix_module_structure.sh > "$RESULTS_DIR/auto_fix.log" 2>&1

        # Retry after fixes
        if cargo check > "$RESULTS_DIR/check_after_fix.log" 2>&1; then
            print_status 0 "Compilation successful after automatic fixes"
        else
            print_status 1 "Compilation still failing after automatic fixes"
            echo "Remaining errors:"

fi

echo ""
echo "🏗️ Step 3: Full Build Validation"
echo "================================"

print_info "Attempting full build..."
if cargo build > "$RESULTS_DIR/build.log" 2>&1; then
    print_status 0 "Full build successful"

    # Check binary creation
    if [ -f "target/debug/proximadb-server" ]; then
        print_status 0 "proximadb-server binary created"
    else
        print_status 1 "proximadb-server binary not found"
    fi

    if [ -f "target/debug/proximadb-bench" ]; then
        print_status 0 "proximadb-bench binary created"
    else
        print_status 1 "proximadb-bench binary not found"
    fi

else
    print_status 1 "Full build failed"
    echo "Build errors:"

fi

echo ""
echo "🧪 Step 4: Test Compilation Validation"
echo "======================================"

print_info "Checking test compilation..."
if cargo test --no-run > "$RESULTS_DIR/test_compile.log" 2>&1; then

    print_status 1 "Test compilation failed"

echo "📊 Step 5: Benchmark Compilation Validation"
echo "==========================================="

if cargo bench --no-run > "$RESULTS_DIR/bench_compile.log" 2>&1; then
    print_status 0 "Benchmark compilation successful"
else
    print_status 1 "Benchmark compilation failed"

echo ""

    # Check binary sizes
    if [ -f "target/release/proximadb-server" ]; then
        server_size=$(du -h target/release/proximadb-server | cut -f1)

==== Pre-commit Hook Setup

#!/bin/bash
# File: .git/hooks/pre-commit

set -e

echo "🔧 ProximaDB Pre-commit Validation"

# Function to print colored output
print_error() {
    echo -e "\033[0;31m❌ $1\033[0m"
}

print_success() {
    echo -e "\033[0;32m✅ $1\033[0m"
}

print_info() {
    echo -e "\033[0;34mℹ️  $1\033[0m"
}

# Check if this is a merge commit (skip hooks for merge commits)
if git rev-parse -q --verify MERGE_HEAD; then
    print_info "Merge commit detected, skipping pre-commit hooks"
    exit 0
fi

print_info "Running pre-commit validation..."

# 1. Check compilation
print_info "Checking compilation..."
if ! cargo check --quiet; then
    print_error "Compilation failed! Please fix errors before committing."
    echo "Run 'cargo check' to see detailed errors."
    exit 1
fi
print_success "Compilation check passed"

# 2. Run tests (only fast unit tests)
print_info "Running unit tests..."
if ! cargo test --lib --quiet; then
    print_error "Unit tests failed! Please fix failing tests before committing."
    echo "Run 'cargo test --lib' to see detailed test failures."
    exit 1
fi
print_success "Unit tests passed"

# 3. Check formatting
print_info "Checking code formatting..."
if ! cargo fmt --check; then
    print_error "Code formatting issues found!"
    echo "Run 'cargo fmt' to fix formatting automatically."
    exit 1
fi
print_success "Code formatting is correct"

# 4. Run clippy (only errors, not warnings)
print_info "Running clippy analysis..."
if ! cargo clippy --quiet -- -D clippy::all -D clippy::pedantic -A clippy::missing_docs_in_private_items; then
    print_error "Clippy found critical issues!"
    echo "Run 'cargo clippy' to see detailed issues."
    exit 1
fi
print_success "Clippy analysis passed"

# 5. Check for common security issues
print_info "Checking for security issues..."
if grep -r "unwrap()" src/ --include="*.rs" | head -1; then
    print_error "Found unwrap() usage which can cause panics!"
    echo "Consider using proper error handling instead of unwrap()."
    # This is a warning, not a blocker
fi

# 6. Check for TODO/FIXME in committed code
print_info "Checking for TODO/FIXME markers..."
todo_count=$(git diff --cached --name-only | xargs grep -l "TODO\|FIXME" | wc -l || echo "0")
if [ "$todo_count" -gt 0 ]; then
    print_info "Found $todo_count files with TODO/FIXME markers"
    echo "This is informational - consider addressing these before release."
fi

print_success "Pre-commit validation completed successfully!"
echo ""

=== Common Error Patterns and Solutions

==== Pattern 1: Missing Module Files
```
error[E0583]: file not found for module `analytics`
```
**Solution**: Create `src/ai/analytics.rs` or remove unused module declaration

==== Pattern 2: Unresolved Imports
```
error[E0432]: unresolved import `crate::storage::tenant::CollectionBridgeConfig`
```
**Solution**: Create the missing type or fix the import path

==== Pattern 3: Type Not Found
```
error[E0412]: cannot find type `SomeType` in this scope
```
**Solution**: Add proper `use` statement or create the missing type
