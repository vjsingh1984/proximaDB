// File: benches/production_validation/comprehensive_qps_benchmark.rs
use criterion::{black_box, criterion_group, criterion_main, Criterion, BenchmarkId};
use proximadb::*;
use std::time::{Duration, Instant};
use tokio::runtime::Runtime;

pub fn comprehensive_qps_benchmark(c: &mut Criterion) {
    let rt = Runtime::new().unwrap();

    // Test different vector dimensions
    let dimensions = vec![128, 256, 512, 768, 1024, 1536];

    // Test different dataset sizes
    let dataset_sizes = vec![1000, 10000, 100000, 1000000];

    // Test different storage engines
    let storage_engines = vec!["SST", "VIPER", "NOVA", "SWIFT", "RAPTOR", "PRISM", "HELIX"];

    for &dim in &dimensions {
        for &size in &dataset_sizes {
            for engine in &storage_engines {
                let benchmark_name = format!("qps_d{}_n{}_engine_{}", dim, size, engine);

                c.bench_with_input(
                    BenchmarkId::new("vector_search_qps", &benchmark_name),
                    &(dim, size, engine),
                    |b, &(dim, size, engine)| {
                        b.to_async(&rt).iter_custom(|iters| async move {
                            let db = setup_database_with_engine(engine).await;
                            let dataset = generate_realistic_dataset(dim, size).await;

                            // Insert test data
                            let insert_start = Instant::now();
                            insert_vectors(&db, &dataset).await;
                            let insert_duration = insert_start.elapsed();

                            // Generate realistic search queries
                            let queries = generate_realistic_queries(&dataset, iters as usize);

                            // Measure search QPS
                            let search_start = Instant::now();

                            for query in queries {
                                black_box(db.vector_search(&query).await.unwrap());
                            }

                            let search_duration = search_start.elapsed();

                            // Calculate actual QPS
                            let qps = iters as f64 / search_duration.as_secs_f64();

                            // Log detailed metrics
                            println!("Benchmark: {} | QPS: {:.1} | Insert Rate: {:.1} vectors/sec | Engine: {}",
                                    benchmark_name,
                                    qps,
                                    dataset.len() as f64 / insert_duration.as_secs_f64(),
                                    engine
                            );

                            // Store metrics in benchmark database
                            store_benchmark_result(BenchmarkResult {
                                benchmark_name: benchmark_name.clone(),
                                qps,
                                dimensions: dim,
                                dataset_size: size,
                                storage_engine: engine.to_string(),
                                insert_rate: dataset.len() as f64 / insert_duration.as_secs_f64(),
                                timestamp: chrono::Utc::now(),
                            }).await;

                            search_duration
                        });
                    },
                );
            }
        }
    }
}

async fn setup_database_with_engine(engine: &str) -> ProximaDB {
    let mut config = ProximaDBConfig::default();
    config.default_storage_engine = engine.to_string();
    config.data_directory = format!("/tmp/proximadb_bench_{}", engine);

    ProximaDB::new(config).await.unwrap()
}

async fn generate_realistic_dataset(dimensions: usize, size: usize) -> Vec<VectorRecord> {
    let mut dataset = Vec::new();

    for i in 0..size {
        let vector = generate_realistic_vector(dimensions);
        let metadata = generate_realistic_metadata(i);

        dataset.push(VectorRecord {
            id: format!("vector_{}", i),
            vector,
            metadata,
        });
    }

    dataset
}

fn generate_realistic_vector(dimensions: usize) -> Vec<f32> {
    use rand::Rng;
    let mut rng = rand::thread_rng();

    // Generate vectors with realistic distributions (not just random)
    // Simulate common vector patterns from real embeddings
    let mut vector = Vec::with_capacity(dimensions);

    for i in 0..dimensions {
        // Create clustered patterns similar to real embeddings
        let base_value = (i as f32 / dimensions as f32) * 2.0 - 1.0;
        let noise = rng.gen_range(-0.3..0.3);
        vector.push(base_value + noise);
    }

    // Normalize vector (common in real embedding systems)
    let magnitude: f32 = vector.iter().map(|x| x * x).sum::<f32>().sqrt();
    if magnitude > 0.0 {
        for val in &mut vector {
            *val /= magnitude;
        }
    }

    vector
}

fn generate_realistic_metadata(index: usize) -> HashMap<String, String> {
    let mut metadata = HashMap::new();

    // Add realistic metadata patterns
    metadata.insert("category".to_string(), format!("category_{}", index % 20));
    metadata.insert("user_id".to_string(), format!("user_{}", index % 1000));
    metadata.insert("timestamp".to_string(), chrono::Utc::now().timestamp().to_string());
    metadata.insert("source".to_string(), "production_benchmark".to_string());

    // Add tenant information for multi-tenant testing
    metadata.insert("tenant_id".to_string(), format!("tenant_{}", index % 10));

    metadata
}

use criterion::{criterion_group, criterion_main, Criterion, BenchmarkId};
use std::sync::Arc;
use tokio::sync::Semaphore;

pub fn multi_tenant_isolation_benchmark(c: &mut Criterion) {
    let rt = tokio::runtime::Runtime::new().unwrap();

    // Test different tenant counts
    let tenant_counts = vec![1, 5, 10, 25, 50, 100];

    for &tenant_count in &tenant_counts {
        c.bench_with_input(
            BenchmarkId::new("multi_tenant_qps", tenant_count),
            &tenant_count,
            |b, &tenant_count| {
                b.to_async(&rt).iter_custom(|_iters| async move {
                    let start = Instant::now();

                    // Setup multi-tenant test environment
                    let db = setup_multi_tenant_database().await;
                    let tenants = create_test_tenants(tenant_count).await;

                    // Create semaphore to limit concurrent operations
                    let semaphore = Arc::new(Semaphore::new(tenant_count));

                    // Setup data for each tenant
                    for tenant in &tenants {
                        setup_tenant_data(&db, tenant).await;
                    }

                    // Run concurrent tenant operations
                    let mut handles = Vec::new();

                    for tenant in tenants {
                        let db_clone = db.clone();
                        let semaphore_clone = semaphore.clone();

                        handles.push(tokio::spawn(async move {
                            let _permit = semaphore_clone.acquire().await.unwrap();

                            // Perform realistic tenant workload
                            let mut tenant_qps = 0;
                            let tenant_start = Instant::now();

                            for _ in 0..100 { // 100 queries per tenant
                                let query = generate_tenant_query(&tenant);
                                match db_clone.tenant_vector_search(&query).await {
                                    Ok(_) => tenant_qps += 1,
                                    Err(e) => eprintln!("Tenant {} query failed: {}", tenant.id, e),
                                }
                            }

                            let tenant_duration = tenant_start.elapsed();
                            let tenant_actual_qps = tenant_qps as f64 / tenant_duration.as_secs_f64();

                            TenantBenchmarkResult {
                                tenant_id: tenant.id,
                                qps: tenant_actual_qps,
                                duration: tenant_duration,
                                successful_queries: tenant_qps,
                            }
                        }));
                    }

                    // Wait for all tenants to complete
                    let tenant_results: Vec<TenantBenchmarkResult> = futures::future::join_all(handles)
                        .await
                        .into_iter()
                        .map(|r| r.unwrap())
                        .collect();

                    let total_duration = start.elapsed();

                    // Calculate multi-tenant performance metrics
                    let total_qps: f64 = tenant_results.iter().map(|r| r.qps).sum();
                    let avg_qps_per_tenant: f64 = total_qps / tenant_count as f64;
                    let min_qps_per_tenant: f64 = tenant_results.iter().map(|r| r.qps).fold(f64::INFINITY, f64::min);
                    let max_qps_per_tenant: f64 = tenant_results.iter().map(|r| r.qps).fold(0.0, f64::max);

                    // Log detailed multi-tenant metrics
                    println!("Multi-Tenant Benchmark | Tenants: {} | Total QPS: {:.1} | Avg QPS/Tenant: {:.1} | Min: {:.1} | Max: {:.1}",
                            tenant_count, total_qps, avg_qps_per_tenant, min_qps_per_tenant, max_qps_per_tenant);

                    // Check tenant isolation (no tenant should get <50% of average)
                    let isolation_threshold = avg_qps_per_tenant * 0.5;
                    let isolated_tenants = tenant_results.iter().all(|r| r.qps >= isolation_threshold);

                    println!("Tenant Isolation Check: {} | Threshold: {:.1} QPS",
                            if isolated_tenants { "PASSED" } else { "FAILED" },
                            isolation_threshold);

                    // Store multi-tenant benchmark results
                    store_multi_tenant_benchmark(MultiTenantBenchmarkResult {
                        tenant_count,
                        total_qps,
                        avg_qps_per_tenant,
                        min_qps_per_tenant,
                        max_qps_per_tenant,
                        isolation_maintained: isolated_tenants,
                        timestamp: chrono::Utc::now(),
                        tenant_results,
                    }).await;

                    total_duration
                });
            },
        );
    }
}

// File: benches/production_validation/cache_efficiency_benchmark.rs
pub fn cache_efficiency_benchmark(c: &mut Criterion) {
    let rt = tokio::runtime::Runtime::new().unwrap();

    c.bench_function("cache_hit_rate_validation", |b| {
        b.to_async(&rt).iter_custom(|_iters| async move {
            let db = setup_database_with_cache().await;
            let cache_orchestrator = db.get_cache_orchestrator();

            // Generate realistic query patterns with repetition
            let unique_queries = generate_cache_test_queries(1000);
            let repeated_queries = simulate_realistic_query_pattern(&unique_queries, 10000);

            let mut cache_hits = 0;
            let mut cache_misses = 0;
            let mut total_queries = 0;

            let start = Instant::now();

            // Execute queries and track cache performance
            for query in &repeated_queries {
                total_queries += 1;

                // Check if query result is cached
                if let Some(_cached_result) = cache_orchestrator.get_cached_result(query).await {
                    cache_hits += 1;
                } else {
                    cache_misses += 1;

                    // Execute query and cache result
                    let result = db.vector_search_internal(query).await.unwrap();
                    cache_orchestrator.cache_result(query.clone(), result).await;
                }
            }

            let duration = start.elapsed();
            let hit_rate = (cache_hits as f64) / (total_queries as f64) * 100.0;
            let qps = total_queries as f64 / duration.as_secs_f64();

            // Log detailed cache metrics
            println!("Cache Performance | Hit Rate: {:.1}% | Hits: {} | Misses: {} | QPS: {:.1}",
                    hit_rate, cache_hits, cache_misses, qps);

            // Validate cache efficiency meets expectations
            let cache_efficiency_target = 85.0; // Target 85%+ hit rate
            let meets_target = hit_rate >= cache_efficiency_target;

            println!("Cache Efficiency Target: {:.1}% | Achieved: {:.1}% | Status: {}",
                    cache_efficiency_target, hit_rate,
                    if meets_target { "PASSED" } else { "FAILED" });

            // Store cache benchmark results
            store_cache_benchmark(CacheBenchmarkResult {
                hit_rate,
                cache_hits,
                cache_misses,
                total_queries,
                qps,
                duration,
                meets_target,
                timestamp: chrono::Utc::now(),
            }).await;

            duration
        });
    });
}

fn simulate_realistic_query_pattern(unique_queries: &[VectorSearchRequest], total_queries: usize) -> Vec<VectorSearchRequest> {
    use rand::Rng;
    let mut rng = rand::thread_rng();
    let mut pattern_queries = Vec::new();

    // Simulate realistic access patterns:
    // - 70% of queries are repeated (cacheable)
    // - 20% are variations of popular queries
    // - 10% are completely new queries

    for _ in 0..total_queries {
        let pattern_type: f32 = rng.gen();

        if pattern_type < 0.7 {
            // Repeat a popular query (high cache hit probability)
            let popular_index = rng.gen_range(0..unique_queries.len().min(100));
            pattern_queries.push(unique_queries[popular_index].clone());
        } else if pattern_type < 0.9 {
            // Variation of a popular query (medium cache hit probability)
            let base_index = rng.gen_range(0..unique_queries.len().min(100));
            let mut variation = unique_queries[base_index].clone();

            // Modify query slightly (change k, add filter, etc.)
            variation.k = rng.gen_range(1..20);
            pattern_queries.push(variation);
        } else {
            // Completely new query (cache miss)
            let new_index = rng.gen_range(0..unique_queries.len());
            pattern_queries.push(unique_queries[new_index].clone());
        }
    }

    pattern_queries
}

            'recommended_production_capacity': 'Unknown'
        }

        if self.benchmark_data['qps_benchmarks']:
            qps_values = [b['qps'] for b in self.benchmark_data['qps_benchmarks']]
            summary['peak_qps'] = max(qps_values)
            summary['average_qps'] = sum(qps_values) / len(qps_values)

        if self.benchmark_data['multi_tenant_benchmarks']:
            mt_results = self.benchmark_data['multi_tenant_benchmarks']
            avg_per_tenant = [b['avg_qps_per_tenant'] for b in mt_results]
            if avg_per_tenant:
                summary['multi_tenant_qps_per_tenant'] = sum(avg_per_tenant) / len(avg_per_tenant)

        if self.benchmark_data['cache_benchmarks']:
            cache_rates = [b['hit_rate'] for b in self.benchmark_data['cache_benchmarks']]
            summary['cache_hit_rate'] = sum(cache_rates) / len(cache_rates)

        # Calculate recommended production capacity (conservative estimate)
        if summary['average_qps'] > 0:
            # Recommend 70% of peak performance for production safety margin
            summary['recommended_production_capacity'] = int(summary['peak_qps'] * 0.7)

        return summary
