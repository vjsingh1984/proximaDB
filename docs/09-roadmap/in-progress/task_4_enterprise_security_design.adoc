// File: src/auth/sso/providers/aws_iam_provider.rs
use aws_config::meta::region::RegionProviderChain;
use aws_sdk_sts::{Client as StsClient, model::AssumeRoleWithWebIdentityRequest};
use jsonwebtoken::{decode, DecodingKey, Validation, Algorithm};

#[derive(Debug, Clone)]
pub struct AWSIAMProvider {
    sts_client: StsClient,
    config: AWSIAMConfig,
    jwk_cache: Arc<RwLock<HashMap<String, DecodingKey>>>,
}

#[derive(Debug, Clone, Deserialize)]
pub struct AWSIAMConfig {
    pub role_arn: String,
    pub web_identity_token_file: Option<String>,
    pub session_duration_seconds: i32,
    pub external_id: Option<String>,
    pub policy: Option<String>,
    pub jwks_uri: String,
}

impl AWSIAMProvider {
    pub async fn new(config: AWSIAMConfig) -> Result<Self, SSOError> {
        let region_provider = RegionProviderChain::default_provider().or_else("us-east-1");
        let aws_config = aws_config::from_env().region(region_provider).load().await;
        let sts_client = StsClient::new(&aws_config);

        Ok(Self {
            sts_client,
            config,
            jwk_cache: Arc::new(RwLock::new(HashMap::new())),
        })
    }
}

#[async_trait]
impl SSOProvider for AWSIAMProvider {
    async fn authenticate(&self, token: &str) -> Result<UserContext, SSOError> {
        // Step 1: Validate JWT token structure
        let header = jsonwebtoken::decode_header(token)
            .map_err(|e| SSOError::InvalidToken(format!("Invalid JWT header: {}", e)))?;

        // Step 2: Get JWK for token validation
        let kid = header.kid
            .ok_or_else(|| SSOError::InvalidToken("Missing kid in JWT header".to_string()))?;

        let decoding_key = self.get_jwk_decoding_key(&kid).await?;

        // Step 3: Validate JWT signature and claims
        let mut validation = Validation::new(Algorithm::RS256);
        validation.set_audience(&["proximadb"]);
        validation.set_issuer(&[&self.config.jwks_uri]);

        let token_data = decode::<AWSJWTClaims>(token, &decoding_key, &validation)
            .map_err(|e| SSOError::TokenValidationFailed(format!("JWT validation failed: {}", e)))?;

        // Step 4: Assume AWS IAM role using the validated token
        let assume_role_request = AssumeRoleWithWebIdentityRequest::builder()
            .role_arn(&self.config.role_arn)
            .role_session_name(format!("proximadb-session-{}", token_data.claims.sub))
            .web_identity_token(token)
            .duration_seconds(self.config.session_duration_seconds)
            .set_external_id(self.config.external_id.clone())
            .set_policy(self.config.policy.clone())
            .build();

        let assume_role_response = self.sts_client
            .assume_role_with_web_identity(assume_role_request)
            .send()
            .await
            .map_err(|e| SSOError::AWSAuthenticationFailed(format!("STS assume role failed: {}", e)))?;

        let credentials = assume_role_response.credentials
            .ok_or_else(|| SSOError::AWSAuthenticationFailed("No credentials returned from STS".to_string()))?;

        // Step 5: Extract user information and permissions
        let user_context = self.build_user_context_from_aws_credentials(
            &credentials,
            &token_data.claims,
        ).await?;

        Ok(user_context)
    }

    async fn validate_permissions(&self, user_context: &UserContext, resource: &str, action: &str) -> Result<bool, SSOError> {
        // Implement AWS IAM policy evaluation
        let policy_result = self.evaluate_aws_iam_policy(user_context, resource, action).await?;
        Ok(policy_result.allowed)
    }

    fn provider_type(&self) -> SSOProviderType {
        SSOProviderType::AWSIAM
    }
}

impl AWSIAMProvider {
    async fn get_jwk_decoding_key(&self, kid: &str) -> Result<DecodingKey, SSOError> {
        // Check cache first
        {
            let cache = self.jwk_cache.read().await;
            if let Some(key) = cache.get(kid) {
                return Ok(key.clone());
            }
        }

        // Fetch JWK from AWS
        let jwks_response = reqwest::get(&self.config.jwks_uri)
            .await
            .map_err(|e| SSOError::JWKSFetchFailed(format!("JWKS fetch failed: {}", e)))?;

        let jwks: JWKS = jwks_response.json().await
            .map_err(|e| SSOError::JWKSParseFailed(format!("JWKS parse failed: {}", e)))?;

        // Find the specific key
        let jwk = jwks.keys.iter()
            .find(|key| key.kid.as_ref() == Some(&kid.to_string()))
            .ok_or_else(|| SSOError::JWKNotFound(format!("JWK with kid '{}' not found", kid)))?;

        // Convert JWK to DecodingKey
        let decoding_key = DecodingKey::from_jwk(jwk)
            .map_err(|e| SSOError::JWKConversionFailed(format!("JWK conversion failed: {}", e)))?;

        // Cache the key
        {
            let mut cache = self.jwk_cache.write().await;
            cache.insert(kid.to_string(), decoding_key.clone());
        }

        Ok(decoding_key)
    }

    async fn build_user_context_from_aws_credentials(
        &self,
        credentials: &aws_sdk_sts::model::Credentials,
        jwt_claims: &AWSJWTClaims,
    ) -> Result<UserContext, SSOError> {
        let user_context = UserContext {
            user_id: jwt_claims.sub.clone(),
            username: jwt_claims.preferred_username.clone().unwrap_or_else(|| jwt_claims.sub.clone()),
            email: jwt_claims.email.clone(),
            roles: self.extract_aws_roles(&jwt_claims.cognito_groups).await?,
            permissions: self.extract_aws_permissions(&jwt_claims).await?,
            tenant_id: jwt_claims.custom_tenant_id.clone(),
            session_id: uuid::Uuid::new_v4().to_string(),
            authenticated_at: chrono::Utc::now(),
            expires_at: chrono::Utc::now() + chrono::Duration::seconds(self.config.session_duration_seconds as i64),
            authentication_method: AuthenticationMethod::SSOAWS,
            aws_credentials: Some(AWSCredentials {
                access_key_id: credentials.access_key_id().unwrap_or_default().to_string(),
                secret_access_key: credentials.secret_access_key().unwrap_or_default().to_string(),
                session_token: credentials.session_token().map(|t| t.to_string()),
                expiration: credentials.expiration().map(|e| e.as_secs_f64() as i64),
            }),
        };

        Ok(user_context)
    }

    async fn evaluate_aws_iam_policy(&self, user_context: &UserContext, resource: &str, action: &str) -> Result<PolicyEvaluationResult, SSOError> {
        // This would integrate with AWS IAM Policy Evaluation
        // For now, implement basic role-based evaluation
        let allowed = user_context.roles.iter().any(|role| {
            self.role_allows_resource_action(role, resource, action)
        });

        Ok(PolicyEvaluationResult {
            allowed,
            reason: if allowed { "Allowed by IAM role".to_string() } else { "Denied by IAM policy".to_string() },
            evaluated_policies: user_context.roles.clone(),
        })
    }
}

#[async_trait]
impl SSOProvider for AzureADProvider {
    async fn authenticate(&self, token: &str) -> Result<UserContext, SSOError> {
        // Step 1: Validate Azure AD JWT token
        let token_data = self.validate_azure_ad_token(token).await?;

        // Step 2: Get user information from Microsoft Graph
        let user_info = self.get_user_info_from_graph(&token_data.claims.oid).await?;

        // Step 3: Get user's group memberships and roles
        let groups = self.get_user_groups(&token_data.claims.oid).await?;
        let roles = self.extract_roles_from_groups(&groups).await?;

        // Step 4: Build ProximaDB user context
        let user_context = UserContext {
            user_id: token_data.claims.oid.clone(),
            username: user_info.user_principal_name.clone(),
            email: user_info.mail.clone().or(Some(user_info.user_principal_name.clone())),
            roles,
            permissions: self.extract_permissions_from_roles(&roles).await?,
            tenant_id: token_data.claims.custom_tenant_id.clone(),
            session_id: uuid::Uuid::new_v4().to_string(),
            authenticated_at: chrono::Utc::now(),
            expires_at: chrono::DateTime::from_timestamp(token_data.claims.exp as i64, 0)
                .unwrap_or_else(|| chrono::Utc::now() + chrono::Duration::hours(1)),
            authentication_method: AuthenticationMethod::SSOAzureAD,
            azure_ad_info: Some(AzureADInfo {
                object_id: token_data.claims.oid.clone(),
                tenant_id: token_data.claims.tid.clone(),
                upn: user_info.user_principal_name.clone(),
                groups: groups.clone(),
            }),
        };

        Ok(user_context)
    }

    async fn validate_permissions(&self, user_context: &UserContext, resource: &str, action: &str) -> Result<bool, SSOError> {
        // Implement Azure AD conditional access policy evaluation
        let policy_result = self.evaluate_azure_ad_policies(user_context, resource, action).await?;
        Ok(policy_result.allowed)
    }

    fn provider_type(&self) -> SSOProviderType {
        SSOProviderType::AzureAD
    }
}

impl AzureADProvider {
    async fn validate_azure_ad_token(&self, token: &str) -> Result<TokenData<AzureADJWTClaims>, SSOError> {
        // Step 1: Get Azure AD JWKS
        let jwks_url = format!(
            "https://login.microsoftonline.com/{}/discovery/v2.0/keys",
            self.config.tenant_id
        );

        let jwks_response = reqwest::get(&jwks_url).await
            .map_err(|e| SSOError::JWKSFetchFailed(format!("Azure AD JWKS fetch failed: {}", e)))?;

        let jwks: JWKS = jwks_response.json().await
            .map_err(|e| SSOError::JWKSParseFailed(format!("Azure AD JWKS parse failed: {}", e)))?;

        // Step 2: Validate JWT
        let header = jsonwebtoken::decode_header(token)
            .map_err(|e| SSOError::InvalidToken(format!("Invalid JWT header: {}", e)))?;

        let kid = header.kid
            .ok_or_else(|| SSOError::InvalidToken("Missing kid in JWT header".to_string()))?;

        let jwk = jwks.keys.iter()
            .find(|key| key.kid.as_ref() == Some(&kid))
            .ok_or_else(|| SSOError::JWKNotFound(format!("JWK with kid '{}' not found", kid)))?;

        let decoding_key = DecodingKey::from_jwk(jwk)
            .map_err(|e| SSOError::JWKConversionFailed(format!("JWK conversion failed: {}", e)))?;

        // Step 3: Validate token
        let mut validation = Validation::new(Algorithm::RS256);
        validation.set_audience(&[&self.config.client_id]);
        validation.set_issuer(&[&format!("https://login.microsoftonline.com/{}/v2.0", self.config.tenant_id)]);

        let token_data = decode::<AzureADJWTClaims>(token, &decoding_key, &validation)
            .map_err(|e| SSOError::TokenValidationFailed(format!("Azure AD JWT validation failed: {}", e)))?;

        Ok(token_data)
    }

    async fn get_user_info_from_graph(&self, object_id: &str) -> Result<GraphUser, SSOError> {
        let user = self.graph_client
            .users()
            .by_user_id(object_id)
            .get()
            .await
            .map_err(|e| SSOError::GraphAPIFailed(format!("Failed to get user info: {}", e)))?;

        Ok(user)
    }

    async fn get_user_groups(&self, object_id: &str) -> Result<Vec<String>, SSOError> {
        let groups = self.graph_client
            .users()
            .by_user_id(object_id)
            .member_of()
            .get()
            .await
            .map_err(|e| SSOError::GraphAPIFailed(format!("Failed to get user groups: {}", e)))?;

        let group_names = groups.value
            .into_iter()
            .filter_map(|group| group.display_name)
            .collect();

        Ok(group_names)
    }
}

use serde::{Deserialize, Serialize};
use uuid::Uuid;
use chrono::{DateTime, Utc};

#[derive(Debug, Clone)]
pub struct AuditLogger {
    storage: Arc<dyn AuditStorage + Send + Sync>,
    config: AuditConfig,
    encryption_key: Arc<EncryptionKey>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AuditEvent {
    pub event_id: String,
    pub timestamp: DateTime<Utc>,
    pub event_type: AuditEventType,
    pub user_context: Option<UserContext>,
    pub resource: AuditResource,
    pub action: String,
    pub result: AuditResult,
    pub details: HashMap<String, serde_json::Value>,
    pub ip_address: Option<String>,
    pub user_agent: Option<String>,
    pub request_id: Option<String>,
    pub tenant_id: Option<String>,
    pub session_id: Option<String>,
    pub risk_score: Option<f64>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum AuditEventType {
    Authentication,
    Authorization,
    DataAccess,
    DataModification,
    SystemConfiguration,
    SecurityEvent,
    ComplianceEvent,
    PerformanceEvent,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AuditResource {
    pub resource_type: String,  // "collection", "vector", "tenant", "user"
    pub resource_id: String,
    pub parent_resource: Option<Box<AuditResource>>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum AuditResult {
    Success,
    Failure { error_code: String, error_message: String },
    Partial { warnings: Vec<String> },
}

impl AuditLogger {
    pub async fn new(config: AuditConfig) -> Result<Self, AuditError> {
        let storage = Self::create_storage_backend(&config).await?;
        let encryption_key = Arc::new(EncryptionKey::from_config(&config.encryption)?);

        Ok(Self {
            storage,
            config,
            encryption_key,
        })
    }

    pub async fn log_event(&self, event: AuditEvent) -> Result<(), AuditError> {
        // Step 1: Validate event
        self.validate_audit_event(&event)?;

        // Step 2: Enrich event with additional metadata
        let enriched_event = self.enrich_audit_event(event).await?;

        // Step 3: Encrypt sensitive fields
        let encrypted_event = self.encrypt_sensitive_fields(enriched_event)?;

        // Step 4: Store in primary audit storage
        self.storage.store_audit_event(&encrypted_event).await?;

        // Step 5: Send to external audit systems if configured
        if let Some(external_config) = &self.config.external_audit {
            self.send_to_external_audit(&encrypted_event, external_config).await?;
        }

        // Step 6: Check for security alerts
        self.evaluate_security_alerts(&encrypted_event).await?;

        Ok(())
    }

    pub async fn log_authentication_event(
        &self,
        user_context: Option<&UserContext>,
        authentication_method: &str,
        result: AuditResult,
        ip_address: Option<String>,
    ) -> Result<(), AuditError> {
        let event = AuditEvent {
            event_id: Uuid::new_v4().to_string(),
            timestamp: Utc::now(),
            event_type: AuditEventType::Authentication,
            user_context: user_context.cloned(),
            resource: AuditResource {
                resource_type: "authentication".to_string(),
                resource_id: "system".to_string(),
                parent_resource: None,
            },
            action: format!("authenticate_{}", authentication_method),
            result,
            details: HashMap::new(),
            ip_address,
            user_agent: None,
            request_id: None,
            tenant_id: user_context.and_then(|u| u.tenant_id.clone()),
            session_id: user_context.map(|u| u.session_id.clone()),
            risk_score: self.calculate_authentication_risk(user_context, &ip_address).await,
        };

        self.log_event(event).await
    }

    pub async fn log_data_access_event(
        &self,
        user_context: &UserContext,
        resource: AuditResource,
        action: &str,
        result: AuditResult,
        metadata: HashMap<String, serde_json::Value>,
    ) -> Result<(), AuditError> {
        let event = AuditEvent {
            event_id: Uuid::new_v4().to_string(),
            timestamp: Utc::now(),
            event_type: AuditEventType::DataAccess,
            user_context: Some(user_context.clone()),
            resource,
            action: action.to_string(),
            result,
            details: metadata,
            ip_address: None, // Would be populated by middleware
            user_agent: None, // Would be populated by middleware
            request_id: None, // Would be populated by middleware
            tenant_id: user_context.tenant_id.clone(),
            session_id: Some(user_context.session_id.clone()),
            risk_score: self.calculate_data_access_risk(user_context, action).await,
        };

        self.log_event(event).await
    }

    async fn enrich_audit_event(&self, mut event: AuditEvent) -> Result<AuditEvent, AuditError> {
        // Add system metadata
        event.details.insert("system_version".to_string(), serde_json::Value::String(env!("CARGO_PKG_VERSION").to_string()));
        event.details.insert("hostname".to_string(), serde_json::Value::String(hostname::get().unwrap().to_string_lossy().to_string()));

        // Add geolocation for IP address if available
        if let Some(ref ip) = event.ip_address {
            if let Ok(location) = self.get_ip_geolocation(ip).await {
                event.details.insert("geolocation".to_string(), serde_json::to_value(location)?);
            }
        }

        // Add threat intelligence data
        if let Some(threat_info) = self.check_threat_intelligence(&event).await? {
            event.details.insert("threat_intelligence".to_string(), serde_json::to_value(threat_info)?);
        }

        Ok(event)
    }

    fn encrypt_sensitive_fields(&self, mut event: AuditEvent) -> Result<AuditEvent, AuditError> {
        // Encrypt sensitive fields for compliance
        if let Some(user_context) = &mut event.user_context {
            if let Some(ref email) = user_context.email {
                let encrypted_email = self.encryption_key.encrypt(email.as_bytes())?;
                user_context.email = Some(base64::encode(encrypted_email));
            }
        }

        // Encrypt IP addresses for privacy compliance
        if let Some(ref ip) = event.ip_address {
            let encrypted_ip = self.encryption_key.encrypt(ip.as_bytes())?;
            event.ip_address = Some(base64::encode(encrypted_ip));
        }

        Ok(event)
    }

    async fn evaluate_security_alerts(&self, event: &AuditEvent) -> Result<(), AuditError> {
        // Check for security alert conditions
        let mut alerts = Vec::new();

        // Failed authentication attempts
        if event.event_type == AuditEventType::Authentication {
            if let AuditResult::Failure { .. } = &event.result {
                let recent_failures = self.count_recent_auth_failures(&event.user_context, Duration::minutes(15)).await?;
                if recent_failures > 5 {
                    alerts.push(SecurityAlert {
                        alert_type: SecurityAlertType::SuspiciousAuthActivity,
                        severity: SecurityAlertSeverity::High,
                        description: format!("Multiple authentication failures: {} in 15 minutes", recent_failures),
                        user_context: event.user_context.clone(),
                        ip_address: event.ip_address.clone(),
                    });
                }
            }
        }

        // Cross-tenant access attempts
        if let Some(ref user_context) = event.user_context {
            if let Some(ref user_tenant) = user_context.tenant_id {
                if let Some(ref resource_tenant) = event.tenant_id {
                    if user_tenant != resource_tenant {
                        alerts.push(SecurityAlert {
                            alert_type: SecurityAlertType::CrossTenantAccess,
                            severity: SecurityAlertSeverity::Critical,
                            description: format!("Cross-tenant access attempt: user tenant {} accessing resource tenant {}", user_tenant, resource_tenant),
                            user_context: Some(user_context.clone()),
                            ip_address: event.ip_address.clone(),
                        });
                    }
                }
            }
        }

        // Send alerts to security monitoring system
        for alert in alerts {
            self.send_security_alert(alert).await?;
        }

        Ok(())
    }
}

#[derive(Debug, Clone)]
pub struct SecurityMonitor {
    threat_detector: Arc<ThreatDetector>,
    incident_responder: Arc<IncidentResponder>,
    threat_intel: Arc<ThreatIntelligence>,
    config: SecurityMonitoringConfig,
}

impl SecurityMonitor {
    pub async fn monitor_security_events(&self) -> Result<(), SecurityError> {
        let mut event_stream = self.get_security_event_stream().await?;

        while let Some(event) = event_stream.next().await {
            match self.analyze_security_event(&event).await {
                Ok(threat_assessment) => {
                    if threat_assessment.risk_level >= RiskLevel::High {
                        self.handle_high_risk_event(event, threat_assessment).await?;
                    }
                }
                Err(e) => {
                    tracing::error!("Failed to analyze security event: {}", e);
                }
            }
        }

        Ok(())
    }

    async fn analyze_security_event(&self, event: &SecurityEvent) -> Result<ThreatAssessment, SecurityError> {
        let mut threat_scores = Vec::new();

        // Analyze authentication anomalies
        if let Some(auth_score) = self.analyze_authentication_anomaly(event).await? {
            threat_scores.push(auth_score);
        }

        // Analyze access patterns
        if let Some(access_score) = self.analyze_access_pattern_anomaly(event).await? {
            threat_scores.push(access_score);
        }

        // Check against threat intelligence
        if let Some(intel_score) = self.threat_intel.check_indicators(event).await? {
            threat_scores.push(intel_score);
        }

        // Calculate overall risk score
        let overall_risk_score = self.calculate_overall_risk(&threat_scores);
        let risk_level = self.determine_risk_level(overall_risk_score);

        Ok(ThreatAssessment {
            event_id: event.event_id.clone(),
            risk_level,
            risk_score: overall_risk_score,
            threat_indicators: threat_scores,
            recommended_actions: self.generate_recommended_actions(&risk_level),
            confidence: self.calculate_confidence(&threat_scores),
        })
    }

    async fn handle_high_risk_event(
        &self,
        event: SecurityEvent,
        threat_assessment: ThreatAssessment,
    ) -> Result<(), SecurityError> {
        tracing::warn!("High-risk security event detected: {:?}", threat_assessment);

        // Immediate automated response
        match threat_assessment.risk_level {
            RiskLevel::Critical => {
                self.incident_responder.execute_critical_response(&event, &threat_assessment).await?;
            }
            RiskLevel::High => {
                self.incident_responder.execute_high_risk_response(&event, &threat_assessment).await?;
            }
            _ => {}
        }

        // Send security alert
        self.send_security_alert(SecurityAlert {
            alert_id: Uuid::new_v4().to_string(),
            timestamp: Utc::now(),
            severity: match threat_assessment.risk_level {
                RiskLevel::Critical => AlertSeverity::Critical,
                RiskLevel::High => AlertSeverity::High,
                _ => AlertSeverity::Medium,
            },
            event: event.clone(),
            threat_assessment: threat_assessment.clone(),
            automated_response: true,
        }).await?;

        Ok(())
    }
}
