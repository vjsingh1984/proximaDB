// File: src/revenue/metering/engine.rs
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::RwLock;
use chrono::{DateTime, Utc, Duration};
use uuid::Uuid;

/// Enterprise usage metering engine for real-time billing
#[derive(Debug, Clone)]
pub struct UsageMeteringEngine {
    /// Real-time usage event storage
    usage_events: Arc<RwLock<HashMap<String, Vec<UsageEvent>>>>,
    /// Aggregated usage metrics per tenant
    usage_aggregates: Arc<RwLock<HashMap<String, UsageAggregate>>>,
    /// Pricing configuration
    pricing_config: Arc<PricingConfig>,
    /// Billing integration
    billing_integration: Option<Arc<dyn BillingProvider>>,
}

/// Usage event for enterprise metering
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct UsageEvent {
    pub event_id: String,
    pub tenant_id: String,
    pub user_id: String,
    pub event_type: UsageEventType,
    pub timestamp: DateTime<Utc>,
    pub resource_consumed: ResourceConsumption,
    pub cost_impact: Option<f64>,
    pub metadata: HashMap<String, String>,
}

/// Types of billable usage events
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum UsageEventType {
    VectorSearch { k: usize, dimensions: usize },
    VectorInsertion { count: usize, dimensions: usize },
    CollectionCreation { estimated_size: u64 },
    AIQuery { provider: String, tokens: u32 },
    DataStorage { bytes: u64 },
    ComputeTime { milliseconds: u64 },
    DataTransfer { bytes: u64 },
}

/// Resource consumption metrics
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ResourceConsumption {
    pub cpu_milliseconds: u64,
    pub memory_bytes: u64,
    pub storage_bytes: u64,
    pub network_bytes: u64,
    pub ai_tokens: u32,
    pub custom_metrics: HashMap<String, f64>,
}

/// Aggregated usage for billing periods
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct UsageAggregate {
    pub tenant_id: String,
    pub billing_period_start: DateTime<Utc>,
    pub billing_period_end: DateTime<Utc>,
    pub total_searches: u64,
    pub total_insertions: u64,
    pub total_ai_queries: u64,
    pub total_storage_bytes: u64,
    pub total_compute_ms: u64,
    pub total_cost: f64,
    pub usage_by_day: HashMap<String, DailyUsage>,
}

/// Daily usage breakdown for analytics
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DailyUsage {
    pub date: String,
    pub searches: u64,
    pub insertions: u64,
    pub ai_queries: u64,
    pub cost: f64,
    pub peak_qps: f64,
}

impl UsageMeteringEngine {
    /// Create new usage metering engine
    pub async fn new(pricing_config: PricingConfig) -> Result<Self, anyhow::Error> {
        Ok(Self {
            usage_events: Arc::new(RwLock::new(HashMap::new())),
            usage_aggregates: Arc::new(RwLock::new(HashMap::new())),
            pricing_config: Arc::new(pricing_config),
            billing_integration: None,
        })
    }

    /// Record billable usage event
    pub async fn record_usage_event(&self, event: UsageEvent) -> Result<(), anyhow::Error> {
        tracing::debug!("💰 Recording usage event: {:?} for tenant {}", event.event_type, event.tenant_id);

        // Calculate cost impact
        let cost = self.calculate_event_cost(&event).await?;
        let mut enriched_event = event;
        enriched_event.cost_impact = Some(cost);

        // Store event
        let mut events = self.usage_events.write().await;
        events.entry(enriched_event.tenant_id.clone())
            .or_insert_with(Vec::new)
            .push(enriched_event.clone());

        // Update aggregates
        self.update_usage_aggregates(&enriched_event).await?;

        // Trigger billing if threshold reached
        self.check_billing_threshold(&enriched_event.tenant_id).await?;

        tracing::info!("✅ Usage event recorded: ${:.4} cost for tenant {}", cost, enriched_event.tenant_id);
        Ok(())
    }

    /// Calculate cost for specific usage event
    async fn calculate_event_cost(&self, event: &UsageEvent) -> Result<f64, anyhow::Error> {
        let pricing = &self.pricing_config;

        let cost = match &event.event_type {
            UsageEventType::VectorSearch { k, dimensions } => {
                pricing.search_cost_per_1k * (*k as f64 / 1000.0) * pricing.dimension_multiplier(*dimensions)
            }
            UsageEventType::VectorInsertion { count, dimensions } => {
                pricing.insertion_cost_per_1k * (*count as f64 / 1000.0) * pricing.dimension_multiplier(*dimensions)
            }
            UsageEventType::CollectionCreation { estimated_size } => {
                pricing.collection_creation_base_cost + (pricing.storage_cost_per_gb * (*estimated_size as f64 / 1_073_741_824.0))
            }
            UsageEventType::AIQuery { provider: _, tokens } => {
                pricing.ai_cost_per_1k_tokens * (*tokens as f64 / 1000.0)
            }
            UsageEventType::DataStorage { bytes } => {
                pricing.storage_cost_per_gb * (*bytes as f64 / 1_073_741_824.0)
            }
            UsageEventType::ComputeTime { milliseconds } => {
                pricing.compute_cost_per_hour * (*milliseconds as f64 / 3_600_000.0)
            }
            UsageEventType::DataTransfer { bytes } => {
                pricing.transfer_cost_per_gb * (*bytes as f64 / 1_073_741_824.0)
            }
        };

        Ok(cost)
    }

    /// Update usage aggregates for billing
    async fn update_usage_aggregates(&self, event: &UsageEvent) -> Result<(), anyhow::Error> {
        let mut aggregates = self.usage_aggregates.write().await;
        let tenant_aggregate = aggregates.entry(event.tenant_id.clone())
            .or_insert_with(|| UsageAggregate::new_for_tenant(&event.tenant_id));

        // Update aggregate counters
        match &event.event_type {
            UsageEventType::VectorSearch { .. } => tenant_aggregate.total_searches += 1,
            UsageEventType::VectorInsertion { count, .. } => tenant_aggregate.total_insertions += *count as u64,
            UsageEventType::AIQuery { .. } => tenant_aggregate.total_ai_queries += 1,
            UsageEventType::DataStorage { bytes } => tenant_aggregate.total_storage_bytes += *bytes,
            UsageEventType::ComputeTime { milliseconds } => tenant_aggregate.total_compute_ms += *milliseconds,
            _ => {}
        }

        // Update total cost
        tenant_aggregate.total_cost += event.cost_impact.unwrap_or(0.0);

        // Update daily usage
        let date_key = event.timestamp.format("%Y-%m-%d").to_string();
        let daily_usage = tenant_aggregate.usage_by_day.entry(date_key.clone())
            .or_insert_with(|| DailyUsage::new(&date_key));

        match &event.event_type {
            UsageEventType::VectorSearch { .. } => daily_usage.searches += 1,
            UsageEventType::VectorInsertion { count, .. } => daily_usage.insertions += *count as u64,
            UsageEventType::AIQuery { .. } => daily_usage.ai_queries += 1,
            _ => {}
        }

        daily_usage.cost += event.cost_impact.unwrap_or(0.0);

        Ok(())
    }

    /// Check if billing threshold is reached
    async fn check_billing_threshold(&self, tenant_id: &str) -> Result<(), anyhow::Error> {
        let aggregates = self.usage_aggregates.read().await;

        if let Some(aggregate) = aggregates.get(tenant_id) {
            // Trigger billing if cost exceeds threshold or time period elapsed
            if aggregate.total_cost > self.pricing_config.billing_threshold ||
               (Utc::now() - aggregate.billing_period_start).num_days() >= self.pricing_config.billing_frequency_days as i64 {

                // Trigger billing process
                if let Some(ref billing_provider) = self.billing_integration {
                    billing_provider.process_billing(tenant_id, aggregate).await?;
                    tracing::info!("💳 Billing triggered for tenant {}: ${:.2}", tenant_id, aggregate.total_cost);
                }
            }
        }

        Ok(())
    }

    /// Get usage summary for customer dashboard
    pub async fn get_usage_summary(&self, tenant_id: &str, period: BillingPeriod) -> Result<UsageSummary, anyhow::Error> {
        let aggregates = self.usage_aggregates.read().await;

        if let Some(aggregate) = aggregates.get(tenant_id) {
            Ok(UsageSummary {
                tenant_id: tenant_id.to_string(),
                period: period.clone(),
                total_cost: aggregate.total_cost,
                usage_breakdown: self.calculate_usage_breakdown(aggregate),
                trending: self.calculate_usage_trends(aggregate),
                recommendations: self.generate_cost_optimization_recommendations(aggregate).await?,
            })
        } else {
            Err(anyhow::anyhow!("No usage data found for tenant: {}", tenant_id))
        }
    }
}

pub struct EnterpriseSubscription {

    pub current_period_start: DateTime<Utc>,

    pub feature_entitlements: PlanFeatures,

/// Subscription status tracking
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum SubscriptionStatus {
    Active,
    PastDue,
    Cancelled,
    Suspended,
    Trial,
}

/// Billing cycle options
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum BillingCycle {
    Monthly,
    Quarterly,
    Annual,
    Custom { days: u32 },
}

impl SubscriptionManager {
    /// Create new subscription for enterprise tenant
    pub async fn create_enterprise_subscription(
        &self,
        tenant_id: &str,
        plan: EnterprisePlan,
        custom_pricing: Option<CustomPricing>,
    ) -> Result<EnterpriseSubscription, anyhow::Error> {
        let subscription = EnterpriseSubscription {
            subscription_id: Uuid::new_v4().to_string(),
            tenant_id: tenant_id.to_string(),
            plan: plan.clone(),
            status: SubscriptionStatus::Active,
            billing_cycle: BillingCycle::Monthly,
            current_period_start: Utc::now(),
            current_period_end: Utc::now() + Duration::days(30),
            usage_limits: plan.limits.clone(),
            feature_entitlements: plan.features.clone(),
            custom_pricing,
        };

        let mut subscriptions = self.active_subscriptions.write().await;
        subscriptions.insert(tenant_id.to_string(), subscription.clone());

        tracing::info!("✅ Created enterprise subscription for tenant {}: {:?}", tenant_id, plan.tier);
        Ok(subscription)
    }

    /// Process billing for subscription
    pub async fn process_subscription_billing(&self, tenant_id: &str, usage: &UsageAggregate) -> Result<Invoice, anyhow::Error> {
        let subscriptions = self.active_subscriptions.read().await;

        let subscription = subscriptions.get(tenant_id)
            .ok_or_else(|| anyhow::anyhow!("No subscription found for tenant: {}", tenant_id))?;

        // Calculate total bill: base subscription + usage overage
        let base_cost = subscription.plan.base_monthly_cost;
        let usage_cost = self.calculate_usage_overage(usage, &subscription.usage_limits)?;
        let subtotal = base_cost + usage_cost;

        // Apply discounts
        let discount = self.calculate_subscription_discount(subtotal, &subscription.plan)?;
        let total_cost = subtotal - discount;

        // Generate invoice
        let invoice = Invoice {
            invoice_id: Uuid::new_v4().to_string(),
            tenant_id: tenant_id.to_string(),
            subscription_id: subscription.subscription_id.clone(),
            billing_period_start: usage.billing_period_start,
            billing_period_end: usage.billing_period_end,
            base_subscription_cost: base_cost,
            usage_cost,
            discount_amount: discount,
            total_amount: total_cost,
            currency: "USD".to_string(),
            due_date: Utc::now() + Duration::days(30),
            line_items: self.generate_invoice_line_items(usage, subscription)?,
        };

        tracing::info!("💵 Generated invoice for tenant {}: ${:.2} (base: ${:.2}, usage: ${:.2})",
                      tenant_id, total_cost, base_cost, usage_cost);

        Ok(invoice)
    }
}

impl CustomerHealthMonitor {
    /// Calculate comprehensive customer health score
    pub async fn calculate_health_score(&self, tenant_id: &str, usage: &UsageAggregate) -> Result<CustomerHealthScore, anyhow::Error> {
        // Usage health (40% weight)
        let usage_health = self.calculate_usage_health(usage).await?;

        // Engagement health (30% weight)
        let engagement_health = self.calculate_engagement_health(tenant_id).await?;

        // Support health (30% weight)
        let support_health = self.calculate_support_health(tenant_id).await?;

        // Overall weighted health score
        let overall_health = (usage_health * 0.4) + (engagement_health * 0.3) + (support_health * 0.3);

        // Determine growth trajectory
        let growth_trajectory = self.analyze_growth_trajectory(usage).await?;

        // Identify risk factors
        let risk_factors = self.identify_risk_factors(tenant_id, overall_health, &growth_trajectory).await?;

        // Identify expansion opportunities
        let expansion_opportunities = self.identify_expansion_opportunities(usage, &growth_trajectory).await?;

        let health_score = CustomerHealthScore {
            tenant_id: tenant_id.to_string(),
            overall_health,
            usage_health,
            engagement_health,
            support_health,
            growth_trajectory,
            risk_factors,
            expansion_opportunities,
            last_updated: Utc::now(),
        };

        // Store for tracking
        let mut scores = self.health_scores.write().await;
        scores.insert(tenant_id.to_string(), health_score.clone());

        tracing::info!("📊 Updated health score for tenant {}: {:.2} ({})",
                      tenant_id, overall_health, self.health_rating(overall_health));

        Ok(health_score)
    }

    /// Calculate usage pattern health
    async fn calculate_usage_health(&self, usage: &UsageAggregate) -> Result<f64, anyhow::Error> {
        let mut health_score = 0.5; // Start at neutral

        // Positive indicators
        if usage.total_searches > 1000 { health_score += 0.2; }
        if usage.total_ai_queries > 100 { health_score += 0.2; }
        if usage.usage_by_day.len() >= 7 { health_score += 0.1; } // Regular usage

        // Growth indicators
        let recent_usage = self.get_recent_usage_trend(usage);
        if recent_usage > 1.1 { health_score += 0.2; } // 10%+ growth

        Ok(health_score.min(1.0))
    }

    /// Identify expansion opportunities
    async fn identify_expansion_opportunities(&self, usage: &UsageAggregate, trajectory: &GrowthTrajectory) -> Result<Vec<ExpansionOpportunity>, anyhow::Error> {
        let mut opportunities = Vec::new();

        // High usage expansion
        if usage.total_searches > 10000 {
            opportunities.push(ExpansionOpportunity {
                opportunity_type: ExpansionType::HigherTier,
                description: "Customer shows high search volume - candidate for enterprise tier upgrade".to_string(),
                estimated_revenue_impact: 5000.0,
                probability: 0.8,
                recommended_actions: vec![
                    "Reach out with enterprise tier benefits".to_string(),
                    "Offer volume discount and advanced features".to_string(),
                ],
            });
        }

        // AI usage expansion
        if usage.total_ai_queries > 500 {
            opportunities.push(ExpansionOpportunity {
                opportunity_type: ExpansionType::AIFeatures,
                description: "Heavy AI usage indicates opportunity for advanced AI features".to_string(),
                estimated_revenue_impact: 2000.0,
                probability: 0.7,
                recommended_actions: vec![
                    "Demo advanced AI capabilities".to_string(),
                    "Offer AI-focused enterprise plan".to_string(),
                ],
            });
        }

        // Growth trajectory expansion
        if matches!(trajectory, GrowthTrajectory::Expanding) {
            opportunities.push(ExpansionOpportunity {
                opportunity_type: ExpansionType::AdditionalSeats,
