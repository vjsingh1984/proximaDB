// Comprehensive mTLS implementation
pub struct MutualTlsManager {
    // Certificate authority management
    ca_manager: Arc<CertificateAuthorityManager>,
    // Certificate storage and rotation
    cert_store: Arc<CertificateStore>,
    // TLS configuration
    tls_config: Arc<TlsConfiguration>,
    // Certificate validation
    cert_validator: Arc<CertificateValidator>,
}

impl MutualTlsManager {
    pub async fn new_with_context(
        context: &SharedContext,
        tls_config: TlsConfiguration,
    ) -> Result<Self, ProximaDBError> {
        // Initialize CA manager
        let ca_manager = CertificateAuthorityManager::new(&tls_config.ca_config).await?;
        
        // Initialize certificate store with encryption
        let cert_store = CertificateStore::new_encrypted(
            &tls_config.cert_store_config,
            &tls_config.encryption_key,
        ).await?;
        
        // Initialize certificate validator
        let cert_validator = CertificateValidator::new(
            ca_manager.clone(),
            &tls_config.validation_config,
        )?;
        
        Ok(Self {
            ca_manager,
            cert_store,
            tls_config: Arc::new(tls_config),
            cert_validator,
        })
    }
    
    pub async fn create_server_tls_config(&self) -> Result<ServerTlsConfig, ProximaDBError> {
        // Load server certificate and private key
        let server_cert = self.cert_store.get_server_certificate().await?;
        let private_key = self.cert_store.get_server_private_key().await?;
        
        // Load CA certificate for client validation
        let ca_cert = self.ca_manager.get_ca_certificate().await?;
        
        // Create TLS acceptor configuration
        let tls_config = ServerTlsConfig::builder()
            .identity(Identity::from_pem(&server_cert, &private_key)?)
            .client_ca_root(Certificate::from_pem(&ca_cert)?)
            .client_auth_required(true)
            .build()?;
        
        Ok(tls_config)
    }
    
    pub async fn create_client_tls_config(
        &self,
        client_id: &str,
    ) -> Result<ClientTlsConfig, ProximaDBError> {
        // Generate or retrieve client certificate
        let client_cert = self.get_or_generate_client_certificate(client_id).await?;
        let client_key = self.cert_store.get_client_private_key(client_id).await?;
        
        // Load CA certificate for server validation
        let ca_cert = self.ca_manager.get_ca_certificate().await?;
        
        // Create TLS connector configuration
        let tls_config = ClientTlsConfig::builder()
            .identity(Identity::from_pem(&client_cert, &client_key)?)
            .ca_certificate(Certificate::from_pem(&ca_cert)?)
            .domain_name(&self.tls_config.server_domain)
            .build()?;
        
        Ok(tls_config)
    }
    
    async fn get_or_generate_client_certificate(
        &self,
        client_id: &str,
    ) -> Result<Vec<u8>, ProximaDBError> {
        // Check if certificate exists and is valid
        if let Some(existing_cert) = self.cert_store.get_client_certificate(client_id).await? {
            if self.cert_validator.is_certificate_valid(&existing_cert).await? {
                return Ok(existing_cert);
            }
        }
        
        // Generate new certificate
        let cert_request = CertificateRequest {
            client_id: client_id.to_string(),
            validity_period: self.tls_config.client_cert_validity,
            key_usage: vec![KeyUsage::DigitalSignature, KeyUsage::KeyEncipherment],
            extended_key_usage: vec![ExtendedKeyUsage::ClientAuth],
        };
        
        let new_cert = self.ca_manager.issue_client_certificate(cert_request).await?;
        
        // Store new certificate
        self.cert_store.store_client_certificate(client_id, &new_cert).await?;
        
        Ok(new_cert)
    }
}

// gRPC server integration
pub async fn create_secure_grpc_server(
    tls_manager: Arc<MutualTlsManager>,
    service_registry: ServiceRegistry,
) -> Result<Server, ProximaDBError> {

        .add_service(service_registry.graph_service)
        .add_service(service_registry.admin_service)

    Ok(server)
}

// JWT/OIDC authentication manager
pub struct RestAuthenticationManager {
    // JWT validation configuration
    jwt_validator: Arc<JwtValidator>,
    // OIDC provider integration
    oidc_client: Arc<OidcClient>,
    // Token cache for performance
    token_cache: Arc<TokenCache>,
    // User session management
    session_manager: Arc<SessionManager>,
}

impl RestAuthenticationManager {
    pub async fn new_with_context(
        context: &SharedContext,
        auth_config: AuthenticationConfig,
    ) -> Result<Self, ProximaDBError> {
        // Initialize JWT validator
        let jwt_validator = JwtValidator::new(&auth_config.jwt_config).await?;
        
        // Initialize OIDC client
        let oidc_client = OidcClient::new(&auth_config.oidc_config).await?;
        
        // Initialize token cache with TTL
        let token_cache = TokenCache::new_with_orchestrator(
            context.orchestrator.clone(),
            auth_config.token_cache_ttl,
        )?;
        
        // Initialize session manager
        let session_manager = SessionManager::new(&auth_config.session_config)?;
        
        Ok(Self {
            jwt_validator,
            oidc_client,
            token_cache,
            session_manager,
        })
    }
    
    pub async fn authenticate_request(
        &self,
        request: &HttpRequest,
    ) -> Result<AuthenticatedUser, ProximaDBError> {
        // Extract token from request
        let token = self.extract_token_from_request(request)?;
        
        // Check token cache first
        if let Some(cached_user) = self.token_cache.get(&token).await? {
            return Ok(cached_user);
        }
        
        // Validate JWT token
        let jwt_claims = self.jwt_validator.validate_token(&token).await?;
        
        // Verify with OIDC provider if configured
        if self.oidc_client.is_configured() {
            self.oidc_client.verify_token(&token, &jwt_claims).await?;
        }
        
        // Create authenticated user
        let authenticated_user = AuthenticatedUser {
            user_id: jwt_claims.sub,
            username: jwt_claims.preferred_username,
            email: jwt_claims.email,
            roles: jwt_claims.roles.unwrap_or_default(),
            permissions: self.resolve_permissions(&jwt_claims.roles).await?,
            tenant_id: jwt_claims.tenant_id,
            session_id: self.session_manager.create_session(&jwt_claims).await?,
        };
        
        // Cache authenticated user
        self.token_cache.put(&token, &authenticated_user).await?;
        
        Ok(authenticated_user)
    }
    
    async fn resolve_permissions(
        &self,
        roles: &Option<Vec<String>>,
    ) -> Result<Vec<Permission>, ProximaDBError> {
        let mut permissions = Vec::new();
        
        if let Some(user_roles) = roles {
            for role in user_roles {
                let role_permissions = self.get_role_permissions(role).await?;
                permissions.extend(role_permissions);
            }
        }
        
        // Deduplicate permissions
        permissions.sort();
        permissions.dedup();
        
        Ok(permissions)
    }
}

// REST middleware integration
pub struct AuthenticationMiddleware {
    auth_manager: Arc<RestAuthenticationManager>,
    excluded_paths: Vec<String>,
}

impl AuthenticationMiddleware {
    pub async fn authenticate(
        &self,
        req: ServiceRequest,
        next: Next<impl MessageBody>,
    ) -> Result<ServiceResponse<impl MessageBody>, Error> {
        // Check if path is excluded from authentication
        if self.is_path_excluded(req.path()) {
            return next.call(req).await;
        }
        
        // Authenticate request
        match self.auth_manager.authenticate_request(req.request()).await {
            Ok(authenticated_user) => {
                // Add user to request extensions
                req.extensions_mut().insert(authenticated_user);
                next.call(req).await
            }
            Err(auth_error) => {
                // Return authentication error
                let response = HttpResponse::Unauthorized()
                    .json(ErrorResponse {
                        error: "Authentication failed".to_string(),
                        details: auth_error.to_string(),
                    });
                Ok(req.into_response(response))
            }
        }
    }
}

// RBAC authorization engine
pub struct RbacAuthorizationEngine {
    // Role definitions storage
    role_store: Arc<dyn RoleStore>,
    // Permission cache for performance
    permission_cache: Arc<PermissionCache>,
    // Audit logger
    audit_logger: Arc<AuditLogger>,
    // Policy evaluation engine
    policy_engine: Arc<PolicyEngine>,
}

impl RbacAuthorizationEngine {
    pub async fn authorize_action(
        &self,
        user: &AuthenticatedUser,
        resource: &Resource,
        action: &Action,
    ) -> Result<AuthorizationResult, ProximaDBError> {
        // Create authorization context
        let auth_context = AuthorizationContext {
            user: user.clone(),
            resource: resource.clone(),
            action: action.clone(),
            timestamp: SystemTime::now(),
            request_id: uuid::Uuid::new_v4(),
        };
        
        // Check cached permissions first
        let cache_key = self.generate_permission_cache_key(&auth_context);
        if let Some(cached_result) = self.permission_cache.get(&cache_key).await? {
            // Log cache hit to audit
            self.audit_logger.log_authorization_cache_hit(&auth_context).await?;
            return Ok(cached_result);
        }
        
        // Evaluate authorization policies
        let authorization_result = self.evaluate_authorization_policies(&auth_context).await?;
        
        // Cache the result
        self.permission_cache.put(&cache_key, &authorization_result).await?;
        
        // Log authorization decision
        self.audit_logger.log_authorization_decision(&auth_context, &authorization_result).await?;
        
        Ok(authorization_result)
    }
    
    async fn evaluate_authorization_policies(
        &self,
        context: &AuthorizationContext,
    ) -> Result<AuthorizationResult, ProximaDBError> {
        // Step 1: Check if user has required role
        let required_permission = self.map_action_to_permission(&context.action, &context.resource)?;
        
        // Step 2: Check user's roles and permissions
        let user_permissions = self.get_user_permissions(&context.user).await?;
        
        // Step 3: Check tenant-scoped permissions
        let tenant_authorized = self.check_tenant_authorization(&context).await?;
        
        // Step 4: Check resource-specific permissions
        let resource_authorized = self.check_resource_authorization(&context).await?;
        
        // Step 5: Evaluate custom policies
        let policy_result = self.policy_engine.evaluate_policies(context).await?;
        
        // Combine all authorization checks
        let is_authorized = user_permissions.contains(&required_permission)
            && tenant_authorized
            && resource_authorized
            && policy_result.allowed;
        
        Ok(AuthorizationResult {
            allowed: is_authorized,
            reason: if is_authorized {
                "Access granted".to_string()
            } else {
                format!("Access denied: {}", self.get_denial_reason(context, &policy_result))
            },
            required_permission,
            evaluated_policies: policy_result.evaluated_policies,
            decision_timestamp: SystemTime::now(),
        })
    }
    
    async fn get_user_permissions(
        &self,
        user: &AuthenticatedUser,
    ) -> Result<Vec<Permission>, ProximaDBError> {
        let mut all_permissions = Vec::new();
        
        // Get permissions from user's roles
        for role_name in &user.roles {
            if let Some(role) = self.role_store.get_role(role_name).await? {
                // Check tenant scope
                if let Some(role_tenant) = &role.tenant_scope {
                    if role_tenant != &user.tenant_id {
                        continue; // Skip role if tenant mismatch
                    }
                }
                all_permissions.extend(role.permissions);
            }
        }
        
        // Add any direct permissions assigned to user
        all_permissions.extend(user.permissions.clone());
        
        // Deduplicate
        all_permissions.sort();
        all_permissions.dedup();
        
        Ok(all_permissions)
    }
}

// Authorization middleware for services
pub struct AuthorizationMiddleware {
    rbac_engine: Arc<RbacAuthorizationEngine>,
}

impl AuthorizationMiddleware {
    pub async fn authorize_vector_operation(
        &self,
        user: &AuthenticatedUser,
        operation: &VectorOperation,
    ) -> Result<(), ProximaDBError> {
        let resource = Resource::Collection {
            collection_id: operation.collection_id.clone(),
            tenant_id: user.tenant_id.clone(),
        };
        
        let action = match operation.operation_type {
            VectorOperationType::Insert => Action::VectorInsert,
            VectorOperationType::Update => Action::VectorUpdate,
            VectorOperationType::Delete => Action::VectorDelete,
            VectorOperationType::Search => Action::VectorSearch,
        };
        
        let auth_result = self.rbac_engine.authorize_action(user, &resource, &action).await?;
        
        if !auth_result.allowed {
            return Err(ProximaDBError::Unauthorized(auth_result.reason));
        }
        
        Ok(())
    }
}

impl AuditLogger {
    pub async fn log_authentication_event(
        &self,
        user_id: &str,
        auth_result: &AuthenticationResult,
        request_context: &RequestContext,
    ) -> Result<(), ProximaDBError> {
        let audit_event = AuditEvent {
            event_id: uuid::Uuid::new_v4().to_string(),
            event_type: AuditEventType::Authentication,
            timestamp: SystemTime::now(),
            user_id: Some(user_id.to_string()),
            tenant_id: auth_result.tenant_id.clone(),
            resource: Resource::System,
            action: Action::Authenticate,
            result: if auth_result.success {
                AuditResult::Success
            } else {
                AuditResult::Failure(auth_result.error_message.clone())
            },
            ip_address: request_context.client_ip.clone(),
            user_agent: request_context.user_agent.clone(),
            session_id: auth_result.session_id.clone(),
            request_id: request_context.request_id.clone(),
            metadata: auth_result.metadata.clone(),
        };
        
        // Encrypt sensitive data
        let encrypted_event = self.encryption_manager.encrypt_audit_event(audit_event).await?;
        
        // Add to batch processor for efficient storage
        self.batch_processor.add_event(encrypted_event).await?;
        
        Ok(())
    }
    
    pub async fn log_data_access_event(
        &self,
        user: &AuthenticatedUser,
        resource: &Resource,
        access_details: &DataAccessDetails,
        request_context: &RequestContext,
    ) -> Result<(), ProximaDBError> {
        let audit_event = AuditEvent {
            event_id: uuid::Uuid::new_v4().to_string(),
            event_type: AuditEventType::DataAccess,
            timestamp: SystemTime::now(),
            user_id: Some(user.user_id.clone()),
            tenant_id: Some(user.tenant_id.clone()),
            resource: resource.clone(),
            action: Action::DataRead,
            result: AuditResult::Success,
            ip_address: request_context.client_ip.clone(),
            user_agent: request_context.user_agent.clone(),
            session_id: Some(user.session_id.clone()),
            request_id: request_context.request_id.clone(),
            metadata: access_details.to_metadata(),
        };
        
        // Add data classification metadata
        let classified_event = self.add_data_classification(audit_event, access_details).await?;
        
        // Encrypt and store
        let encrypted_event = self.encryption_manager.encrypt_audit_event(classified_event).await?;
        self.batch_processor.add_event(encrypted_event).await?;
        
        Ok(())
    }
    
    pub async fn log_security_event(
        &self,
        security_event: SecurityEvent,
        severity: SecuritySeverity,
    ) -> Result<(), ProximaDBError> {
        let audit_event = AuditEvent {
            event_id: uuid::Uuid::new_v4().to_string(),
            event_type: AuditEventType::SecurityEvent,
            timestamp: SystemTime::now(),
            user_id: security_event.user_id,
            tenant_id: security_event.tenant_id,
            resource: security_event.affected_resource,
            action: security_event.security_action,
            result: AuditResult::SecurityAlert(severity),
            ip_address: security_event.source_ip,
            user_agent: security_event.user_agent,
            session_id: security_event.session_id,
            request_id: security_event.request_id,
            metadata: security_event.security_metadata,
        };
        
        // High-priority processing for security events
        if severity >= SecuritySeverity::High {
            // Immediate storage for critical security events
            let encrypted_event = self.encryption_manager.encrypt_audit_event(audit_event).await?;
            self.audit_store.store_immediate(encrypted_event).await?;
        } else {
            // Regular batch processing for lower severity
            let encrypted_event = self.encryption_manager.encrypt_audit_event(audit_event).await?;
            self.batch_processor.add_event(encrypted_event).await?;
        }
        
        Ok(())
    }
}

        tenant_id: &str,

        let quotas = self.quota_store.get_tenant_quotas(tenant_id).await?;
        let current_usage = self.usage_tracker.get_current_usage(tenant_id).await?;
        
        // Check if any quota is approaching limits (e.g., 80% threshold)
        for (quota_type, quota_value) in &quotas.quotas {
            if let Some(current_value) = current_usage.get(quota_type) {
                let usage_percentage = (*current_value as f64) / (quota_value.value as f64);

                    self.send_quota_warning(tenant_id, quota_type, usage_percentage).await?;

                if usage_percentage >= 0.95 {
                    // Send critical quota warning
                    self.send_critical_quota_warning(tenant_id, quota_type, usage_percentage).await?;
                }
            }
        }
        
        Ok(())
    }

        &self,
        quotas: &TenantQuotas,
        current_usage: &CurrentUsage,
        operation: &Operation,
    ) -> Result<Vec<QuotaCheck>, ProximaDBError> {
        let mut quota_checks = Vec::new();
        
        // Check storage quota
        if let Some(storage_quota) = &quotas.max_storage_gb {
            let estimated_storage_increase = operation.estimate_storage_usage();
            let new_storage_usage = current_usage.storage_gb + estimated_storage_increase;
            
            quota_checks.push(QuotaCheck {
                quota_type: QuotaType::Storage,
                current_value: current_usage.storage_gb,
                quota_limit: storage_quota.value,
                estimated_new_value: new_storage_usage,
                allowed: new_storage_usage <= storage_quota.value,
                enforcement_policy: storage_quota.enforcement_policy.clone(),
            });
        }
        
        // Check QPS quota
        if let Some(qps_quota) = &quotas.max_qps {
            let current_qps = self.calculate_current_qps(current_usage).await?;
            
            quota_checks.push(QuotaCheck {
                quota_type: QuotaType::QPS,
                current_value: current_qps as u64,
                quota_limit: qps_quota.value,
                estimated_new_value: (current_qps + 1.0) as u64,
                allowed: current_qps < qps_quota.value as f64,
                enforcement_policy: qps_quota.enforcement_policy.clone(),
            });
        }
        
        // Check API calls quota
        if let Some(api_quota) = &quotas.max_api_calls_per_day {
            let daily_api_calls = self.get_daily_api_calls(current_usage).await?;
            
            quota_checks.push(QuotaCheck {
                quota_type: QuotaType::APICalls,
                current_value: daily_api_calls,
                quota_limit: api_quota.value,
                estimated_new_value: daily_api_calls + 1,
                allowed: daily_api_calls < api_quota.value,
                enforcement_policy: api_quota.enforcement_policy.clone(),
            });
        }
        
        Ok(quota_checks)
    }

            }

        let tracer_provider = TracerProvider::builder()

                KeyValue::new("operation.type", "vector_search"),

        let result = async {
            let _guard = span.clone();
            operation.await
        }.await;
        
        let duration = start_time.elapsed().as_secs_f64();
        
        // Record metrics and span attributes based on result
        match &result {
            Ok(_) => {
                // Record success metrics

                    KeyValue::new("status", "success"),
                    KeyValue::new("tenant_id", tenant_id.to_string()),

    // SLA tracking
    sla_tracker: Arc<SlaTracker>,
    // Alert thresholds
    alert_thresholds: Arc<AlertThresholdStore>,
}

#[derive(Debug, Clone)]
pub struct SlaDefinition {
    pub sla_id: String,
    pub tenant_id: Option<String>, // None for global SLAs
    pub sla_type: SlaType,
    pub target_value: f64,
    pub measurement_window: Duration,
    pub breach_threshold: f64, // Percentage of target (e.g., 0.95 for 95%)
}

#[derive(Debug, Clone)]
pub enum SlaType {
    ResponseTime { percentile: f64 }, // e.g., P99 < 100ms
    Availability { uptime_percentage: f64 }, // e.g., 99.9% uptime
    Throughput { min_qps: f64 }, // e.g., >= 1000 QPS
    ErrorRate { max_error_percentage: f64 }, // e.g., < 0.1% errors
}

impl SlaMonitor {
    pub async fn check_sla_compliance(
        &self,
        tenant_id: Option<&str>,
    ) -> Result<Vec<SlaComplianceReport>, ProximaDBError> {
        let mut compliance_reports = Vec::new();
        
        // Get applicable SLAs
        let slas = self.sla_definitions.get_applicable_slas(tenant_id).await?;
        
        for sla in slas {
            let compliance_report = self.evaluate_sla_compliance(&sla).await?;
            
            // Check for SLA breaches
            if compliance_report.is_breach {
                self.handle_sla_breach(&sla, &compliance_report).await?;
            }
            
            compliance_reports.push(compliance_report);
        }
        
        Ok(compliance_reports)
    }
    
    async fn evaluate_sla_compliance(
        &self,
        sla: &SlaDefinition,
    ) -> Result<SlaComplianceReport, ProximaDBError> {
        let measurement_start = SystemTime::now() - sla.measurement_window;
        
        let actual_value = match &sla.sla_type {
            SlaType::ResponseTime { percentile } => {
                self.sla_tracker.calculate_response_time_percentile(
                    sla.tenant_id.as_deref(),
                    *percentile,
                    measurement_start,
                ).await?
            }
            SlaType::Availability { .. } => {
                self.sla_tracker.calculate_availability_percentage(
                    sla.tenant_id.as_deref(),
                    measurement_start,
                ).await?
            }
            SlaType::Throughput { .. } => {
                self.sla_tracker.calculate_average_throughput(
                    sla.tenant_id.as_deref(),
                    measurement_start,
                ).await?
            }
            SlaType::ErrorRate { .. } => {
                self.sla_tracker.calculate_error_rate_percentage(
                    sla.tenant_id.as_deref(),
                    measurement_start,
                ).await?
            }
        };
        
        let compliance_percentage = self.calculate_compliance_percentage(&sla.sla_type, actual_value, sla.target_value);
        let is_breach = compliance_percentage < sla.breach_threshold;
        
        Ok(SlaComplianceReport {
            sla_id: sla.sla_id.clone(),
            tenant_id: sla.tenant_id.clone(),
            sla_type: sla.sla_type.clone(),
            target_value: sla.target_value,
            actual_value,
            compliance_percentage,
            is_breach,
            measurement_window: sla.measurement_window,
            evaluation_timestamp: SystemTime::now(),
        })
    }
}

impl DistributedTracingManager {
    pub async fn start_distributed_operation(
        &self,
        operation_name: &str,
        parent_context: Option<Context>,
    ) -> Result<DistributedOperationContext, ProximaDBError> {
        // Generate correlation ID for the operation
        let correlation_id = self.correlation_manager
            .correlation_id_generator
            .generate_correlation_id();
        
        // Create or extend trace context
        let trace_context = match parent_context {
            Some(parent) => self.context_propagator.create_child_context(parent)?,
            None => self.context_propagator.create_root_context(&correlation_id)?,
        };
        
        // Start root span
        let root_span = self.create_root_span(operation_name, &correlation_id, &trace_context)?;
        
        // Create distributed operation context
        let operation_context = DistributedOperationContext {
            correlation_id: correlation_id.clone(),
            trace_context,
            root_span,
            operation_start_time: SystemTime::now(),
            operation_name: operation_name.to_string(),
        };
        
        // Store context for cross-service access
        self.correlation_manager
            .request_context_store
            .store_context(&correlation_id, &operation_context)
            .await?;
        
        Ok(operation_context)
    }
    
    pub async fn trace_storage_engine_operation<F, R>(
        &self,
        engine_name: &str,
        operation_name: &str,
        parent_context: &DistributedOperationContext,
        operation: F,
    ) -> Result<R, ProximaDBError>
    where
        F: Future<Output = Result<R, ProximaDBError>>,
    {
        // Create child span for storage engine operation
        let span = self.create_storage_engine_span(
            engine_name,
            operation_name,
            &parent_context.correlation_id,
            &parent_context.trace_context,
        )?;
        
        let start_time = Instant::now();
        
        // Execute operation with tracing
        let result = async {
            let _guard = span.clone();
            
            // Add storage engine specific attributes
            span.set_attribute(KeyValue::new("storage.engine", engine_name.to_string()));
            span.set_attribute(KeyValue::new("storage.operation", operation_name.to_string()));
            span.set_attribute(KeyValue::new("correlation.id", parent_context.correlation_id.clone()));
            
            operation.await
        }.await;
        
        let duration = start_time.elapsed();
        
        // Update span with operation results
        match &result {
            Ok(_) => {
                span.set_attribute(KeyValue::new("operation.status", "success"));
                span.set_attribute(KeyValue::new("operation.duration_ms", duration.as_millis() as i64));
                span.set_status(Status::Ok);
            }
            Err(error) => {
                span.set_attribute(KeyValue::new("operation.status", "error"));
                span.set_attribute(KeyValue::new("error.type", error.error_type()));
                span.set_attribute(KeyValue::new("error.message", error.to_string()));
                span.set_status(Status::error(error.to_string()));
            }
        }
        
        span.end();
        
        // Record cross-engine correlation if needed
        self.correlation_manager
            .trace_linker
            .record_engine_operation(&parent_context.correlation_id, engine_name, operation_name, duration)
            .await?;
        
        result
    }
    
    pub async fn trace_cross_service_call<F, R>(
        &self,
        service_name: &str,
        method_name: &str,
        parent_context: &DistributedOperationContext,
        service_call: F,
    ) -> Result<R, ProximaDBError>
    where
        F: Future<Output = Result<R, ProximaDBError>>,
    {
        // Create span for cross-service call
        let span = self.create_cross_service_span(
            service_name,
            method_name,
            &parent_context.correlation_id,
            &parent_context.trace_context,
        )?;
        
        // Inject trace context for downstream service
        let mut headers = HashMap::new();
        self.context_propagator.inject_context(&parent_context.trace_context, &mut headers)?;
        
        let start_time = Instant::now();
        
        // Execute service call with trace context
        let result = async {
            let _guard = span.clone();
            
            span.set_attribute(KeyValue::new("service.name", service_name.to_string()));
            span.set_attribute(KeyValue::new("service.method", method_name.to_string()));
            span.set_attribute(KeyValue::new("correlation.id", parent_context.correlation_id.clone()));
            
            // Add trace headers to the service call context
            service_call.await
        }.await;
        
        let duration = start_time.elapsed();
        
        // Update span with call results
        self.update_cross_service_span(&span, &result, duration)?;
        span.end();
        
        result
    }
}
