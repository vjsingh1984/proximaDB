== Core Concepts & Clarifications

=== Key Design Principles
1. **Engine per Collection**: Each collection is bound to a specific storage engine at creation time
2. **No Cross-Engine Routing**: AdvancedSearchOptimizer works within the collection's engine
3. **Collection Config Driven**: Optimization decisions based on collection configuration
4. **Unified Infrastructure**: Leverage existing cache, metrics, and filesystem components
5. **Query-Time Focus**: Optimize execution path based on available precomputed data

=== What AdvancedSearchOptimizer Actually Does
* Examines collection config for available indexes and quantization
* Decides optimal execution strategy within the engine
* Manages cascading quantization refinement
* Handles fallback to flat search when beneficial
* Caches optimization decisions in UnifiedCache

pub struct AdvancedSearchOptimizer {
    cache_orchestrator: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    axis_manager: Arc<AxisManager>,
    decision_cache: Arc<DecisionCache>,
}

#[derive(Clone, Debug)]
pub struct OptimizationDecision {
    pub strategy: ExecutionStrategy,
    pub estimated_latency_ms: f32,
    pub confidence: f32,
    pub cache_key: Option<String>,
    pub reasoning: String,
}

#[derive(Clone, Debug)]
pub enum ExecutionStrategy {
    /// Use precomputed quantized vectors
    UsePrecomputedQuantization {
        level: QuantizationLevel,
        cascade: bool,
    },

    /// Use HNSW or other index from AXIS
    UseIndex {
        index_type: String,  // From collection config
        parameters: IndexParameters,
    },

    /// Perform runtime quantization
    RuntimeQuantization {
        level: QuantizationLevel,
        cache_result: bool,
    },

    /// Use cached results if available
    UseCachedResults {
        cache_key: String,
    },

    /// Flat search (brute force)
    FlatSearch {
        batch_size: usize,
        use_simd: bool,
    },

    /// Hybrid approach
    Hybrid {
        primary: Box<ExecutionStrategy>,
        fallback: Box<ExecutionStrategy>,
    },
}

impl AdvancedSearchOptimizer {
    pub async fn optimize(
        &self,
        request: &SearchRequest,
        collection_config: &CollectionConfig,
    ) -> Result<OptimizationDecision> {
        let start = std::time::Instant::now();

        // Record metric
        self.metrics.increment_counter("search.optimization.attempts", 1.0);

        // Check cache first
        if let Some(cached_decision) = self.check_decision_cache(request, collection_config).await? {
            self.metrics.increment_counter("search.optimization.cache_hits", 1.0);
            return Ok(cached_decision);
        }

        // Analyze available resources
        let resources = self.analyze_resources(collection_config).await?;

        // Make optimization decision
        let decision = self.make_decision(request, collection_config, &resources).await?;

        // Cache the decision
        self.cache_decision(request, collection_config, &decision).await?;

        // Record metrics
        let latency = start.elapsed().as_micros() as f64;
        self.metrics.record_histogram("search.optimization.latency_us", latency);

        Ok(decision)
    }

    async fn analyze_resources(
        &self,
        config: &CollectionConfig,
    ) -> Result<AvailableResources> {
        let mut resources = AvailableResources::default();

        // Check for quantization columns
        if let Some(quant_config) = &config.quantization {
            resources.has_precomputed_quantization = quant_config.precompute;
            resources.quantization_levels = quant_config.levels.clone();

            // Check if quantized data actually exists in storage
            // TODO: Add storage check when implementing
        }

        // Check for indexes
        for index_config in &config.index_configs {
            let index_exists = self.axis_manager
                .has_index(&config.name, &index_config.index_type)
                .await?;

            if index_exists {
                resources.available_indexes.push(IndexInfo {
                    index_type: index_config.index_type.clone(),
                    is_ready: index_exists,
                    parameters: index_config.parameters.clone(),
                });
            }
        }

        // Check cache status
        let cache_stats = self.cache_orchestrator.get_statistics().await?;
        resources.cache_hit_ratio = cache_stats.hit_ratio;
        resources.available_memory_gb = cache_stats.available_memory_gb;

        Ok(resources)
    }

    async fn make_decision(
        &self,
        request: &SearchRequest,
        config: &CollectionConfig,
        resources: &AvailableResources,
    ) -> Result<OptimizationDecision> {
        // Priority order for optimization strategies

        // 1. Check for cached results
        let cache_key = self.generate_cache_key(request, config);
        if self.cache_orchestrator.has_cached_results(&cache_key).await? {
            return Ok(OptimizationDecision {
                strategy: ExecutionStrategy::UseCachedResults { cache_key },
                estimated_latency_ms: 0.1,
                confidence: 1.0,
                cache_key: Some(cache_key),
                reasoning: "Using cached results".to_string(),
            });
        }

        // 2. Use HNSW index if available and beneficial
        if let Some(hnsw_index) = resources.available_indexes
            .iter()
            .find(|idx| idx.index_type == "HNSW" && idx.is_ready) {

            if self.should_use_index(request, hnsw_index, config) {
                return Ok(OptimizationDecision {
                    strategy: ExecutionStrategy::UseIndex {
                        index_type: "HNSW".to_string(),
                        parameters: hnsw_index.parameters.clone(),
                    },
                    estimated_latency_ms: self.estimate_index_latency(request, hnsw_index),
                    confidence: 0.9,
                    cache_key: None,
                    reasoning: "HNSW index available and beneficial".to_string(),
                });
            }
        }

        // 3. Use precomputed quantization if available
        if resources.has_precomputed_quantization && !resources.quantization_levels.is_empty() {
            let best_level = self.select_quantization_level(
                request,
                &resources.quantization_levels,
                config,
            )?;

            return Ok(OptimizationDecision {
                strategy: ExecutionStrategy::UsePrecomputedQuantization {
                    level: best_level,
                    cascade: request.top_k <= 100,  // Cascade for small k
                },
                estimated_latency_ms: self.estimate_quantized_latency(request, best_level),
                confidence: 0.85,
                cache_key: None,
                reasoning: format!("Using precomputed {:?} quantization", best_level),
            });
        }

        // 4. Runtime quantization if beneficial
        if self.should_use_runtime_quantization(request, config, resources) {
            let level = self.select_runtime_quantization_level(request, resources)?;

            return Ok(OptimizationDecision {
                strategy: ExecutionStrategy::RuntimeQuantization {
                    level,
                    cache_result: true,
                },
                estimated_latency_ms: self.estimate_runtime_quant_latency(request, level),
                confidence: 0.75,
                cache_key: None,
                reasoning: format!("Runtime {:?} quantization", level),
            });
        }

        // 5. Default to flat search
        Ok(OptimizationDecision {
            strategy: ExecutionStrategy::FlatSearch {
                batch_size: self.calculate_batch_size(request, resources),
                use_simd: true,  // Always use SIMD if available
            },
            estimated_latency_ms: self.estimate_flat_search_latency(request, config),
            confidence: 0.7,
            cache_key: None,
            reasoning: "Flat search (no optimizations available)".to_string(),
        })
    }

    fn should_use_index(&self, request: &SearchRequest, index: &IndexInfo, config: &CollectionConfig) -> bool {
        // Use index if:
        // - Collection has more than 10K vectors
        // - Not using complex filters that index doesn't support
        // - k is reasonable (not too large)

        let estimated_vectors = config.estimated_count.unwrap_or(0);

        estimated_vectors > 10_000 &&
        request.top_k <= 1000 &&
        !self.has_complex_filters(request)
    }

    fn select_quantization_level(
        &self,
        request: &SearchRequest,
        available_levels: &[String],
        config: &CollectionConfig,
    ) -> Result<QuantizationLevel> {
        // Select based on request requirements and data characteristics

        // High precision needed?
        if request.min_score.unwrap_or(0.0) > 0.95 {
            // Use least aggressive quantization
            if available_levels.contains(&"pq16".to_string()) {
                return Ok(QuantizationLevel::PQ(16));
            }
            if available_levels.contains(&"int8".to_string()) {
                return Ok(QuantizationLevel::Int8);
            }
        }

        // Large result set?
        if request.top_k > 100 {
            // Use more precise quantization
            if available_levels.contains(&"pq8".to_string()) {
                return Ok(QuantizationLevel::PQ(8));
            }
            if available_levels.contains(&"int8".to_string()) {
                return Ok(QuantizationLevel::Int8);
            }
        }

        // Default: use most aggressive available
        if available_levels.contains(&"binary".to_string()) {
            return Ok(QuantizationLevel::Binary);
        }
        if available_levels.contains(&"int8".to_string()) {
            return Ok(QuantizationLevel::Int8);
        }

        // Fallback to first available
        Ok(self.parse_quantization_level(&available_levels[0])?)
    }

}

use crate::storage::engines::traits::UnifiedStorageEngine;
use crate::compute::quantization::unified::UnifiedQuantizationEngine;

/// Extension trait for search optimization
#[async_trait]
pub trait OptimizedSearchExecution {
    async fn execute_with_strategy(
        &self,
        request: SearchRequest,
        strategy: ExecutionStrategy,
        config: &CollectionConfig,
    ) -> Result<SearchResponse>;
}

/// Common implementation for all engines
pub struct OptimizedSearchExecutor {
    engine: Arc<dyn UnifiedStorageEngine>,
    quantization_engine: Arc<UnifiedQuantizationEngine>,
    cache_orchestrator: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
}

impl OptimizedSearchExecutor {
    pub async fn execute(
        &self,
        request: SearchRequest,
        decision: OptimizationDecision,
        config: &CollectionConfig,
    ) -> Result<SearchResponse> {
        let start = Instant::now();

        // Record decision metrics
        self.metrics.increment_counter(
            &format!("search.strategy.{:?}", decision.strategy),
            1.0
        );

        let response = match decision.strategy {
            ExecutionStrategy::UsePrecomputedQuantization { level, cascade } => {
                self.search_with_precomputed(request, level, cascade).await?
            },

            ExecutionStrategy::UseIndex { index_type, parameters } => {
                self.search_with_index(request, &index_type, parameters).await?
            },

            ExecutionStrategy::RuntimeQuantization { level, cache_result } => {
                self.search_with_runtime_quantization(request, level, cache_result).await?
            },

            ExecutionStrategy::UseCachedResults { cache_key } => {
                self.get_cached_results(&cache_key).await?
            },

            ExecutionStrategy::FlatSearch { batch_size, use_simd } => {
                self.flat_search(request, batch_size, use_simd).await?
            },

            ExecutionStrategy::Hybrid { primary, fallback } => {
                self.hybrid_search(request, *primary, *fallback).await?
            },
        };

        // Record performance metrics
        let latency = start.elapsed().as_millis() as f64;
        self.metrics.record_histogram("search.execution.latency_ms", latency);
        self.metrics.record_histogram(
            &format!("search.execution.{:?}.latency_ms", decision.strategy),
            latency
        );

        // Cache results if beneficial
        if self.should_cache_results(&request, &response) {
            self.cache_results(&request, &response).await?;
        }

        Ok(response)
    }

    async fn search_with_precomputed(
        &self,
        request: SearchRequest,
        level: QuantizationLevel,
        cascade: bool,
    ) -> Result<SearchResponse> {
        // Use precomputed quantized vectors stored in collection

        if cascade {
            // Start with binary, refine with better quantization
            let mut candidates = self.search_quantized_level(
                &request,
                QuantizationLevel::Binary,
                request.top_k * 10,
            ).await?;

            if level != QuantizationLevel::Binary {
                // Refine with target level
                candidates = self.refine_with_quantization(
                    candidates,
                    &request,
                    level,
                ).await?;
            }

            // Optional: final reranking with full vectors
            if request.use_exact_reranking.unwrap_or(false) {
                candidates = self.rerank_exact(candidates, &request).await?;
            }

            Ok(SearchResponse { results: candidates, ..Default::default() })
        } else {
            // Direct search with specified quantization level
            let results = self.search_quantized_level(&request, level, request.top_k).await?;
            Ok(SearchResponse { results, ..Default::default() })
        }
    }

    async fn search_with_runtime_quantization(
        &self,
        request: SearchRequest,
        level: QuantizationLevel,
        cache_result: bool,
    ) -> Result<SearchResponse> {
        // Quantize at query time
        let query_quantized = self.quantization_engine
            .quantize(&request.query_vector, level)
            .await?;

        // Scan and quantize data vectors on the fly
        let results = self.engine
            .scan_with_runtime_quantization(
                &request.collection_id,
                query_quantized,
                level,
                request.top_k,
            )
            .await?;

        // Cache quantized vectors for future use if beneficial
        if cache_result && self.should_cache_quantized(&request) {
            self.cache_quantized_vectors(&request.collection_id, level).await?;
        }

        Ok(SearchResponse { results, ..Default::default() })
    }
}

use crate::compute::quantization::unified::UnifiedQuantizationEngine;
use crate::compute::quantization::global_cache::GlobalQuantizationCache;
use crate::storage::cache::orchestrator::CacheOrchestrator;
use crate::metrics::unified::UnifiedMetrics;

pub struct QuantizationPrecomputeService {
    engine: Arc<UnifiedQuantizationEngine>,
    codebook_cache: Arc<GlobalQuantizationCache>,
    cache_orchestrator: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
}

impl QuantizationPrecomputeService {
    /// Called during flush operation
    pub async fn precompute_for_flush(
        &self,
        records: Vec<VectorRecord>,
        config: &CollectionConfig,
    ) -> Result<Vec<VectorRecord>> {
        if !config.quantization.as_ref()?.precompute {
            return Ok(records);  // No precomputation needed
        }

        let quant_config = config.quantization.as_ref().unwrap();
        let levels = self.parse_levels(&quant_config.levels)?;

        // Get or create codebooks
        let codebooks = self.ensure_codebooks(&config.name, &levels).await?;

        // Process in parallel batches
        let chunk_size = self.calculate_chunk_size(&records);
        let mut handles = Vec::new();

        for chunk in records.chunks(chunk_size) {
            let engine = self.engine.clone();
            let levels_clone = levels.clone();
            let codebooks_clone = codebooks.clone();
            let metrics = self.metrics.clone();
            let chunk_vec = chunk.to_vec();

            let handle = tokio::spawn(async move {
                Self::process_chunk(chunk_vec, levels_clone, engine, codebooks_clone, metrics).await
            });
            handles.push(handle);
        }

        // Collect results
        let mut precomputed = Vec::new();
        for handle in handles {
            match handle.await? {
                Ok(chunk_results) => precomputed.extend(chunk_results),
                Err(e) => {
                    // Log error but continue
                    self.metrics.increment_counter("quantization.precompute.errors", 1.0);
                    tracing::warn!("Precomputation failed for chunk: {}", e);
                    // TODO: Consider partial success handling
                }
            }
        }

        // Cache the precomputed data
        self.cache_precomputed(&config.name, &precomputed).await?;

        Ok(precomputed)
    }

    async fn process_chunk(
        chunk: Vec<VectorRecord>,
        levels: Vec<QuantizationLevel>,
        engine: Arc<UnifiedQuantizationEngine>,
        codebooks: Arc<CodebookSet>,
        metrics: Arc<UnifiedMetrics>,
    ) -> Result<Vec<VectorRecord>> {
        let mut results = Vec::new();

        for mut record in chunk {
            let start = Instant::now();

            // Skip if already quantized
            if record.quantized.is_some() {
                results.push(record);
                continue;
            }

            let mut quantized = QuantizedVectors::default();
            let mut quality_scores = Vec::new();

            for level in &levels {
                let (quant_data, quality) = engine.quantize_with_quality(
                    &record.values,
                    level,
                    codebooks.as_ref(),
                ).await?;

                match level {
                    QuantizationLevel::Binary => {
                        quantized.binary = Some(quant_data.to_bytes());
                        quality_scores.push(quality);
                    },
                    QuantizationLevel::Int8 => {
                        quantized.int8 = Some(quant_data.to_bytes());
                        quality_scores.push(quality);
                    },
                    QuantizationLevel::PQ(bits) => {
                        let pq_data = PQData {
                            codes: quant_data.to_bytes(),
                            codebook_id: codebooks.get_pq_id(*bits),
                            num_subvectors: codebooks.get_pq_m(*bits),
                            bits_per_code: *bits,
                        };

                        match bits {
                            4 => quantized.pq4 = Some(pq_data),
                            8 => quantized.pq8 = Some(pq_data),
                            16 => quantized.pq16 = Some(pq_data),
                            _ => {}
                        }
                        quality_scores.push(quality);
                    },
                }
            }

            // Add metadata
            record.quantized = Some(quantized);
            record.quant_metadata = Some(QuantizationMetadata {
                original_norm: compute_norm(&record.values),
                quality_scores,
                computation_time_us: start.elapsed().as_micros() as i64,
                algorithm_version: env!("CARGO_PKG_VERSION").to_string(),
            });

            results.push(record);

            // Update metrics
            metrics.increment_counter("quantization.precompute.vectors", 1.0);
            metrics.record_histogram(
                "quantization.precompute.latency_us",
                start.elapsed().as_micros() as f64
            );
        }

        Ok(results)
    }

    async fn cache_precomputed(
        &self,
        collection_name: &str,
        records: &[VectorRecord],
    ) -> Result<()> {
        // Cache quantized data for fast access
        let cache_key = format!("quantized:{}", collection_name);

        // Use unified cache orchestrator
        self.cache_orchestrator
            .cache_quantized_batch(&cache_key, records)
            .await?;

        Ok(())
    }
}

/// Handles runtime quantization when precomputed data is not available
pub struct RuntimeQuantizationHandler {
    quantization_engine: Arc<UnifiedQuantizationEngine>,
    cache_orchestrator: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
}

impl RuntimeQuantizationHandler {
    pub async fn handle_missing_quantization(
        &self,
        request: &SearchRequest,
        config: &CollectionConfig,
    ) -> Result<QuantizationStrategy> {
        // Check if we should compute at runtime
        if !self.should_quantize_runtime(request, config) {
            return Ok(QuantizationStrategy::Skip);
        }

        // Check cache for previously computed quantization
        let cache_key = format!("runtime_quant:{}:{:?}",
                               request.collection_id,
                               request.query_vector.len());

        if let Some(cached) = self.cache_orchestrator.get(&cache_key).await? {
            self.metrics.increment_counter("quantization.runtime.cache_hits", 1.0);
            return Ok(QuantizationStrategy::UseCached(cached));
        }

        // Compute quantization
        let level = self.select_runtime_level(request, config)?;
        let quantized = self.quantization_engine
            .quantize(&request.query_vector, level)
            .await?;

        // Cache for future use
        self.cache_orchestrator
            .put(&cache_key, &quantized, Duration::from_secs(3600))
            .await?;

        Ok(QuantizationStrategy::Compute(quantized))
    }

    fn should_quantize_runtime(&self, request: &SearchRequest, config: &CollectionConfig) -> bool {
        // Quantize at runtime if:
        // 1. Collection is large enough
        // 2. No precomputed quantization available
        // 3. Query characteristics suggest benefit

        let estimated_size = config.estimated_count.unwrap_or(0);
        estimated_size > 100_000 &&
        request.top_k <= 100
    }
