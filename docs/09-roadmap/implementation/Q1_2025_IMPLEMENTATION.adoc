use crate::index::axis::AxisManager;
use crate::storage::engines::traits::UnifiedStorageEngine;

pub struct CostEstimator {
    axis_manager: Arc<AxisManager>,
    cache_stats: Arc<CacheStatistics>,
}

impl CostEstimator {
    pub async fn estimate_cost(
        &self,
        query: &SearchQuery,
        engine: &EngineType,
    ) -> Result<CostEstimate> {
        // 1. Check HNSW index availability
        let has_hnsw = self.axis_manager.has_index(
            &query.collection_id,
            IndexType::HNSW
        ).await?;

        // 2. Estimate I/O cost based on cache hit ratio
        let cache_hit_ratio = self.cache_stats.get_hit_ratio();

        // 3. Calculate engine-specific costs
        let engine_cost = match engine {
            EngineType::SST => self.estimate_sst_cost(query, has_hnsw),
            EngineType::VIPER => self.estimate_viper_cost(query, has_hnsw),
            EngineType::NOVA => self.estimate_nova_cost(query, has_hnsw),
            // ... other engines
        };

        Ok(CostEstimate {
            cpu_cycles: engine_cost.cpu,
            memory_bytes: engine_cost.memory,
            io_operations: engine_cost.io * (1.0 - cache_hit_ratio),
            estimated_latency_ms: engine_cost.total_ms,
        })
    }
}

        // NEW: Get optimizer from context

        // NEW: Check if we should use optimized path
        if let Some(optimized_plan) = optimizer.create_plan(&request).await? {

        }

        // Existing search logic as fallback

    }

    // NEW: Optimized search execution
    async fn execute_optimized_search(
        &self,
        plan: OptimizedSearchPlan,
    ) -> Result<SearchResponse> {
        match plan.strategy {
            SearchStrategy::UseHNSW => {
                let hnsw_results = self.axis_manager
                    .search_hnsw(&plan.query)
                    .await?;
                self.rerank_with_exact_distance(hnsw_results).await
            },
            SearchStrategy::UseQuantized(level) => {
                self.search_with_quantization(plan.query, level).await
            },
            SearchStrategy::Hybrid(primary, secondary) => {
                self.execute_hybrid_search(primary, secondary).await
            },
        }
    }
}

== Quantized Vector Precomputation

=== Overview

==== Step 2: Create Precompute Service
File: `src/compute/quantization/precompute.rs`

[source,rust]
----
use crate::compute::quantization::unified::UnifiedQuantizationEngine;
use crate::storage::cache::global::GlobalQuantizationCache;

pub struct QuantizationPrecomputeService {
    engine: Arc<UnifiedQuantizationEngine>,
    cache: Arc<GlobalQuantizationCache>,
}

impl QuantizationPrecomputeService {
    pub async fn precompute_batch(
        &self,
        records: Vec<VectorRecord>,
        config: &QuantizationConfig,
    ) -> Result<Vec<VectorRecord>> {
        // Get or create codebook for this collection
        let codebook = self.cache
            .get_or_create_codebook(&config.collection_id)
            .await?;

        // Process in parallel batches
        let chunk_size = 1000;
        let mut handles = Vec::new();

        for chunk in records.chunks(chunk_size) {
            let engine = self.engine.clone();
            let cb = codebook.clone();
            let levels = config.levels.clone();

            let handle = tokio::spawn(async move {
                Self::precompute_chunk(chunk, &levels, engine, cb).await
            });
            handles.push(handle);
        }

        // Collect results
        let mut precomputed = Vec::new();
        for handle in handles {
            precomputed.extend(handle.await??);
        }

        Ok(precomputed)
    }

    async fn precompute_chunk(
        chunk: &[VectorRecord],
        levels: &[QuantizationLevel],
        engine: Arc<UnifiedQuantizationEngine>,
        codebook: Arc<Codebook>,
    ) -> Result<Vec<VectorRecord>> {
        let mut results = Vec::new();

        for record in chunk {
            let mut quantized = QuantizedVectors::default();

            for level in levels {
                match level {
                    QuantizationLevel::Binary => {
                        quantized.binary = Some(
                            engine.quantize_binary(&record.values)?
                        );
                    },
                    QuantizationLevel::Int8 => {
                        quantized.int8 = Some(
                            engine.quantize_int8(&record.values)?
                        );
                    },
                    QuantizationLevel::PQ(bits) => {
                        let pq_data = engine.quantize_pq(
                            &record.values,
                            *bits,
                            &codebook
                        )?;
                        match bits {
                            4 => quantized.pq4 = Some(pq_data),
                            8 => quantized.pq8 = Some(pq_data),
                            16 => quantized.pq16 = Some(pq_data),
                            32 => quantized.pq32 = Some(pq_data),
                            _ => {}
                        }
                    },
                }
            }

            quantized.codebook_id = Some(codebook.id.clone());

            let mut new_record = record.clone();
            new_record.quantized = Some(quantized);
            results.push(new_record);
        }

        Ok(results)
    }
}

impl SSTEngine {
    async fn do_flush(&self, params: FlushParameters) -> Result<FlushResult> {
        let records = self.write_buffer.drain_batch(params.batch_size)?;

        // NEW: Precompute quantized vectors if enabled
        let records = if params.collection_config.quantization.enabled {
            let precompute = QuantizationPrecomputeService::new(
                self.quantization_engine.clone()
            );
            precompute.precompute_batch(
                records,
                &params.collection_config.quantization
            ).await?
        } else {
            records
        };

        // Write to storage with quantized data
        let sstable = SSTableBuilder::new()
            .add_records(records)
            .with_compression(params.compression)
            .build()?;

        self.filesystem.write_sstable(&sstable).await?;

        Ok(FlushResult {
            records_flushed: records.len(),
            bytes_written: sstable.size_bytes(),
            has_quantized: params.collection_config.quantization.enabled,
        })
    }
}

impl SSTEngine {
    async fn search_with_quantization(
        &self,
        request: SearchRequest,
        level: QuantizationLevel,
    ) -> Result<SearchResponse> {
        // Quantize query vector once
        let query_quantized = self.quantization_engine
            .quantize(&request.query_vector, level)
            .await?;

        // Scan using precomputed quantized vectors
        let mut candidates = Vec::new();

        for sstable in self.get_sstables().await? {
            // Read only quantized data, not full vectors
            let quantized_scan = sstable
                .scan_quantized(level)
                .await?;

            for (id, quantized_vector) in quantized_scan {
                let distance = match level {
                    QuantizationLevel::Binary => {
                        compute_hamming_distance(&query_quantized, &quantized_vector)
                    },
                    QuantizationLevel::Int8 => {
                        compute_int8_distance(&query_quantized, &quantized_vector)
                    },
                    QuantizationLevel::PQ(_) => {
                        compute_pq_distance(&query_quantized, &quantized_vector)
                    },
                };

                candidates.push((id, distance));
            }
        }

        // Re-rank top candidates with full precision if needed
        candidates.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap());
        candidates.truncate(request.top_k * 10); // Overselect

        let final_results = if request.use_exact_reranking {
            self.rerank_with_full_vectors(candidates, &request.query_vector).await?
        } else {
            candidates.into_iter()
                .take(request.top_k)
                .collect()
        };

        Ok(SearchResponse {
            results: final_results,
            search_method: format!("quantized_{:?}", level),
        })
    }
}

=== Success Metrics

==== Performance Targets
- **Search Latency**: < 10µs for cached quantized searches
- **Throughput**: > 100K queries/sec with quantization
- **Storage Overhead**: < 30% increase with all quantization levels
- **Precomputation Time**: < 100ms per 10K vectors

==== Quality Metrics
- **Recall@10**: > 95% for INT8, > 90% for binary
- **Test Coverage**: > 90% for new code
- **Documentation**: Complete API docs and examples

=== Common Pitfalls to Avoid

1. **Don't block flush operations**: Use async precomputation
2. **Don't store redundant data**: Deduplicate identical quantized vectors
3. **Don't ignore memory limits**: Implement streaming for large batches
4. **Don't break compatibility**: Support reading old data without quantized vectors
5. **Don't hardcode strategies**: Make optimizer configurable

=== Rollback Plan

If issues arise:
1. Feature flag to disable optimizer: `search.optimizer.enabled = false`
2. Fallback to standard search when optimizer fails
3. Skip precomputation if it fails during flush
4. Keep existing search code paths intact
