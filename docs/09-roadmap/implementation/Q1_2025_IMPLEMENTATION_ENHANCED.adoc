pub struct AdvancedSearchOptimizer {

/// Optimization plan that engines will execute
#[derive(Clone, Debug)]
pub struct OptimizedSearchPlan {
    pub strategy: SearchStrategy,
    pub engine_hint: Option<EngineType>,
    pub index_type: Option<IndexType>,
    pub quantization_level: Option<QuantizationLevel>,
    pub parallelism: usize,
    pub timeout_ms: u64,
    pub cost_estimate: CostEstimate,
}

#[derive(Clone, Debug)]
pub enum SearchStrategy {
    /// Use HNSW index for approximate search
    UseHNSW {
        ef_search: usize,
        rerank_top_k: Option<usize>,
    },
    /// Use quantized vectors for fast filtering
    UseQuantized {
        level: QuantizationLevel,
        cascade: bool,  // Use cascading refinement
    },
    /// Hybrid approach combining multiple strategies
    Hybrid {
        primary: Box<SearchStrategy>,
        secondary: Box<SearchStrategy>,
        fusion_method: FusionMethod,
    },
    /// Direct brute-force search (for small datasets)
    BruteForce {
        batch_size: usize,
    },
    /// Distributed search across multiple shards
    Distributed {
        shard_selector: ShardSelector,
        aggregation: AggregationMethod,
    },
}

impl AdvancedSearchOptimizer {
    pub async fn create_plan(
        &self,
        request: &SearchRequest,
    ) -> Result<Option<OptimizedSearchPlan>> {
        // Collect current system state
        let system_state = self.collect_system_state().await?;

        // Check optimization cache for similar queries
        if let Some(cached_plan) = self.optimization_cache
            .read()
            .await
            .get_plan(request, &system_state)? {
            return Ok(Some(cached_plan));
        }

        // Estimate costs for different strategies
        let strategies = self.generate_candidate_strategies(request, &system_state)?;
        let mut evaluated_strategies = Vec::new();

        for strategy in strategies {
            let cost = self.cost_estimator
                .estimate_cost(request, &strategy, &system_state)
                .await?;
            evaluated_strategies.push((strategy, cost));
        }

        // Select best strategy
        let best = self.select_best_strategy(evaluated_strategies, request)?;

        // Cache the decision
        if let Some(ref plan) = best {
            self.optimization_cache
                .write()
                .await
                .cache_plan(request, plan.clone())?;
        }

        Ok(best)
    }

    async fn collect_system_state(&self) -> Result<SystemState> {
        let cache_stats = self.statistics_collector.get_cache_statistics().await?;
        let index_stats = self.statistics_collector.get_index_statistics().await?;
        let load_stats = self.statistics_collector.get_system_load().await?;

        Ok(SystemState {
            cache_hit_ratio: cache_stats.hit_ratio,
            available_memory_gb: load_stats.available_memory_gb,
            cpu_usage_percent: load_stats.cpu_usage_percent,
            active_indexes: index_stats.active_indexes,
            total_vectors: index_stats.total_vectors,
        })
    }
}

use crate::storage::engines::impls::viper::engine::ViperEngine;

pub struct CostEstimator {

    cache_orchestrator: Arc<CacheOrchestrator>,
    historical_data: Arc<RwLock<HistoricalPerformance>>,

impl CostEstimator {
    pub async fn estimate_cost(
        &self,
        query: &SearchRequest,
        strategy: &SearchStrategy,
        state: &SystemState,
    ) -> Result<CostEstimate> {
        let base_cost = self.estimate_base_cost(query, strategy).await?;

        // Apply ProximaDB-specific adjustments
        let adjusted_cost = self.apply_adjustments(base_cost, state, query).await?;

        // Learn from historical data
        let predicted_cost = self.apply_ml_prediction(adjusted_cost, query).await?;

        Ok(predicted_cost)
    }

    async fn estimate_base_cost(
        &self,
        query: &SearchRequest,
        strategy: &SearchStrategy,
    ) -> Result<BaseCost> {
        match strategy {
            SearchStrategy::UseHNSW { ef_search, .. } => {
                // HNSW cost model based on ProximaDB's AXIS implementation
                let index_stats = self.axis_manager
                    .get_index_statistics(&query.collection_id)
                    .await?;

                let nodes_to_visit = self.estimate_hnsw_nodes(*ef_search, index_stats.levels);
                let memory_access = nodes_to_visit * index_stats.node_size_bytes;

                Ok(BaseCost {
                    cpu_cycles: nodes_to_visit as f64 * 1000.0,  // Distance computations
                    memory_bytes: memory_access,
                    io_operations: if state.cache_hit_ratio > 0.8 { 0 } else { nodes_to_visit / 100 },
                    network_bytes: 0,
                })
            },

            SearchStrategy::UseQuantized { level, cascade } => {
                // Quantization cost model
                let reduction_factor = match level {
                    QuantizationLevel::Binary => 32.0,
                    QuantizationLevel::Int8 => 4.0,
                    QuantizationLevel::PQ(bits) => 32.0 / (*bits as f32),
                };

                let base_memory = query.top_k * query.dimension * 4;  // f32 size
                let quantized_memory = (base_memory as f32 / reduction_factor) as usize;

                Ok(BaseCost {
                    cpu_cycles: if *cascade { 50000.0 } else { 20000.0 },
                    memory_bytes: quantized_memory,
                    io_operations: quantized_memory / 4096,  // Page-aligned reads
                    network_bytes: 0,
                })
            },

            _ => self.estimate_fallback_cost(query, strategy).await,
        }
    }

    fn apply_adjustments(
        &self,
        base: BaseCost,
        state: &SystemState,
        query: &SearchRequest,
    ) -> Result<CostEstimate> {
        // Memory pressure adjustment
        let memory_multiplier = if state.available_memory_gb < 2.0 {
            2.0  // High memory pressure, likely swapping
        } else {
            1.0
        };

        // CPU contention adjustment
        let cpu_multiplier = 1.0 + (state.cpu_usage_percent / 100.0);

        // Cache benefit
        let cache_benefit = state.cache_hit_ratio;
        let effective_io = base.io_operations as f32 * (1.0 - cache_benefit);

        // Convert to latency estimate
        let estimated_latency_ms =
            (base.cpu_cycles / 3_000_000_000.0) * 1000.0 * cpu_multiplier +  // 3GHz CPU assumption
            (base.memory_bytes as f64 / 50_000_000_000.0) * 1000.0 * memory_multiplier +  // 50GB/s memory
            (effective_io * 0.1) as f64;  // 0.1ms per I/O operation

        Ok(CostEstimate {
            cpu_cycles: base.cpu_cycles,
            memory_bytes: base.memory_bytes,
            io_operations: effective_io,
            estimated_latency_ms,
            confidence: 0.85,  // Will be refined with ML
        })
    }
}

    async fn execute_optimized_search(
        &self,
        plan: OptimizedSearchPlan,
        request: SearchRequest,
    ) -> Result<SearchResponse>;

/// Macro to implement optimized search for all engines
macro_rules! impl_optimized_search {
    ($engine:ty, $name:literal) => {
        #[async_trait]
        impl OptimizedSearchExecutor for $engine {
            async fn execute_optimized_search(
                &self,
                plan: OptimizedSearchPlan,
                request: SearchRequest,
            ) -> Result<SearchResponse> {
                // Log optimization decision
                tracing::debug!(
                    engine = $name,
                    strategy = ?plan.strategy,
                    "Executing optimized search"
                );

                match plan.strategy {
                    SearchStrategy::UseHNSW { ef_search, rerank_top_k } => {
                        self.search_with_hnsw(request, ef_search, rerank_top_k).await
                    },
                    SearchStrategy::UseQuantized { level, cascade } => {
                        self.search_with_quantization(request, level, cascade).await
                    },
                    SearchStrategy::Hybrid { primary, secondary, fusion_method } => {
                        self.search_hybrid(request, *primary, *secondary, fusion_method).await
                    },
                    SearchStrategy::BruteForce { batch_size } => {
                        self.search_brute_force(request, batch_size).await
                    },
                    SearchStrategy::Distributed { shard_selector, aggregation } => {
                        self.search_distributed(request, shard_selector, aggregation).await
                    },
                }
            }
        }
    };
}

use crate::compute::quantization::unified::UnifiedQuantizationEngine;
use crate::compute::quantization::global_cache::GlobalQuantizationCache;
use crate::compute::quantization::selection::QuantizationSelector;
use crate::proto::proximadb::{VectorRecord, QuantizedVectors, QuantizationMetadata};

pub struct QuantizationPrecomputeService {
    engine: Arc<UnifiedQuantizationEngine>,
    cache: Arc<GlobalQuantizationCache>,
    selector: Arc<QuantizationSelector>,
    parallel_executor: Arc<ParallelExecutor>,
}

impl QuantizationPrecomputeService {
    /// Main entry point for batch precomputation
    pub async fn precompute_batch(
        &self,
        records: Vec<VectorRecord>,
        config: &QuantizationConfig,
    ) -> Result<Vec<VectorRecord>> {
        // Analyze batch characteristics
        let batch_stats = self.analyze_batch(&records)?;

        // Select optimal quantization levels based on data
        let selected_levels = self.selector
            .select_levels(&batch_stats, &config.constraints)?;

        // Get or create codebooks
        let codebooks = self.ensure_codebooks(&config.collection_id, &selected_levels).await?;

        // Process in parallel with proper resource management
        let chunk_size = self.calculate_optimal_chunk_size(&batch_stats);
        let mut futures = Vec::new();

        for chunk in records.chunks(chunk_size) {
            let engine = self.engine.clone();
            let levels = selected_levels.clone();
            let cbs = codebooks.clone();
            let chunk_vec = chunk.to_vec();

            let future = self.parallel_executor.spawn(async move {
                Self::process_chunk(chunk_vec, levels, engine, cbs).await
            });
            futures.push(future);
        }

        // Collect results with error handling
        let mut precomputed = Vec::new();
        for future in futures {
            match future.await {
                Ok(Ok(chunk_results)) => precomputed.extend(chunk_results),
                Ok(Err(e)) => {
                    // Log error but continue with other chunks
                    tracing::warn!("Chunk precomputation failed: {}", e);
                    // Return original records for failed chunk
                },
                Err(e) => {
                    tracing::error!("Task join error: {}", e);
                }
            }
        }

        Ok(precomputed)
    }

    async fn process_chunk(
        chunk: Vec<VectorRecord>,
        levels: Vec<QuantizationLevel>,
        engine: Arc<UnifiedQuantizationEngine>,
        codebooks: Arc<CodebookSet>,
    ) -> Result<Vec<VectorRecord>> {
        let mut results = Vec::new();

        for mut record in chunk {
            let start = Instant::now();
            let original_norm = compute_l2_norm(&record.values);

            let mut quantized = QuantizedVectors::default();
            let mut quality_scores = Vec::new();

            for level in &levels {
                match level {
                    QuantizationLevel::Binary => {
                        let (binary_data, quality) = engine
                            .quantize_binary_with_quality(&record.values)?;
                        quantized.binary = Some(binary_data);
                        quality_scores.push(quality);
                    },

                    QuantizationLevel::Int8 => {
                        let (int8_data, quality) = engine
                            .quantize_int8_with_quality(&record.values)?;
                        quantized.int8 = Some(int8_data);
                        quality_scores.push(quality);
                    },

                    QuantizationLevel::PQ(bits) => {
                        let codebook = codebooks.get_pq_codebook(*bits)?;
                        let (pq_data, quality) = engine
                            .quantize_pq_with_quality(&record.values, codebook)?;

                        let pq_message = PQData {
                            codes: pq_data,
                            codebook_id: codebook.id.clone(),
                            num_subvectors: codebook.m,
                            bits_per_code: *bits,
                        };

                        match bits {
                            4 => quantized.pq4 = Some(pq_message),
                            8 => quantized.pq8 = Some(pq_message),
                            16 => quantized.pq16 = Some(pq_message),
                            32 => quantized.pq32 = Some(pq_message),
                            _ => {},
                        }
                        quality_scores.push(quality);
                    },

                    QuantizationLevel::Scalar(bits) => {
                        let (scalar_data, params, quality) = engine
                            .quantize_scalar_with_quality(&record.values, *bits)?;
                        quantized.scalar = Some(ScalarQuantized {
                            values: scalar_data,
                            scale: params.scale,
                            offset: params.offset,
                            bits: *bits,
                        });
                        quality_scores.push(quality);
                    },
                }
            }

            // Add metadata
            let metadata = QuantizationMetadata {
                original_norm,
                quality_scores,
                computation_time_us: start.elapsed().as_micros() as i64,
                algorithm_version: env!("CARGO_PKG_VERSION").to_string(),
            };

            record.quantized = Some(quantized);
            record.quant_metadata = Some(metadata);
            results.push(record);
        }

        Ok(results)
    }

    fn calculate_optimal_chunk_size(&self, stats: &BatchStatistics) -> usize {
        // Dynamic chunk size based on available resources
        let available_memory = self.get_available_memory();
        let per_vector_memory = stats.avg_dimension * 4 * stats.quantization_levels;
        let optimal_chunk = (available_memory / per_vector_memory).min(1000).max(10);

        optimal_chunk
    }
}

// SST Engine specific implementation
impl QuantizedStorageEngine for SstEngine {
    async fn store_quantized(
        &self,
        records: Vec<VectorRecord>,
        strategy: QuantizedStorageStrategy,
    ) -> Result<()> {
        match strategy {
            QuantizedStorageStrategy::Inline => {
                // Store everything in SSTable
                let sstable = SSTableBuilder::new()
                    .with_records(records)
                    .with_quantized_index(true)  // Enable special index
                    .build()?;

                self.write_sstable(sstable).await
            },

            QuantizedStorageStrategy::Separate { .. } => {
                // Split storage
                let (original, quantized) = self.split_records(records)?;

                // Store original vectors
                let main_sst = SSTableBuilder::new()
                    .with_records(original)
                    .build()?;
                self.write_sstable(main_sst).await?;

                // Store quantized in separate file
                let quant_sst = QuantizedSSTableBuilder::new()
                    .with_quantized(quantized)
                    .with_compression(CompressionType::LZ4)
                    .build()?;
                self.write_quantized_sstable(quant_sst).await
            },

            _ => Err(anyhow!("Strategy not supported for SST engine"))
        }
    }
}

// VIPER Engine specific implementation (columnar)
impl QuantizedStorageEngine for ViperEngine {
    async fn store_quantized(
        &self,
        records: Vec<VectorRecord>,
        strategy: QuantizedStorageStrategy,
    ) -> Result<()> {
        // VIPER uses columnar storage - perfect for separate quantized columns
        let mut parquet_builder = ParquetBuilder::new();

        // Add original vector column
        parquet_builder.add_column("vectors", self.extract_vectors(&records)?);

        // Add quantized columns based on available levels
        if records[0].quantized.is_some() {
            if let Some(binary) = &records[0].quantized.as_ref().unwrap().binary {
                parquet_builder.add_binary_column(
                    "quantized_binary",
                    self.extract_binary_quantized(&records)?
                );
            }

            if let Some(int8) = &records[0].quantized.as_ref().unwrap().int8 {
                parquet_builder.add_fixed_len_byte_array_column(
                    "quantized_int8",
                    self.extract_int8_quantized(&records)?
                );
            }

            // Add PQ columns with dictionary encoding for codebook references
            if let Some(pq8) = &records[0].quantized.as_ref().unwrap().pq8 {
                parquet_builder.add_dictionary_encoded_column(
                    "quantized_pq8",
                    self.extract_pq_quantized(&records, 8)?
                );
            }
        }

        let parquet_file = parquet_builder.build()?;
        self.write_parquet(parquet_file).await
    }
}

pub struct QuantizedSearchEngine {
    storage: Arc<dyn UnifiedStorageEngine>,
    distance_compute: Arc<UnifiedDistanceCompute>,
    cascade_controller: Arc<CascadeController>,
}

impl QuantizedSearchEngine {
    /// Perform search using precomputed quantized vectors
    pub async fn search_with_precomputed(
        &self,
        request: SearchRequest,
        level: QuantizationLevel,
        cascade: bool,
    ) -> Result<SearchResponse> {
        let start = Instant::now();

        // Quantize query vector once
        let query_quantized = self.quantize_query(&request.query_vector, level).await?;

        // Phase 1: Scan with quantized vectors
        let candidates = if cascade {
            self.cascade_search(&query_quantized, &request).await?
        } else {
            self.single_level_search(&query_quantized, level, &request).await?
        };

        // Phase 2: Rerank with full precision if needed
        let final_results = if request.use_exact_reranking && !candidates.is_empty() {
            self.rerank_exact(candidates, &request.query_vector).await?
        } else {
            candidates
        };

        Ok(SearchResponse {
            results: final_results,
            search_time_ms: start.elapsed().as_millis() as f32,
            metadata: hashmap! {
                "quantization_level" => level.to_string(),
                "cascade_used" => cascade.to_string(),
                "candidates_before_rerank" => candidates.len().to_string(),
            },
        })
    }

    async fn cascade_search(
        &self,
        query_quantized: &QuantizedQuery,
        request: &SearchRequest,
    ) -> Result<Vec<SearchResult>> {
        // Start with most aggressive quantization
        let mut candidates = self.single_level_search(
            query_quantized,
            QuantizationLevel::Binary,
            request,
        ).await?;

        // Take top candidates for refinement
        candidates.truncate(request.top_k * 10);

        // Refine with INT8
        if !candidates.is_empty() {
            candidates = self.refine_with_level(
                candidates,
                query_quantized,
                QuantizationLevel::Int8,
            ).await?;
            candidates.truncate(request.top_k * 5);
        }

        // Final refinement with PQ if available
        if !candidates.is_empty() && query_quantized.has_pq() {
            candidates = self.refine_with_level(
                candidates,
                query_quantized,
                QuantizationLevel::PQ(8),
            ).await?;
        }

        Ok(candidates)
    }

    async fn single_level_search(
        &self,
        query: &QuantizedQuery,
        level: QuantizationLevel,
        request: &SearchRequest,
    ) -> Result<Vec<SearchResult>> {
        // Get appropriate scanner for the level
        let scanner = self.storage
            .scan_quantized(&request.collection_id, level, None)
            .await?;

        // Parallel distance computation
        let mut heap = BinaryHeap::new();
        let batch_size = 1000;

        while let Some(batch) = scanner.next_batch(batch_size).await? {
            let distances = self.distance_compute
                .compute_batch_quantized(query, &batch, level)
                .await?;

            for (record, distance) in batch.iter().zip(distances.iter()) {
                if heap.len() < request.top_k {
                    heap.push(SearchResult {
                        id: record.id.clone(),
                        distance: *distance,
                        metadata: record.metadata.clone(),
                    });
                } else if let Some(worst) = heap.peek() {
                    if distance < &worst.distance {
                        heap.pop();
                        heap.push(SearchResult {
                            id: record.id.clone(),
                            distance: *distance,
                            metadata: record.metadata.clone(),
                        });
                    }
                }
            }
        }

        // Convert heap to sorted vector
        let mut results: Vec<_> = heap.into_iter().collect();
        results.sort_by(|a, b| a.distance.partial_cmp(&b.distance).unwrap());

        Ok(results)
    }
}

pub struct OptimizationMetrics {
    // Counters
    optimization_decisions: Counter,
    cache_hits: Counter,
    cache_misses: Counter,

    // Histograms
    decision_latency: Histogram,
    search_latency_by_strategy: HashMap<String, Histogram>,
    quantization_quality: Histogram,

    // Gauges
    active_indexes: Gauge,
    quantized_vectors_ratio: Gauge,
    memory_usage: Gauge,
}

impl OptimizationMetrics {
    pub fn record_optimization_decision(&self, strategy: &SearchStrategy, latency_us: u64) {
        self.optimization_decisions.inc();
        self.decision_latency.observe(latency_us as f64);

        let strategy_name = format!("{:?}", strategy);
        self.search_latency_by_strategy
            .entry(strategy_name)
            .or_insert_with(|| Histogram::new())
            .observe(latency_us as f64);
    }

    pub fn export_prometheus(&self) -> String {
        // Export metrics in Prometheus format
        format!(
            "# HELP search_optimization_decisions Total optimization decisions made\n\
             # TYPE search_optimization_decisions counter\n\
             search_optimization_decisions {}\n\
             # HELP search_decision_latency_microseconds Decision latency\n\
             # TYPE search_decision_latency_microseconds histogram\n\
             {}",
            self.optimization_decisions.get(),
            self.decision_latency.export()
        )
    }
}
