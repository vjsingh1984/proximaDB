**Enhanced implementation guide for Q1 2025: Advanced Search Optimization and Quantized Vector Precomputation with ProximaDB-specific integration**

[IMPORTANT]
====
**Key Architecture Decision**: Quantization is computed ONLY during flush and compaction operations.
Proto messages (VectorRecord) remain unchanged. Quantized vectors are stored internally using
a tuple/wrapper approach that pairs original records with optional quantization data.
This separation ensures:

* Proto messages stay clean and backward compatible
* Quantization is transparent to API clients
* Engines can optimize storage format independently
* Compaction can recalculate quantization for merged data
====

/// Optimization plan that engines will execute
#[derive(Clone, Debug)]
pub struct OptimizedSearchPlan {
    pub strategy: SearchStrategy,
    pub engine_hint: Option<EngineType>,
    pub index_type: Option<IndexType>,
    pub quantization_level: Option<QuantizationLevel>,
    pub parallelism: usize,
    pub timeout_ms: u64,

#[derive(Clone, Debug)]
pub enum SearchStrategy {
    /// Use HNSW index for approximate search
    UseHNSW {
        ef_search: usize,
        rerank_top_k: Option<usize>,
    },
    /// Use quantized vectors for fast filtering
    UseQuantized {
        level: QuantizationLevel,
        cascade: bool,  // Use cascading refinement
    },
    /// Hybrid approach combining multiple strategies
    Hybrid {
        primary: Box<SearchStrategy>,
        secondary: Box<SearchStrategy>,
        fusion_method: FusionMethod,
    },
    /// Direct brute-force search (for small datasets)
    BruteForce {
        batch_size: usize,
    },
    /// Distributed search across multiple shards
    Distributed {
        shard_selector: ShardSelector,
        aggregation: AggregationMethod,
    },
}

impl AdvancedSearchOptimizer {
    pub async fn create_plan(
        &self,
        request: &SearchRequest,
    ) -> Result<Option<OptimizedSearchPlan>> {
        // Collect current system state
        let system_state = self.collect_system_state().await?;

        // Check optimization cache for similar queries
        if let Some(cached_plan) = self.optimization_cache
            .read()
            .await
            .get_plan(request, &system_state)? {
            return Ok(Some(cached_plan));
        }

        // Estimate costs for different strategies
        let strategies = self.generate_candidate_strategies(request, &system_state)?;
        let mut evaluated_strategies = Vec::new();

        for strategy in strategies {
            let cost = self.cost_estimator
                .estimate_cost(request, &strategy, &system_state)
                .await?;
            evaluated_strategies.push((strategy, cost));
        }

        // Select best strategy
        let best = self.select_best_strategy(evaluated_strategies, request)?;

        // Cache the decision
        if let Some(ref plan) = best {
            self.optimization_cache
                .write()
                .await
                .cache_plan(request, plan.clone())?;
        }

        Ok(best)
    }

    async fn collect_system_state(&self) -> Result<SystemState> {
        let cache_stats = self.statistics_collector.get_cache_statistics().await?;
        let index_stats = self.statistics_collector.get_index_statistics().await?;
        let load_stats = self.statistics_collector.get_system_load().await?;

        Ok(SystemState {
            cache_hit_ratio: cache_stats.hit_ratio,
            available_memory_gb: load_stats.available_memory_gb,
            cpu_usage_percent: load_stats.cpu_usage_percent,
            active_indexes: index_stats.active_indexes,
            total_vectors: index_stats.total_vectors,
        })
    }
}

use crate::storage::engines::impls::sst::SstEngine;
use crate::storage::engines::impls::viper::engine::ViperEngine;
use crate::storage::engines::impls::nova::engine::NovaEngine;

pub struct CostEstimator {
    axis_manager: Arc<AxisManager>,
    cache_orchestrator: Arc<CacheOrchestrator>,
    historical_data: Arc<RwLock<HistoricalPerformance>>,
}

impl CostEstimator {
    pub async fn estimate_cost(
        &self,
        query: &SearchRequest,
        strategy: &SearchStrategy,
        state: &SystemState,
    ) -> Result<CostEstimate> {
        let base_cost = self.estimate_base_cost(query, strategy).await?;

        // Apply ProximaDB-specific adjustments
        let adjusted_cost = self.apply_adjustments(base_cost, state, query).await?;

        // Learn from historical data
        let predicted_cost = self.apply_ml_prediction(adjusted_cost, query).await?;

        Ok(predicted_cost)
    }

    async fn estimate_base_cost(
        &self,
        query: &SearchRequest,
        strategy: &SearchStrategy,
    ) -> Result<BaseCost> {
        match strategy {
            SearchStrategy::UseHNSW { ef_search, .. } => {
                // HNSW cost model based on ProximaDB's AXIS implementation
                let index_stats = self.axis_manager
                    .get_index_statistics(&query.collection_id)
                    .await?;

                let nodes_to_visit = self.estimate_hnsw_nodes(*ef_search, index_stats.levels);
                let memory_access = nodes_to_visit * index_stats.node_size_bytes;

                Ok(BaseCost {
                    cpu_cycles: nodes_to_visit as f64 * 1000.0,  // Distance computations
                    memory_bytes: memory_access,
                    io_operations: if state.cache_hit_ratio > 0.8 { 0 } else { nodes_to_visit / 100 },
                    network_bytes: 0,
                })
            },

            SearchStrategy::UseQuantized { level, cascade } => {
                // Quantization cost model
                let reduction_factor = match level {
                    QuantizationLevel::Binary => 32.0,
                    QuantizationLevel::Int8 => 4.0,
                    QuantizationLevel::PQ(bits) => 32.0 / (*bits as f32),
                };

                let base_memory = query.top_k * query.dimension * 4;  // f32 size
                let quantized_memory = (base_memory as f32 / reduction_factor) as usize;

                Ok(BaseCost {
                    cpu_cycles: if *cascade { 50000.0 } else { 20000.0 },
                    memory_bytes: quantized_memory,
                    io_operations: quantized_memory / 4096,  // Page-aligned reads
                    network_bytes: 0,
                })
            },

            _ => self.estimate_fallback_cost(query, strategy).await,
        }
    }

    fn apply_adjustments(
        &self,
        base: BaseCost,
        state: &SystemState,
        query: &SearchRequest,
    ) -> Result<CostEstimate> {
        // Memory pressure adjustment
        let memory_multiplier = if state.available_memory_gb < 2.0 {
            2.0  // High memory pressure, likely swapping
        } else {
            1.0
        };

        // CPU contention adjustment
        let cpu_multiplier = 1.0 + (state.cpu_usage_percent / 100.0);

        // Cache benefit
        let cache_benefit = state.cache_hit_ratio;
        let effective_io = base.io_operations as f32 * (1.0 - cache_benefit);

        // Convert to latency estimate
        let estimated_latency_ms =
            (base.cpu_cycles / 3_000_000_000.0) * 1000.0 * cpu_multiplier +  // 3GHz CPU assumption
            (base.memory_bytes as f64 / 50_000_000_000.0) * 1000.0 * memory_multiplier +  // 50GB/s memory
            (effective_io * 0.1) as f64;  // 0.1ms per I/O operation

        Ok(CostEstimate {
            cpu_cycles: base.cpu_cycles,
            memory_bytes: base.memory_bytes,
            io_operations: effective_io,
            estimated_latency_ms,
            confidence: 0.85,  // Will be refined with ML
        })
    }
}

        &self,
        plan: OptimizedSearchPlan,
        request: SearchRequest,

/// Macro to implement optimized search for all engines
macro_rules! impl_optimized_search {
    ($engine:ty, $name:literal) => {
        #[async_trait]
        impl OptimizedSearchExecutor for $engine {
            async fn execute_optimized_search(
                &self,
                plan: OptimizedSearchPlan,
                request: SearchRequest,
            ) -> Result<SearchResponse> {
                // Log optimization decision
                tracing::debug!(
                    engine = $name,
                    strategy = ?plan.strategy,
                    "Executing optimized search"
                );

                match plan.strategy {
                    SearchStrategy::UseHNSW { ef_search, rerank_top_k } => {
                        self.search_with_hnsw(request, ef_search, rerank_top_k).await
                    },
                    SearchStrategy::UseQuantized { level, cascade } => {
                        self.search_with_quantization(request, level, cascade).await
                    },
                    SearchStrategy::Hybrid { primary, secondary, fusion_method } => {
                        self.search_hybrid(request, *primary, *secondary, fusion_method).await
                    },
                    SearchStrategy::BruteForce { batch_size } => {
                        self.search_brute_force(request, batch_size).await
                    },
                    SearchStrategy::Distributed { shard_selector, aggregation } => {
                        self.search_distributed(request, shard_selector, aggregation).await
                    },
                }
            }
        }
    };
}

#[derive(Debug, Clone)]
pub struct QuantizedVector {
    pub vector_id: String,
    pub binary: Option<Vec<u8>>,      // 1 bit per dimension
    pub int8: Option<Vec<i8>>,         // 8 bits per dimension

pub struct PQData {
    pub codes: Vec<u8>,                // PQ codes
    pub codebook_id: String,           // Reference to codebook
    pub num_subvectors: u32,           // M parameter
    pub bits_per_code: u32,            // Bits per subvector
}

#[derive(Debug, Clone)]
pub struct QuantizationMetadata {
    pub original_norm: f32,            // L2 norm of original vector
    pub quality_scores: Vec<f32>,      // Quality score per quantization level
    pub computation_time_us: u64,      // Time taken to compute
    pub algorithm_version: String,     // Version for compatibility
}

// Wrapper for flush operations that pairs original with quantized
#[derive(Debug, Clone)]
pub struct VectorWithQuantization {
    pub record: VectorRecord,          // Original proto message (unchanged)
    pub quantized: Option<QuantizedVector>, // Optional quantization (internal only)
}

pub type QuantizedVectorPair = (VectorRecord, Option<QuantizedVector>);

// For batch operations during flush
pub struct QuantizedBatch {
    pub records: Vec<VectorRecord>,
    pub quantized: Vec<Option<QuantizedVector>>,

    // Ensure parallel arrays stay in sync
    pub fn iter(&self) -> impl Iterator<Item = (&VectorRecord, &Option<QuantizedVector>)> {
        self.records.iter().zip(self.quantized.iter())
    }

    pub fn into_pairs(self) -> Vec<(VectorRecord, Option<QuantizedVector>)> {
        self.records.into_iter().zip(self.quantized.into_iter()).collect()
    }
}

use crate::compute::quantization::unified::UnifiedQuantizationEngine;
use crate::compute::quantization::global_cache::GlobalQuantizationCache;
use crate::compute::quantization::selection::QuantizationSelector;
use crate::proto::proximadb::VectorRecord;  // Proto unchanged
use crate::storage::formats::quantized_vector::{QuantizedVector, QuantizedBatch};

pub struct QuantizationPrecomputeService {
    engine: Arc<UnifiedQuantizationEngine>,
    cache: Arc<GlobalQuantizationCache>,
    selector: Arc<QuantizationSelector>,
    parallel_executor: Arc<ParallelExecutor>,
}

impl QuantizationPrecomputeService {
    /// Main entry point for flush - returns paired vectors and quantization
    pub async fn quantize_for_flush(
        &self,
        records: Vec<VectorRecord>,
        config: &QuantizationConfig,
    ) -> Result<QuantizedBatch> {
        // Analyze batch characteristics
        let batch_stats = self.analyze_batch(&records)?;

        // Select optimal quantization levels based on data
        let selected_levels = self.selector
            .select_levels(&batch_stats, &config.constraints)?;

        // Get or create codebooks
        let codebooks = self.ensure_codebooks(&config.collection_id, &selected_levels).await?;

        // Process in parallel with proper resource management
        let chunk_size = self.calculate_optimal_chunk_size(&batch_stats);
        let mut futures = Vec::new();

        for chunk in records.chunks(chunk_size) {
            let engine = self.engine.clone();
            let levels = selected_levels.clone();
            let cbs = codebooks.clone();
            let chunk_vec = chunk.to_vec();

            let future = self.parallel_executor.spawn(async move {
                Self::process_chunk(chunk_vec, levels, engine, cbs).await
            });
            futures.push(future);
        }

        // Collect results with error handling
        let mut all_records = Vec::new();
        let mut all_quantized = Vec::new();

        for (i, future) in futures.into_iter().enumerate() {
            match future.await {
                Ok(Ok(chunk_results)) => {
                    for (record, quantized) in chunk_results {
                        all_records.push(record);
                        all_quantized.push(quantized);
                    }
                },
                Ok(Err(e)) => {
                    // Log error but continue with other chunks
                    tracing::warn!("Chunk {} precomputation failed: {}", i, e);
                    // Add original records without quantization for failed chunk
                    let chunk = &records[i * chunk_size..(i + 1) * chunk_size.min(records.len())];
                    for record in chunk {
                        all_records.push(record.clone());
                        all_quantized.push(None);  // No quantization for failed chunks
                    }
                },
                Err(e) => {
                    tracing::error!("Task join error: {}", e);
                }
            }
        }

        Ok(QuantizedBatch {
            records: all_records,
            quantized: all_quantized,
        })
    }

    async fn process_chunk(
        chunk: Vec<VectorRecord>,
        levels: Vec<QuantizationLevel>,
        engine: Arc<UnifiedQuantizationEngine>,
        codebooks: Arc<CodebookSet>,
    ) -> Result<Vec<(VectorRecord, Option<QuantizedVector>)>> {
        let mut results = Vec::new();

        for record in chunk {  // Don't mutate the proto record
            let start = Instant::now();
            let original_norm = compute_l2_norm(&record.values);

            let mut quantized = QuantizedVector {
                vector_id: record.id.clone(),
                binary: None,
                int8: None,
                pq: None,
                metadata: QuantizationMetadata::default(),
            };
            let mut quality_scores = Vec::new();

            for level in &levels {
                match level {
                    QuantizationLevel::Binary => {
                        let (binary_data, quality) = engine
                            .quantize_binary_with_quality(&record.values)?;
                        quantized.binary = Some(binary_data);
                        quality_scores.push(quality);
                    },

                    QuantizationLevel::Int8 => {
                        let (int8_data, quality) = engine
                            .quantize_int8_with_quality(&record.values)?;
                        quantized.int8 = Some(int8_data.into_iter().map(|u| u as i8).collect());
                        quality_scores.push(quality);
                    },

                    QuantizationLevel::PQ(bits) => {
                        let codebook = codebooks.get_pq_codebook(*bits)?;
                        let (pq_data, quality) = engine
                            .quantize_pq_with_quality(&record.values, codebook)?;

                        quantized.pq = Some(PQData {
                            codes: pq_data,
                            codebook_id: codebook.id.clone(),
                            num_subvectors: codebook.m,
                            bits_per_code: *bits,
                        });
                        quality_scores.push(quality);
                    },

                    QuantizationLevel::Scalar(bits) => {
                        // Handle scalar quantization if needed
                        // For now, we'll skip as it's less common
                        tracing::debug!("Scalar quantization not implemented yet");
                    },
                }
            }

            // Set metadata
            quantized.metadata = QuantizationMetadata {
                original_norm,
                quality_scores,
                computation_time_us: start.elapsed().as_micros() as u64,
                algorithm_version: env!("CARGO_PKG_VERSION").to_string(),
            };

            // Return tuple of original record and quantization
            results.push((record, Some(quantized)));
        }

        Ok(results)
    }

    fn calculate_optimal_chunk_size(&self, stats: &BatchStatistics) -> usize {
        // Dynamic chunk size based on available resources
        let available_memory = self.get_available_memory();
        let per_vector_memory = stats.avg_dimension * 4 * stats.quantization_levels;
        let optimal_chunk = (available_memory / per_vector_memory).min(1000).max(10);

        optimal_chunk
    }
}

/// Trait for engines to implement quantized storage
#[async_trait]
pub trait QuantizedStorageEngine {
    /// Store vectors with precomputed quantization (paired approach)
    async fn store_quantized(
        &self,
        batch: QuantizedBatch,  // Uses paired records and quantization
        storage_strategy: QuantizedStorageStrategy,
    ) -> Result<()>;

    /// Retrieve only quantized representations
    async fn scan_quantized(
        &self,
        collection: &str,
        level: QuantizationLevel,
        range: Option<Range<String>>,
    ) -> Result<QuantizedScanIterator>;

    /// Get storage overhead for quantized vectors
    async fn get_quantized_storage_stats(&self) -> Result<QuantizedStorageStats>;
}

#[derive(Clone, Debug)]
pub enum QuantizedStorageStrategy {
    /// Store quantized vectors inline with original
    Inline,

    /// Store in separate files for memory efficiency
    Separate {
        original_file: PathBuf,
        quantized_dir: PathBuf,
    },

    /// Store in columnar format (for VIPER/NOVA)
    Columnar {
        schema: QuantizedSchema,
    },

    /// Tiered storage (hot data inline, cold separate)
    Tiered {
        hot_threshold: Duration,
        inline_levels: Vec<QuantizationLevel>,
        separate_levels: Vec<QuantizationLevel>,
    },
}

// SST Engine specific implementation
impl QuantizedStorageEngine for SstEngine {
    async fn store_quantized(
        &self,
        batch: QuantizedBatch,
        strategy: QuantizedStorageStrategy,
    ) -> Result<()> {
        match strategy {
            QuantizedStorageStrategy::Inline => {
                // Store everything in SSTable with quantization alongside
                let mut builder = SSTableBuilder::new();

                // Add records with their quantization
                for (record, quantized) in batch.into_pairs() {
                    builder.add_record(record);
                    if let Some(q) = quantized {
                        builder.add_quantization(&q);
                    }
                }

                let sstable = builder
                    .with_quantized_index(true)  // Enable special index
                    .build()?;

                self.write_sstable(sstable).await
            },

            QuantizedStorageStrategy::Separate { .. } => {
                // Store original vectors
                let main_sst = SSTableBuilder::new()
                    .with_records(batch.records)
                    .build()?;
                self.write_sstable(main_sst).await?;

                // Store quantized in separate file
                let quant_sst = QuantizedSSTableBuilder::new()
                    .with_quantized_vectors(batch.quantized)
                    .with_compression(CompressionType::LZ4)
                    .build()?;
                self.write_quantized_sstable(quant_sst).await
            },

            _ => Err(anyhow!("Strategy not supported for SST engine"))
        }
    }
}

// VIPER Engine specific implementation (columnar)
impl QuantizedStorageEngine for ViperEngine {
    async fn store_quantized(
        &self,
        batch: QuantizedBatch,
        strategy: QuantizedStorageStrategy,
    ) -> Result<()> {
        // VIPER uses columnar storage - perfect for separate quantized columns
        let mut parquet_builder = ParquetBuilder::new();

        // Add original vector column using constants
        parquet_builder.add_fixed_list_column(
            constants::FIELD_VECTOR_FP32,
            self.extract_vectors(&batch.records)?
        );

        // Add quantized columns based on what's available
        // Check first item to determine which quantization types are present
        if let Some(first_quant) = batch.quantized.first() {
            if let Some(ref q) = first_quant {
                // Binary quantization column
                if q.binary.is_some() {
                    let binary_data = batch.quantized.iter()
                        .filter_map(|opt| opt.as_ref().and_then(|q| q.binary.clone()))
                        .collect();
                    parquet_builder.add_binary_column(
                        constants::FIELD_VECTOR_BINARY,
                        binary_data
                    );
                }

                // INT8 quantization column
                if q.int8.is_some() {
                    let int8_data = batch.quantized.iter()
                        .filter_map(|opt| opt.as_ref().and_then(|q| q.int8.clone()))
                        .collect();
                    parquet_builder.add_fixed_len_byte_array_column(
                        constants::FIELD_VECTOR_INT8,
                        int8_data
                    );
                }

                // PQ columns with proper column names
                if q.pq.is_some() {
                    let pq_data = batch.quantized.iter()
                        .filter_map(|opt| opt.as_ref().and_then(|q| q.pq.clone()))
                        .collect();
                    parquet_builder.add_dictionary_encoded_column(
                        constants::FIELD_VECTOR_PQ,
                        pq_data
                    );
                }
            }
        }

        let parquet_file = parquet_builder.build()?;
        self.write_parquet(parquet_file).await
    }
}

    storage: Arc<dyn UnifiedStorageEngine>,
    distance_compute: Arc<UnifiedDistanceCompute>,
    cascade_controller: Arc<CascadeController>,
}

impl QuantizedSearchEngine {
    /// Perform search using precomputed quantized vectors
    pub async fn search_with_precomputed(
        &self,
        request: SearchRequest,
        level: QuantizationLevel,
        cascade: bool,
    ) -> Result<SearchResponse> {
        let start = Instant::now();

        // Quantize query vector once
        let query_quantized = self.quantize_query(&request.query_vector, level).await?;

        // Phase 1: Scan with quantized vectors
        let candidates = if cascade {
            self.cascade_search(&query_quantized, &request).await?
        } else {
            self.single_level_search(&query_quantized, level, &request).await?
        };

        // Phase 2: Rerank with full precision if needed
        let final_results = if request.use_exact_reranking && !candidates.is_empty() {
            self.rerank_exact(candidates, &request.query_vector).await?
        } else {
            candidates
        };

        Ok(SearchResponse {
            results: final_results,
            search_time_ms: start.elapsed().as_millis() as f32,
            metadata: hashmap! {
                "quantization_level" => level.to_string(),
                "cascade_used" => cascade.to_string(),
                "candidates_before_rerank" => candidates.len().to_string(),
            },
        })
    }

    async fn cascade_search(
        &self,
        query_quantized: &QuantizedQuery,
        request: &SearchRequest,
    ) -> Result<Vec<SearchResult>> {
        // Start with most aggressive quantization
        let mut candidates = self.single_level_search(
            query_quantized,
            QuantizationLevel::Binary,
            request,
        ).await?;

        // Take top candidates for refinement
        candidates.truncate(request.top_k * 10);

        // Refine with INT8
        if !candidates.is_empty() {
            candidates = self.refine_with_level(
                candidates,
                query_quantized,
                QuantizationLevel::Int8,
            ).await?;
            candidates.truncate(request.top_k * 5);
        }

        // Final refinement with PQ if available
        if !candidates.is_empty() && query_quantized.has_pq() {
            candidates = self.refine_with_level(
                candidates,
                query_quantized,
                QuantizationLevel::PQ(8),
            ).await?;
        }

        Ok(candidates)
    }

    async fn single_level_search(
        &self,
        query: &QuantizedQuery,
        level: QuantizationLevel,
        request: &SearchRequest,
    ) -> Result<Vec<SearchResult>> {
        // Get appropriate scanner for the level
        let scanner = self.storage
            .scan_quantized(&request.collection_id, level, None)
            .await?;

        // Parallel distance computation
        let mut heap = BinaryHeap::new();
        let batch_size = 1000;

        while let Some(batch) = scanner.next_batch(batch_size).await? {
            let distances = self.distance_compute
                .compute_batch_quantized(query, &batch, level)
                .await?;

            for (record, distance) in batch.iter().zip(distances.iter()) {
                if heap.len() < request.top_k {
                    heap.push(SearchResult {
                        id: record.id.clone(),
                        distance: *distance,
                        metadata: record.metadata.clone(),
                    });
                } else if let Some(worst) = heap.peek() {
                    if distance < &worst.distance {
                        heap.pop();
                        heap.push(SearchResult {
                            id: record.id.clone(),
                            distance: *distance,
                            metadata: record.metadata.clone(),
                        });
                    }
                }
            }
        }

        // Convert heap to sorted vector
        let mut results: Vec<_> = heap.into_iter().collect();
        results.sort_by(|a, b| a.distance.partial_cmp(&b.distance).unwrap());

        Ok(results)
    }
}

// Extension to UnifiedStorageEngine trait for quantization-aware flush
#[async_trait]
pub trait QuantizedFlush: UnifiedStorageEngine {
    /// Flush with quantization precomputation
    async fn flush_with_quantization(
        &self,
        records: Vec<VectorRecord>,
        collection_config: Option<&Collection>,
    ) -> Result<FlushResult> {
        // Check if quantization is enabled
        let quantization_enabled = collection_config
            .and_then(|c| c.config.as_ref())
            .and_then(|cfg| cfg.quantization.as_ref())
            .map(|q| q.enabled)
            .unwrap_or(false);

        if !quantization_enabled {
            // Fall back to regular flush
            return self.flush_internal(records).await;
        }

        // Get quantization service
        let quant_service = QuantizationPrecomputeService::global();

        // Extract quantization config
        let quant_config = collection_config
            .and_then(|c| c.config.as_ref())
            .and_then(|cfg| cfg.quantization.as_ref())
            .unwrap();

        // Quantize the batch
        let quantized_batch = quant_service
            .quantize_for_flush(records, quant_config)
            .await?;

        // Store with engine-specific strategy
        match self.engine_type() {
            EngineType::SST | EngineType::SWIFT | EngineType::RAPTOR => {
                // Row-based engines: inline storage
                self.store_quantized(
                    quantized_batch,
                    QuantizedStorageStrategy::Inline,
                ).await
            },
            EngineType::VIPER | EngineType::NOVA | EngineType::HELIX => {
                // Columnar engines: separate columns
                self.store_quantized(
                    quantized_batch,
                    QuantizedStorageStrategy::Columnar {
                        schema: self.get_columnar_schema()?,
                    },
                ).await
            },
        }
    }
}

use crate::storage::engines::core::formats::columnar::unified_compaction::UnifiedCompactionFramework;
use crate::compute::quantization::storage_engine::StorageQuantizationEngine;

pub struct QuantizationAwareCompaction {
    framework: Arc<UnifiedCompactionFramework>,
    quant_engine: Arc<StorageQuantizationEngine>,
    memory_pool: Arc<VectorMemoryPool>,
}

impl QuantizationAwareCompaction {
    /// Compact files with quantization recalculation
    pub async fn compact_with_requantization(
        &self,
        files: Vec<String>,
        collection_config: &Collection,
    ) -> Result<String> {
        // Check if quantization is enabled
        let quant_config = collection_config
            .config.as_ref()
            .and_then(|c| c.quantization.as_ref());

        if quant_config.is_none() || !quant_config.unwrap().enabled {
            // Fall back to regular compaction
            return self.framework.compact_files(files).await;
        }

        let quant_config = quant_config.unwrap();

        // Step 1: Load all vectors from files (not just metadata!)
        let mut all_records = Vec::new();
        for file_path in &files {
            let records = self.load_all_vectors(file_path).await?;
            all_records.extend(records);
        }

        info!("Loaded {} vectors for compaction with requantization", all_records.len());

        // Step 2: Process in batches to control memory
        let batch_size = 10_000;
        let mut compacted_batches = Vec::new();

        for batch in all_records.chunks(batch_size) {
            // Get memory from pool for SIMD operations
            let mut memory_buffer = self.memory_pool.allocate(batch.len()).await?;

            // Recalculate quantization for merged data
            let quantized_batch = self.requantize_batch(
                batch,
                quant_config,
                &mut memory_buffer,
            ).await?;

            compacted_batches.push(quantized_batch);

            // Return memory to pool
            self.memory_pool.release(memory_buffer).await;
        }

        // Step 3: Write compacted file with new quantization
        let output_path = self.write_compacted_file(compacted_batches).await?;

        // Step 4: Clean up old files
        for file_path in files {
            self.mark_for_deletion(&file_path).await?;
        }

        Ok(output_path)
    }

    async fn requantize_batch(
        &self,
        batch: &[VectorRecord],
        config: &QuantizationConfig,
        memory: &mut VectorMemoryBuffer,
    ) -> Result<QuantizedBatch> {
        // Extract vectors into SIMD-aligned memory
        memory.load_vectors(batch)?;

        let mut quantized_vectors = Vec::with_capacity(batch.len());

        // Binary quantization - recalculate thresholds for merged data
        if config.enable_binary {
            let binary_threshold = self.calculate_binary_threshold(memory)?;
            for (i, record) in batch.iter().enumerate() {
                let binary = self.quant_engine
                    .quantize_binary_with_threshold(&record.values, binary_threshold)?;
                quantized_vectors[i].binary = Some(binary);
            }
        }

        // INT8 quantization - find new min/max for merged data
        if config.enable_int8 {
            let (min, max) = memory.compute_min_max()?;
            for (i, record) in batch.iter().enumerate() {
                let int8 = self.quant_engine
                    .quantize_int8_with_range(&record.values, min, max)?;
                quantized_vectors[i].int8 = Some(int8);
            }
        }

        // PQ quantization - retrain codebooks on merged data
        if config.enable_pq {
            // Sample vectors for training
            let sample_size = (batch.len() / 10).min(1000).max(100);
            let samples = self.sample_vectors(batch, sample_size);

            // Train new codebooks
            let codebooks = self.quant_engine
                .train_pq_codebooks(&samples, config.pq_segments, config.pq_bits)?;

            // Apply to all vectors
            for (i, record) in batch.iter().enumerate() {
                let pq = self.quant_engine
                    .quantize_pq_with_codebooks(&record.values, &codebooks)?;
                quantized_vectors[i].pq = Some(pq);
            }
        }

        Ok(QuantizedBatch {
            records: batch.to_vec(),
            quantized: quantized_vectors,
        })
    }

    fn calculate_binary_threshold(&self, memory: &VectorMemoryBuffer) -> Result<f32> {
        // Use SIMD to compute optimal threshold
        let mean = memory.compute_mean_simd()?;
        let std_dev = memory.compute_std_dev_simd()?;
        Ok(mean + 0.5 * std_dev)  // Adaptive threshold
    }
}
