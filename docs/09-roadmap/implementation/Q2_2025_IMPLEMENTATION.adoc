#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct GraphNode {
    pub id: String,
    pub labels: Vec<String>,
    pub properties: HashMap<String, Value>,
    pub vector: Option<Vec<f32>>,  // Optional embedding
    pub edges_out: Vec<EdgeRef>,
    pub edges_in: Vec<EdgeRef>,

pub struct GraphEdge {
    pub id: String,
    pub from_node: String,
    pub to_node: String,
    pub edge_type: String,
    pub properties: HashMap<String, Value>,
    pub weight: f32,
}

pub struct GraphStorage {
    nodes: Arc<dyn UnifiedStorageEngine>,
    edges: Arc<dyn UnifiedStorageEngine>,
    node_index: Arc<AxisManager>,  // Reuse AXIS for vector similarity
}

impl GraphStorage {
    pub async fn create_node(&self, node: GraphNode) -> Result<String> {
        // Store node with optional vector embedding
        let record = VectorRecord {
            id: node.id.clone(),
            values: node.vector.unwrap_or_default(),
            metadata: self.node_to_metadata(&node),
            timestamp: Utc::now().timestamp_millis(),
            ..Default::default()
        };

        // Store in appropriate engine based on access pattern
        let engine = if node.vector.is_some() {
            // Use NOVA for vector-enabled nodes
            self.nodes.get_engine(EngineType::NOVA)?
        } else {
            // Use SST for pure graph nodes
            self.nodes.get_engine(EngineType::SST)?
        };

        engine.insert(vec![record]).await?;

        // Index node for fast lookup
        self.index_node(&node).await?;

        Ok(node.id)
    }

    pub async fn create_edge(&self, edge: GraphEdge) -> Result<String> {
        // Validate nodes exist
        self.validate_nodes(&edge.from_node, &edge.to_node).await?;

        // Store edge
        let edge_record = self.edge_to_record(&edge);
        self.edges.insert(vec![edge_record]).await?;

        // Update node edge lists
        self.update_node_edges(&edge).await?;

        Ok(edge.id)
    }

    async fn index_node(&self, node: &GraphNode) -> Result<()> {
        // Create label index
        for label in &node.labels {
            self.node_index.add_to_label_index(label, &node.id).await?;
        }

        // Create property indexes
        for (key, value) in &node.properties {
            if self.is_indexed_property(key) {
                self.node_index.add_to_property_index(key, value, &node.id).await?;
            }
        }

        // Index vector if present
        if let Some(vector) = &node.vector {
            self.node_index.add_vector(&node.id, vector).await?;
        }

        Ok(())
    }
}

pub struct TraversalEngine {
    storage: Arc<GraphStorage>,
    cache: Arc<TraversalCache>,
}

impl TraversalEngine {
    pub async fn bfs(
        &self,
        start_node: &str,
        max_depth: usize,
        filter: Option<TraversalFilter>,
    ) -> Result<Vec<TraversalPath>> {
        let mut visited = HashSet::new();
        let mut queue = VecDeque::new();
        let mut paths = Vec::new();

        queue.push_back((start_node.to_string(), 0, vec![start_node.to_string()]));
        visited.insert(start_node.to_string());

        while let Some((node_id, depth, path)) = queue.pop_front() {
            if depth >= max_depth {
                continue;
            }

            // Get neighbors with caching
            let neighbors = self.get_neighbors(&node_id, &filter).await?;

            for neighbor in neighbors {
                if !visited.contains(&neighbor.id) {
                    visited.insert(neighbor.id.clone());

                    let mut new_path = path.clone();
                    new_path.push(neighbor.id.clone());

                    paths.push(TraversalPath {
                        nodes: new_path.clone(),
                        total_weight: self.calculate_path_weight(&new_path).await?,
                        depth: depth + 1,
                    });

                    queue.push_back((neighbor.id, depth + 1, new_path));
                }
            }
        }

        Ok(paths)
    }

    pub async fn shortest_path(
        &self,
        start: &str,
        end: &str,
        weight_fn: Option<WeightFunction>,
    ) -> Result<Option<TraversalPath>> {
        // Dijkstra's algorithm with A* optimization if vectors available
        let mut distances = HashMap::new();
        let mut previous = HashMap::new();
        let mut heap = BinaryHeap::new();

        distances.insert(start.to_string(), 0.0);
        heap.push(HeapNode {
            node_id: start.to_string(),
            cost: 0.0,
        });

        // Use vector similarity for A* heuristic if available
        let end_vector = self.storage.get_node_vector(end).await?;

        while let Some(HeapNode { node_id, cost }) = heap.pop() {
            if node_id == end {
                return Ok(Some(self.reconstruct_path(&previous, end)));
            }

            if cost > *distances.get(&node_id).unwrap_or(&f32::MAX) {
                continue;
            }

            let neighbors = self.storage.get_neighbors(&node_id).await?;

            for neighbor in neighbors {
                let edge_weight = weight_fn
                    .as_ref()
                    .map(|f| f(&node_id, &neighbor.id))
                    .unwrap_or(1.0);

                let new_cost = cost + edge_weight;

                // A* heuristic using vector similarity
                let heuristic = if let (Some(neighbor_vec), Some(end_vec)) =
                    (neighbor.vector.as_ref(), end_vector.as_ref()) {
                    1.0 - cosine_similarity(neighbor_vec, end_vec)
                } else {
                    0.0
                };

                let priority = new_cost + heuristic;

                if new_cost < *distances.get(&neighbor.id).unwrap_or(&f32::MAX) {
                    distances.insert(neighbor.id.clone(), new_cost);
                    previous.insert(neighbor.id.clone(), node_id.clone());
                    heap.push(HeapNode {
                        node_id: neighbor.id,
                        cost: priority,
                    });
                }
            }
        }

        Ok(None)
    }

    pub async fn pagerank(
        &self,
        damping_factor: f32,
        iterations: usize,
    ) -> Result<HashMap<String, f32>> {
        // Distributed PageRank implementation
        let all_nodes = self.storage.get_all_node_ids().await?;
        let num_nodes = all_nodes.len() as f32;

        let mut ranks = HashMap::new();
        for node_id in &all_nodes {
            ranks.insert(node_id.clone(), 1.0 / num_nodes);
        }

        for _ in 0..iterations {
            let mut new_ranks = HashMap::new();

            for node_id in &all_nodes {
                let mut rank = (1.0 - damping_factor) / num_nodes;

                let incoming = self.storage.get_incoming_edges(node_id).await?;
                for edge in incoming {
                    let source_outgoing_count = self.storage
                        .get_outgoing_count(&edge.from_node)
                        .await? as f32;

                    let source_rank = ranks.get(&edge.from_node).unwrap_or(&0.0);
                    rank += damping_factor * source_rank / source_outgoing_count;
                }

                new_ranks.insert(node_id.clone(), rank);
            }

            ranks = new_ranks;
        }

        Ok(ranks)
    }
}

impl QueryRoot {
    async fn node(&self, ctx: &Context<'_>, id: String) -> Result<Option<GraphNodeQL>> {
        let storage = ctx.data::<Arc<GraphStorage>>()?;

        match storage.get_node(&id).await {
            Ok(node) => Ok(Some(node.into())),
            Err(_) => Ok(None),
        }
    }

    async fn search_nodes(
        &self,
        ctx: &Context<'_>,
        labels: Option<Vec<String>>,
        properties: Option<serde_json::Value>,
        vector: Option<Vec<f32>>,
        limit: Option<i32>,
    ) -> Result<Vec<GraphNodeQL>> {
        let storage = ctx.data::<Arc<GraphStorage>>()?;

        let mut query = NodeQuery::new();

        if let Some(labels) = labels {
            query = query.with_labels(labels);
        }

        if let Some(props) = properties {
            query = query.with_properties(props);
        }

        if let Some(vec) = vector {
            query = query.similar_to_vector(vec, 0.8);
        }

        let results = storage.search_nodes(query, limit.unwrap_or(10)).await?;
        Ok(results.into_iter().map(Into::into).collect())
    }

    async fn traverse(
        &self,
        ctx: &Context<'_>,
        start_node: String,
        pattern: String,  // Cypher-like pattern
        max_depth: i32,
    ) -> Result<Vec<TraversalResult>> {
        let engine = ctx.data::<Arc<TraversalEngine>>()?;

        let pattern = parse_traversal_pattern(&pattern)?;
        let paths = engine.traverse(&start_node, pattern, max_depth as usize).await?;

        Ok(paths.into_iter().map(Into::into).collect())
    }
}

        if let Some(text) = doc.text {

                self.graph_index.create_edge(rel).await?;
            }
        }

        Ok(())
    }

use tantivy::{Index, IndexWriter, Document, Field};

pub struct TextSearchEngine {
    index: Index,
    writer: Arc<Mutex<IndexWriter>>,
    analyzer: TextAnalyzer,
}

impl TextSearchEngine {
    pub async fn index_text(&self, doc_id: &str, text: &str) -> Result<()> {
        // Tokenize and analyze text
        let tokens = self.analyzer.analyze(text)?;

        // Create inverted index entry
        let mut doc = Document::new();
        doc.add_text(self.fields.id, doc_id);
        doc.add_text(self.fields.content, text);

        // Add analyzed tokens
        for token in tokens {
            doc.add_text(self.fields.tokens, &token);
        }

        // Add to index
        self.writer.lock().await.add_document(doc)?;

        Ok(())
    }

    pub async fn search_text(
        &self,
        query: &str,
        limit: usize,
    ) -> Result<Vec<TextSearchResult>> {
        let searcher = self.index.reader()?.searcher();
        let query = self.build_query(query)?;

        let top_docs = searcher.search(&query, &TopDocs::with_limit(limit))?;

        let mut results = Vec::new();
        for (_score, doc_address) in top_docs {
            let doc = searcher.doc(doc_address)?;
            let doc_id = doc.get_first(self.fields.id)
                .and_then(|v| v.as_text())
                .ok_or_else(|| anyhow!("Missing doc id"))?;

            results.push(TextSearchResult {
                doc_id: doc_id.to_string(),
                score: _score,
                highlights: self.generate_highlights(&doc, query)?,
            });
        }

        Ok(results)
    }

    fn build_query(&self, query_str: &str) -> Result<Box<dyn Query>> {
        // Support multiple query types
        if query_str.contains("AND") || query_str.contains("OR") {
            // Boolean query
            self.build_boolean_query(query_str)
        } else if query_str.contains("*") || query_str.contains("?") {
            // Wildcard query
            self.build_wildcard_query(query_str)
        } else if query_str.starts_with("\"") && query_str.ends_with("\"") {
            // Phrase query
            self.build_phrase_query(&query_str[1..query_str.len()-1])
        } else {
            // Default: fuzzy matching
            self.build_fuzzy_query(query_str)
        }
    }
}

pub struct FusionEngine {
    strategies: HashMap<String, Box<dyn FusionStrategy>>,
}

pub trait FusionStrategy: Send + Sync {
    async fn fuse(
        &self,
        results: MultiModalResults,
        weights: &FusionWeights,
    ) -> Result<Vec<FusedResult>>;
}

pub struct RecipocalRankFusion;

impl FusionStrategy for RecipocalRankFusion {
    async fn fuse(
        &self,
        results: MultiModalResults,
        weights: &FusionWeights,
    ) -> Result<Vec<FusedResult>> {
        let mut doc_scores = HashMap::new();

        // RRF formula: score = sum(1 / (k + rank))
        let k = 60.0;

        // Process vector results
        for (rank, result) in results.vector_results.iter().enumerate() {
            let score = weights.vector * (1.0 / (k + rank as f32 + 1.0));
            *doc_scores.entry(result.id.clone()).or_insert(0.0) += score;
        }

        // Process text results
        for (rank, result) in results.text_results.iter().enumerate() {
            let score = weights.text * (1.0 / (k + rank as f32 + 1.0));
            *doc_scores.entry(result.doc_id.clone()).or_insert(0.0) += score;
        }

        // Process metadata results
        for (rank, result) in results.metadata_results.iter().enumerate() {
            let score = weights.metadata * (1.0 / (k + rank as f32 + 1.0));
            *doc_scores.entry(result.id.clone()).or_insert(0.0) += score;
        }

        // Process graph results (with relationship bonus)
        for (rank, result) in results.graph_results.iter().enumerate() {
            let base_score = weights.graph * (1.0 / (k + rank as f32 + 1.0));
            let relationship_bonus = result.path_length as f32 * 0.1;
            let score = base_score * (1.0 + relationship_bonus);
            *doc_scores.entry(result.node_id.clone()).or_insert(0.0) += score;
        }

        // Sort by fused score
        let mut fused: Vec<_> = doc_scores
            .into_iter()
            .map(|(id, score)| FusedResult { id, score, sources: vec![] })
            .collect();

        fused.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());

        Ok(fused)
    }
}

pub struct LearnedFusion {
    model: Arc<RerankingModel>,
}

impl FusionStrategy for LearnedFusion {
    async fn fuse(
        &self,
        results: MultiModalResults,
        weights: &FusionWeights,
    ) -> Result<Vec<FusedResult>> {
        // Use learned model to rerank
        let features = self.extract_features(&results)?;
        let scores = self.model.predict(&features).await?;

        let mut fused = Vec::new();
        for (idx, doc_id) in self.get_unique_docs(&results).iter().enumerate() {
            fused.push(FusedResult {
                id: doc_id.clone(),
                score: scores[idx],
                sources: self.get_sources(doc_id, &results),
            });
        }

        fused.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());
        Ok(fused)
    }
}

pub struct MultiModalSearchEngine {
    index: Arc<MultiModalIndex>,
    fusion: Arc<FusionEngine>,
    reranker: Option<Arc<CrossModalReranker>>,
}

impl MultiModalSearchEngine {
    pub async fn search(
        &self,
        query: MultiModalQuery,
    ) -> Result<MultiModalSearchResponse> {
        // Parallel search across modalities
        let (vector_results, text_results, metadata_results, graph_results) = tokio::join!(
            self.search_vectors(&query),
            self.search_text(&query),
            self.search_metadata(&query),
            self.search_graph(&query),
        );

        let results = MultiModalResults {
            vector_results: vector_results?,
            text_results: text_results?,
            metadata_results: metadata_results?,
            graph_results: graph_results?,
        };

        // Fuse results
        let mut fused = self.fusion
            .fuse(results.clone(), &query.fusion_weights)
            .await?;

        // Cross-modal reranking if available
        if let Some(reranker) = &self.reranker {
            fused = reranker.rerank(fused, &query, &results).await?;
        }

        // Apply result limit
        fused.truncate(query.limit);

        Ok(MultiModalSearchResponse {
            results: fused,
            search_time_ms: start.elapsed().as_millis() as f32,
            modalities_used: self.get_used_modalities(&query),
        })
    }

    async fn search_vectors(&self, query: &MultiModalQuery) -> Result<Vec<VectorResult>> {
        if let Some(vector) = &query.vector {
            self.index.vector_index
                .search_similar(vector, query.vector_limit.unwrap_or(100))
                .await
        } else if let Some(text) = &query.text {
            // Generate vector from text using embedding model
            let vector = self.generate_embedding(text).await?;
            self.index.vector_index
                .search_similar(&vector, query.vector_limit.unwrap_or(100))
                .await
        } else {
            Ok(Vec::new())
        }
    }
}
