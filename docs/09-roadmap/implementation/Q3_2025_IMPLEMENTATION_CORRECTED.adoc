use crate::storage::engines::traits::UnifiedStorageEngine;
use crate::storage::cache::orchestrator::CacheOrchestrator;
use crate::metrics::unified::UnifiedMetrics;
use crate::models::collection::CollectionConfig;

/// Row-level security that operates within collection boundaries
pub struct CollectionRLS {
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    policy_cache: Arc<RwLock<HashMap<String, Vec<RLSPolicy>>>>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct RLSPolicy {
    pub name: String,
    pub collection: String,
    pub operation: Operation,
    pub predicate: SecurityPredicate,
    pub enabled: bool,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum SecurityPredicate {
    /// User can only see their own data
    OwnerOnly { metadata_field: String },
    
    /// Role-based access using metadata fields
    RoleBased { 
        metadata_field: String,
        allowed_values: Vec<String>,
    },
    
    /// Department isolation
    DepartmentIsolation {
        user_dept_field: String,
        record_dept_field: String,
    },
    
    /// Time-based access
    TimeBasedAccess {
        expiry_field: String,
    },
    
    /// Combination of predicates
    And(Vec<SecurityPredicate>),
    Or(Vec<SecurityPredicate>),
}

impl CollectionRLS {
    /// Apply RLS filter to search request based on collection config
    pub async fn apply_security_filter(
        &self,
        request: &mut SearchRequest,
        user_context: &UserContext,
        collection_config: &CollectionConfig,
    ) -> Result<()> {
        self.metrics.record_operation("security.rls.apply").await;
        
        // Check if RLS is enabled for collection
        if !collection_config.security.rls_enabled {
            return Ok(());
        }

        // Get policies from cache or config
        let policies = self.get_policies_for_collection(&collection_config.name).await?;
        
        for policy in policies {
            if !policy.enabled || !policy.applies_to_operation(&request.operation) {
                continue;
            }

            // Build metadata filter based on predicate
            let filter = self.build_metadata_filter(
                &policy.predicate,
                user_context,
                collection_config,
            )?;
            
            // Add to existing filters
            request.metadata_filter = Some(match request.metadata_filter.clone() {
                Some(existing) => MetadataFilter::And(vec![existing, filter]),
                None => filter,
            });
        }

        // Cache the filtered request pattern for optimization
        let cache_key = format!("rls_filter:{}:{}", collection_config.name, user_context.user_id);
        self.cache.put(&cache_key, &request.metadata_filter).await?;

        Ok(())
    }

    fn build_metadata_filter(
        &self,
        predicate: &SecurityPredicate,
        user_context: &UserContext,
        config: &CollectionConfig,
    ) -> Result<MetadataFilter> {
        match predicate {
            SecurityPredicate::OwnerOnly { metadata_field } => {
                // Filter to records where metadata field equals user ID
                Ok(MetadataFilter::Equals {
                    field: metadata_field.clone(),
                    value: Value::String(user_context.user_id.clone()),
                })
            },
            SecurityPredicate::RoleBased { metadata_field, allowed_values } => {
                // Check if user has any of the allowed roles
                let user_roles = &user_context.roles;
                let has_access = allowed_values.iter().any(|v| user_roles.contains(v));
                
                if has_access {
                    Ok(MetadataFilter::AlwaysTrue)
                } else {
                    // Filter to empty result set
                    Ok(MetadataFilter::AlwaysFalse)
                }
            },
            SecurityPredicate::DepartmentIsolation { user_dept_field, record_dept_field } => {
                // User can only see records from their department
                let user_dept = user_context.attributes.get(user_dept_field)
                    .ok_or_else(|| anyhow!("User department not found"))?;
                
                Ok(MetadataFilter::Equals {
                    field: record_dept_field.clone(),
                    value: user_dept.clone(),
                })
            },
            SecurityPredicate::TimeBasedAccess { expiry_field } => {
                // Filter out expired records
                Ok(MetadataFilter::GreaterThan {
                    field: expiry_field.clone(),
                    value: Value::Number(Utc::now().timestamp()),
                })
            },
            SecurityPredicate::And(predicates) => {
                let filters = predicates
                    .iter()
                    .map(|p| self.build_metadata_filter(p, user_context, config))
                    .collect::<Result<Vec<_>>>()?;
                Ok(MetadataFilter::And(filters))
            },
            SecurityPredicate::Or(predicates) => {
                let filters = predicates
                    .iter()
                    .map(|p| self.build_metadata_filter(p, user_context, config))
                    .collect::<Result<Vec<_>>>()?;
                Ok(MetadataFilter::Or(filters))
            },
        }
    }

    /// Get policies for a collection, using cache
    async fn get_policies_for_collection(&self, collection: &str) -> Result<Vec<RLSPolicy>> {
        // Check cache first
        let cache_key = format!("rls_policies:{}", collection);
        if let Some(policies) = self.cache.get::<Vec<RLSPolicy>>(&cache_key).await? {
            return Ok(policies);
        }

        // Load from configuration
        let policies = {
            let policy_cache = self.policy_cache.read().await;
            policy_cache.get(collection).cloned().unwrap_or_default()
        };
        
        // Cache for future use
        self.cache.put(&cache_key, &policies).await?;
        
        Ok(policies)
    }
}

use crate::storage::persistence::filesystem::unified::UnifiedCachingFilesystem;
use ring::aead::{Aad, BoundKey, Nonce, NonceSequence, SealingKey, UnboundKey};

/// Field-level encryption that works with collection storage
pub struct FieldEncryption {
    filesystem: Arc<UnifiedCachingFilesystem>,
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    key_store: Arc<KeyStore>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct FieldEncryptionConfig {
    pub collection: String,
    pub fields: Vec<EncryptedField>,
    pub key_id: String,
    pub algorithm: String,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct EncryptedField {
    pub metadata_path: String,  // Path in metadata map
    pub encryption_type: EncryptionType,
    pub searchable: bool,  // Whether to maintain searchable index
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum EncryptionType {
    Deterministic,  // Same plaintext -> same ciphertext
    Randomized,     // Different ciphertext each time
}

impl FieldEncryption {
    /// Encrypt fields before storage in engine
    pub async fn encrypt_record(
        &self,
        record: &mut VectorRecord,
        config: &CollectionConfig,
    ) -> Result<()> {
        if !config.security.field_encryption_enabled {
            return Ok(());
        }

        self.metrics.record_operation("security.encryption.field").await;
        
        let encryption_config = &config.security.encryption_config;
        let key = self.key_store.get_key(&encryption_config.key_id).await?;
        
        for field in &encryption_config.fields {
            if let Some(value) = self.get_metadata_value(&record.metadata, &field.metadata_path) {
                let encrypted = match field.encryption_type {
                    EncryptionType::Deterministic => {
                        // Use deterministic encryption for searchable fields
                        self.encrypt_deterministic(value, &key, &field.metadata_path).await?
                    },
                    EncryptionType::Randomized => {
                        // Use randomized encryption for maximum security
                        self.encrypt_randomized(value, &key).await?
                    },
                };
                
                // Replace original value with encrypted
                self.set_metadata_value(
                    &mut record.metadata,
                    &field.metadata_path,
                    encrypted,
                )?;
                
                // Maintain searchable index if configured
                if field.searchable {
                    self.update_searchable_index(record, field, value).await?;
                }
            }
        }
        
        // Mark record as encrypted
        record.metadata.insert(
            "__encrypted".to_string(),
            Value::Bool(true),
        );
        
        record.metadata.insert(
            "__encryption_version".to_string(),
            Value::String(encryption_config.version.clone()),
        );
        
        Ok(())
    }

    /// Decrypt fields after retrieval from engine
    pub async fn decrypt_record(
        &self,
        record: &mut VectorRecord,
        config: &CollectionConfig,
    ) -> Result<()> {
        // Check if record is encrypted
        let is_encrypted = record.metadata
            .get("__encrypted")
            .and_then(|v| v.as_bool())
            .unwrap_or(false);
            
        if !is_encrypted {
            return Ok(());
        }

        self.metrics.record_operation("security.decryption.field").await;
        
        let encryption_config = &config.security.encryption_config;
        let key = self.key_store.get_key(&encryption_config.key_id).await?;
        
        for field in &encryption_config.fields {
            if let Some(encrypted_value) = self.get_metadata_value(&record.metadata, &field.metadata_path) {
                let decrypted = self.decrypt_value(encrypted_value, &key).await?;
                
                self.set_metadata_value(
                    &mut record.metadata,
                    &field.metadata_path,
                    decrypted,
                )?;
            }
        }
        
        // Remove encryption markers
        record.metadata.remove("__encrypted");
        record.metadata.remove("__encryption_version");
        
        Ok(())
    }

    async fn encrypt_deterministic(
        &self,
        value: &Value,
        key: &[u8],
        path: &str,
    ) -> Result<Value> {
        // Use path as additional data for deterministic IV
        let iv = self.derive_deterministic_iv(value, path);
        let plaintext = serde_json::to_vec(value)?;
        
        // Check cache for deterministic encryption
        let cache_key = format!("enc_det:{}:{}", path, hash(&plaintext));
        if let Some(cached) = self.cache.get::<Vec<u8>>(&cache_key).await? {
            return Ok(Value::Bytes(cached));
        }
        
        let ciphertext = self.aes_gcm_encrypt(&plaintext, key, &iv)?;
        
        // Cache deterministic result
        self.cache.put(&cache_key, &ciphertext).await?;
        
        Ok(Value::Bytes(ciphertext))
    }

    /// Maintain searchable index for encrypted fields
    async fn update_searchable_index(
        &self,
        record: &VectorRecord,
        field: &EncryptedField,
        plaintext_value: &Value,
    ) -> Result<()> {
        // Create blind index for searching
        let blind_index = self.create_blind_index(plaintext_value)?;
        
        // Store in separate metadata field
        let index_field = format!("__search_index_{}", field.metadata_path);
        record.metadata.insert(index_field, Value::String(blind_index));
        
        Ok(())
    }

    fn create_blind_index(&self, value: &Value) -> Result<String> {
        // Create searchable blind index using HMAC
        // This allows equality searches without revealing plaintext
        let value_bytes = serde_json::to_vec(value)?;
        let hmac = hmac_sha256(&value_bytes, &self.key_store.get_blind_index_key()?);
        Ok(base64::encode(&hmac[..16]))  // Use first 16 bytes for index
    }
}

use crate::metrics::unified::{UnifiedMetrics, MetricEvent};

/// Audit logging that leverages UnifiedMetrics
pub struct AuditLogger {
    metrics: Arc<UnifiedMetrics>,
    cache: Arc<CacheOrchestrator>,
    compliance_config: ComplianceConfig,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct AuditEvent {
    pub timestamp: i64,
    pub user_id: String,
    pub collection: String,
    pub operation: String,
    pub resource_id: Option<String>,
    pub result: AuditResult,
    pub metadata: HashMap<String, Value>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum AuditResult {
    Success,
    Denied { reason: String },
    Failed { error: String },
}

impl AuditLogger {
    /// Log security event using metrics system
    pub async fn log_event(&self, event: AuditEvent) -> Result<()> {
        // Convert to metric event
        let metric_event = MetricEvent {
            name: format!("audit.{}.{}", event.collection, event.operation),
            value: 1.0,
            tags: hashmap!{
                "user_id" => event.user_id.clone(),
                "collection" => event.collection.clone(),
                "operation" => event.operation.clone(),
                "result" => format!("{:?}", event.result),
            },
            timestamp: event.timestamp,
            metadata: event.metadata,
        };
        
        // Record via unified metrics
        self.metrics.record_event(metric_event).await;
        
        // Check compliance requirements
        if self.requires_immediate_persistence(&event) {
            self.persist_audit_event(&event).await?;
        }
        
        // Detect anomalies
        if self.is_suspicious(&event).await {
            self.trigger_alert(&event).await?;
        }
        
        Ok(())
    }

    /// Log collection access
    pub async fn log_access(
        &self,
        user_context: &UserContext,
        collection: &str,
        operation: &str,
        success: bool,
    ) -> Result<()> {
        let event = AuditEvent {
            timestamp: Utc::now().timestamp_millis(),
            user_id: user_context.user_id.clone(),
            collection: collection.to_string(),
            operation: operation.to_string(),
            resource_id: None,
            result: if success {
                AuditResult::Success
            } else {
                AuditResult::Failed { error: "Access denied".to_string() }
            },
            metadata: hashmap!{
                "ip_address" => Value::String(user_context.ip_address.clone()),
                "user_agent" => Value::String(user_context.user_agent.clone()),
            },
        };
        
        self.log_event(event).await
    }

    async fn is_suspicious(&self, event: &AuditEvent) -> bool {
        // Check for anomalous patterns
        match &event.result {
            AuditResult::Failed { error } if error.contains("authentication") => {
                // Check failed auth rate
                let key = format!("failed_auth:{}", event.user_id);
                let count = self.cache.get::<u64>(&key).await.unwrap_or(Some(0)).unwrap_or(0);
                
                // Update counter
                self.cache.put(&key, &(count + 1)).await.ok();
                
                count > 5  // Alert after 5 failed attempts
            },
            AuditResult::Success if event.operation == "bulk_export" => {
                // Check for large exports
                event.metadata.get("record_count")
                    .and_then(|v| v.as_u64())
                    .map(|count| count > 100000)
                    .unwrap_or(false)
            },
            _ => false,
        }
    }

    fn requires_immediate_persistence(&self, event: &AuditEvent) -> bool {
        // Compliance requirements
        matches!(&self.compliance_config.mode, ComplianceMode::Strict) ||
        event.operation == "delete" ||
        event.operation == "modify_security_policy"
    }

    async fn persist_audit_event(&self, event: &AuditEvent) -> Result<()> {
        // Store in audit log collection
        let audit_record = VectorRecord {
            id: uuid::Uuid::new_v4().to_string(),
            values: vec![],  // No vector for audit logs
            metadata: serde_json::to_value(event)?.as_object().unwrap().clone(),
            timestamp: event.timestamp,
            ..Default::default()
        };
        
        // Store in special audit collection
        // TODO: Store via collection service
        
        Ok(())
    }

            }
        }

== Distributed Operations

Implement distribution at the collection level, respecting engine boundaries while enabling horizontal scaling.

* Leverage UnifiedMetrics for distributed monitoring
* Shard routing at collection service level
* No cross-engine data movement

/// Sharding that respects collection-engine binding
pub struct CollectionShardManager {
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    shard_registry: Arc<RwLock<HashMap<String, Vec<ShardInfo>>>>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct ShardInfo {
    pub shard_id: String,
    pub collection: String,
    pub engine_type: EngineType,
    pub node_id: String,
    pub key_range: KeyRange,
    pub replicas: Vec<ReplicaInfo>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct KeyRange {
    pub start_hash: u64,
    pub end_hash: u64,
}

impl CollectionShardManager {
    /// Create sharded collection
    pub async fn create_sharded_collection(
        &self,
        config: CollectionConfig,
        shard_count: usize,
    ) -> Result<()> {
        self.metrics.record_operation("distributed.create_sharded_collection").await;
        
        // All shards use same engine type as collection
        let engine_type = config.engine_type;
        
        // Calculate shard ranges
        let shard_ranges = self.calculate_shard_ranges(shard_count);
        
        // Create shards
        let mut shards = Vec::new();
        for (i, range) in shard_ranges.iter().enumerate() {
            let shard = ShardInfo {
                shard_id: format!("{}_{}", config.name, i),
                collection: config.name.clone(),
                engine_type,
                node_id: self.select_node_for_shard(i, shard_count).await?,
                key_range: range.clone(),
                replicas: self.select_replicas(&config).await?,
            };
            shards.push(shard);
        }
        
        // Register shards
        {
            let mut registry = self.shard_registry.write().await;
            registry.insert(config.name.clone(), shards.clone());
        }
        
        // Initialize each shard on its node
        for shard in shards {
            self.initialize_shard(&shard, &config).await?;
        }
        
        Ok(())
    }

    /// Route request to appropriate shards
    pub async fn route_request(
        &self,
        collection: &str,
        request: &SearchRequest,
    ) -> Result<Vec<ShardInfo>> {
        let shards = {
            let registry = self.shard_registry.read().await;
            registry.get(collection).cloned()
                .ok_or_else(|| anyhow!("Collection not found"))?
        };
        
        // If request has routing key, use it
        if let Some(routing_key) = &request.routing_key {
            let hash = self.hash_key(routing_key);
            let shard = shards.iter()
                .find(|s| s.key_range.contains(hash))
                .ok_or_else(|| anyhow!("No shard for key"))?;
            return Ok(vec![shard.clone()]);
        }
        
        // Otherwise, query all shards
        Ok(shards)
    }

    /// Execute distributed search
    pub async fn distributed_search(
        &self,
        collection: &str,
        request: SearchRequest,
    ) -> Result<SearchResponse> {
        let start = Instant::now();
        
        // Route to shards
        let shards = self.route_request(collection, &request).await?;
        
        // Execute in parallel
        let futures = shards.iter().map(|shard| {
            let req = request.clone();
            let shard = shard.clone();
            async move {
                self.search_shard(&shard, req).await
            }
        });
        
        let results = futures::future::try_join_all(futures).await?;
        
        // Merge results
        let merged = self.merge_search_results(results, request.top_k);
        
        self.metrics.record_histogram(
            "distributed.search.latency",
            start.elapsed().as_millis() as f64,
        ).await;
        
        Ok(merged)
    }

    async fn search_shard(
        &self,
        shard: &ShardInfo,
        request: SearchRequest,
    ) -> Result<SearchResponse> {
        // Check if shard is local
        if shard.node_id == self.get_node_id() {
            // Local shard, execute directly
            let engine = self.get_shard_engine(shard).await?;
            engine.search(request).await
        } else {
            // Remote shard, use RPC
            let client = self.get_node_client(&shard.node_id).await?;
            client.search_shard(&shard.shard_id, request).await
        }
    }

    fn merge_search_results(
        &self,
        shard_results: Vec<SearchResponse>,
        top_k: usize,
    ) -> SearchResponse {
        let mut all_results = Vec::new();
        
        for response in shard_results {
            all_results.extend(response.results);
        }
        
        // Sort by distance/score
        all_results.sort_by(|a, b| a.distance.partial_cmp(&b.distance).unwrap());
        all_results.truncate(top_k);
        
        SearchResponse {
            results: all_results,
            search_time_ms: 0,  // Will be set by caller
        }
    }

    pub async fn rebalance_shards(&self, collection: &str) -> Result<()> {

        // - Move shards between nodes for load balancing
        // - Keep same engine type
        // - Update routing table

    }

/// Replication that maintains collection-engine binding
pub struct EngineReplication {
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    wal: Arc<WriteAheadLog>,
}

impl EngineReplication {
    /// Replicate write operation to shard replicas
    pub async fn replicate_write(
        &self,
        shard: &ShardInfo,
        operation: WriteOperation,
        consistency: ConsistencyLevel,
    ) -> Result<()> {
        self.metrics.record_operation("distributed.replicate_write").await;
        
        // Write to WAL first
        let wal_position = self.wal.append(&operation).await?;
        
        // Determine required acknowledgments
        let required_acks = match consistency {
            ConsistencyLevel::One => 1,
            ConsistencyLevel::Quorum => (shard.replicas.len() + 1) / 2 + 1,
            ConsistencyLevel::All => shard.replicas.len() + 1,
        };
        
        // Send to replicas in parallel
        let futures = shard.replicas.iter().map(|replica| {
            let op = operation.clone();
            let replica = replica.clone();
            async move {
                self.replicate_to_node(&replica, op).await
            }
        });
        
        // Wait for required acks
        let results = futures::future::join_all(futures).await;
        let successful_acks = results.iter().filter(|r| r.is_ok()).count() + 1;  // +1 for primary
        
        if successful_acks < required_acks {
            return Err(anyhow!(
                "Insufficient replicas: {} of {} required",
                successful_acks,
                required_acks
            ));
        }
        
        Ok(())
    }

    async fn replicate_to_node(
        &self,
        replica: &ReplicaInfo,
        operation: WriteOperation,
    ) -> Result<()> {
        // Use existing filesystem for replication
        let replication_path = format!("/replication/{}/{}", replica.node_id, operation.id);
        
        // Serialize operation
        let data = bincode::serialize(&operation)?;
        
        // Write via unified filesystem (handles remote nodes)
        let filesystem = self.get_filesystem().await?;
        filesystem.write(&replication_path, &data).await?;
        
        Ok(())
    }

    /// Handle node failure
    pub async fn handle_node_failure(&self, failed_node: &str) -> Result<()> {
        self.metrics.record_operation("distributed.handle_node_failure").await;
        
        // Find affected shards
        let affected_shards = self.find_shards_on_node(failed_node).await?;
        
        for shard in affected_shards {
            if shard.node_id == failed_node {
                // Primary failed, promote replica
                self.promote_replica(&shard).await?;
            } else {
                // Replica failed, create new replica
                self.create_replacement_replica(&shard, failed_node).await?;
            }
        }
        
        Ok(())
    }

   - Regular security audits of access patterns
   - Compliance reporting automation

2. **Performance**:
   - Cache RLS policies per user session
   - Use deterministic encryption for searchable fields
   - Batch audit log writes
   - Optimize metadata filters at engine level

3. **Distributed Operations**:
   - Start with single-node, add distribution when needed
   - Monitor shard balance and rebalance periodically
   - Use local replicas for read performance
