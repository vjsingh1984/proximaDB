== AutoML Integration

=== Overview
Implement automated optimization that works within ProximaDB's collection-bound engine architecture, leveraging existing infrastructure.

=== Key Architecture Principles
* Collections remain bound to their engines - no cross-engine migration
* Use existing UnifiedMetrics for workload analysis
* Leverage CollectionConfig for optimization decisions
* Integrate with UnifiedCache for performance tracking
* Build on existing quantization and index infrastructure

/// AutoML optimizer that works within collection boundaries
pub struct CollectionOptimizer {
    metrics: Arc<UnifiedMetrics>,
    cache: Arc<CacheOrchestrator>,
    quantization_engine: Arc<UnifiedQuantizationEngine>,

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct OptimizationHistory {
    pub collection: String,
    pub engine_type: EngineType,  // Fixed - cannot change
    pub optimizations_applied: Vec<OptimizationRecord>,
    pub performance_metrics: PerformanceMetrics,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct OptimizationRecord {
    pub timestamp: i64,
    pub optimization_type: OptimizationType,
    pub parameters: HashMap<String, Value>,
    pub impact: PerformanceImpact,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum OptimizationType {
    IndexCreation { index_type: String },
    QuantizationEnabled { levels: Vec<String> },
    CacheStrategy { strategy: String },
    FlushThreshold { new_threshold: usize },
    CompactionSchedule { schedule: String },
}

impl CollectionOptimizer {
    /// Analyze collection and suggest optimizations
    pub async fn analyze_collection(
        &self,
        collection_config: &CollectionConfig,
    ) -> Result<OptimizationPlan> {
        self.metrics.record_operation("automl.analyze_collection").await;
        
        // Get workload statistics from UnifiedMetrics
        let stats = self.get_workload_stats(&collection_config.name).await?;
        
        // Analyze within engine constraints
        let mut recommendations = Vec::new();
        
        // Index optimization (within engine)
        if let Some(index_rec) = self.analyze_index_needs(&stats, collection_config).await? {
            recommendations.push(index_rec);
        }
        
        // Quantization optimization
        if let Some(quant_rec) = self.analyze_quantization_needs(&stats, collection_config).await? {
            recommendations.push(quant_rec);
        }
        
        // Cache optimization
        if let Some(cache_rec) = self.analyze_cache_patterns(&stats, collection_config).await? {
            recommendations.push(cache_rec);
        }
        
        // Engine-specific tuning
        if let Some(engine_rec) = self.tune_engine_parameters(&stats, collection_config).await? {
            recommendations.push(engine_rec);
        }
        
        Ok(OptimizationPlan {
            collection: collection_config.name.clone(),
            engine_type: collection_config.engine_type,  // Cannot change
            recommendations,
            estimated_improvement: self.estimate_improvement(&recommendations),
        })
    }

    async fn get_workload_stats(&self, collection: &str) -> Result<WorkloadStats> {
        // Query UnifiedMetrics for collection statistics
        let cache_key = format!("workload_stats:{}", collection);
        
        if let Some(cached) = self.cache.get::<WorkloadStats>(&cache_key).await? {
            return Ok(cached);
        }
        
        let stats = WorkloadStats {
            query_rate: self.metrics.get_rate("search", collection).await?,
            insert_rate: self.metrics.get_rate("insert", collection).await?,
            avg_latency: self.metrics.get_avg("latency", collection).await?,
            p99_latency: self.metrics.get_percentile("latency", collection, 99.0).await?,
            cache_hit_rate: self.metrics.get_rate("cache_hit", collection).await?,
            common_filters: self.analyze_filter_patterns(collection).await?,
            vector_dimensions_used: self.analyze_dimension_usage(collection).await?,
            time_patterns: self.analyze_temporal_patterns(collection).await?,
        };
        
        self.cache.put(&cache_key, &stats).await?;
        Ok(stats)
    }

    async fn analyze_index_needs(
        &self,
        stats: &WorkloadStats,
        config: &CollectionConfig,
    ) -> Result<Option<Recommendation>> {
        // Check if collection already has indexes
        let existing_indexes = &config.index_config.indexes;
        
        // Analyze query patterns
        let query_patterns = self.extract_query_patterns(stats);
        
        // Recommend index only if beneficial
        if stats.query_rate > 100.0 && stats.p99_latency > 50.0 {
            // Check which index type would work best with the engine
            let recommended_index = match config.engine_type {
                EngineType::SST | EngineType::SWIFT => {
                    // Row-based engines benefit from HNSW
                    if !existing_indexes.contains(&"hnsw".to_string()) {
                        Some("hnsw")
                    } else {
                        None
                    }
                },
                EngineType::VIPER | EngineType::NOVA => {
                    // Columnar engines benefit from IVF
                    if !existing_indexes.contains(&"ivf".to_string()) {
                        Some("ivf")
                    } else {
                        None
                    }
                },
                EngineType::RAPTOR => {
                    // Adaptive engine can use either
                    if stats.query_rate > stats.insert_rate * 10.0 {
                        Some("hnsw")  // Read-heavy
                    } else {
                        Some("lsh")   // Balanced
                    }
                },
                EngineType::HELIX => {
                    // Spatial engine benefits from spatial indexes
                    Some("spatial_hnsw")
                },
            };
            
            if let Some(index_type) = recommended_index {
                return Ok(Some(Recommendation {
                    type_: RecommendationType::CreateIndex,
                    description: format!("Create {} index for improved query performance", index_type),
                    config_changes: hashmap!{
                        "index.type" => Value::String(index_type.to_string()),
                        "index.parameters" => self.get_index_parameters(index_type, stats),
                    },
                    estimated_impact: PerformanceImpact {
                        latency_reduction: 0.5,  // 50% reduction
                        throughput_increase: 2.0,  // 2x throughput
                    },
                }));
            }
        }
        
        Ok(None)
    }

    async fn analyze_quantization_needs(
        &self,
        stats: &WorkloadStats,
        config: &CollectionConfig,
    ) -> Result<Option<Recommendation>> {
        // Check if quantization is already enabled
        if config.quantization.enabled {
            // Possibly optimize quantization levels
            return self.optimize_quantization_levels(stats, config).await;
        }
        
        // Recommend quantization for read-heavy workloads
        if stats.query_rate > stats.insert_rate * 5.0 {
            // Determine best quantization levels based on accuracy requirements
            let levels = if stats.avg_latency > 100.0 {
                // Need aggressive optimization
                vec!["binary", "int8"]
            } else if stats.avg_latency > 50.0 {
                // Moderate optimization
                vec!["int8", "pq8"]
            } else {
                // Light optimization
                vec!["pq16"]
            };
            
            return Ok(Some(Recommendation {
                type_: RecommendationType::EnableQuantization,
                description: "Enable quantization for faster search".to_string(),
                config_changes: hashmap!{
                    "quantization.enabled" => Value::Bool(true),
                    "quantization.levels" => Value::Array(levels.iter().map(|l| Value::String(l.to_string())).collect()),
                    "quantization.precompute" => Value::Bool(true),
                },
                estimated_impact: PerformanceImpact {
                    latency_reduction: 0.7,
                    throughput_increase: 3.0,
                },
            }));
        }
        
        Ok(None)
    }

    async fn analyze_cache_patterns(
        &self,
        stats: &WorkloadStats,
        config: &CollectionConfig,
    ) -> Result<Option<Recommendation>> {
        // Analyze cache effectiveness
        if stats.cache_hit_rate < 0.5 && stats.query_rate > 50.0 {
            // Poor cache hit rate, analyze why
            let cache_analysis = self.analyze_cache_misses(&config.name).await?;
            
            let recommendation = match cache_analysis.miss_reason {
                CacheMissReason::CacheTooSmall => {
                    Recommendation {
                        type_: RecommendationType::AdjustCache,
                        description: "Increase cache size for better hit rate".to_string(),
                        config_changes: hashmap!{
                            "cache.size_mb" => Value::Number(cache_analysis.recommended_size),
                        },
                        estimated_impact: PerformanceImpact {
                            latency_reduction: 0.3,
                            throughput_increase: 1.5,
                        },
                    }
                },
                CacheMissReason::PoorEvictionPolicy => {
                    Recommendation {
                        type_: RecommendationType::AdjustCache,
                        description: "Switch to adaptive cache eviction".to_string(),
                        config_changes: hashmap!{
                            "cache.eviction_policy" => Value::String("adaptive".to_string()),
                        },
                        estimated_impact: PerformanceImpact {
                            latency_reduction: 0.2,
                            throughput_increase: 1.3,
                        },
                    }
                },
                _ => return Ok(None),
            };
            
            return Ok(Some(recommendation));
        }
        
        Ok(None)
    }

    async fn tune_engine_parameters(
        &self,
        stats: &WorkloadStats,
        config: &CollectionConfig,
    ) -> Result<Option<Recommendation>> {
        // Engine-specific tuning
        match config.engine_type {
            EngineType::SST => {
                // Tune SST flush thresholds
                if stats.insert_rate > 100.0 {
                    return Ok(Some(Recommendation {
                        type_: RecommendationType::TuneEngine,
                        description: "Adjust SST flush threshold for write-heavy workload".to_string(),
                        config_changes: hashmap!{
                            "engine.sst.flush_threshold" => Value::Number(10000),
                            "engine.sst.compaction_interval" => Value::Number(300),
                        },
                        estimated_impact: PerformanceImpact {
                            latency_reduction: 0.1,
                            throughput_increase: 1.2,
                        },
                    }));
                }
            },
            EngineType::VIPER => {
                // Tune VIPER column compression
                if stats.query_rate > stats.insert_rate * 10.0 {
                    return Ok(Some(Recommendation {
                        type_: RecommendationType::TuneEngine,
                        description: "Enable aggressive compression for read-heavy workload".to_string(),
                        config_changes: hashmap!{
                            "engine.viper.compression" => Value::String("zstd".to_string()),
                            "engine.viper.compression_level" => Value::Number(9),
                        },
                        estimated_impact: PerformanceImpact {
                            latency_reduction: 0.0,  // No latency improvement
                            throughput_increase: 1.1,  // Slight throughput gain from reduced I/O
                        },
                    }));
                }
            },
            // Add other engine-specific optimizations
            _ => {}
        }
        
        Ok(None)
    }

    /// Apply optimization plan
    pub async fn apply_optimization(
        &self,
        plan: &OptimizationPlan,
        collection_service: &CollectionService,
    ) -> Result<()> {
        self.metrics.record_operation("automl.apply_optimization").await;
        
        for recommendation in &plan.recommendations {
            // Apply configuration changes
            let mut config = collection_service.get_collection_config(&plan.collection).await?;
            
            // TODO: Apply recommendation changes to config

            // based on the recommendation.config_changes
            
            // Track optimization
            self.record_optimization(&plan.collection, recommendation).await?;
        }
        
        Ok(())
    }

    async fn record_optimization(
        &self,
        collection: &str,
        recommendation: &Recommendation,
    ) -> Result<()> {
        let record = OptimizationRecord {
            timestamp: Utc::now().timestamp_millis(),

        let mut history = self.optimization_history.write().await;
        history.entry(collection.to_string())
            .or_insert_with(|| OptimizationHistory {
                collection: collection.to_string(),
                engine_type: EngineType::SST,  // Will be set properly
                optimizations_applied: Vec::new(),
                performance_metrics: PerformanceMetrics::default(),
            })
            .optimizations_applied.push(record);
        
        Ok(())
    }

pub struct AdaptiveQuantizationSelector {
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    quantization_engine: Arc<UnifiedQuantizationEngine>,
}

impl AdaptiveQuantizationSelector {
    /// Select optimal quantization level for a query
    pub async fn select_level(
        &self,
        query: &SearchRequest,
        collection_config: &CollectionConfig,
    ) -> Result<Option<QuantizationLevel>> {
        // Check if quantization is enabled
        if !collection_config.quantization.enabled {
            return Ok(None);
        }
        
        // Check for precomputed quantization
        if collection_config.quantization.precomputed {
            // Use precomputed levels based on query characteristics
            return self.select_precomputed_level(query, collection_config).await;
        }
        
        // Dynamic selection based on query difficulty
        let difficulty = self.estimate_query_difficulty(query).await?;
        
        let level = match difficulty {
            QueryDifficulty::Easy => {
                // Use aggressive quantization
                Some(QuantizationLevel::Binary)
            },
            QueryDifficulty::Medium => {
                // Balance speed and accuracy
                Some(QuantizationLevel::Int8)
            },
            QueryDifficulty::Hard => {
                // Prioritize accuracy
                if query.top_k <= 10 {
                    Some(QuantizationLevel::PQ8)
                } else {
                    None  // Use full precision
                }
            },
        };
        
        // Record decision for learning
        self.record_quantization_decision(query, level.clone()).await?;
        
        Ok(level)
    }

    async fn select_precomputed_level(
        &self,
        query: &SearchRequest,
        config: &CollectionConfig,
    ) -> Result<Option<QuantizationLevel>> {
        // Check which precomputed levels are available
        let available_levels = &config.quantization.precomputed_levels;
        
        // Select based on query requirements
        if query.top_k <= 10 && available_levels.contains(&"binary".to_string()) {
            Ok(Some(QuantizationLevel::Binary))
        } else if query.top_k <= 100 && available_levels.contains(&"int8".to_string()) {
            Ok(Some(QuantizationLevel::Int8))
        } else if available_levels.contains(&"pq8".to_string()) {
            Ok(Some(QuantizationLevel::PQ8))
        } else {
            Ok(None)
        }
    }

    // TODO: Implement ML-based difficulty estimation when ready
    async fn estimate_query_difficulty(&self, query: &SearchRequest) -> Result<QueryDifficulty> {
        // For now, use heuristics
        if query.top_k <= 10 && query.metadata_filter.is_none() {
            Ok(QueryDifficulty::Easy)
        } else if query.top_k <= 100 {
            Ok(QueryDifficulty::Medium)
        } else {
            Ok(QueryDifficulty::Hard)
        }
    }

use crate::services::collection_service::CollectionService;
use crate::proto::proximadb::VectorRecord;

/// Embedding service that uses ProximaDB collections
pub struct ProximaDBEmbeddingService {
    collection_service: Arc<CollectionService>,
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
    providers: HashMap<String, Arc<dyn EmbeddingProvider>>,
}

#[async_trait]
pub trait EmbeddingProvider: Send + Sync {
    async fn generate(&self, text: &str) -> Result<Vec<f32>>;
    async fn generate_batch(&self, texts: &[String]) -> Result<Vec<Vec<f32>>>;
    fn dimension(&self) -> usize;
    fn model_name(&self) -> &str;
}

impl ProximaDBEmbeddingService {
    /// Generate and store embeddings in a collection
    pub async fn embed_and_store(
        &self,
        documents: Vec<Document>,
        collection_name: &str,
        provider_name: &str,
    ) -> Result<Vec<String>> {
        self.metrics.record_operation("llm.embed_and_store").await;
        
        // Get collection config
        let config = self.collection_service.get_collection_config(collection_name).await?;
        
        // Get embedding provider
        let provider = self.providers.get(provider_name)
            .ok_or_else(|| anyhow!("Provider not found"))?;
        
        // Verify dimension matches
        if config.dimension != provider.dimension() {
            return Err(anyhow!("Dimension mismatch: collection {} vs provider {}", 
                config.dimension, provider.dimension()));
        }
        
        // Process in batches
        let mut records = Vec::new();
        let batch_size = 100;
        
        for chunk in documents.chunks(batch_size) {
            // Extract texts
            let texts: Vec<String> = chunk.iter().map(|d| d.content.clone()).collect();
            
            // Check cache for existing embeddings
            let (cached, uncached) = self.separate_cached(&texts, provider_name).await?;
            
            // Generate embeddings for uncached
            let mut embeddings = cached;
            if !uncached.is_empty() {
                let new_embeddings = provider.generate_batch(&uncached).await?;
                
                // Cache new embeddings
                for (text, embedding) in uncached.iter().zip(new_embeddings.iter()) {
                    let cache_key = format!("embedding:{}:{}", provider_name, hash(text));
                    self.cache.put(&cache_key, embedding).await?;
                    embeddings.push(embedding.clone());
                }
            }
            
            // Create VectorRecords
            for (doc, embedding) in chunk.iter().zip(embeddings.iter()) {
                let record = VectorRecord {
                    id: doc.id.clone(),
                    values: embedding.clone(),
                    metadata: hashmap!{
                        "content" => Value::String(doc.content.clone()),
                        "title" => Value::String(doc.title.clone()),
                        "source" => Value::String(doc.source.clone()),
                        "embedding_model" => Value::String(provider.model_name().to_string()),
                        "timestamp" => Value::Number(Utc::now().timestamp()),
                    },
                    timestamp: Utc::now().timestamp_millis(),
                    ..Default::default()
                };
                records.push(record);
            }
        }
        
        // Store in collection using its bound engine
        let response = self.collection_service.insert_records(collection_name, records).await?;
        
        Ok(response.ids)
    }

    async fn separate_cached(
        &self,
        texts: &[String],
        provider: &str,
    ) -> Result<(Vec<Vec<f32>>, Vec<String>)> {
        let mut cached = Vec::new();
        let mut uncached = Vec::new();
        
        for text in texts {
            let cache_key = format!("embedding:{}:{}", provider, hash(text));
            if let Some(embedding) = self.cache.get::<Vec<f32>>(&cache_key).await? {
                cached.push(embedding);
            } else {
                uncached.push(text.clone());
            }
        }
        
        Ok((cached, uncached))
    }
}

/// RAG pipeline that uses ProximaDB collections
pub struct CollectionRAGPipeline {
    collection_service: Arc<CollectionService>,
    embedding_service: Arc<ProximaDBEmbeddingService>,
    llm_provider: Arc<dyn LLMProvider>,
    cache: Arc<CacheOrchestrator>,
    metrics: Arc<UnifiedMetrics>,
}

impl CollectionRAGPipeline {
    /// Execute RAG query against a collection
    pub async fn query(
        &self,
        question: &str,
        collection_name: &str,
        config: RAGConfig,
    ) -> Result<RAGResponse> {
        let start = Instant::now();
        self.metrics.record_operation("llm.rag.query").await;
        
        // Step 1: Check semantic cache
        if let Some(cached) = self.check_semantic_cache(question, collection_name).await? {
            self.metrics.record_cache_hit("llm.rag.semantic_cache").await;
            return Ok(cached);
        }
        
        // Step 2: Generate question embedding
        let question_embedding = self.embedding_service
            .providers.get(&config.embedding_provider)
            .ok_or_else(|| anyhow!("Provider not found"))?
            .generate(question)
            .await?;
        
        // Step 3: Retrieve relevant documents from collection
        let search_request = SearchRequest {
            query_vector: question_embedding,
            top_k: config.retrieval_top_k,
            metadata_filter: config.metadata_filter,
            ..Default::default()
        };
        
        let search_response = self.collection_service
            .search(collection_name, search_request)
            .await?;
        
        // Step 4: Build context from retrieved documents
        let context = self.build_context(search_response.results, &config).await?;
        
        // Step 5: Generate response
        let prompt = self.build_rag_prompt(question, &context, &config)?;
        
        let llm_response = self.llm_provider.generate(LLMRequest {
            prompt,
            max_tokens: config.max_response_tokens,
            temperature: config.temperature,
            ..Default::default()
        }).await?;
        
        // Step 6: Create response
        let response = RAGResponse {
            answer: llm_response.text,
            sources: context.sources,
            confidence: self.calculate_confidence(&context, &llm_response),
            latency_ms: start.elapsed().as_millis() as u64,
            tokens_used: llm_response.tokens_used,
        };
        
        // Step 7: Cache response
        self.cache_response(question, collection_name, &response).await?;
        
        Ok(response)
    }

    async fn check_semantic_cache(
        &self,
        question: &str,
        collection: &str,
    ) -> Result<Option<RAGResponse>> {
        // Use a special cache collection for semantic similarity
        let cache_collection = format!("{}_rag_cache", collection);
        
        // Check if cache collection exists
        if !self.collection_service.collection_exists(&cache_collection).await? {
            return Ok(None);
        }
        
        // Generate question embedding
        let embedding = self.embedding_service
            .providers.get("default")
            .ok_or_else(|| anyhow!("Default provider not found"))?
            .generate(question)
            .await?;
        
        // Search for similar questions
        let search_request = SearchRequest {
            query_vector: embedding,
            top_k: 1,
            metadata_filter: Some(MetadataFilter::GreaterThan {
                field: "timestamp".to_string(),
                value: Value::Number((Utc::now() - Duration::hours(24)).timestamp()),
            }),
            ..Default::default()
        };
        
        let results = self.collection_service
            .search(&cache_collection, search_request)
            .await?;
        
        if let Some(result) = results.results.first() {
            // Check similarity threshold
            if result.distance > 0.95 {  // Very similar
                // Deserialize cached response
                if let Some(response_str) = result.metadata.get("response") {
                    let cached_response: RAGResponse = serde_json::from_str(
                        response_str.as_str().unwrap_or("")
                    )?;
                    return Ok(Some(cached_response));
                }
            }
        }
        
        Ok(None)
    }

    async fn build_context(
        &self,
        results: Vec<SearchResult>,
        config: &RAGConfig,
    ) -> Result<RAGContext> {
        let mut context_parts = Vec::new();
        let mut sources = Vec::new();
        
        for (i, result) in results.iter().take(config.context_top_k).enumerate() {
            // Extract content from metadata
            let content = result.metadata.get("content")
                .and_then(|v| v.as_str())
                .unwrap_or("");
            
            let source = result.metadata.get("source")
                .and_then(|v| v.as_str())
                .unwrap_or("unknown");
            
            context_parts.push(format!("[{}] {}", i + 1, content));
            sources.push(Source {
                id: result.id.clone(),
                title: result.metadata.get("title")
                    .and_then(|v| v.as_str())
                    .unwrap_or("").to_string(),
                url: source.to_string(),
                relevance: result.score,
            });
        }
        
        Ok(RAGContext {
            text: context_parts.join("\n\n"),
            sources,
            total_tokens: self.estimate_tokens(&context_parts.join(" ")),
        })
    }

    fn build_rag_prompt(
        &self,
        question: &str,
        context: &RAGContext,
        config: &RAGConfig,
    ) -> Result<String> {
        let prompt = format!(
            "{}\n\nContext:\n{}\n\nQuestion: {}\n\nAnswer:",
            config.system_prompt.as_deref().unwrap_or(
                "Answer the question based on the provided context. \
                 If the answer cannot be found in the context, say so."
            ),
            context.text,
            question
        );
        
        Ok(prompt)
    }

    async fn cache_response(
        &self,
        question: &str,
        collection: &str,
        response: &RAGResponse,
    ) -> Result<()> {
        // Create cache collection if it doesn't exist
        let cache_collection = format!("{}_rag_cache", collection);
        
        if !self.collection_service.collection_exists(&cache_collection).await? {
            // Create cache collection with same dimension as main collection
            let main_config = self.collection_service
                .get_collection_config(collection)
                .await?;
            
            let cache_config = CollectionConfig {
                name: cache_collection.clone(),
                dimension: main_config.dimension,
                engine_type: EngineType::SST,  // Use SST for cache (fast writes)
                ..Default::default()
            };
            
            self.collection_service.create_collection(cache_config).await?;
        }
        
        // Generate embedding for question
        let embedding = self.embedding_service
            .providers.get("default")
            .ok_or_else(|| anyhow!("Default provider not found"))?
            .generate(question)
            .await?;
        
        // Store in cache collection
        let cache_record = VectorRecord {
            id: uuid::Uuid::new_v4().to_string(),
            values: embedding,
            metadata: hashmap!{
                "question" => Value::String(question.to_string()),
                "response" => Value::String(serde_json::to_string(response)?),
                "timestamp" => Value::Number(Utc::now().timestamp()),
            },
            timestamp: Utc::now().timestamp_millis(),
            ..Default::default()
        };
        
        self.collection_service.insert_records(&cache_collection, vec![cache_record]).await?;
        
        Ok(())
    }
}
