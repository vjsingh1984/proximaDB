impl SparsityAnalyzer {
    pub fn detect_sparsity(&self, vector: &[f32]) -> SparsityInfo {
        let zero_count = vector.iter().filter(|&&x| x.abs() < f32::EPSILON).count();
        let sparsity = zero_count as f32 / vector.len() as f32;
        SparsityInfo {
            sparsity_ratio: sparsity,
            non_zero_count: vector.len() - zero_count,
            detected_at: Instant::now(),
        }
    }

pub fn sparse_l2_distance(a: &[f32], b: &[f32]) -> f32 {
    let mut sum = 0.0f32;

        if a[i] == 0.0 && b[i] == 0.0 {

        let diff = a[i] - b[i];

// src/storage/batch/strategy_selector.rs
pub struct BatchProcessingStrategy {
    parallel_threshold: usize,
    enable_preallocation: bool,
    rayon_threads: usize,
}

impl BatchProcessingStrategy {
    pub fn should_use_parallel(&self, batch_size: usize) -> bool {
        batch_size >= self.parallel_threshold
    }

    pub fn process_batch<T, F>(&self, items: Vec<T>, func: F) -> Vec<R>
    where
        F: Fn(&T) -> R + Sync,
    {
        if self.should_use_parallel(items.len()) {
            // Use Rayon
            items.par_iter().map(func).collect()
        } else {
            // Use sequential
            items.iter().map(func).collect()
        }
    }
}
----

use std::sync::OnceLock;

pub struct LazyMetadata {
    /// Always-loaded fields (inline)
    pub inline_fields: HashMap<String, SqlValue>,

    /// Lazily-loaded fields
    lazy_fields: OnceLock<HashMap<String, SqlValue>>,

    /// Loader function
    loader: Arc<dyn Fn() -> HashMap<String, SqlValue> + Send + Sync>,
}

impl LazyMetadata {
    pub fn get_field(&self, key: &str) -> Option<&SqlValue> {
        // Check inline first
        if let Some(value) = self.inline_fields.get(key) {
            return Some(value);
        }

        // Lazy load remaining fields if needed
        let lazy_fields = self.lazy_fields.get_or_init(|| {
            (self.loader)()
        });

        lazy_fields.get(key)
    }

    pub fn materialize_all(&mut self) -> &HashMap<String, SqlValue> {
        self.lazy_fields.get_or_init(|| (self.loader)())
    }
}
