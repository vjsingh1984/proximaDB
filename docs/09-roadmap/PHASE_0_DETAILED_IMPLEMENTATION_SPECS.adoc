impl GraphService {
    /// Evaluate numeric comparison operations
    fn evaluate_numeric_comparison(
        &self,
        op: PropertyFilterOperator,
        extracted_values: &[serde_json::Value],
        expected_value: &serde_json::Value,
    ) -> bool {
        extracted_values.iter().any(|v| {
            let (v_num, expected_num) = match (self.extract_number(v), self.extract_number(expected_value)) {
                (Some(v), Some(e)) => (v, e),
                _ => return false,
            };
            
            match op {
                PropertyFilterOperator::PROPERTY_FILTER_OPERATOR_GREATER_THAN => v_num > expected_num,
                PropertyFilterOperator::PROPERTY_FILTER_OPERATOR_GREATER_EQUAL => v_num >= expected_num,
                PropertyFilterOperator::PROPERTY_FILTER_OPERATOR_LESS_THAN => v_num < expected_num,
                PropertyFilterOperator::PROPERTY_FILTER_OPERATOR_LESS_EQUAL => v_num <= expected_num,
                _ => false,
            }
        })
    }
    
    /// Extract numeric value from JSON value
    fn extract_number(&self, value: &serde_json::Value) -> Option<f64> {
        match value {
            serde_json::Value::Number(n) => n.as_f64(),
            serde_json::Value::String(s) => s.parse::<f64>().ok(),
            _ => None,
        }
    }
    
    /// Evaluate regex matching
    fn evaluate_regex_match(
        &self,
        extracted_values: &[serde_json::Value],
        expected_value: &serde_json::Value,
        negate: bool,
    ) -> bool {
        if let Some(pattern) = expected_value.as_str() {
            if let Ok(regex) = regex::Regex::new(pattern) {
                let matches = extracted_values.iter().any(|v| {
                    if let Some(s) = v.as_str() {
                        regex.is_match(s)
                    } else {
                        false
                    }
                });
                if negate { !matches } else { matches }
            } else {
                false
            }
        } else {
            false
        }
    }
}

//! Common utilities for SqlValue handling across storage engines

use crate::proto::proximadb_v1::SqlValue;
use serde_json::Value as JsonValue;

/// Unified SqlValue pattern matching utility
pub struct SqlValueMatcher;

impl SqlValueMatcher {
    /// Convert SqlValue to string representation
    pub fn to_string(value: &SqlValue) -> String {
        match value {
            SqlValue::Text(s) => s.clone(),
            SqlValue::Integer(i) => i.to_string(),
            SqlValue::Real(f) => f.to_string(),
            SqlValue::Blob(b) => format!("blob({} bytes)", b.len()),
            SqlValue::Null => "NULL".to_string(),
        }
    }
    
    /// Convert SqlValue to JSON Value
    pub fn to_json_value(value: &SqlValue) -> JsonValue {
        match value {
            SqlValue::Text(s) => JsonValue::String(s.clone()),
            SqlValue::Integer(i) => JsonValue::Number(serde_json::Number::from(*i)),
            SqlValue::Real(f) => {
                if let Some(num) = serde_json::Number::from_f64(*f) {
                    JsonValue::Number(num)
                } else {
                    JsonValue::Null
                }
            },
            SqlValue::Blob(b) => {
                // Encode binary data as base64 string
                JsonValue::String(crate::utils::encoding::base64_encode(b))
            },
            SqlValue::Null => JsonValue::Null,
        }
    }
    
    /// Extract numeric value if possible
    pub fn to_number(value: &SqlValue) -> Option<f64> {
        match value {
            SqlValue::Integer(i) => Some(*i as f64),
            SqlValue::Real(f) => Some(*f),
            SqlValue::Text(s) => s.parse().ok(),
            SqlValue::Blob(_) | SqlValue::Null => None,
        }
    }
    
    /// Check if SqlValue represents a "truthy" value
    pub fn is_truthy(value: &SqlValue) -> bool {
        match value {
            SqlValue::Text(s) => !s.is_empty(),
            SqlValue::Integer(i) => *i != 0,
            SqlValue::Real(f) => *f != 0.0,
            SqlValue::Blob(b) => !b.is_empty(),
            SqlValue::Null => false,
        }
    }
    
    /// Compare two SqlValues for equality
    pub fn equals(a: &SqlValue, b: &SqlValue) -> bool {
        match (a, b) {
            (SqlValue::Text(a), SqlValue::Text(b)) => a == b,
            (SqlValue::Integer(a), SqlValue::Integer(b)) => a == b,
            (SqlValue::Real(a), SqlValue::Real(b)) => (a - b).abs() < f64::EPSILON,
            (SqlValue::Blob(a), SqlValue::Blob(b)) => a == b,
            (SqlValue::Null, SqlValue::Null) => true,
            // Cross-type numeric comparisons
            (SqlValue::Integer(i), SqlValue::Real(f)) => (*i as f64 - f).abs() < f64::EPSILON,
            (SqlValue::Real(f), SqlValue::Integer(i)) => (f - *i as f64).abs() < f64::EPSILON,
            _ => false,
        }
    }
    
    /// Compare SqlValues with ordering
    pub fn compare(a: &SqlValue, b: &SqlValue) -> std::cmp::Ordering {
        use std::cmp::Ordering;
        
        match (a, b) {
            (SqlValue::Text(a), SqlValue::Text(b)) => a.cmp(b),
            (SqlValue::Integer(a), SqlValue::Integer(b)) => a.cmp(b),
            (SqlValue::Real(a), SqlValue::Real(b)) => a.partial_cmp(b).unwrap_or(Ordering::Equal),
            (SqlValue::Blob(a), SqlValue::Blob(b)) => a.cmp(b),
            (SqlValue::Null, SqlValue::Null) => Ordering::Equal,
            // Cross-type numeric comparisons
            (SqlValue::Integer(i), SqlValue::Real(f)) => (*i as f64).partial_cmp(f).unwrap_or(Ordering::Equal),
            (SqlValue::Real(f), SqlValue::Integer(i)) => f.partial_cmp(&(*i as f64)).unwrap_or(Ordering::Equal),
            // Type ordering: Null < Integer < Real < Text < Blob
            (SqlValue::Null, _) => Ordering::Less,
            (_, SqlValue::Null) => Ordering::Greater,
            (SqlValue::Integer(_), SqlValue::Real(_)) => Ordering::Less,
            (SqlValue::Real(_), SqlValue::Integer(_)) => Ordering::Greater,
            (SqlValue::Integer(_), SqlValue::Text(_)) => Ordering::Less,
            (SqlValue::Text(_), SqlValue::Integer(_)) => Ordering::Greater,
            // Add more type ordering as needed
            _ => Ordering::Equal,
        }
    }
}

/// Macro for consistent SqlValue pattern matching
#[macro_export]
macro_rules! match_sql_value {
    ($value:expr, {
        Text($text:ident) => $text_body:expr,
        Integer($int:ident) => $int_body:expr,
        Real($real:ident) => $real_body:expr,
        Blob($blob:ident) => $blob_body:expr,
        Null => $null_body:expr,
    }) => {
        match $value {
            SqlValue::Text($text) => $text_body,
            SqlValue::Integer($int) => $int_body,
            SqlValue::Real($real) => $real_body,
            SqlValue::Blob($blob) => $blob_body,
            SqlValue::Null => $null_body,
        }
    };
}

                    true
                }
            },

    }

fn check_block_condition(&self, block: &HierarchicalBlock, condition: &FilterCondition) -> bool {
    if let Some(block_metadata) = &block.metadata {
        if let Some(field_value) = block_metadata.get(&condition.field) {

        }

fn evaluate_condition_on_sql_value(&self, value: &SqlValue, condition: &FilterCondition) -> bool {
    match condition.operator {
        FilterOperator::Equals => SqlValueMatcher::equals(value, &condition.value),
        FilterOperator::GreaterThan => {
            use std::cmp::Ordering;
            SqlValueMatcher::compare(value, &condition.value) == Ordering::Greater
        },
        FilterOperator::LessThan => {
            use std::cmp::Ordering;
            SqlValueMatcher::compare(value, &condition.value) == Ordering::Less
        },
        FilterOperator::GreaterEqual => {
            use std::cmp::Ordering;
            matches!(SqlValueMatcher::compare(value, &condition.value), Ordering::Greater | Ordering::Equal)

//! Unified type conversion utilities for ProximaDB

use std::collections::HashMap;
use std::sync::Arc;
use serde_json::Value as JsonValue;

/// Utility for common type conversions used throughout ProximaDB
pub struct TypeConverter;

impl TypeConverter {
    /// Convert HashMap<String, String> to HashMap<String, JsonValue>
    pub fn string_map_to_json_map(map: HashMap<String, String>) -> HashMap<String, JsonValue> {
        map.into_iter()
            .map(|(k, v)| (k, JsonValue::String(v)))
            .collect()
    }
    
    /// Convert HashMap<String, String> reference to HashMap<String, JsonValue>
    pub fn string_map_ref_to_json_map(map: &HashMap<String, String>) -> HashMap<String, JsonValue> {
        map.iter()
            .map(|(k, v)| (k.clone(), JsonValue::String(v.clone())))
            .collect()
    }
    
    /// Convert HashMap<String, JsonValue> to HashMap<String, String>
    /// Non-string values are serialized to their JSON representation
    pub fn json_map_to_string_map(map: HashMap<String, JsonValue>) -> HashMap<String, String> {
        map.into_iter()
            .map(|(k, v)| {
                let string_value = match v {
                    JsonValue::String(s) => s,
                    JsonValue::Number(n) => n.to_string(),
                    JsonValue::Bool(b) => b.to_string(),
                    JsonValue::Null => "null".to_string(),
                    _ => v.to_string(), // Serialize complex types as JSON strings
                };
                (k, string_value)
            })
            .collect()
    }
    
    /// Safe Arc dereferencing with lifetime management
    pub fn deref_arc<T>(arc_value: &Arc<T>) -> &T {
        arc_value.as_ref()
    }
    
    /// Convert Vec<Arc<T>> to Vec<&T> for batch processing
    pub fn deref_arc_vec<T>(arc_vec: &[Arc<T>]) -> Vec<&T> {
        arc_vec.iter().map(|arc| arc.as_ref()).collect()
    }
    
    /// Safe conversion from Option<Arc<T>> to Option<&T>
    pub fn deref_arc_option<T>(arc_option: &Option<Arc<T>>) -> Option<&T> {
        arc_option.as_ref().map(|arc| arc.as_ref())
    }
}

/// Trait for types that can be converted to HashMap<String, JsonValue>
pub trait ToJsonMap {
    fn to_json_map(&self) -> HashMap<String, JsonValue>;
}

/// Trait for types that can be converted from HashMap<String, JsonValue>
pub trait FromJsonMap {
    type Error;
    fn from_json_map(map: HashMap<String, JsonValue>) -> Result<Self, Self::Error>
    where
        Self: Sized;
}

// Implementations for common types
impl ToJsonMap for HashMap<String, String> {
    fn to_json_map(&self) -> HashMap<String, JsonValue> {
        TypeConverter::string_map_ref_to_json_map(self)
    }
}

impl ToJsonMap for HashMap<String, JsonValue> {
    fn to_json_map(&self) -> HashMap<String, JsonValue> {
        self.clone()
    }
}

/// Macro for safe Arc dereferencing in pattern matching
#[macro_export]
macro_rules! deref_arc {
    ($arc_expr:expr) => {
        $arc_expr.as_ref()
    };
}

/// Macro for consistent metadata conversion
#[macro_export]
macro_rules! convert_metadata {
    (string_to_json, $map:expr) => {
        $crate::core::type_conversions::TypeConverter::string_map_to_json_map($map)
    };
    (string_ref_to_json, $map:expr) => {
        $crate::core::type_conversions::TypeConverter::string_map_ref_to_json_map($map)
    };
    (json_to_string, $map:expr) => {
        $crate::core::type_conversions::TypeConverter::json_map_to_string_map($map)
    };
}
