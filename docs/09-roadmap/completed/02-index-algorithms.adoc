pub struct HNSWIndex {
    config: HNSWConfig,
    layers: DashMap<(usize, InternalNodeId), Vec<InternalNodeId>>,
    entry_point: RwLock<Option<InternalNodeId>>,
    id_mapping: Arc<GlobalIDIndex>,
    distance_computer: Arc<UnifiedDistanceCompute>,
    stats: Arc<IndexStats>,
}

impl HNSWIndex {
    pub async fn add(&self, id: String, vector: Vec<f32>) -> Result<()> {
        let internal_id = self.id_mapping.register(id)?;
        let level = self.get_random_level();

        // Multi-layer graph construction
        for layer in 0..=level {
            let neighbors = self.search_layer(vector, layer, self.config.ef_construction);
            self.prune_connections(neighbors, self.config.m);
            self.layers.insert((layer, internal_id), neighbors);
        }
        Ok(())
    }

    pub async fn search(&self, query: Vec<f32>, k: usize) -> Result<Vec<SearchResult>> {
        // Sophisticated multi-layer search with pruning
        let candidates = self.search_layer(query, 0, self.config.ef);
        self.rerank_with_pruning(candidates, k)
    }
}

pub struct LSHIndex {
    hash_tables: Vec<HashTable>,
    projections: Vec<ProjectionMatrix>,
    config: LSHConfig,
}

impl LSHIndex {
    pub fn hash(&self, vector: &[f32]) -> Vec<BucketId> {
        self.projections.iter()
            .map(|proj| proj.project_and_hash(vector))
            .collect()
    }
}

fn build_tree(&mut self, indices: Vec<usize>) {
    if indices.len() <= self.config.max_leaf_size {
        self.create_leaf(indices);
        return;
    }

    let (left, right) = self.random_split(indices);
    let split_plane = self.compute_hyperplane(left, right);
    self.nodes.push(Node::Internal(split_plane));

    self.build_tree(left);
    self.build_tree(right);
}
