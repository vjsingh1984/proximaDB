=== Your First AI-Powered Collection
[source,python]
----
from proximadb import ProximaDBClient

# Connect (Python client auto-installs: pip install proximadb)
client = ProximaDBClient("localhost:5678")

# Create a collection (auto-selects optimal storage engine)
collection = client.create_collection(
    name="documents",
    dimension=1536,  # OpenAI ada-002 embeddings
    engine="auto"    # Let ProximaDB optimize
)

# Add your data
documents = [
    {"id": "1", "text": "Q3 revenue increased 40% to $2.3M"},
    {"id": "2", "text": "Customer satisfaction scores improved"},
    {"id": "3", "text": "New enterprise deals in pipeline"}
]

collection.add_documents(documents)

# Ask questions naturally
insights = collection.query(
    "What's driving our growth?",
    top_k=10,
    include_context=True
)

print(insights.answer)
# → "Q3 growth is driven by 40% revenue increase to $2.3M,
#    supported by improved customer satisfaction and
#    strong enterprise pipeline."
----

=== Production-Ready Configuration
[source,toml]
----
# config/config.toml
[server]
http_port = 5678
grpc_port = 5679

[storage]
default_engine = "auto"  # Automatically optimizes
data_dir = "/data"

[ai]
providers = ["openai", "anthropic"]  # Your AI providers
default_model = "gpt-4"

[performance]
enable_simd = true      # Hardware acceleration
enable_gpu = true       # GPU support
quantization = "auto"   # Automatic compression
