import proximadb

# Enterprise client with authentication
client = proximadb.Client(
    endpoint="https://proximadb.company.com:5678",
    api_key="your-enterprise-api-key",
    auth_method="jwt",
    enable_tls=True
)

# Create collection with enterprise features
collection = client.create_collection(
    name="enterprise_knowledge",
    dimension=1536,
    distance_metric="cosine",
    security_policy="enterprise",
    compliance_mode="gdpr_hipaa"
)

# Unified intelligence query
results = client.unified_search(
    collection="enterprise_knowledge",
    query="find strategic insights about market opportunities",
    include_relationships=True,
    include_knowledge_context=True,
    security_filter={"department": "strategy", "clearance": "executive"}
)
