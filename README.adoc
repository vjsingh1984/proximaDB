----

**Want to see it work?** Try this Python example:

[source,python]
----
from proximadb import connect_rest

# Connect and create your first collection
client = connect_rest("http://localhost:5678")
collection = client.create_collection("my_ai_app", dimension=384)

# Insert your embeddings (works with any embedding model)
embeddings = [[0.1, 0.2, ...], [0.3, 0.4, ...]]  # Your vectors here
metadata = [{"doc": "page1.pdf"}, {"doc": "page2.pdf"}]
client.insert_vectors("my_ai_app", embeddings, metadata)

# Search with powerful SQL syntax - now with AND/OR/NOT!
results = client.search_sql("""
    SELECT vector_id, metadata 
    FROM my_ai_app 
    WHERE similarity(vector, [0.1, 0.2, ...]) > 0.8 
    AND (metadata->>'type' = 'important' OR metadata->>'priority' > 5)
    AND metadata->>'status' != 'archived'
""")

print(f"Found {len(results)} similar documents!")
----
