== 🚀 5-Minute Quick Start

=== **Install and Run**

[source,bash]
----
# Clone and build
git clone https://github.com/vjsingh1984/proximadb.git
cd proximadb
cargo build --release

# Start server (ports 5678 REST, 5679 gRPC)
./target/release/proximadb-server --config config/config.toml

# Verify health
curl http://localhost:5678/health
----

=== **Your First AI Query**

[source,python]
----
import proximadb

# Connect to ProximaDB
client = proximadb.Client("localhost:5678")

# Create a knowledge collection
collection = client.create_collection("business_docs", dimension=1536)

# Add documents with embeddings
documents = [
    {"text": "Q3 sales increased 25% driven by enterprise customers",
     "metadata": {"type": "sales_report", "quarter": "Q3"}},
    {"text": "Customer satisfaction scores highest in premium tier",
     "metadata": {"type": "survey", "category": "satisfaction"}}
]

# ProximaDB automatically handles embeddings
collection.insert_documents(documents)

# Ask questions in natural language
answer = client.ask("What drove our Q3 growth?")
print(answer)
# Result: "Q3 sales increased 25% driven by enterprise customers,
#          based on analysis of your sales reports..."

# Vector similarity search
similar = collection.search("enterprise customer growth", top_k=5)

# Graph relationships
relationships = collection.find_related("sales_report", "satisfaction")
----

=== **Enterprise Configuration**

[source,toml]
----
# config/production.toml
[server]
host = "0.0.0.0"
rest_port = 5678
grpc_port = 5679

[storage]
# Choose your optimization
engine = "sst"         # Real-time: SST, Analytics: VIPER
data_path = "/data/proximadb"
max_memory = "32GB"

[ai]
# Configure your LLM providers
default_provider = "openai"
providers = ["openai", "anthropic", "azure_openai"]

[performance]
# Hardware acceleration
enable_simd = true
threads = 16
cache_size = "8GB"
----
