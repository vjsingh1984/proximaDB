### Phase 3: Atomic HNSW Update Strategy

```rust
pub struct HNSWCompactionManager {
    global_index: Arc<RwLock<HNSWIndex>>,
    vector_mapping: Arc<RwLock<VectorLocationMap>>,
    compaction_lock: Arc<Mutex<()>>, // Serialize compactions
}

impl HNSWCompactionManager {
    pub async fn compact_files(&self, input_files: Vec<FileId>) -> Result<FileId> {
        let _compaction_guard = self.compaction_lock.lock().await;
        
        // Phase 1: Analyze cross-file connections
        let affected_connections = self.analyze_cross_file_connections(&input_files).await?;
        
        // Phase 2: Create compaction plan with stable IDs
        let compaction_plan = self.create_compaction_plan(&input_files, &affected_connections).await?;
        
        // Phase 3: Write new file with sorted/merged data
        let new_file_id = self.write_compacted_file(&compaction_plan).await?;
        
        // Phase 4: Atomic HNSW update (critical section)
        {
            let mut hnsw = self.global_index.write().await;
            let mut mapping = self.vector_mapping.write().await;
            
            // Update vector location mapping
            for (vector_id, new_location) in &compaction_plan.new_locations {
                mapping.update_location(*vector_id, new_location.clone())?;
            }
            
            // Update HNSW connections that span the compacted region
            self.update_cross_file_connections(&mut hnsw, &compaction_plan)?;
            
            // Update entry points if needed
            self.update_entry_points(&mut hnsw, &compaction_plan)?;
        }
        
        // Phase 5: Activate new file for queries
        self.activate_compacted_file(new_file_id, &input_files).await?;
        
        // Phase 6: Background cleanup
        tokio::spawn(async move {
            self.cleanup_old_files(input_files).await;
        });
        
        Ok(new_file_id)
    }
    
    async fn analyze_cross_file_connections(&self, files: &[FileId]) -> Result<CrossFileConnections> {
        let hnsw = self.global_index.read().await;
        let mapping = self.vector_mapping.read().await;
        
        let mut connections = CrossFileConnections::new();
        
        // For each vector in the files being compacted
        for file_id in files {
            let vector_ids = mapping.get_vectors_in_file(file_id)?;
            
            for vector_id in vector_ids {
                // Get all connections for this vector
                let neighbors = hnsw.get_connections(*vector_id)?;
                
                for neighbor_id in neighbors {
                    let neighbor_location = mapping.get_location(neighbor_id)?;
                    
                    // If neighbor is in a different file, this is a cross-file connection
                    if !files.contains(&neighbor_location.file_id) {
                        connections.add_external_connection(*vector_id, neighbor_id);
                    }
                }
            }
        }
        
        Ok(connections)
    }
}
```

### Phase 5: Detailed Query Handling During Compaction

```rust
pub enum FileRegistryState {
    Normal {
        active_files: HashMap<FileId, FileMetadata>,
    },
    Compacting {
        old_files: HashMap<FileId, FileMetadata>,
        new_file: Option<FileMetadata>,
        compaction_id: Uuid,
        transition_mapping: HashMap<VectorId, (VectorLocation, VectorLocation)>,
    },
    Transitioning {
        new_file: FileMetadata,
        draining_files: HashMap<FileId, FileMetadata>,
        cutover_time: Instant,
    },
}

impl ArtusQueryEngine {
    pub async fn search_with_compaction_awareness(
        &self,
        query: &[f32],
        predicates: &[Predicate],
        options: &SearchOptions,
    ) -> Result<SearchResult> {
        let registry_state = self.file_registry.read().await;
        
        match &*registry_state {
            FileRegistryState::Normal { active_files } => {
                // Standard query path
                self.standard_search(query, predicates, options, active_files).await
            }
            
            FileRegistryState::Compacting { old_files, new_file, transition_mapping, .. } => {
                // Dual-file query path
                self.compaction_aware_search(
                    query, 
                    predicates, 
                    options, 
                    old_files, 
                    new_file, 
                    transition_mapping
                ).await
            }
            
            FileRegistryState::Transitioning { new_file, draining_files, cutover_time } => {
                // Prefer new file, fallback to old files for missing data
                self.transitioning_search(
                    query, 
                    predicates, 
                    options, 
                    new_file, 
                    draining_files, 
                    *cutover_time
                ).await
            }
        }
    }
    
    async fn compaction_aware_search(
        &self,
        query: &[f32],
        predicates: &[Predicate],
        options: &SearchOptions,
        old_files: &HashMap<FileId, FileMetadata>,
        new_file: &Option<FileMetadata>,
        transition_mapping: &HashMap<VectorId, (VectorLocation, VectorLocation)>,
    ) -> Result<SearchResult> {
        let mut all_results = Vec::new();
        
        // Query old files for vectors not yet moved
        for (file_id, metadata) in old_files {
            let results = self.query_file(file_id, query, predicates, options).await?;
            all_results.extend(results.rows);
        }
        
        // Query new file if it exists
        if let Some(new_metadata) = new_file {
            let results = self.query_file(&new_metadata.file_id, query, predicates, options).await?;
            all_results.extend(results.rows);
        }
        
        // Deduplicate based on stable vector IDs
        all_results.sort_by(|a, b| a.id.cmp(&b.id));
        all_results.dedup_by(|a, b| a.id == b.id);
        
        // Re-sort by score and apply limit
        all_results.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());
        all_results.truncate(options.top_k);
        
        Ok(SearchResult { rows: all_results })
    }
}
```

### Phase 7: Background HNSW Maintenance

```rust
pub struct HNSWMaintenanceScheduler {
    maintenance_queue: Arc<Mutex<VecDeque<MaintenanceTask>>>,
    worker_pool: Arc<ThreadPool>,
    global_index: Arc<RwLock<HNSWIndex>>,
}

#[derive(Debug)]
pub enum MaintenanceTask {
    UpdateConnections {
        vector_id: VectorId,
        old_location: VectorLocation,
        new_location: VectorLocation,
        priority: Priority,
    },
    RebuildLocalGraph {
        file_id: FileId,
        affected_vectors: Vec<VectorId>,
    },
    ValidateConnections {
        sample_vectors: Vec<VectorId>,
        expected_quality: f32,
    },
    OptimizeGraph {
        region: GraphRegion,
        optimization_level: OptimizationLevel,
    },
}

impl HNSWMaintenanceScheduler {
    pub async fn schedule_compaction_maintenance(
        &self,
        compaction_plan: &CompactionPlan,
    ) -> Result<()> {
        // High priority: Update critical entry points
        for entry_point in &compaction_plan.affected_entry_points {
            self.maintenance_queue.lock().await.push_front(
                MaintenanceTask::UpdateConnections {
                    vector_id: entry_point.vector_id,
                    old_location: entry_point.old_location.clone(),
                    new_location: entry_point.new_location.clone(),
                    priority: Priority::Critical,
                }
            );
        }
        
        // Medium priority: Update cross-file connections
        for connection in &compaction_plan.cross_file_connections {
            self.maintenance_queue.lock().await.push_back(
                MaintenanceTask::UpdateConnections {
                    vector_id: connection.source_id,
                    old_location: connection.old_location.clone(),
                    new_location: connection.new_location.clone(),
                    priority: Priority::Normal,
                }
            );
        }
        
        // Low priority: Optimize local graph structure
        self.maintenance_queue.lock().await.push_back(
            MaintenanceTask::RebuildLocalGraph {
                file_id: compaction_plan.new_file_id,
                affected_vectors: compaction_plan.moved_vectors.clone(),
            }
        );
        
        // Background: Validate graph quality
        self.maintenance_queue.lock().await.push_back(
            MaintenanceTask::ValidateConnections {
                sample_vectors: compaction_plan.sample_for_validation(),
                expected_quality: 0.95,
            }
        );
        
        Ok(())
    }
}
```

## Key Design Decisions

### 1. Stable Vector IDs
- **Vector IDs remain constant** across compaction
- **Location mapping** updated atomically
- **Query consistency** maintained during transitions

### 2. Dual-Phase Compaction
- **Phase 1**: Write new file, update mapping
- **Phase 2**: Update HNSW, switch traffic
- **Atomic updates** in critical sections

### 3. Query Compatibility
- **Compaction-aware queries** handle multiple file states
- **Result deduplication** based on stable IDs
- **Graceful degradation** during updates

### 4. HNSW Update Strategies
- **Conservative**: Safe but slower updates
- **Optimistic**: Fast but riskier updates  
- **Hybrid**: Balanced approach with monitoring

### 5. Background Maintenance
- **Prioritized queue** for update tasks
- **Quality validation** after major changes
- **Incremental optimization** of graph structure

This design ensures that:

1. **Vector IDs remain stable** across compaction cycles
2. **HNSW graph consistency** is maintained during updates
3. **Query availability** continues during compaction
4. **Performance degradation** is minimized and bounded
5. **Recovery is automatic** with quality validation

The hybrid HNSW + ArtusV2 model successfully handles compaction while maintaining both query performance and graph structure integrity.
