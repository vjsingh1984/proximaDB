from proximadb import ChunkingConfig, ChunkingStrategy
from typing import List, Dict, Any
import re
from lxml import etree

class SECFilingParser:
    """
    Advanced parser that handles XBRL, HTML, and TXT formats
    with intelligent structure preservation
    """
    
    def __init__(self):
        self.xbrl_namespaces = {
            'xbrli': 'http://www.xbrl.org/2003/instance',
            'us-gaap': 'http://fasb.org/us-gaap/',
            'dei': 'http://xbrl.sec.gov/dei/'
        }
        
    def parse_filing(self, content: str, format: str) -> Dict[str, Any]:
        """
        Intelligently parse filing regardless of format issues
        """
        if format == 'xbrl':
            return self._parse_xbrl_with_fallback(content)
        elif format == 'html':
            return self._parse_html_with_recovery(content)
        else:  # txt
            return self._parse_txt_with_patterns(content)
    
    def _parse_xbrl_with_fallback(self, content: str) -> Dict[str, Any]:
        """
        Parse XBRL with multiple fallback strategies
        """
        sections = {}
        
        try:
            # Primary: Parse as structured XML
            root = etree.fromstring(content.encode())
            
            # Extract contexts (time periods, segments)
            contexts = self._extract_xbrl_contexts(root)
            
            # Extract facts with context
            facts = self._extract_xbrl_facts(root, contexts)
            
            # Build hierarchical structure
            sections = self._build_xbrl_hierarchy(facts)
            
        except etree.XMLSyntaxError:
            # Fallback 1: Extract as text blocks
            sections = self._extract_xbrl_text_blocks(content)
            
        except Exception:
            # Fallback 2: Pattern-based extraction
            sections = self._pattern_extract_xbrl(content)
        
        return {
            'format': 'xbrl',
            'sections': sections,
            'metadata': self._extract_xbrl_metadata(content)
        }
    
    def _parse_html_with_recovery(self, content: str) -> Dict[str, Any]:
        """
        Parse HTML with table structure recovery
        """
        from bs4 import BeautifulSoup, NavigableString
        
        soup = BeautifulSoup(content, 'html.parser')
        sections = {}
        
        # Strategy 1: Find semantic sections
        for header in soup.find_all(['h1', 'h2', 'h3']):
            section_name = header.get_text(strip=True)
            section_content = []
            
            for sibling in header.find_next_siblings():
                if sibling.name and sibling.name.startswith('h'):
                    break
                    
                if sibling.name == 'table':
                    # Extract table with structure preservation
                    table_data = self._extract_html_table(sibling)
                    section_content.append({
                        'type': 'table',
                        'data': table_data
                    })
                else:
                    text = sibling.get_text(strip=True)
                    if text:
                        section_content.append({
                            'type': 'text',
                            'data': text
                        })
            
            sections[section_name] = section_content
        
        # Strategy 2: Pattern-based section detection
        if not sections:
            sections = self._detect_sections_by_pattern(soup.get_text())
        
        return {
            'format': 'html',
            'sections': sections,
            'tables': self._extract_all_tables(soup)
        }
    
    def _parse_txt_with_patterns(self, content: str) -> Dict[str, Any]:
        """
        Parse TXT format using pattern recognition
        """
        sections = {}
        lines = content.split('\n')
        
        # Detect ASCII art tables
        table_regions = self._detect_ascii_tables(lines)
        
        # Detect section headers (ITEM 1, PART I, etc.)
        section_pattern = re.compile(
            r'^(ITEM|PART|SECTION)\s+([0-9IVX]+[A-Z]?)\s*[.-]?\s*(.+)$',
            re.IGNORECASE
        )
        
        current_section = None
        current_content = []
        
        for i, line in enumerate(lines):
            # Check if line is part of a table
            if any(i >= start and i <= end for start, end in table_regions):
                if current_section and current_content:
                    # Parse the table
                    table_start, table_end = next(
                        (s, e) for s, e in table_regions if i >= s and i <= e
                    )
                    table_text = '\n'.join(lines[table_start:table_end+1])
                    current_content.append({
                        'type': 'ascii_table',
                        'data': self._parse_ascii_table(table_text)
                    })
                continue
            
            # Check for section header
            match = section_pattern.match(line.strip())
            if match:
                # Save previous section
                if current_section:
                    sections[current_section] = current_content
                
                # Start new section
                current_section = f"{match.group(1)} {match.group(2)}: {match.group(3)}"
                current_content = []
            else:
                # Add to current section
                if line.strip():
                    current_content.append({
                        'type': 'text',
                        'data': line.strip()
                    })
        
        # Save last section
        if current_section:
            sections[current_section] = current_content
        
        return {
            'format': 'txt',
            'sections': sections
        }
    
    def _extract_html_table(self, table_element) -> pd.DataFrame:
        """
        Extract HTML table preserving structure
        """
        rows = []
        headers = []
        
        # Extract headers
        for th in table_element.find_all('th'):
            headers.append(th.get_text(strip=True))
        
        # Extract rows
        for tr in table_element.find_all('tr'):
            row = []
            for td in tr.find_all('td'):
                row.append(td.get_text(strip=True))
            if row:
                rows.append(row)
        
        # Create DataFrame
        if headers and rows:
            return pd.DataFrame(rows, columns=headers[:len(rows[0])])
        elif rows:
            return pd.DataFrame(rows)
        else:
            return pd.DataFrame()
    
    def _detect_ascii_tables(self, lines: List[str]) -> List[tuple]:
        """
        Detect ASCII art tables in text
        """
        table_regions = []
        in_table = False
        table_start = -1
        
        for i, line in enumerate(lines):
            # Simple heuristic: lines with multiple pipes or dashes
            is_table_line = (
                line.count('|') >= 2 or
                line.count('-') >= 10 or
                line.count('=') >= 10
            )
            
            if is_table_line and not in_table:
                table_start = i
                in_table = True
            elif not is_table_line and in_table:
                table_regions.append((table_start, i-1))
                in_table = False
        
        return table_regions

class HybridEmbeddingStrategy:
    """
    Use different models for different document sections
    """
    
    def __init__(self):
        from sentence_transformers import SentenceTransformer
        import openai
        
        # Primary: Finance-specific model for financial sections
        self.finbert = SentenceTransformer('yiyanghkust/finbert-tone')
        
        # Secondary: General model for narrative sections
        self.e5_large = SentenceTransformer('intfloat/e5-large-v2')
        
        # Fallback: OpenAI for complex/mixed content
        self.openai_client = openai.Client()
        
        # Lightweight: For metadata and short text
        self.minilm = SentenceTransformer('all-MiniLM-L6-v2')
    
    def get_embedding(self, text: str, section_type: str) -> List[float]:
        """
        Choose model based on section type
        """
        if section_type in ['financial_statements', 'notes_to_financials']:
            # Use FinBERT for financial data
            return self.finbert.encode(text).tolist()
            
        elif section_type in ['risk_factors', 'mda']:
            # Use E5-Large for narrative analysis
            instruction = "Represent this financial disclosure for retrieval: "
            return self.e5_large.encode(instruction + text).tolist()
            
        elif section_type in ['business_overview', 'legal_proceedings']:
            # Use OpenAI for general content
            response = self.openai_client.embeddings.create(
                model="text-embedding-3-small",
                input=text
            )
            return response.data[0].embedding
            
        else:
            # Use MiniLM for metadata/short text
            return self.minilm.encode(text).tolist()
    
    def create_multi_vector_representation(
        self,
        text: str,
        section_type: str
    ) -> Dict[str, List[float]]:
        """
        Create multiple embeddings for different purposes
        """
        return {
            'semantic': self.get_embedding(text, section_type),
            'keyword': self._create_keyword_embedding(text),
            'entity': self._create_entity_embedding(text),
            'numeric': self._create_numeric_embedding(text)
        }
    
    def _create_keyword_embedding(self, text: str) -> List[float]:
        """
        TF-IDF based keyword embedding
        """
        from sklearn.feature_extraction.text import TfidfVectorizer
        
        # Extract financial keywords
        keywords = self._extract_financial_keywords(text)
        
        # Create sparse embedding
        vectorizer = TfidfVectorizer(max_features=768)
        sparse_vector = vectorizer.fit_transform([' '.join(keywords)])
        
        # Convert to dense
        return sparse_vector.toarray()[0].tolist()
    
    def _create_entity_embedding(self, text: str) -> List[float]:
        """
        Entity-based embedding using NER
        """
        import spacy
        
        nlp = spacy.load("en_core_web_sm")
        doc = nlp(text)
        
        entities = []
        for ent in doc.ents:
            if ent.label_ in ['ORG', 'MONEY', 'PERCENT', 'DATE']:
                entities.append(f"{ent.label_}:{ent.text}")
        
        # Embed entities
        if entities:
            entity_text = ' '.join(entities)
            return self.minilm.encode(entity_text).tolist()
        else:
            return [0.0] * 384
    
    def _create_numeric_embedding(self, text: str) -> List[float]:
        """
        Extract and embed numeric patterns
        """
        import re
        
        # Extract numbers with context
        number_pattern = re.compile(
            r'(\$?[\d,]+\.?\d*)\s*(million|billion|thousand|%)?'
        )
        
        numbers = []
        for match in number_pattern.finditer(text):
            value = match.group(1).replace(',', '').replace('$', '')
            unit = match.group(2) or ''
            
            # Normalize to standard scale
            try:
                num = float(value)
                if 'billion' in unit.lower():
                    num *= 1e9
                elif 'million' in unit.lower():
                    num *= 1e6
                elif 'thousand' in unit.lower():
                    num *= 1e3
                
                numbers.append(num)
            except ValueError:
                continue
        
        # Create statistical embedding
        if numbers:
            import numpy as np
            stats = [
                np.mean(numbers),
                np.std(numbers),
                np.min(numbers),
                np.max(numbers),
                len(numbers)
            ]
            # Pad to fixed size
            return stats + [0.0] * (128 - len(stats))
        else:
            return [0.0] * 128

        """

        from datasets import Dataset
        
        tokenizer = AutoTokenizer.from_pretrained('bert-base-uncased')
        
        texts = []
        labels = []
        
        # Collect filing texts with labels
        for filing_path in Path(filings_dir).glob('*.txt'):
            filing_type = self._extract_filing_type(filing_path)
            content = filing_path.read_text()
            
            # Split into sentences
            sentences = self._split_into_sentences(content)
            
            for sentence in sentences:
                texts.append(sentence)
                labels.append(filing_type)
        
        # Tokenize
        encodings = tokenizer(
            texts,
            truncation=True,
            padding=True,
            max_length=512
        )
        
        # Create dataset
        dataset = Dataset.from_dict({
            'input_ids': encodings['input_ids'],
            'attention_mask': encodings['attention_mask'],
            'labels': labels
        })
        
        return dataset

        Fine-tune BERT on SEC data

            AutoModelForSequenceClassification,
            TrainingArguments,

            'bert-base-uncased',

from proximadb import ChunkingStrategy, ChunkingConfig
from proximadb.chunking import TextChunker

class SECFilingChunker:
    """
    Advanced chunking that handles messy SEC filing formats
    """
    
    def __init__(self):
        # Different strategies for different content types
        self.strategies = {
            'table': ChunkingConfig(
                strategy=ChunkingStrategy.FIXED_SIZE,
                chunk_size=4096,  # Large chunks for tables
                chunk_overlap=0,
                preserve_tables=True,
                table_format='markdown'
            ),
            
            'narrative': ChunkingConfig(
                strategy=ChunkingStrategy.RECURSIVE,
                chunk_size=1024,
                chunk_overlap=128,
                split_on=['\n\n', '\n', '. ', ', '],
                preserve_sentences=True,
                add_context=True,
                context_size=100
            ),
            
            'risk_factors': ChunkingConfig(
                strategy=ChunkingStrategy.SEMANTIC,
                chunk_size=768,
                chunk_overlap=100,
                similarity_threshold=0.7,
                min_chunk_size=256,
                max_chunk_size=1536
            ),
            
            'financial_notes': ChunkingConfig(
                strategy=ChunkingStrategy.PARAGRAPH,
                preserve_paragraphs=True,
                include_headers=True,
                min_chunk_size=200,
                max_chunk_size=2000
            )
        }
    
    def chunk_filing(self, filing_content: Dict[str, Any]) -> List[TextChunk]:
        """
        Intelligently chunk filing based on content type
        """
        all_chunks = []
        
        for section_name, section_content in filing_content['sections'].items():
            # Determine section type
            section_type = self._classify_section(section_name)
            
            # Get appropriate strategy
            config = self.strategies.get(section_type, self.strategies['narrative'])
            chunker = TextChunker(config)
            
            # Process based on content type
            for item in section_content:
                if item['type'] == 'table':
                    # Special handling for tables
                    chunks = self._chunk_table(item['data'], section_name)
                    
                elif item['type'] == 'ascii_table':
                    # Convert ASCII table to structured format first
                    structured = self._structure_ascii_table(item['data'])
                    chunks = self._chunk_table(structured, section_name)
                    
                else:  # text
                    # Standard text chunking
                    chunks = chunker.chunk_text(
                        item['data'],
                        source_id=f"{filing_content['ticker']}_{filing_content['form_type']}",
                        metadata={
                            'section': section_name,
                            'section_type': section_type,
                            'format': filing_content['format']
                        }
                    )
                
                all_chunks.extend(chunks)
        
        return all_chunks
    
    def _chunk_table(self, table_data: pd.DataFrame, section_name: str) -> List[TextChunk]:
        """
        Chunk tabular data intelligently
        """
        chunks = []
        
        # Strategy 1: Keep small tables together
        if len(table_data) <= 10:
            chunk_text = table_data.to_markdown()
            chunks.append(TextChunk(
                chunk_id=f"table_{hash(chunk_text)}",
                text=chunk_text,
                metadata={
                    'type': 'table',
                    'rows': len(table_data),
                    'columns': list(table_data.columns),
                    'section': section_name
                }
            ))
        
        # Strategy 2: Split large tables by logical groups
        else:
            # Look for natural breakpoints (subtotals, categories)
            breakpoints = self._find_table_breakpoints(table_data)
            
            for i, (start, end) in enumerate(breakpoints):
                sub_table = table_data.iloc[start:end]
                chunk_text = sub_table.to_markdown()
                
                chunks.append(TextChunk(
                    chunk_id=f"table_{hash(chunk_text)}_{i}",
                    text=chunk_text,
                    metadata={
                        'type': 'table_segment',
                        'segment': i,
                        'total_segments': len(breakpoints),
                        'section': section_name
                    }
                ))
        
        return chunks
